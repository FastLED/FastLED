229fl::size basic_string::write(const char* str, fl::size n) {
230 fl::size newLen = mLength + n;
231
232 // Handle non-owning storage
233 if (isNonOwning()) {
234 const char* existingData = constData();
235 fl::size existingLen = mLength;
236 if (newLen + 1 <= mInlineCapacity) {
237 if (existingLen > 0 && existingData) {
238 fl::memcpy(inlineBufferPtr(), existingData, existingLen);
239 }
240 fl::memcpy(inlineBufferPtr() + existingLen, str, n);
241 inlineBufferPtr()[newLen] = '\0';
242 mStorage.reset();
243 mLength = newLen;
244 } else {
245 NotNullStringHolderPtr newData = NotNullStringHolderPtr(fl::make_shared<StringHolder>(newLen));
246 if (existingLen > 0 && existingData) {
247 fl::memcpy(newData->data(), existingData, existingLen);
248 }
249 fl::memcpy(newData->data() + existingLen, str, n);
250 newData->data()[newLen] = '\0';
251 mStorage = newData;
252 mLength = newLen;
253 }
254 return mLength;
255 }
256
257 if (hasHeapData() && heapData().get().use_count() <= 1) {
258 NotNullStringHolderPtr& heap = heapData();
259 if (!heap->hasCapacity(newLen)) {
260 // Check if str points into our buffer (self-referential write).
261 // grow() uses realloc which can relocate the buffer.
262 const char* bufStart = heap->data();
263 fl::size grow_length = fl::max(3, newLen * 3 / 2);
264 if (str >= bufStart && str < bufStart + mLength + 1) {
265 fl::size offset = static_cast<fl::size>(str - bufStart);
266 heap->grow(grow_length);
267 str = heap->data() + offset; // update to new location
268 } else {
269 heap->grow(grow_length);
270 }
271 }
272 fl::memcpy(heap->data() + mLength, str, n);
273 mLength = newLen;
274 heap->data()[mLength] = '\0';
275 return mLength;
276 } else if (hasHeapData()) {
277 // Copy-on-write
278 NotNullStringHolderPtr newData = NotNullStringHolderPtr(fl::make_shared<StringHolder>(newLen));
279 {
280 const NotNullStringHolderPtr& heap = heapData();
281 fl::memcpy(newData->data(), heap->data(), mLength);
282 fl::memcpy(newData->data() + mLength, str, n);
283 newData->data()[newLen] = '\0';
284 mStorage = newData;
285 mLength = newLen;
286 }
287 return mLength;
288 }
289 if (newLen + 1 <= mInlineCapacity) {
290 FL_DISABLE_WARNING_PUSH
291 FL_DISABLE_WARNING(array-bounds)
292 fl::memcpy(inlineBufferPtr() + mLength, str, n);
293 FL_DISABLE_WARNING_POP
294 mLength = newLen;
295 inlineBufferPtr()[mLength] = '\0';
296 return mLength;
297 }
298 // Transition from inline to heap
299 NotNullStringHolderPtr newData = NotNullStringHolderPtr(fl::make_shared<StringHolder>(newLen));
300 {
301 fl::memcpy(newData->data(), inlineBufferPtr(), mLength);
302 fl::memcpy(newData->data() + mLength, str, n);
303 newData->data()[newLen] = '\0';
304 mStorage = newData;
305 mLength = newLen;
306 }
307 return mLength;
308}
