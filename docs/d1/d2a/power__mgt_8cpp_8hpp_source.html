78// Source order, not wire order: RGB, then W or warm-W/cool-W. An undeclared
79// white gets the maximum representable emitter draw, so a managed channel
80// cannot silently under-budget it. Legacy estimates keep their old fallback.
81static void emitter_power_weights(fl::u8 count, fl::u8 (&weights)[5]) {
82 const PowerModelRGB& rgb = gPowerModel();
83 weights[0] = rgb.red_mW;
84 weights[1] = rgb.green_mW;
85 weights[2] = rgb.blue_mW;
86 weights[3] = 0;
87 weights[4] = 0;
88 if (count == 4) {
89 const fl::u8 white = gWhiteEmitterPower().mW;
90 weights[3] = white != 0 ? white : 255;
91 } else if (count == 5) {
92 const RgbwwEmitterPower& rgbww = gRgbwwEmitterPower();
93 if (rgbww.declared) {
94 weights[0] = rgbww.model.red_mW;
95 weights[1] = rgbww.model.green_mW;
96 weights[2] = rgbww.model.blue_mW;
97 weights[3] = rgbww.model.warm_white_mW;
98 weights[4] = rgbww.model.white_mW;
99 } else {
100 weights[3] = 255;
101 weights[4] = 255;
102 }
103 }
104}

124 for (fl::size i = 0; i < kPowerScalingTableSize; ++i) {
125 forward[i] = static_cast<fl::u8>(i);
126 reverse[i] = static_cast<fl::u8>(i);

138static void rebuild_power_scaling_tables(float exponent) {
139 PowerScalingState& state = gPowerScaling();
140 if (!(exponent > 0.0f) ||
141 fl::almost_equal(exponent, kLinearPowerScalingExponent, kPowerScalingExponentEpsilon)) {
142 state.reset_identity();
143 return;
144 }
145
146 // Forward LUT: source brightness -> scaled brightness via pow(x/255, exponent)
147 state.forward[0] = 0;
148 for (fl::size i = 1; i < kPowerScalingTableSize; ++i) {
149 float normalized = static_cast<float>(i) / 255.0f;
150 // Route through fl::powf / fl::lroundf (libm-gated via FL_MATH_USE_LIBM,
151 // see fl/math/math.cpp.hpp). Use the float overloads — fl::pow takes
152 // double, so passing floats would auto-promote and re-anchor the
153 // double-precision soft-FP chain that #3002 is trying to avoid.
154 int mapped = static_cast<int>(
155 fl::lroundf(fl::powf(normalized, exponent) * 255.0f));
156 state.forward[i] = static_cast<fl::u8>(fl::clamp(mapped, 0, 255));
157 }
158 state.forward[255] = 255;
159 state.max_step = 1;
160 for (fl::size i = 1; i < kPowerScalingTableSize; ++i) {
161 const int step = state.forward[i] - state.forward[i - 1];
162 if (step > state.max_step) {
163 state.max_step = static_cast<fl::u8>(step);
164 }
165 }
166
167 // Reverse LUT: scaled brightness -> largest source whose forward value is
168 // <= the scaled value. Floor-inverse, so `unmap_power_value()` never
169 // rounds a budgeted scaled brightness *up* past the power budget.
170 state.reverse[0] = 0;
171 int source = 0;
172 for (int scaled = 1; scaled < static_cast<int>(kPowerScalingTableSize); ++scaled) {
173 while (source < 255 && state.forward[source + 1] <= scaled) {
174 ++source;
175 }
176 state.reverse[scaled] = static_cast<fl::u8>(source);
177 }
178}
179#endif

240// Span-based version (primary implementation)
241fl::u32 calculate_unscaled_power_mW(fl::span<const CRGB> leds) {
242 fl::u32 red32 = 0, green32 = 0, blue32 = 0;
243
244 // Iterate using span's safe indexing
245 for(size_t i = 0; i < leds.size(); i++) {
246 red32 += map_power_value(leds[i].r);
247 green32 += map_power_value(leds[i].g);
248 blue32 += map_power_value(leds[i].b);
249 }
250
251 red32 *= gPowerModel().red_mW;
252 green32 *= gPowerModel().green_mW;
253 blue32 *= gPowerModel().blue_mW;
254
255 red32 >>= 8;
256 green32 >>= 8;
257 blue32 >>= 8;
258
259 fl::u32 total = red32 + green32 + blue32 + (gPowerModel().dark_mW * leds.size());
260
261 return total;
262}

264fl::u32 calculate_unscaled_emitter_power_mW(fl::span<const fl::u8> codes,
265 fl::u8 emitter_count) {
266 if (emitter_count < 3 || emitter_count > 5 ||
267 codes.size() % emitter_count != 0) {
268 return 0;
269 }
270 fl::u8 weights[5];
271 emitter_power_weights(emitter_count, weights);
272 fl::u32 sums[5] = {};
273 for (fl::size i = 0; i < codes.size(); i += emitter_count) {
274 for (fl::u8 channel = 0; channel < emitter_count; ++channel) {
275 sums[channel] += map_power_value(codes[i + channel]);
276 }
277 }
278 fl::u32 total = static_cast<fl::u32>(gPowerModel().dark_mW) *
279 (codes.size() / emitter_count);
280 for (fl::u8 channel = 0; channel < emitter_count; ++channel) {
281 total += (sums[channel] * weights[channel]) >> 8;
282 }
283 return total;
284}

292// Four emitters, using the conversion the encoder will actually run.
293//
294// The three-emitter overload reads the source triple the sketch wrote. For an
295// RGBW controller that triple is an input to `rgb_2_rgbw`, not a description
296// of the drives, and the two disagree by more than rounding: at full white on
297// the shipped default model the source triple reads 249 while the strip draws
298// 348 under `kRGBWMaxBrightness` and 99 under `kRGBWExactColors`. #4156 R3.
299//
300// Scales are passed as 255 because this is the demand at full brightness --
301// the caller scales the result. That is exact for `kRGBWExactColors` and
302// `kRGBWMaxBrightness`, whose conversions commute with the scale to within
303// 0.4% over the measured cases. It is not exact for `kRGBWBoostedWhite`,
304// which reallocates between emitters as the scale drops: demand there runs up
305// to 12.1% above the linear projection at low brightness. Charging four
306// emitters instead of three is the larger correction by far, and the residual
307// is stated rather than hidden -- closing it needs demand evaluated per
308// candidate brightness, which is a pixel walk per step of the limiter's
309// search.
310fl::u32 calculate_unscaled_power_mW(fl::span<const CRGB> leds, const fl::Rgbw& rgbw) {
311 const fl::u8 white_mW = gWhiteEmitterPower().mW;
312 // `!rgbw.active()` is a fast path, not a different answer: the only
313 // inactive mode is `kRGBWInvalid`, which `rgb_2_rgbw` dispatches to
314 // null-white -- RGB passed through with w = 0 -- so the loop below would
315 // return the same total. Removing the test changes nothing observable
316 // and costs one conversion per pixel on every plain RGB controller in
317 // the sketch, which the estimator walks once per frame.
318 //
319 // The `white_mW == 0` half *is* behavioural: with no declared white
320 // there is nothing to charge the fourth diode at.
321 if (!rgbw.active() || white_mW == 0) {
322 if (rgbw.active()) {
323 FL_WARN_ONCE("power: controller is in RGBW mode but the power model "
324 "declares no white emitter, so the budget covers three "
325 "of its four diodes. Call "
326 "FastLED.setPowerModel(PowerModelRGBW(...)) to fix it.");
327 }
328 return calculate_unscaled_power_mW(leds);
329 }
330
331 fl::u32 red32 = 0, green32 = 0, blue32 = 0, white32 = 0;
332 for (fl::size i = 0; i < leds.size(); i++) {
333 fl::u8 r = 0, g = 0, b = 0, w = 0;
334 fl::rgb_2_rgbw(rgbw, leds[i].r, leds[i].g, leds[i].b, 255, 255, 255,
335 &r, &g, &b, &w);
336 red32 += map_power_value(r);
337 green32 += map_power_value(g);
338 blue32 += map_power_value(b);
339 white32 += map_power_value(w);
340 }
341
342 red32 *= gPowerModel().red_mW;
343 green32 *= gPowerModel().green_mW;
344 blue32 *= gPowerModel().blue_mW;
345 white32 *= white_mW;
346
347 red32 >>= 8;
348 green32 >>= 8;
349 blue32 >>= 8;
350 white32 >>= 8;
351
352 return red32 + green32 + blue32 + white32 +
353 (gPowerModel().dark_mW * leds.size());
354}

357// The part of the estimate a brightness scalar cannot touch: every controller
358// IC draws its quiescent current whether or not an emitter is lit, and the MCU
359// draws its own regardless of what the strip is doing. #4156 R4:
360//
361// Fixed idle consumption cannot be reduced by multiplying LED flux.
362//
363// Scaling that baseline along with the emitters is what let the limiter
364// under-report. At 300 WS2812s on the default model the baseline is 1500 mW,
365// and a 2000 mW budget used to be answered with a brightness that draws 3461.
366static fl::u32 fixed_power_mW(fl::u32 led_count) {
367 return static_cast<fl::u32>(gPowerModel().dark_mW) * led_count;
368}
369
370// Largest brightness whose *total* demand -- baseline included -- stays inside
371// the budget. Zero when the baseline alone is already over it: no brightness
372// meets the budget then, and answering with a lit strip would promise a bound
373// that cannot be held at any setting.
374static fl::u8 brightness_within_budget(fl::u32 fixed_mW, fl::u32 controllable_mW,
375 fl::u8 target_brightness,
376 fl::u32 max_power_mW) {
377 const fl::u32 requested_mW =
378 fixed_mW + scale_power_for_brightness(controllable_mW, target_brightness);
379 if (requested_mW <= max_power_mW) {
380 return target_brightness;
381 }
382 if (controllable_mW == 0 || max_power_mW <= fixed_mW) {
383 // Nothing left to scale, or the baseline alone is already over the
384 // budget. Either way no brightness meets it, and returning here is
385 // also what keeps the subtraction below from underflowing and the
386 // division from dividing by zero on an all-dark strip.
387 return 0;
388 }
389 // The ratio is taken over the whole 0-255 scaled range, because
390 // `controllable_mW` is the demand at *full* brightness. Scaling it by the
391 // requested brightness first would answer "what fraction of the request
392 // fits", which is smaller than "what brightness fits" by exactly that
393 // fraction -- and the step-down below only ever decreases, so it cannot
394 // recover it. The cap is where the request comes back in.
395 //
396 // The cap cannot actually fire from here: the caller only reaches this
397 // branch when the request is already over budget, which is the same
398 // inequality. It is kept because that argument runs through
399 // `map_power_value`'s rounding, and a helper that clamps to its own
400 // argument is cheaper than depending on that.
401 const fl::u32 headroom_mW = max_power_mW - fixed_mW;
402 const fl::u8 target_scaled = map_power_value(target_brightness);
403 fl::u64 allowed_scaled =
404 (static_cast<fl::u64>(255) * headroom_mW) / controllable_mW;
405 if (allowed_scaled > target_scaled) {
406 allowed_scaled = target_scaled;
407 }
408 const fl::u32 recommended_scaled = static_cast<fl::u32>(allowed_scaled);
409 fl::u8 recommended = unmap_power_value(static_cast<fl::u8>(recommended_scaled));
410
411 // Then check the answer instead of trusting the division. Inverting the
412 // ratio in scaled space is off by the rounding of whichever `scale32by8`
413 // the platform compiled -- the FASTLED_SCALE8_FIXED form computes
414 // `i * (s + 1) >> 8`, so the closed form lands about `controllable/256`
415 // over the budget, which was 176 mW of a 5000 mW budget on the default
416 // model. Demand is monotone in brightness, so stepping down until it fits
417 // is exact whatever the rounding, and terminates: 0 always fits here,
418 // because the branch above already returned for a budget under baseline.
419 while (recommended > 0 &&
420 fixed_mW + scale_power_for_brightness(controllable_mW, recommended) >
421 max_power_mW) {
422 --recommended;
423 }
424 return recommended;
425}

440fl::u32 controller_unscaled_power_mW(const fl::CLEDController& controller) {
441 const fl::span<const CRGB> leds(controller.leds(),
442 static_cast<fl::size>(controller.size()));
443#if FL_COLOR_PIPELINE_SHARED
444 // Through the pipeline hook, which exists whenever a pipeline does; naming
445 // the estimator directly would link the colour pipeline into sketches that
446 // limit power but never bind a profile. `pipeline` is this estimate's own
447 // reference, so reconfiguring the controller mid-pass cannot free the
448 // pipeline it is walking (#4440).
449 const fl::shared_ptr<fl::StreamingPipelineQ16> pipeline = controller.colorPipeline();
450 const fl::ColorPipelineHooks& hooks = fl::colorPipelineHooks();
451 if (pipeline && hooks.unscaledPowerMilliwatts != nullptr) {
452 // The callback charges already-solved physical emitter bytes. In
453 // particular, the RGBW legacy estimator must not convert them again.
454 const fl::u8 emitters = controller.getRgbww().active()
455 ? 5 : (controller.getRgbw().active() ? 4 : 3);
456 return hooks.unscaledPowerMilliwatts(
457 *pipeline, leds, emitters, &calculate_unscaled_emitter_power_mW);
458 }
459#endif
460 // Below the large-memory tier only this is compiled: a managed channel
461 // there is charged for its source (FL_COLOR_PIPELINE_SHARED).
462 return calculate_unscaled_power_mW(leds, controller.getRgbw());
463}

465// Two output codes: the most one `BINARY_DITHER` offset can lift a lit channel
466// once scaled. `d <= 256/s` scales back to at most one code, and the
467// FASTLED_SCALE8_FIXED form `(x * (s + 1)) >> 8` can carry one more; an
468// exhaustive check over every source, scale and offset finds 2 as the maximum.
469static constexpr fl::u32 kDitherReserveCodes = 2;

471#if FL_PLATFORM_HAS_TINY_MEMORY

476 FL_UNUSED(leds);
477 return 0;

485fl::u32 dither_reserve_mW(fl::span<const CRGB> leds) {
486 fl::u32 lit_r = 0, lit_g = 0, lit_b = 0;
487 for (fl::size i = 0; i < leds.size(); ++i) {
488 // A zero channel is never dithered: `dither()` returns 0 for it.
489 lit_r += leds[i].r != 0 ? 1u : 0u;
490 lit_g += leds[i].g != 0 ? 1u : 0u;
491 lit_b += leds[i].b != 0 ? 1u : 0u;
492 }
493 const PowerModelRGB& model = gPowerModel();
494 const fl::u64 per_code =
495 static_cast<fl::u64>(lit_r) * model.red_mW +
496 static_cast<fl::u64>(lit_g) * model.green_mW +
497 static_cast<fl::u64>(lit_b) * model.blue_mW;
498 // Same >> 8 as the estimator, rounded up: this is a bound, and truncating
499 // it would hand back the under-charge it exists to remove.
500 const fl::u64 scaled = per_code * kDitherReserveCodes * max_power_step();
501 const fl::u64 reserve = (scaled + 255) >> 8;
502 // Saturate rather than wrap: an under-stated reserve is the failure this
503 // exists to prevent.
504 return reserve > 0xFFFFFFFFu ? 0xFFFFFFFFu : static_cast<fl::u32>(reserve);
505}

507fl::u32 controller_dither_reserve_mW(const fl::CLEDController& controller) {
508 if (controller.getDither() != BINARY_DITHER) {
509 return 0;
510 }
511#if FL_COLOR_PIPELINE_SHARED
512 // A managed channel never takes the legacy offsets (C5); with
513 // BINARY_DITHER it runs the pipeline's temporal dither instead, which
514 // emits floor(x) or floor(x) + 1 of the exact code -- at most one code
515 // above the rounded code its estimate charges. The solve can light a
516 // channel the source left dark, so every channel of every lit pixel is
517 // reserved one code.
518 if (controller.colorPipeline()) {
519 const fl::span<const CRGB> leds(controller.leds(),
520 static_cast<fl::size>(controller.size()));
521 fl::u32 lit = 0;
522 for (fl::size i = 0; i < leds.size(); ++i) {
523 lit += (leds[i].r | leds[i].g | leds[i].b) != 0 ? 1u : 0u;
524 }
525 const fl::u8 emitters = controller.getRgbww().active()
526 ? 5 : (controller.getRgbw().active() ? 4 : 3);
527 fl::u8 weights[5];
528 emitter_power_weights(emitters, weights);
529 fl::u32 sum_weights = 0;
530 for (fl::u8 i = 0; i < emitters; ++i) {
531 sum_weights += weights[i];
532 }
533 const fl::u64 per_code = static_cast<fl::u64>(lit) * sum_weights;
534 const fl::u64 reserve = (per_code * max_power_step() + 255) >> 8;
535 return reserve > 0xFFFFFFFFu ? 0xFFFFFFFFu
536 : static_cast<fl::u32>(reserve);
537 }
538#endif
539#if !defined(FL_IS_AVR)
540 // Legacy RGBW and RGBWW conversions read the raw pixel, not the
541 // dithered one. Managed channels have already returned above.
542 if (controller.getRgbw().active() || controller.getRgbww().active()) {
543 return 0;
544 }
545#endif
546 return dither_reserve_mW(fl::span<const CRGB>(
547 controller.leds(), static_cast<fl::size>(controller.size())));
548}

554// Candidate-specific legacy demand. Unlike the old unity-demand projection,
555// this runs the same brightness premix and white extraction as PixelController.
556// Dither can raise a lit channel by at most two codes, so charge that upward
557// choice without touching the global/frame phase.
558fl::u32 legacyFramePowerMilliwatts(fl::CLEDController& controller,
559 fl::u8 brightness) FL_NO_EXCEPT {
560 const fl::u32 count = static_cast<fl::u32>(controller.size());
561 const fl::u8 emitters = controller.getRgbww().active() ? 5 :
562 (controller.getRgbw().active() ? 4 : 3);
563 fl::u8 weights[5];
564 emitter_power_weights(emitters, weights);
565 fl::u64 sums[5] = {};
566 PixelController<RGB> pixels(
567 controller.leds(), controller.size(),
568 controller.getAdjustmentData(brightness), DISABLE_DITHER);
569 for (fl::u32 i = 0; i < count; ++i) {
570 fl::u8 codes[5] = {};
571 if (emitters == 5) {
572 pixels.loadAndScaleRGBWWUnordered(controller.getRgbww(),
573 &codes[0], &codes[1], &codes[2], &codes[3], &codes[4]);
574 } else if (emitters == 4) {
575 pixels.loadAndScaleRGBWUnordered(controller.getRgbw(),
576 &codes[0], &codes[1], &codes[2], &codes[3]);
577 } else {
578 pixels.loadAndScaleRGB(&codes[0], &codes[1], &codes[2]);
579 }
580 const CRGB& source = controller.leds()[i];
581 const bool dithered = controller.getDither() == BINARY_DITHER &&
582 (source.r | source.g | source.b) != 0 && brightness != 0;
583 for (fl::u8 c = 0; c < emitters; ++c) {
584 const fl::u8 upper = dithered
585 ? static_cast<fl::u8>(codes[c] > 253 ? 255 : codes[c] + 2)
586 : codes[c];
587 sums[c] += map_power_value(upper);
588 }
589 pixels.advanceData();
590 }
591 fl::u64 total = static_cast<fl::u64>(gPowerModel().dark_mW) * count;
592 for (fl::u8 c = 0; c < emitters; ++c) {
593 total += (sums[c] * weights[c] + 255u) >> 8;
594 }
595 return total > 0xFFFFFFFFu ? 0xFFFFFFFFu : static_cast<fl::u32>(total);
596}

606 fl::vector<ManagedFramePower> managed;

610bool captureFramePowerSnapshot(FramePowerSnapshot* snapshot) FL_NO_EXCEPT {
611 const fl::PowerFrameHooks& hooks = fl::powerFrameHooks();
612 fl::size managed_count = 0;
613 fl::size legacy_count = 0;
614 for (fl::CLEDController* controller = fl::CLEDController::head();
615 controller != nullptr; controller = controller->next()) {
616 if (!controller->getEnabled()) continue;
617 if (controller->colorPipeline() && hooks.buildPowerHistogram) {
618 ++managed_count;
619 } else {
620 ++legacy_count;
621 }
622 }
623 // Allocate exactly once, then construct histograms in their final slots.
624 // fl::vector silently leaves size unchanged when allocation fails.
625 snapshot->managed.resize(managed_count);
626 snapshot->legacy.resize(legacy_count);
627 if (snapshot->managed.size() != managed_count ||
628 snapshot->legacy.size() != legacy_count) return false;
629 fl::size managed_index = 0;
630 fl::size legacy_index = 0;
631 for (fl::CLEDController* controller = fl::CLEDController::head();
632 controller != nullptr; controller = controller->next()) {
633 if (!controller->getEnabled()) continue;
634 const fl::shared_ptr<fl::StreamingPipelineQ16> pipeline =
635 controller->colorPipeline();
636 if (pipeline && hooks.buildPowerHistogram != nullptr) {
637 if (managed_index >= managed_count) return false;
638 ManagedFramePower& entry = snapshot->managed[managed_index++];
639 entry.pipeline = pipeline;
640 entry.codec = fl::powerChannelCodecPolicy(*controller);
641 const fl::u8 emitters = controller->getRgbww().active() ? 5 :
642 (controller->getRgbw().active() ? 4 : 3);
643 emitter_power_weights(emitters, entry.weights);
644 hooks.buildPowerHistogram(
645 *pipeline,
646 fl::span<const CRGB>(controller->leds(), controller->size()),
647 emitters, &entry.histogram);
648 } else {
649 if (legacy_index >= legacy_count) return false;
650 snapshot->legacy[legacy_index++] = controller;
651 }
652 }
653 // A binding/list mutation during capture would invalidate the model.
654 return managed_index == managed_count && legacy_index == legacy_count;
655}

657fl::u32 framePowerAtFluxMilliwatts(const FramePowerSnapshot& snapshot,
658 fl::u32 flux_q16,
659 fl::u8 legacy_brightness) FL_NO_EXCEPT {
660 fl::u64 total = gMCU_mW;
661 const fl::FluxScalar flux = fl::FluxScalar::fromRawQ16(
662 static_cast<fl::i32>(flux_q16));
663 const fl::PowerFrameHooks& hooks = fl::powerFrameHooks();
664 for (fl::size i = 0; i < snapshot.managed.size(); ++i) {
665 const ManagedFramePower& entry = snapshot.managed[i];
666 const fl::u64 numerator = hooks.histogramPowerNumerator(
667 *entry.pipeline, entry.histogram, flux, entry.codec,
668 entry.weights,
669 &map_power_value);
670 total += static_cast<fl::u64>(gPowerModel().dark_mW) *
671 entry.histogram.pixels + ((numerator + 255u) >> 8);
672 }
673 for (fl::size i = 0; i < snapshot.legacy.size(); ++i) {
674 total += legacyFramePowerMilliwatts(*snapshot.legacy[i],
675 legacy_brightness);
676 }
677 return total > 0xFFFFFFFFu ? 0xFFFFFFFFu : static_cast<fl::u32>(total);
678}

682fl::FramePowerPlan fl::calculateFramePowerPlan(
683 fl::u8 requested_brightness, fl::u32 budget_mW) FL_NO_EXCEPT {
684 FramePowerPlan plan;
685 plan.mcu_mW = gMCU_mW;
686 FramePowerSnapshot snapshot;
687 if (!captureFramePowerSnapshot(&snapshot)) {
688 // Never solve against a partial set of enabled channels. Zero is the
689 // minimum controllable output; the unknown demand is not certified.
690 plan.flux_q16 = 0;
691 plan.legacy_brightness = 0;
692 plan.modeled_mW = 0xFFFFFFFFu;
693 plan.infeasible = true;
694 plan.limited = requested_brightness != 0;
695 return plan;
696 }
697 const fl::u32 request_flux = fl::FluxScalar::fromBrightness(
698 requested_brightness).rawQ16();
699 const fl::u32 zero_demand = framePowerAtFluxMilliwatts(snapshot, 0, 0);
700 if (zero_demand > budget_mW) {
701 plan.flux_q16 = 0;
702 plan.legacy_brightness = 0;
703 plan.modeled_mW = zero_demand;
704 plan.infeasible = true;
705 plan.limited = requested_brightness != 0;
706 return plan;
707 }
708 fl::u32 low = 0;
709 fl::u32 high = request_flux;
710 const auto legacy_at_flux = [request_flux, requested_brightness](
711 fl::u32 candidate) -> fl::u8 {
712 return candidate == request_flux ? requested_brightness
713 : static_cast<fl::u8>((static_cast<fl::u64>(candidate) * 255u) >> 16);
714 };
715 while (low < high) {
716 const fl::u32 candidate = low + (high - low + 1) / 2;
717 if (framePowerAtFluxMilliwatts(snapshot, candidate,
718 legacy_at_flux(candidate)) <= budget_mW) {
719 low = candidate;
720 } else {
721 high = candidate - 1;
722 }
723 }
724 plan.flux_q16 = low;
725 plan.limited = low < request_flux;
726 // Preserve the caller's exact byte when no limiting was needed. Q16's
727 // representation of b/255 is rounded, so projecting it back with a floor
728 // can otherwise turn even an unlimited b=1 into zero.
729 plan.legacy_brightness = legacy_at_flux(low);
730 plan.modeled_mW = framePowerAtFluxMilliwatts(
731 snapshot, low, plan.legacy_brightness);
732 return plan;
733}

814// sets brightness to
815// - no more than target_brightness
816// - no more than max_mW milliwatts
817fl::u8 calculate_max_brightness_for_power_mW( fl::u8 target_brightness, fl::u32 max_power_mW)
818{
819 // gMCU_mW and every controller's dark current are baseline: present at any
820 // brightness, and so kept out of the part the scalar acts on (#4156 R4).
821 fl::u32 fixed_mW = gMCU_mW;
822 fl::u32 controllable_mW = 0;
823
824 CLEDController *pCur = CLEDController::head();
825 while(pCur) {
826 const fl::u32 count = pCur->size();
827 // Through its RGBW conversion, and for a bound colour profile the
828 // solved drives rather than the source (#4344).
829 const fl::u32 unscaled_mW = controller_unscaled_power_mW(*pCur);
830 const fl::u32 dark_mW = fixed_power_mW(count);
831 // Dither's upward reserve does not scale with brightness, so it is
832 // charged with the baseline (#4342).
833 fixed_mW += dark_mW + controller_dither_reserve_mW(*pCur);
834 controllable_mW += unscaled_mW > dark_mW ? unscaled_mW - dark_mW : 0;
835 pCur = pCur->next();
836 }
837
838#if POWER_DEBUG_PRINT == 1
839 Serial.print("power demand at full brightness mW = ");
840 Serial.println( fixed_mW + controllable_mW);
841#endif
842
843 const fl::u32 requested_power_mW =
844 fixed_mW + scale_power_for_brightness(controllable_mW, target_brightness);
845#if POWER_DEBUG_PRINT == 1
846 if( target_brightness != 255 ) {
847 Serial.print("power demand at scaled brightness mW = ");
848 Serial.println( requested_power_mW);
849 }
850 Serial.print("power limit mW = ");
851 Serial.println( max_power_mW);
852#endif
853
854 if( requested_power_mW <= max_power_mW) {
855#if POWER_LED > 0
856 if( gMaxPowerIndicatorLEDPinNumber ) {
857 Pin(gMaxPowerIndicatorLEDPinNumber).lo(); // turn the LED off
858 }
859#endif
860#if POWER_DEBUG_PRINT == 1
861 Serial.print("demand is under the limit");
862#endif
863 return target_brightness;
864 }
865
866 const fl::u8 recommended_brightness = brightness_within_budget(
867 fixed_mW, controllable_mW, target_brightness, max_power_mW);
868#if POWER_DEBUG_PRINT == 1
869 Serial.print("recommended brightness # = ");
870 Serial.println( recommended_brightness);
871
872 fl::u32 resultant_power_mW =
873 fixed_mW + scale_power_for_brightness(controllable_mW, recommended_brightness);
874 Serial.print("resultant power demand mW = ");
875 Serial.println( resultant_power_mW);
876
877 Serial.println();
878#endif
879
880#if POWER_LED > 0
881 if( gMaxPowerIndicatorLEDPinNumber ) {
882 Pin(gMaxPowerIndicatorLEDPinNumber).hi(); // turn the LED on
883 }
884#endif
885
886 return recommended_brightness;
887}
