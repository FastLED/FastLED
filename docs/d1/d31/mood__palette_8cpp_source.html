21MoodPalette paletteFor(const RoomSense &s, float hueDrift) {
22 MoodPalette p;
23 // Travel the short way round the wheel from cool to warm (through
24 // magenta/red), which is the emotionally right path: sad -> tense ->
25 // happy, not sad -> green -> happy.
26 const float w = clampf(s.warmth, 0.0f, 1.0f);
27 float hue = kCoolHue + (kWarmHue + 256.0f - kCoolHue) * w;
28 hue += hueDrift * 256.0f;
29 while (hue >= 256.0f)
30 hue -= 256.0f;
31 while (hue < 0.0f)
32 hue += 256.0f;
33 p.hueBase = static_cast<fl::u8>(hue);
34
35 const float a = clampf(s.arousal, 0.0f, 1.0f);
36 p.hueSpread =
37 static_cast<fl::u8>(kMinSpread + (kMaxSpread - kMinSpread) * a);
38 p.saturation = static_cast<fl::u8>(kMinSat + (kMaxSat - kMinSat) * a);
39 return p;
40}

42RGBf paletteColor(const MoodPalette &p, float t) {
43 // Hue is quantised to the 256-step wheel, but neighbouring hues differ
44 // by about one unit per channel at full value, so a slow drift reads as
45 // continuous. Everything after this point stays in float.
46 const float tt = clampf(t, 0.0f, 1.0f);
47 const fl::u8 hue = static_cast<fl::u8>(
48 p.hueBase + static_cast<fl::u8>(tt * static_cast<float>(p.hueSpread)));
49 return toRGBf(CRGB(CHSV(hue, p.saturation, 255)));
50}
