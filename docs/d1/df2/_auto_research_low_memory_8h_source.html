191// Period-stat helpers for `pinToggleRx`. Mirrors the FlexIO RX benchmark
192// logic in `examples/AutoResearch/AutoResearchRemote.cpp::flexioRxBenchmark`
193// but inline so we don't pull in the AutoResearch ObjectFLED bus.
194struct LowMemPinTogglePeriodStats {
195 fl::u32 periods;
196 fl::u32 mean_ns;
197 fl::u32 sigma_ns;
198 fl::u32 min_ns;
199 fl::u32 max_ns;
200};
201
202inline LowMemPinTogglePeriodStats computeLowMemPeriodStats(
203 const fl::EdgeTime* edges, fl::size edge_count) FL_NO_EXCEPT {
204 LowMemPinTogglePeriodStats s{0, 0, 0, 0, 0};
205 fl::u64 sum = 0;
206 fl::u64 sum_sq = 0;
207 fl::u32 min_ns = 0xFFFFFFFFu;
208 fl::u32 max_ns = 0;
209 fl::u32 count = 0;
210 for (fl::size i = 0; i + 1 < edge_count; i += 2) {
211 const fl::u32 period_ns =
212 static_cast<fl::u32>(edges[i].ns) +
213 static_cast<fl::u32>(edges[i + 1].ns);
214 if (period_ns == 0) continue;
215 sum += period_ns;
216 sum_sq += static_cast<fl::u64>(period_ns) * static_cast<fl::u64>(period_ns);
217 if (period_ns < min_ns) min_ns = period_ns;
218 if (period_ns > max_ns) max_ns = period_ns;
219 ++count;
220 }
221 if (count == 0) return s;
222 const fl::u32 mean = static_cast<fl::u32>(sum / static_cast<fl::u64>(count));
223 const fl::u64 mean64 = static_cast<fl::u64>(mean);
224 const fl::u64 var = (sum_sq / static_cast<fl::u64>(count)) - (mean64 * mean64);
225 fl::u32 sig = 0;
226 while (static_cast<fl::u64>(sig + 1) * static_cast<fl::u64>(sig + 1) <= var) ++sig;
227 s.periods = count;
228 s.mean_ns = mean;
229 s.sigma_ns = sig;
230 s.min_ns = (min_ns == 0xFFFFFFFFu) ? 0u : min_ns;
231 s.max_ns = max_ns;
232 return s;
233}
234#endif // FL_IS_ARM_LPC && !FASTLED_AUTORESEARCH_IEEE754_MODE && !FAULT_TEST

237FL_NO_RETURN FL_NO_INLINE void triggerLowMemoryNullFault() FL_NO_EXCEPT {
238 volatile fl::u32* const p =
239 reinterpret_cast<volatile fl::u32*>(0u); // ok reinterpret cast - fault test
240 *p = 0x3302u;
241 while (true) {}

244FL_DISABLE_WARNING_PUSH
245FL_DISABLE_WARNING(infinite-recursion)
246FL_NO_INLINE int recurseLowMemoryStackBlow(int depth) FL_NO_EXCEPT {
247 volatile fl::u8 pad[64] = {0};
248 pad[static_cast<fl::u8>(depth) & 0x3Fu] = static_cast<fl::u8>(depth);
249 // LPC845/M0+ has no stack-limit trap; unbounded recursion can corrupt the
250 // runtime before the HardFault handler can emit. Burn stack frames, then
251 // force the HardFault path while the diagnostic surface is still intact.
252 if (depth >= 32) {
253 triggerLowMemoryNullFault();
254 }
255 return recurseLowMemoryStackBlow(depth + 1) + pad[depth & 0x3F];
256}
257FL_DISABLE_WARNING_POP
258#endif // FL_IS_ARM_LPC_845 && FASTLED_AUTORESEARCH_FAULT_TEST

320 // pinToggleRx (FastLED #3021 Phase 1) — bit-bang square wave on tx_pin
321 // and capture SCT edges on rx_pin. CSV stats out.
322#if !defined(FASTLED_AUTORESEARCH_LPC_DMA_LOOPBACK)
323 // Left out of the RX-DMA loopback builds (--ws2812-loopback/--pwm-dma-cl):
324 // with the DMA capture linked, the 64 KB LPC845 has no room for this
325 // unrelated ~1 KB diagnostic. --pin-toggle-rx builds keep it.
326 remote.bind("pinToggleRx",
327 [](int tx_pin, int rx_pin, int freq_hz, int duration_ms) -> fl::string {
328 if (tx_pin < 0 || rx_pin < 0 || freq_hz <= 0 || duration_ms <= 0) {
329 return fl::string("0,0,0,0,0,0,0");
330 }
331 const fl::u32 expected_edges =
332 2u * static_cast<fl::u32>(freq_hz) *
333 static_cast<fl::u32>(duration_ms) / 1000u;
334 // Keep both the RX vector and diagnostic copy buffer small on
335 // 16 KB SRAM LPC8xx parts. Larger static buffers regressed early
336 // init by leaving too little room for stack/heap (FastLED #3200),
337 // while stack buffers previously overflowed during this handler
338 // (FastLED #3125).
339 constexpr fl::u32 kLowMemEdgeBufSize = 64u;
340 fl::u32 cap = expected_edges + (expected_edges / 2u);
341 if (cap < 2u) cap = 2u;
342 if (cap > kLowMemEdgeBufSize) cap = kLowMemEdgeBufSize;
343
344 auto rx = fl::LpcSctRxChannel::create(rx_pin);
345 if (!rx) return fl::string("0,0,0,0,0,0,0");
346
347 fl::RxConfig cfg;
348 cfg.buffer_size = cap;
349 cfg.start_low = true;
350 if (!rx->begin(cfg)) return fl::string("0,0,0,0,0,0,0");
351
352 pinMode(tx_pin, OUTPUT);
353 digitalWrite(tx_pin, LOW);
354
355 const fl::u32 half_us = 500000u / static_cast<fl::u32>(freq_hz);
356 const fl::u32 cycles = static_cast<fl::u32>(freq_hz) *
357 static_cast<fl::u32>(duration_ms) / 1000u;
358
359 for (fl::u32 i = 0; i < cycles; ++i) {
360 digitalWrite(tx_pin, HIGH);
361 delayMicroseconds(half_us);
362 rx->pollOnce();
363 digitalWrite(tx_pin, LOW);
364 delayMicroseconds(half_us);
365 rx->pollOnce();
366 if ((i & 0x7Fu) == 0u) {
367 fl::Watchdog::instance().feed();
368 }
369 }
370 rx->wait(1);
371
372 // Static storage keeps this off the stack; the small fixed size
373 // keeps .bss below the LPC845-BRK early-init failure threshold.
374 fl::EdgeTime* edges_buf = autoresearch::low_memory::pinToggleState().edges;
375 const fl::size n_read = rx->getRawEdgeTimes(
376 fl::span<fl::EdgeTime>(edges_buf, sizeof(edges_buf) / sizeof(edges_buf[0])));
377
378 LowMemPinTogglePeriodStats stats = computeLowMemPeriodStats(edges_buf, n_read);
379
380 fl::sstream s;
381 const bool success = (stats.periods > 0);
382 s << (success ? 1 : 0) << ',' << static_cast<fl::u32>(n_read) << ','
383 << stats.periods << ',' << stats.mean_ns << ',' << stats.sigma_ns << ','
384 << stats.min_ns << ',' << stats.max_ns;
385 return s.str();

389#if defined(FASTLED_AUTORESEARCH_LPC_WS2812)
390 // ws2812SctTest (FastLED #3021 Phase 2) -- WS2812 byte-match loopback.
391 remote.bind("ws2812SctTest",
392 [](int test_case, int tx_pin, int rx_pin, int capture_ms) -> fl::string {
393 if (test_case < 0 || test_case > 4 ||
394 tx_pin < 0 || rx_pin < 0 || capture_ms <= 0) {
395 return fl::string("0,0,0,0,0,0,0,0");
396 }
397 int num_leds = 1;
398 switch (test_case) {
399 case 1: num_leds = 3; break;
400 case 4: num_leds = 100; break;
401 default: num_leds = 1; break;
402 }
403 autoresearch::low_memory::Ws2812State& ws_state =
404 autoresearch::low_memory::ws2812State();
405 CRGB* leds_buf = ws_state.leds;
406 for (int i = 0; i < num_leds; ++i) {
407 switch (test_case) {
408 case 0: leds_buf[i] = CRGB(0xFF, 0x00, 0x00); break;
409 case 1: {
410 if (i == 0) leds_buf[i] = CRGB(0xFF, 0x00, 0x00);
411 else if (i == 1) leds_buf[i] = CRGB(0x00, 0xFF, 0x00);
412 else leds_buf[i] = CRGB(0x00, 0x00, 0xFF);
413 break;
414 }
415 case 2: leds_buf[i] = CRGB(0x00, 0x00, 0x00); break;
416 case 3: leds_buf[i] = CRGB(0xFF, 0xFF, 0xFF); break;
417 case 4: {
418 const uint8_t r = (i % 3 == 0) ? 0xFFu : 0u;
419 const uint8_t g = (i % 3 == 1) ? 0xFFu : 0u;
420 const uint8_t b = (i % 3 == 2) ? 0xFFu : 0u;
421 leds_buf[i] = CRGB(r, g, b);
422 break;
423 }
424 }
425 }
426 const int expected_bytes = num_leds * 3;
427 uint8_t* expected_buf = ws_state.expected;
428 for (int i = 0; i < num_leds; ++i) {
429 expected_buf[i * 3 + 0] = leds_buf[i].g;
430 expected_buf[i * 3 + 1] = leds_buf[i].r;
431 expected_buf[i * 3 + 2] = leds_buf[i].b;
432 }
433 auto rx = fl::LpcSctRxChannel::create(rx_pin);
434 if (!rx) return fl::string("0,0,0,0,0,0,0,0");
435 fl::RxConfig cfg;
436 const fl::u32 expected_edges = (fl::u32)num_leds * 24u * 2u;
437 fl::u32 cap = expected_edges + (expected_edges / 2u);
438 // The low-memory sketch cannot afford the old 2048-edge reserve
439 // on a 16 KB SRAM LPC845. 192 edges covers the 1- and 3-LED
440 // sanity cases and fails the larger diagnostic cases without
441 // starving setup()/loop() stack.
442 constexpr fl::u32 kLowMemWs2812EdgeCapacity = 192u;
443 if (cap < 48u) cap = 48u;
444 if (cap > kLowMemWs2812EdgeCapacity) cap = kLowMemWs2812EdgeCapacity;
445 cfg.buffer_size = cap;
446 cfg.start_low = true;
447 if (!rx->begin(cfg)) return fl::string("0,0,0,0,0,0,0,0");
448
449 constexpr int kFixedTxPin = 10; // P0_10
450 if (tx_pin != kFixedTxPin) {
451 return fl::string("0,0,0,0,0,0,0,0");
452 }
453 if (!ws_state.fastled_initialized) {
454 FastLED.addLeds<WS2812, kFixedTxPin, GRB>(leds_buf, 100);
455 ws_state.fastled_initialized = true;
456 }
457 FastLED.show();
458 rx->wait(capture_ms);
459
460 uint8_t* decoded_buf = ws_state.decoded;
461 for (int i = 0; i < expected_bytes; ++i) decoded_buf[i] = 0;
462
463 fl::ChipsetTiming4Phase timing =
464 fl::make4PhaseTiming(
465 fl::to_runtime_timing<fl::WS2812ChipsetTiming>(),
466 /*reset_us=*/0u);
467
468 auto dec = rx->decode(timing, fl::span<u8>(decoded_buf, expected_bytes));
469 const fl::u32 decoded_bytes = dec.ok() ? dec.value() : 0u;
470
471 fl::u32 matched = 0;
472 fl::u32 mismatched = 0;
473 for (fl::u32 i = 0; i < decoded_bytes && i < (fl::u32)expected_bytes; ++i) {
474 if (decoded_buf[i] == expected_buf[i]) ++matched;
475 else ++mismatched;
476 }
477 const fl::u32 missing = (fl::u32)expected_bytes - decoded_bytes;
478 mismatched += missing;
479
480 // Diagnostic-only raw-edge sample. Keep this small in .bss; the
481 // decoder uses the RX channel's bounded capture vector above.
482 constexpr fl::size kLowMemWs2812ProbeSize = 64u;
483 fl::EdgeTime* probe = ws_state.probe;
484 const fl::size edges_captured = rx->getRawEdgeTimes(
485 fl::span<fl::EdgeTime>(probe, kLowMemWs2812ProbeSize));
486
487 const bool success = (mismatched == 0 && decoded_bytes == (fl::u32)expected_bytes);
488
489 fl::sstream s;
490 s << (success ? 1 : 0) << ',' << test_case << ',' << num_leds << ','
491 << expected_bytes << ',' << decoded_bytes << ','
492 << matched << ',' << mismatched << ','
493 << static_cast<fl::u32>(edges_captured);
494 return s.str();
495 });
