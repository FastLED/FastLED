111void AutoResearchRemoteControl::bindBenchmarkMethods(fl::Remote& remote) {
112 // ====== AutoResearch perf-instrumentation sanity probes ======
113 // Task 2 of meta #3113. Three probes verify autoresearch timing
114 // produces trustworthy data BEFORE we use it to gate optimizations.
115 // Fail-loud: any probe failing → host reporter MUST stamp result
116 // UNTRUSTED and refuse to feed wave2dPerf into comparison tables.
117
118 // Probe 2a: memcpy throughput. Caller invokes with { bytes,
119 // iterations }; returns MB/s on SRAM. Host compares vs per-platform
120 // vendor floor (e.g. ESP32-S3 ~175 MB/s) before trusting wave2dPerf.
121 remote.bind("perfProbeMemcpy", [](const fl::json& args) -> fl::json {
122 int bytes = 4096;
123 int iterations = 1000;
124 if (args.is_array() && args.size() >= 1 && args[0].is_object()) {
125 const fl::json &cfg = args[0];
126 if (cfg.contains("bytes") && cfg["bytes"].is_int()) {
127 bytes = static_cast<int>(cfg["bytes"].as_int().value());
128 }
129 if (cfg.contains("iterations") && cfg["iterations"].is_int()) {
130 iterations = static_cast<int>(cfg["iterations"].as_int().value());
131 }
132 }
133 fl::json response = fl::json::object();
134 response.set("bytes", static_cast<int64_t>(bytes));
135 response.set("iterations", static_cast<int64_t>(iterations));
136 if (bytes < 64 || bytes > 8192 || iterations < 1 || iterations > 100000) {
137 response.set("success", false);
138 response.set("error", "out_of_range");
139 return response;
140 }
141 // Heap-allocated on first use: 16 KB of static buffers
142 // overflowed ESP32-S2's dram0 bss by 3.7 KB (FastLED#3576
143 // Phase 4 compile matrix).
144 PerfProbeMemcpyState& state = perfProbeMemcpyState();
145 fl::vector<uint8_t>& src_vec = state.source;
146 fl::vector<uint8_t>& dst_vec = state.destination;
147 if (src_vec.size() < 8192) {
148 src_vec.resize(8192);
149 dst_vec.resize(8192);
150 }
151 uint8_t *src_buf = src_vec.data();
152 uint8_t *dst_buf = dst_vec.data();
153 for (int i = 0; i < bytes; ++i) {
154 src_buf[i] = static_cast<uint8_t>(i & 0xFF);
155 }
156 const fl::u32 t0 = fl::micros();
157 for (int i = 0; i < iterations; ++i) {
158 FL_BUILTIN_MEMCPY(dst_buf, src_buf, bytes);
159 }
160 const fl::u32 t1 = fl::micros();
161 const fl::u32 total_us = t1 - t0;
162 const double total_bytes = static_cast<double>(bytes) * iterations;
163 const double mb_per_s = (total_us > 0)
164 ? (total_bytes / total_us) * (1e6 / (1024.0 * 1024.0))
165 : 0.0;
166 response.set("success", true);
167 response.set("total_us", static_cast<int64_t>(total_us));
168 response.set("mb_per_s", mb_per_s);
169 return response;
170 });
171
172 // Probe 2b: nop-loop calibration. Volatile fence prevents the
173 // compiler optimizing the loop away. Host computes
174 // (total_us * cpu_mhz / iterations) and checks [1.0, 2.5]
175 // cycles-per-nop. Out-of-band → timer broken or IRQ jitter.
176 remote.bind("perfProbeNop", [](const fl::json& args) -> fl::json {
177 int iterations = 100000;
178 if (args.is_array() && args.size() >= 1 && args[0].is_object()) {
179 const fl::json &cfg = args[0];
180 if (cfg.contains("iterations") && cfg["iterations"].is_int()) {
181 iterations = static_cast<int>(cfg["iterations"].as_int().value());
182 }
183 }
184 fl::json response = fl::json::object();
185 response.set("iterations", static_cast<int64_t>(iterations));
186 if (iterations < 1000 || iterations > 10000000) {
187 response.set("success", false);
188 response.set("error", "out_of_range");
189 return response;
190 }
191 volatile int counter = 0;
192 const fl::u32 t0 = fl::micros();
193 for (int i = 0; i < iterations; ++i) {
194#if defined(__GNUC__) || defined(__clang__)
195 __asm__ volatile("nop");
196#else
197 counter += 1;
198#endif
199 }
200 const fl::u32 t1 = fl::micros();
201 (void)counter;
202 const fl::u32 total_us = t1 - t0;
203 const double us_per_iter = (iterations > 0)
204 ? (static_cast<double>(total_us) / iterations) : 0.0;
205 response.set("success", true);
206 response.set("total_us", static_cast<int64_t>(total_us));
207 response.set("us_per_iter", us_per_iter);
208 return response;
209 });
210
211 // Probe 2d: bit-bang cost attribution (FastLED#4203). The portable
212 // BIT_BANG driver stretches every WS2812 bit by a constant ~2-3us, which
213 // makes its output undecodable. The captures prove the cost is constant
214 // per bit but cannot say which call carries it, so measure the candidates
215 // against a common loop baseline instead of guessing:
216 // nop - empty-loop floor for this board
217 // delayNs - fl::delayNanoseconds(ns): runtime u64 divide, and on
218 // ESP32 an esp_clk_cpu_freq() query on every call
219 // delayNsHz - same with the clock frequency hoisted out; the gap
220 // against delayNs is exactly the per-call clock query
221 // writeByte - DigitalMultiWrite8::writeByte(): nibble LUT lookups plus
222 // two out-of-line applyNibble() calls
223 // A bit costs 3x delay + 3x writeByte, so these numbers reconstruct the
224 // measured per-bit overhead directly.
225 remote.bind("perfProbeBitBangCost", [](const fl::json& args) -> fl::json {
226 int iterations = 20000;
227 int ns = 400; // WS2812 T0H — the budget being blown
228 int pin = -1; // -1 keeps GPIO untouched (default, safe)
229 if (args.is_array() && args.size() >= 1 && args[0].is_object()) {
230 const fl::json &cfg = args[0];
231 if (cfg.contains("iterations") && cfg["iterations"].is_int()) {
232 iterations = static_cast<int>(cfg["iterations"].as_int().value());
233 }
234 if (cfg.contains("ns") && cfg["ns"].is_int()) {
235 ns = static_cast<int>(cfg["ns"].as_int().value());
236 }
237 if (cfg.contains("pin") && cfg["pin"].is_int()) {
238 pin = static_cast<int>(cfg["pin"].as_int().value());
239 }
240 }
241 fl::json response = fl::json::object();
242 response.set("iterations", static_cast<int64_t>(iterations));
243 response.set("ns", static_cast<int64_t>(ns));
244 response.set("pin", static_cast<int64_t>(pin));
245 if (iterations < 1000 || iterations > 1000000 || ns < 0) {
246 response.set("success", false);
247 response.set("error", "out_of_range");
248 return response;
249 }
250 // Six timed loops each run `iterations` times, so the wall time is
251 // roughly 6 * iterations * ns. AutoResearch's loop watchdog is 5 s;
252 // cap the estimate well under it so a large ns cannot turn this probe
253 // into a watchdog reset on an unattended bench.
254 constexpr fl::u64 kMaxProbeNs = 2000000000ULL; // 2 s
255 const fl::u64 estimated_ns =
256 static_cast<fl::u64>(iterations) * static_cast<fl::u64>(ns) * 6ULL;
257 if (estimated_ns > kMaxProbeNs) {
258 response.set("success", false);
259 response.set("error", "duration_budget_exceeded");
260 response.set("estimated_ns", static_cast<int64_t>(estimated_ns));
261 response.set("budget_ns", static_cast<int64_t>(kMaxProbeNs));
262 return response;
263 }
264
265 const fl::u32 requested_ns = static_cast<fl::u32>(ns);
266 const fl::u32 hz = static_cast<fl::u32>(FL_CPU_FREQUENCY());
267
268 // Baseline: same loop shape, no payload. An empty memory barrier
269 // keeps the loop alive without adding work of its own -- the previous
270 // `volatile int` read-modify-write costs a load, add and store to
271 // memory every iteration, which inflates nop_per and therefore
272 // under-reports every overhead computed by subtracting it.
273 const fl::u32 nop_t0 = fl::micros();
274 for (int i = 0; i < iterations; ++i) {
275 __asm__ __volatile__("" ::: "memory");
276 }
277 const fl::u32 nop_us = fl::micros() - nop_t0;
278
279 const fl::u32 delay_t0 = fl::micros();
280 for (int i = 0; i < iterations; ++i) {
281 fl::delayNanoseconds(requested_ns);
282 }
283 const fl::u32 delay_us = fl::micros() - delay_t0;
284
285 const fl::u32 delay_hz_t0 = fl::micros();
286 for (int i = 0; i < iterations; ++i) {
287 fl::delayNanoseconds(requested_ns, hz);
288 }
289 const fl::u32 delay_hz_us = fl::micros() - delay_hz_t0;
290
291 // Compile-time NS: cycles_from_ns_*() folds to a constant, so the
292 // runtime u64 divide disappears while the platform busy-wait still
293 // runs. If this is fast, the divide carried the cost; if it is still
294 // slow, the cost is inside the busy-wait. Fixed at 400ns (WS2812 T0H)
295 // because a template argument cannot come from the RPC payload -- the
296 // `ns` parameter only steers the runtime variants above.
297 // Runtime form pinned to 400 ns. ns_conversion_us subtracts the
298 // compile-time 400 ns loop, and the compile-time form cannot take its
299 // duration from the RPC payload -- so comparing it against the
300 // caller-supplied `ns` above would difference two different delays
301 // and misattribute the gap to conversion cost.
302 const fl::u32 delay_hz400_t0 = fl::micros();
303 for (int i = 0; i < iterations; ++i) {
304 fl::delayNanoseconds(400u, hz);
305 }
306 const fl::u32 delay_hz400_us = fl::micros() - delay_hz400_t0;
307
308 const fl::u32 delay_ct_t0 = fl::micros();
309 for (int i = 0; i < iterations; ++i) {
310 fl::delayNanoseconds<400>();
311 }
312 const fl::u32 delay_ct_us = fl::micros() - delay_ct_t0;
313
314 // Pure cycle-counted loop: no ns->cycles conversion and no platform
315 // busy-wait wrapper. This is the floor a hand-rolled bit loop could
316 // reach. The board reports cpu_hz=125000000, so 400ns is 50 cycles;
317 // 60 is kept as a small deliberate overshoot. delaycycles<> is specialized
318 // only up to 50 with no generic fallback, so 60 is composed from two
319 // specializations rather than written as delaycycles<60>(), which is
320 // an undefined reference at link time.
321 const fl::u32 delay_cyc_t0 = fl::micros();
322 for (int i = 0; i < iterations; ++i) {
323 fl::delaycycles<50>();
324 fl::delaycycles<10>();
325 }
326 const fl::u32 delay_cyc_us = fl::micros() - delay_cyc_t0;
327
328 // pin < 0 leaves every slot inactive, so applyNibble performs no GPIO
329 // write at all — that isolates the LUT and call overhead from the
330 // cost of the pin write itself.
331 fl::Pins8 pins8;
332 for (int i = 0; i < 8; ++i) {
333 pins8.pins[i] = -1;
334 }
335 if (pin >= 0) {
336 fl::pinMode(pin, fl::PinMode::Output);
337 pins8.pins[0] = pin;
338 }
339 fl::DigitalMultiWrite8 writer;
340 writer.init(pins8);
341 const fl::u32 wb_t0 = fl::micros();
342 for (int i = 0; i < iterations; ++i) {
343 writer.writeByte(static_cast<fl::u8>(i & 0xFF));
344 }
345 const fl::u32 wb_us = fl::micros() - wb_t0;
346 if (pin >= 0) {
347 fl::digitalWrite(pin, fl::PinValue::Low);
348 }
349
350 const double denom = static_cast<double>(iterations);
351 const double nop_per = static_cast<double>(nop_us) / denom;
352 const double delay_per = static_cast<double>(delay_us) / denom;
353 const double delay_hz_per = static_cast<double>(delay_hz_us) / denom;
354 const double wb_per = static_cast<double>(wb_us) / denom;
355
356 response.set("success", true);
357 response.set("cpu_hz", static_cast<int64_t>(hz));
358 response.set("nop_us_per_iter", nop_per);
359 response.set("delay_us_per_iter", delay_per);
360 response.set("delay_hz_us_per_iter", delay_hz_per);
361 response.set("write_byte_us_per_iter", wb_per);
362 // Cost above the loop floor, i.e. what the caller actually pays.
363 response.set("delay_overhead_us", delay_per - nop_per);
364 response.set("delay_hz_overhead_us", delay_hz_per - nop_per);
365 response.set("write_byte_overhead_us", wb_per - nop_per);
366 const double delay_ct_per = static_cast<double>(delay_ct_us) / denom;
367 const double delay_cyc_per = static_cast<double>(delay_cyc_us) / denom;
368 response.set("delay_ct_us_per_iter", delay_ct_per);
369 response.set("delay_cycles_us_per_iter", delay_cyc_per);
370 response.set("delay_ct_overhead_us", delay_ct_per - nop_per);
371 response.set("delay_cycles_overhead_us", delay_cyc_per - nop_per);
372 // Cost attributable to the runtime ns->cycles conversion: the gap
373 // between the runtime and compile-time forms, both at 400 ns.
374 const double delay_hz400_per =
375 static_cast<double>(delay_hz400_us) / iterations;
376 response.set("delay_hz400_us_per_iter", delay_hz400_per);
377 response.set("ns_conversion_us", delay_hz400_per - delay_ct_per);
378 response.set("clock_query_us", delay_per - delay_hz_per);
379 // One clockless bit issues three delays and three writeByte calls.
380 response.set("predicted_bit_overhead_us",
381 3.0 * ((delay_per - nop_per) + (wb_per - nop_per)));
382 return response;
383 });
