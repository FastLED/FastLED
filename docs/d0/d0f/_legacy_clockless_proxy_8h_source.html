26// GPIO6/7/8 on the classic ESP32 (esp32dev / WROOM modules) are
27// reserved for SPI flash (SD2/SD3/CMD/CLK/SD0/SD1 — pins 6-11) and
28// FastLED's `FASTLED_UNUSABLE_PIN_MASK` for esp32dev marks pins
29// {6, 7, 8, 9, 10, 11, 20} as `validpin() == false`. Instantiating any
30// clockless driver bound to those pins fails the FastLED static_assert
31// at compile time. Other ESP32 variants (S2/S3/C-series, P4) repurpose
32// GPIO6-8 and accept them as valid output pins, and every non-ESP
33// family treats them as normal digital pins. The classic-ESP32
34// sentinel is `FL_IS_ESP_32DEV` (defined in `src/platforms/esp/is_esp.h`);
35// the prior #3447 attempt used `FL_IS_ESP_32` which is not a real macro,
36// so the gate never fired and esp32dev compile still hit the
37// static_assert. LegacyClocklessProxy iterates pins 0-8 so all three
38// of 6/7/8 must be excluded for the classic ESP32 build.
39#if defined(FL_IS_ESP_32DEV)
40#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_6 0
41#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_7 0
42#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_8 0
43#else
44#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_6 1
45#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_7 1
46#define AUTORESEARCH_LEGACY_SUPPORTS_PIN_8 1
47#endif

173public:
174 LegacyClocklessProxy(
175 int pin, CRGB* leds, int numLeds,
176 LegacyClocklessChipset chipset = LegacyClocklessChipset::WS2812B,
177 bool rgbw = false) {
178 switch (pin) {
179 case 0: mController = create<0>(leds, numLeds, chipset, rgbw); break;
180 case 1: mController = create<1>(leds, numLeds, chipset, rgbw); break;
181 case 2: mController = create<2>(leds, numLeds, chipset, rgbw); break;
182 case 3: mController = create<3>(leds, numLeds, chipset, rgbw); break;
183 case 4: mController = create<4>(leds, numLeds, chipset, rgbw); break;
184 case 5: mController = create<5>(leds, numLeds, chipset, rgbw); break;
185#if AUTORESEARCH_LEGACY_SUPPORTS_PIN_6
186 case 6: mController = create<6>(leds, numLeds, chipset, rgbw); break;
187#endif
188#if AUTORESEARCH_LEGACY_SUPPORTS_PIN_7
189 case 7: mController = create<7>(leds, numLeds, chipset, rgbw); break;
190#endif
191#if AUTORESEARCH_LEGACY_SUPPORTS_PIN_8
192 case 8: mController = create<8>(leds, numLeds, chipset, rgbw); break;
193#endif
194#if AUTORESEARCH_LEGACY_SUPPORTS_PIN_22
195 case 22: mController = create<22>(leds, numLeds, chipset, rgbw); break;

197 default: break; // mController stays nullptr
198 }
