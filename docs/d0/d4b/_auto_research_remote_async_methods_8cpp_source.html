56 // Test: Basic coroutine creation and completion
57// Decline the await-based coroutine tests where no coroutine can run.
58//
59// `fl::platforms::await()` (platforms/await.h) polls
60// `while (!promise.is_completed()) { promise.update(); suspendMainthread(); }`
61// with no bound. On RP the coroutine backend is null, so the producer
62// coroutine that would resolve the promise never executes and that loop never
63// exits: the sketch stops answering RPC entirely and only comes back when the
64// watchdog resets the board. Confirmed on an RP2350W, each of the three
65// tested *in isolation* with a `ping` after it: the call returns no response,
66// the following `ping` returns nothing either, and the board re-enumerates
67// reporting `lastResetCause: WATCHDOG`.
68//
69// Only the three tests that actually call `fl::task::await()` are guarded --
70// Await and AwaitError have one call site each, ChainedAwait two. The two
71// promise-callback tests call it zero times: they wait on
72// `while (producer.isRunning() && millis() - start < 5000)`, which is bounded
73// and exits immediately on RP because a coroutine that never starts is never
74// running. They measure 0 ms and 1 ms. They first looked like hangs only
75// because they were called *after* `testCoroutineAwait` had already wedged
76// the board -- "returned null" and "was called after something else died"
77// are indistinguishable from a transcript.
78//
79// `testCoroutineAll` already declines on FL_IS_RP, but these binds are
80// unconditional and `rpc.discover` advertises them, so the RPC surface invites
81// exactly the call that wedges the board. The three non-await tests
82// (Basic/Stop/Concurrent) are deliberately left callable: they return
83// `taskRan: false` promptly and are the evidence that the backend is null.
84//
85// This guards the symptom. The underlying hazard is that an unbounded
86// `await()` hangs wherever a promise cannot be resolved; see FastLED#3832.
87#if defined(FL_IS_RP)
88#define AUTORESEARCH_DECLINE_AWAIT_ON_RP(name) \
89 do { \
90 fl::json r = fl::json::object(); \
91 r.set("success", false); \
92 r.set("supported", false); \
93 r.set("backend", "null"); \
94 r.set("reason", \
95 "RP2xxx selects the null coroutine backend; " name \
96 " would block forever in await() and wedge the board " \
97 "until the watchdog resets it"); \
98 return r; \
99 } while (0)
100#else
101#define AUTORESEARCH_DECLINE_AWAIT_ON_RP(name) do { } while (0)
102#endif

214 // Test: Consumer coroutine awaits promise, producer coroutine fulfills it.
215 // Verifies fl::task::await() truly blocks until the producer resolves the promise.
216 // Main thread does NOT touch the promise — only the producer coroutine does.
217 remote.bind("testCoroutineAwait", [](const fl::json& args) -> fl::json {
218 (void)args;
219 AUTORESEARCH_DECLINE_AWAIT_ON_RP("testCoroutineAwait");
220 fl::json r = fl::json::object();
221
222 auto promise_ptr = fl::make_shared<fl::task::Promise<int>>(fl::task::Promise<int>::create());
223 fl::atomic<bool> consumer_started(false);
224 fl::atomic<bool> consumer_finished(false);
225 fl::atomic<int> consumer_value(0);
226 fl::atomic<bool> consumer_ok(false);
227 fl::atomic<bool> producer_started(false);
228 fl::atomic<bool> producer_finished(false);
229
230 // Consumer coroutine: starts first, calls fl::task::await() which should block
231 // until the producer coroutine resolves the promise
232 fl::task::CoroutineConfig consumer_cfg;
233 consumer_cfg.func = [promise_ptr, &consumer_started, &consumer_finished,
234 &consumer_value, &consumer_ok]() {
235 consumer_started.store(true);
236 // This should block the coroutine until producer fulfills the promise
237 auto result = fl::task::await(*promise_ptr);
238 if (result.ok()) {
239 consumer_ok.store(true);
240 consumer_value.store(result.value());
241 }
242 consumer_finished.store(true);
243 };
244 consumer_cfg.name = "await_consumer";
245 auto consumer = fl::task::coroutine(consumer_cfg);
246
247 // Small delay so consumer enters fl::task::await() before producer starts
248 delay(50);
249
250 // Producer coroutine: simulates async work, then resolves the promise
251 fl::task::CoroutineConfig producer_cfg;
252 producer_cfg.func = [promise_ptr, &producer_started, &producer_finished]() {
253 producer_started.store(true);
254 delay(200); // simulate real async work (e.g. sensor read, network I/O)
255 promise_ptr->complete_with_value(42);
256 producer_finished.store(true);
257 };
258 producer_cfg.name = "await_producer";
259 auto producer = fl::task::coroutine(producer_cfg);
260
261 // Main thread waits for both coroutines to finish (polling only)
262 uint32_t start = millis();
263 while ((!consumer_finished.load() || producer.isRunning()) && (millis() - start) < 5000) {
264 delay(10);
265 }
266
267 // Verify: consumer was truly blocked — it should not finish before producer
268 bool passed = consumer_started.load() && consumer_finished.load()
269 && consumer_ok.load() && (consumer_value.load() == 42)
270 && producer_started.load() && producer_finished.load();
271 r.set("success", passed);
272 r.set("consumerStarted", consumer_started.load());
273 r.set("consumerFinished", consumer_finished.load());
274 r.set("consumerOk", consumer_ok.load());
275 r.set("consumerValue", static_cast<int64_t>(consumer_value.load()));
276 r.set("producerStarted", producer_started.load());
277 r.set("producerFinished", producer_finished.load());
278 r.set("durationMs", static_cast<int64_t>(millis() - start));
279 return r;
