108void printStreamRaw(const char* messageType, const fl::json& data) {
109 // Build pure JSONL message: RESULT: {"type":"...", ...data}
110 fl::json output = fl::json::object();
111 output.set("type", messageType);
112
113 // Copy all fields from data into output
114 if (data.is_object()) {
115 auto keys = data.keys();
116 for (fl::size i = 0; i < keys.size(); i++) {
117 output.set(keys[i].c_str(), data[keys[i]]);
118 }
119 }
120
121 // Use fl:: serial transport for consistent formatting
122 fl::string formatted = fl::formatJsonResponse(output, "RESULT: ");
123 Serial.println(formatted.c_str()); // ok serial - ok autoresearch rpc serial - stream response boundary
124}

129// Return codes:
130// 0 = SUCCESS

133// 3 = INVALID_ARGS

135enum class ReturnCode : int {
136 SUCCESS = 0,
137 TEST_FAILED = 1,
138 HARDWARE_ERROR = 2,
139 INVALID_ARGS = 3
140};

142fl::json makeResponse(bool success, ReturnCode returnCode, const char* message,
143 const fl::json& data = fl::json()) {
144 fl::json r = fl::json::object();
145 r.set("success", success);
146 r.set("returnCode", static_cast<int64_t>(static_cast<int>(returnCode)));
147 r.set("message", message);
148 if (!data.is_null() && data.has_value()) {
149 r.set("data", data);
150 }
151 return r;
152}

156fl::json AutoResearchRemoteControl::findConnectedPinsImpl(const fl::json& args) {
157 fl::json response = fl::json::object();
158
159 // Parse optional arguments: [{startPin: int, endPin: int, autoApply: bool}]
160 //
161 // FastLED #3446: default range stays at the historically-safe 0-8
162 // window so an unparameterised call cannot accidentally drive a
163 // boot/strap/USB pin into reset. Callers that need to sweep higher
164 // pins (e.g. the user-reported (33, 34) short) should issue
165 // multiple `findConnectedPins` calls with overlapping 8-pin windows
166 // (0-8, 8-16, 16-24, ...) from the host side. The Python wrapper
167 // `run_pin_discovery_segmented` in `ci/autoresearch/gpio.py` does
168 // exactly that.
169 int start_pin = 0;
170 int end_pin = 8;
171 bool auto_apply = true; // If true, automatically apply found pins
172
173 if (args.is_array() && args.size() >= 1 && args[0].is_object()) {
174 fl::json config = args[0];
175 if (config.contains("startPin") && config["startPin"].is_int()) {
176 start_pin = static_cast<int>(config["startPin"].as_int().value());
177 }
178 if (config.contains("endPin") && config["endPin"].is_int()) {
179 end_pin = static_cast<int>(config["endPin"].as_int().value());
180 }
181 if (config.contains("autoApply") && config["autoApply"].is_bool()) {
182 auto_apply = config["autoApply"].as_bool().value();
183 }
184 }
185
186 // Validate range
187 if (start_pin < 0 || start_pin > 48 || end_pin < 0 || end_pin > 48 || start_pin >= end_pin) {
188 response.set("error", "InvalidRange");
189 response.set("message", "Pin range must be 0-48 with startPin < endPin");
190 return response;
191 }
192
193 // FastLED #3446: defer to FastLED's existing per-chip pin map
194 // (`FASTLED_UNUSABLE_PIN_MASK` + `SOC_GPIO_VALID_*_MASK`) rather
195 // than hard-coding pin numbers here. The mask already covers
196 // SPI-flash pads, USB-JTAG pads, chip-strap pins, and packaging-
197 // reserved gaps for every supported ESP32 variant, and is the
198 // single source of truth `_ESPPIN<P>::validpin()` uses too. Any
199 // pin not in `_FL_VALID_PIN_MASK` is unsafe to drive; any pin in
200 // `SOC_GPIO_VALID_GPIO_MASK` but NOT `SOC_GPIO_VALID_OUTPUT_GPIO_MASK`
201 // is input-only (drivable in the forward direction only).
202 //
203 // We surface skipped pins in the response so the host log says
204 // exactly which pins were forbidden and why — turning silent
205 // "no jumper found" into actionable "pins 6-11 are flash, 20 is
206 // USB-JTAG, 34-39 are input-only".
207 //
208 // The fallthrough for non-ESP32 hosts is "allow everything" —
209 // platform-specific gating lives on those platforms' own
210 // `findConnectedPins` paths.
211 auto isFastLedValidOutputPin = [](int p) -> bool {
212#if defined(FL_IS_ESP32)
213 if (p < 0 || p >= 64) return false;
214 return (_FL_VALID_PIN_MASK & (1ULL << p)) != 0;
215#else
216 (void)p;
217 return true;
218#endif
219 };
220
221 auto isFastLedInputOnlyPin = [](int p) -> bool {
222#if defined(FL_IS_ESP32)
223 if (p < 0 || p >= 64) return false;
224 constexpr u64 inputOnlyMask =
225 u64(SOC_GPIO_VALID_GPIO_MASK) & ~u64(SOC_GPIO_VALID_OUTPUT_GPIO_MASK);
226 return (inputOnlyMask & (1ULL << p)) != 0;
227#else
228 (void)p;
229 return false;
230#endif
231 };
232
233 auto isFastLedReservedPin = [&](int p) -> bool {
234 // Reserved means "in FASTLED_UNUSABLE_PIN_MASK", i.e. flash /
235 // USB-JTAG / strap pins. Distinct from input-only (those are
236 // probe-as-RX-only, not "skip entirely").
237#if defined(FL_IS_ESP32)
238 if (p < 0 || p >= 64) return false;
239 return (FASTLED_UNUSABLE_PIN_MASK & (1ULL << p)) != 0;
240#else
241 (void)p;
242 return false;
243#endif
244 };
245
246 fl::json skipped_pins = fl::json::array();
247 auto recordSkip = [&](int p, const char* reason) {
248 fl::json entry = fl::json::object();
249 entry.set("pin", static_cast<int64_t>(p));
250 entry.set("reason", reason);
251 skipped_pins.push_back(entry);
252 };
253
254 FL_DBG("[PIN PROBE] Searching for connected pin pairs in range " << start_pin << "-" << end_pin);
255
256 // Helper lambda to test if two pins are connected
257 auto testPinPair = [](int tx, int rx) -> bool {
258 // Test 1: TX drives LOW, RX has pullup → RX should read LOW if connected
259 pinMode(tx, OUTPUT);
260 pinMode(rx, INPUT_PULLUP);
261 digitalWrite(tx, LOW);
262 delay(2); // Allow signal to settle
263 int rx_when_tx_low = digitalRead(rx);
264
265 // Test 2: TX drives HIGH → RX should read HIGH if connected
266 digitalWrite(tx, HIGH);
267 delay(2); // Allow signal to settle
268 int rx_when_tx_high = digitalRead(rx);
269
270 // Restore pins to safe state
271 pinMode(tx, INPUT);
272 pinMode(rx, INPUT);
273
274 return (rx_when_tx_low == LOW) && (rx_when_tx_high == HIGH);
275 };
276
277 // Search for connected adjacent pin pairs (n, n+1)
278 int found_tx = -1;
279 int found_rx = -1;
280 fl::json tested_pairs = fl::json::array();
281
282 for (int pin = start_pin; pin < end_pin; pin++) {
283 int tx_candidate = pin;
284 int rx_candidate = pin + 1;
285
286 // FastLED #3446: hands off the platform's reserved pads
287 // (SPI flash 6-11, USB-JTAG 20, …) — driving them locks the
288 // chip up or corrupts the live flash transaction. The
289 // FASTLED_UNUSABLE_PIN_MASK is per-chip authoritative.
290 if (isFastLedReservedPin(tx_candidate)) {
291 recordSkip(tx_candidate, "reserved-by-FastLED");
292 continue;
293 }
294 if (isFastLedReservedPin(rx_candidate)) {
295 recordSkip(rx_candidate, "reserved-by-FastLED");
296 continue;
297 }
298
299 fl::json pair = fl::json::object();
300 pair.set("tx", static_cast<int64_t>(tx_candidate));
301 pair.set("rx", static_cast<int64_t>(rx_candidate));
302
303 // Test TX→RX direction. Skip when the TX side can't actually
304 // drive output (e.g. input-only GPIO34-39 on classic ESP32).
305 if (!isFastLedValidOutputPin(tx_candidate)) {
306 recordSkip(tx_candidate, "input-only");
307 } else {
308 bool connected_forward = testPinPair(tx_candidate, rx_candidate);
309 if (connected_forward) {
310 pair.set("connected", true);
311 pair.set("direction", "forward");
312 tested_pairs.push_back(pair);
313 found_tx = tx_candidate;
314 found_rx = rx_candidate;
315 FL_DBG("[PIN PROBE] Found connected pair: TX=" << found_tx << " -> RX=" << found_rx);
316 break;
317 }
318 }
319
320 // Test RX→TX direction (reversed). Same drive-capability gate:
321 // if the would-be-driver is input-only, skip — `pinMode(p, OUTPUT)`
322 // is silently ignored and the test produces a false negative.
323 if (isFastLedInputOnlyPin(rx_candidate)) {
324 recordSkip(rx_candidate, "input-only");
325 tested_pairs.push_back(pair);
326 continue;
327 }
328 if (!isFastLedValidOutputPin(rx_candidate)) {
329 recordSkip(rx_candidate, "not-output-capable");
330 tested_pairs.push_back(pair);
331 continue;
332 }
333 bool connected_reverse = testPinPair(rx_candidate, tx_candidate);
334 if (connected_reverse) {
335 pair.set("connected", true);
336 pair.set("direction", "reverse");
337 tested_pairs.push_back(pair);
338 found_tx = rx_candidate; // Swap since reversed
339 found_rx = tx_candidate;
340 FL_DBG("[PIN PROBE] Found connected pair (reversed): TX=" << found_tx << " -> RX=" << found_rx);
341 break;
342 }
343
344 pair.set("connected", false);
345 tested_pairs.push_back(pair);
346 }
347
348 // NOTE: testedPairs array omitted - causes heap exhaustion on ESP32 (21+ objects = ~1500 bytes)
349 // Validation script doesn't use this data, only needs {found, txPin, rxPin}
350 // response.set("testedPairs", tested_pairs);
351 fl::json search_range = fl::json::array();
352 search_range.push_back(static_cast<int64_t>(start_pin));
353 search_range.push_back(static_cast<int64_t>(end_pin));
354 response.set("searchRange", search_range);
355
356 // FastLED #3446: surface every pin we declined to drive (and why)
357 // so the host log can render "Skipped GPIO 6/7/8/9/10/11 (SPI flash),
358 // 20 (USB-JTAG), 34-39 (input-only)". A small list — at most the
359 // bad-pin count for the platform — so this doesn't bloat the
360 // response heap budget the way `tested_pairs` did.
361 response.set("skippedPins", skipped_pins);
362
363 if (found_tx >= 0 && found_rx >= 0) {
364 response.set("success", true);
365 response.set("found", true);
366 response.set("txPin", static_cast<int64_t>(found_tx));
367 response.set("rxPin", static_cast<int64_t>(found_rx));
368
369 // Auto-apply the found pins if requested
370 if (auto_apply) {
371 int old_tx = mState->pin_tx;
372 int old_rx = mState->pin_rx;
373 bool tx_changed = (found_tx != old_tx);
374 bool rx_changed = (found_rx != old_rx);
375
376 fl::shared_ptr<fl::RxChannel> replacement_rx;
377 if (rx_changed) {
378 if (mState->rx_factory) {
379 replacement_rx = mState->rx_factory(found_rx);
380 }
381 if (!replacement_rx) {
382 response.set("success", false);
383 response.set("error", "RxChannelCreationFailed");
384 response.set("message", "Failed to recreate RX channel on discovered pin");
385 response.set("autoApplied", false);
386 return response;
387 }
388 }
389
390 // Commit only after the replacement channel is ready, preserving
391 // the prior channel and validation state on factory failure.
392 if (tx_changed || rx_changed) {
393 mState->invalidateLedRxValidation();
394 }
395 mState->pin_tx = found_tx;
396 mState->pin_rx = found_rx;
397 if (rx_changed) {
398 mState->rx_channel = replacement_rx;
399 }
400
401 FL_DBG("[PIN PROBE] Auto-applied pins: TX=" << found_tx << ", RX=" << found_rx);
402 response.set("autoApplied", true);
403 response.set("previousTxPin", static_cast<int64_t>(old_tx));
404 response.set("previousRxPin", static_cast<int64_t>(old_rx));
405 } else {
406 response.set("autoApplied", false);
407 response.set("message", "Use setPins to apply the found pins");
408 }
409 } else {
410 response.set("success", true); // Function succeeded, just no pins found
411 response.set("found", false);
412 response.set("message", "No connected pin pairs found. Please connect a jumper wire between adjacent GPIO pins.");
413 }
414
415 return response;
416}

468 getBleState().ble_server_active = false;
469 }

499fl::json AutoResearchRemoteControl::startBleRemote() {
500 if (mBleRemote) {
501 fl::json response = fl::json::object();
502 response.set("success", true);
503 response.set("message", "BLE remote already active");
504 response.set("device_name", AUTORESEARCH_BLE_DEVICE_NAME);
505 return response;
506 }
507
508 // Create BLE GATT server (heap-allocates transport state)
509 // On stub platforms, createTransport returns nullptr and logs FL_ERROR.
510 mBleState = fl::net::ble::createTransport(AUTORESEARCH_BLE_DEVICE_NAME);
511 if (!mBleState) {
512 fl::json response = fl::json::object();
513 response.set("success", false);
514 response.set("error", "BLE not available on this platform");
515 return response;
516 }
517
518 // Get transport lambdas that capture mBleState
519 auto callbacks = fl::net::ble::getTransportCallbacks(mBleState);
520
521 // Create BLE Remote instance with BLE transport
522 mBleRemote = fl::make_unique<fl::Remote>(callbacks.first, callbacks.second);
523
524 // Register RPC methods on the BLE remote
525 registerAllMethods(mBleRemote.get());
526
527 mState->ble_server_active = true;
528 getBleState().ble_server_active = true;
529
530 fl::json response = fl::json::object();
531 response.set("success", true);
532 response.set("device_name", AUTORESEARCH_BLE_DEVICE_NAME);
533 response.set("service_uuid", FL_BLE_SERVICE_UUID);
534 response.set("rx_uuid", FL_BLE_CHAR_RX_UUID);
535 response.set("tx_uuid", FL_BLE_CHAR_TX_UUID);
536 return response;
537}
