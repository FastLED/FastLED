55// Platform-specific clockless timing configuration
56//
57// IMPORTANT: All platforms use nanosecond-based timing at the API level.
58// Chipset timing values (T1, T2, T3) are specified in nanoseconds (e.g., T1=250ns, T2=625ns, T3=375ns)
59//
60// Platforms that need cycle-based timing internally (e.g., AVR, ARM, ESP8266) perform
61// nanosecond-to-cycle conversion within their clockless driver implementation using F_CPU or equivalent.
62//
63// Platform-specific clockless drivers are defined in:
64// - ESP32/ESP8266: platforms/esp/clockless.h
65// - All ARM platforms: led_sysdefs_arm_*.h + platform-specific clockless headers
66// - AVR platforms: led_sysdefs_avr.h + platforms/avr/clockless_*.h
67// - Other platforms: respective led_sysdefs_*.h files

191template <
192 typename CONTROLLER,
193 EOrder RGB_ORDER = GRB> // Default on WS2812>
194class RGBWEmulatedController
195 : public CPixelLEDController<RGB_ORDER, CONTROLLER::LANES_VALUE,
196 CONTROLLER::MASK_VALUE> {
197 public:
198 // ControllerT is a helper class. It subclasses the device controller class
199 // and has three methods to call the three protected methods we use.
200 // This is janky, but redeclaring public methods protected in a derived class
201 // is janky, too.
202
203 // N.B., byte order must be RGB.
204 typedef CONTROLLER ControllerBaseT;
205 class ControllerT : public CONTROLLER {
206 friend class RGBWEmulatedController<CONTROLLER, RGB_ORDER>;
207 void *callBeginShowLeds(int size) { return ControllerBaseT::beginShowLeds(size); }
208 void callShow(CRGB *data, int nLeds, fl::u8 brightness) {
209 ControllerBaseT::show(data, nLeds, brightness);
210 }
211 void callEndShowLeds(void *data) { ControllerBaseT::endShowLeds(data); }
212 };
213
214
215 static const int LANES = CONTROLLER::LANES_VALUE;
216 static const fl::u32 MASK = CONTROLLER::MASK_VALUE;
217
218 // The delegated controller must do no reordering.
219 FL_STATIC_ASSERT(RGB == CONTROLLER::RGB_ORDER_VALUE, "The delegated controller MUST NOT do reordering");
220
223 RGBWEmulatedController(const Rgbw& rgbw = RgbwDefault()) {
224 this->setRgbw(rgbw);
225 };
226
228 ~RGBWEmulatedController() = default;
229
230 virtual void *beginShowLeds(int size) override {
231 return mController.callBeginShowLeds(Rgbw::size_as_rgb(size));
232 }
233
234 virtual void endShowLeds(void *data) override {
235 return mController.callEndShowLeds(data);
236 }
237
245 virtual void showPixels(PixelController<RGB_ORDER, LANES, MASK> &pixels) override {
246 // Ensure buffer is large enough
247 ensureBuffer(pixels.size());
248 Rgbw rgbw = this->getRgbw();
249 fl::u8 *data = fl::bit_cast_ptr<fl::u8>(mRGBWPixels.get());
250 while (pixels.has(1)) {
251 pixels.stepDithering();
252 pixels.loadAndScaleRGBW(rgbw, data, data + 1, data + 2, data + 3);
253 data += 4;
254 pixels.advanceData();
255 }
256
257 // Force the device controller to a state where it passes data through
258 // unmodified: color correction, color temperature, dither, and brightness
259 // (passed as an argument to show()). Temporarily enable the controller,
260 // show the LEDs, and disable it again.
261 //
262 // The device controller is in the global controller list, so if we
263 // don't keep it disabled, it will refresh again with unknown brightness,
264 // temperature, etc.
265
266 mController.setCorrection(CRGB(255, 255, 255));
267 mController.setTemperature(CRGB(255, 255, 255));
268 mController.setDither(DISABLE_DITHER);
269
270 mController.setEnabled(true);
271 mController.callShow(mRGBWPixels.get(), Rgbw::size_as_rgb(pixels.size()), 255);
272 mController.setEnabled(false);
273 }
274
275 private:
279 void init() override {
280 mController.init();
281 mController.setEnabled(false);
282 }
283
288 void ensureBuffer(fl::i32 num_leds) {
289 if (num_leds != mNumRGBLeds) {
290 mNumRGBLeds = num_leds;
291 // The delegate controller expects the raw pixel byte data in multiples of 3.
292 // In the case of src data not a multiple of 3, then we need to
293 // add pad bytes so that the delegate controller doesn't walk off the end
294 // of the array and invoke a buffer overflow panic.
295 fl::u32 new_size = Rgbw::size_as_rgb(num_leds);
296 mRGBWPixels.reset(new CRGB[new_size]); // ok bare allocation (array new)
297 // showPixels may never clear the last two pixels.
298 for (fl::u32 i = 0; i < new_size; i++) {
299 mRGBWPixels[i] = CRGB(0, 0, 0);
300 }
301
302 mController.setLeds(mRGBWPixels.get(), new_size);
303 }
304 }
305
306 fl::unique_ptr<CRGB[]> mRGBWPixels;
307 fl::i32 mNumRGBLeds = 0;
308 fl::i32 mNumRGBWLeds = 0;
309 ControllerT mController;
310};

669// WS2816 - is an emulated controller that emits 48 bit pixels by forwarding
670// them to a platform specific WS2812 controller. The WS2812 controller
671// has to output twice as many 24 bit pixels.
672template <int DATA_PIN, EOrder RGB_ORDER = GRB>
673class WS2816Controller
674 : public CPixelLEDController<RGB_ORDER,
675 WS2812Controller800Khz<DATA_PIN, RGB>::LANES_VALUE,
676 WS2812Controller800Khz<DATA_PIN, RGB>::MASK_VALUE> {
677
678public:
679
680 // ControllerT is a helper class. It subclasses the device controller class
681 // and has three methods to call the three protected methods we use.
682 // This is janky, but redeclaring public methods protected in a derived class
683 // is janky, too.
684
685 // N.B., byte order must be RGB.
686 typedef WS2812Controller800Khz<DATA_PIN, RGB> ControllerBaseT;
687 class ControllerT : public ControllerBaseT {
688 friend class WS2816Controller<DATA_PIN, RGB_ORDER>;
689 void *callBeginShowLeds(int size) { return ControllerBaseT::beginShowLeds(size); }
690 void callShow(CRGB *data, int nLeds, fl::u8 brightness) {
691 ControllerBaseT::show(data, nLeds, brightness);
692 }
693 void callEndShowLeds(void *data) { ControllerBaseT::endShowLeds(data); }
694 };
695
696 static const int LANES = ControllerT::LANES_VALUE;
697 static const fl::u32 MASK = ControllerT::MASK_VALUE;
698
699 WS2816Controller() {}
700 ~WS2816Controller() {
701 mController.setLeds(nullptr, 0);
702 }
703
704 virtual void *beginShowLeds(int size) override {
705 mController.setEnabled(true);
706 void *result = mController.callBeginShowLeds(2 * size);
707 mController.setEnabled(false);
708 return result;
709 }
710
711 virtual void endShowLeds(void *data) override {
712 mController.setEnabled(true);
713 mController.callEndShowLeds(data);
714 mController.setEnabled(false);
715 }
716
717 virtual void showPixels(PixelController<RGB_ORDER, LANES, MASK> &pixels) override {
718 // Ensure buffer is large enough
719 ensureBuffer(pixels.size());
720
721 // Create PixelIterator from PixelController (WS2816 is RGB-only, no W channel)
722 fl::PixelIterator pixel_iter = pixels.as_iterator(fl::Rgbw());
723
724 // Create 16-bit RGB iterator range (handles RGB reordering, scaling, brightness)
725 auto rgb16_range = fl::makeScaledPixelRangeRGB16(&pixel_iter);
726
727 // Encode 16-bit RGB pixels to dual 8-bit CRGB format
728 fl::encodeWS2816(rgb16_range.first, rgb16_range.second, mData.get());
729
730 // Ensure device controller won't modify color values
731 mController.setCorrection(CRGB(255, 255, 255));
732 mController.setTemperature(CRGB(255, 255, 255));
733 mController.setDither(DISABLE_DITHER);
734
735 // Output the data stream
736 mController.setEnabled(true);
737 mController.callShow(mData.get(), 2 * pixels.size(), 255);
738 mController.setEnabled(false);
739 }
740
741private:
742 void init() override {
743 mController.init();
744 mController.setEnabled(false);
745 }
746
747 void ensureBuffer(int size_8bit) {
748 int size_16bit = 2 * size_8bit;
749 if (mController.size() != size_16bit) {
750
751 auto new_leds = fl::make_unique<CRGB[]>(size_16bit);
752 mData = fl::move(new_leds);
753 mController.setLeds(mData.get(), size_16bit);
754 }
755 }
756
757 fl::unique_ptr<CRGB[]> mData;
758 ControllerT mController;
759};
