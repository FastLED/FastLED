204// PixelIterator is turns a PixelController<> into a concrete object that can be used to iterate
205// over pixels and transform them into driver data. See PixelController<>::as_iterator() for how
206// to create a PixelIterator.
207// Note: This is designed for micro-controllers with a lot of memory. DO NOT use this in the core library
208// as a PixelIterator consumes a *lot* more instruction data than an instance of PixelController<RGB_ORDER>.
209// This iterator is designed for code in src/platforms/**.
210class PixelIterator {
211 public:
212 template<typename PixelControllerT>
213 PixelIterator(PixelControllerT* pc, const Rgbw& rgbw,
214 Rgbww rgbww = RgbwwInvalid::value()) FL_NO_EXCEPT
215 : mPixelController(pc), mRgbw(rgbw), mRgbww(rgbww) {
216 // Manually build up a vtable.
217 // Wait... what? Stupid nerds trying to show off how smart they are...
218 // Why not just use a virtual function?!
219 //
220 // Before you think this then you should know that the alternative straight
221 // forward way is to have a virtual interface class that PixelController inherits from.
222 // ...and that was already tried. And if you try to do this yourself
223 // this then let me tell you what is going to happen...
224 //
225 // EVERY SINGLE PLATFORM THAT HAS A COMPILED BINARY SIZE CHECK WILL IMMEDIATELY
226 // FAIL AS THE BINARY BLOWS UP BY 10-30%!!! It doesn't matter if only one PixelController
227 // with a vtable is used, gcc seems not to de-virtualize the calls. And we really care
228 // about binary size since FastLED needs to run on those tiny little microcontrollers like
229 // the Attiny85 (and family) which are in the sub $1 range used for commercial products.
230 //
231 // So to satisfy these tight memory requirements we make the dynamic dispatch used in PixelIterator
232 // an optional zero-cost abstraction which doesn't affect the binary size for platforms that
233 // don't use it. So that's why we are using this manual construction of the vtable that is built
234 // up using template magic. If your platform has lots of memory then you'll gladly trade
235 // a sliver of memory for the convenience of having a concrete implementation of
236 // PixelController that you can use without having to make all your driver code a template.
237 //
238 // Btw, this pattern in C++ is called the "type-erasure pattern". It allows non virtual
239 // polymorphism by leveraging the C++ template system to ensure type safety.
240 typedef PixelControllerVtable<PixelControllerT> Vtable;
241 mLoadAndScaleRGBW = RgbwLoadBinder<PixelControllerT>::ordered();
242 mLoadAndScaleRGBWW = RgbwLoadBinder<PixelControllerT>::orderedWW();
243 mLoadAndScaleRGBWUnordered = RgbwLoadBinder<PixelControllerT>::unordered();
244 mLoadAndScaleRGBWWUnordered = RgbwLoadBinder<PixelControllerT>::unorderedWW();
245 mLoadAndScaleRGB = &Vtable::loadAndScaleRGB;
246#if !FL_PLATFORM_HAS_TINY_MEMORY
247 mLoadAndScaleRGB16 = WideLoadBinder<PixelControllerT>::get();
248#endif
249 // NOTE: mLoadAndScale_APA102_HD removed - use fl::loadAndScale_APA102_HD<RGB_ORDER>() from apa102.h encoder
250 // NOTE: mLoadAndScale_WS2816_HD removed - use fl::loadAndScale_WS2816_HD<RGB_ORDER>() from ws2816.h encoder
251 mStepDithering = &Vtable::stepDithering;
252 mAdvanceData = &Vtable::advanceData;
253 mSize = &Vtable::size;
254 mHas = &Vtable::has;
255 #if FASTLED_HD_COLOR_MIXING
256 mLoadRGBScaleAndBrightness = &Vtable::loadRGBScaleAndBrightness;
257 mGetHdScale = &Vtable::getHdScale;
258 #endif
259 }

299 void loadAndScaleRGB(u8 *r_out, u8 *g_out, u8 *b_out) FL_NO_EXCEPT {
300 // The source's own order first (identity for PixelController<RGB>),
301 // then this iterator's. `loadAndScale<SLOT>` is per source channel --

307 *g_out = c[mOrder[1]];

623 template <typename CONTAINER_UIN8_T>
624 void writeHD108(CONTAINER_UIN8_T* out, bool managed = false) FL_NO_EXCEPT {
625 auto back_ins = fl::back_inserter(*out);
626
627 // One loop for both sources. This used to be two: the managed path
628 // written out by hand here, and the 8-bit path routed through
629 // `encodeHD108` over `makeScaledPixelRangeRGB`. Each carried its own
630 // copy of the framing -- the 8-byte start frame, the per-LED emit of
631 // header + three big-endian channels, the latch -- and together they
632 // made this the largest encoder body in a Blink build (481 B on
633 // esp32dev, ahead of `writeWS2812`, the only one Blink runs). The only
634 // thing that differed was where the 16-bit channels came from, so
635 // that is now the only branch (FastLED#4402).
636 //
637 // Byte-identical to the iterator path by construction:
638 // `ScaledPixelIteratorRGB::advance()` is exactly `has(1)`,
639 // `loadAndScaleRGB` into wire order, `stepDithering`, `advanceData`,
640 // once per pixel; the bytes it hands `encodeHD108` are captured before
641 // the step, which is what this loop does too.
642 //
643 // Inline rather than an `encodeHD108_wide` in hd108.h: that header is
644 // included *by* this one, before `PixelIterator` exists, and a
645 // concrete `PixelIterator&` parameter is not a dependent type -- so
646 // its body would be checked there and fail (#4326).
647#if FL_PLATFORM_HAS_TINY_MEMORY
648 // No wide source exists on TINY, so the flag selects nothing there.
649 FL_UNUSED(managed);
650#endif
651 for (int i = 0; i < 8; i++) {
652 *back_ins++ = 0x00; // start frame
653 }
654 // `hd108BrightnessHeader` discards its argument and pins every gain
655 // at 31, so this is 0xFF 0xFF whatever is passed (#4402).
656 u8 f0, f1;
657 hd108BrightnessHeader(0, &f0, &f1);
658 fl::size num_leds = 0;
659 while (has(1)) {
660 u16 r16, g16, b16;
661#if !FL_PLATFORM_HAS_TINY_MEMORY
662 if (managed) {
663 // The managed source has already quantized its device drive
664 // once, to 16 bits. `hd108GammaCorrect` exists to widen an
665 // 8-bit pixel to the 16 this chipset carries; a fixed 2.8
666 // curve on top of a solved drive is the second shaping stage
667 // B1 and section 6 of the spec forbid after the device
668 // solve, and unlike UCS7604's `mGamma.value_or(2.8f)` this
669 // one is hardcoded -- no caller could have chosen otherwise.
670 loadAndScaleRGB16(&r16, &g16, &b16);
671 } else
672#endif
673 {
674 u8 b0, b1, b2;
675 loadAndScaleRGB(&b0, &b1, &b2); // wire order
676 r16 = hd108GammaCorrect(b0);
677 g16 = hd108GammaCorrect(b1);
678 b16 = hd108GammaCorrect(b2);
679 }
680 *back_ins++ = f0;
681 *back_ins++ = f1;
682 *back_ins++ = static_cast<u8>(r16 >> 8);
683 *back_ins++ = static_cast<u8>(r16 & 0xFF);
684 *back_ins++ = static_cast<u8>(g16 >> 8);
685 *back_ins++ = static_cast<u8>(g16 & 0xFF);
686 *back_ins++ = static_cast<u8>(b16 >> 8);
687 *back_ins++ = static_cast<u8>(b16 & 0xFF);
688 stepDithering();
689 advanceData();
690 ++num_leds;
691 }
692 const fl::size latch = num_leds / 2 + 4;
693 for (fl::size i = 0; i < latch; i++) {
694 *back_ins++ = 0xFF; // end frame
695 }
696 }
697
698 private:
699 // vtable emulation
700 void* mPixelController = nullptr;
701 Rgbw mRgbw;
702 Rgbww mRgbww;
703 loadAndScaleRGBWFunction mLoadAndScaleRGBW = nullptr;
704 loadAndScaleRGBWWFunction mLoadAndScaleRGBWW = nullptr;
705 loadAndScaleRGBFunction mLoadAndScaleRGB = nullptr;
706 loadAndScaleRGBWUnorderedFunction mLoadAndScaleRGBWUnordered = nullptr;
707 loadAndScaleRGBWWUnorderedFunction mLoadAndScaleRGBWWUnordered = nullptr;
708#if !FL_PLATFORM_HAS_TINY_MEMORY
709 loadAndScaleRGB16Function mLoadAndScaleRGB16 = nullptr;
710#endif
711 // Wire byte i comes from source channel mOrder[i]. Identity unless
712 // setColorOrder() is called; PixelIteratorAny is the caller.
713 u8 mOrder[3] = {0, 1, 2};
714 // NOTE: mLoadAndScale_APA102_HD removed - use fl::loadAndScale_APA102_HD<RGB_ORDER>() from apa102.h encoder

813inline void ScaledPixelIteratorRGB16::advance() FL_NO_EXCEPT {
814 if (!mPixels) {
815 mHasValue = false;
816 return;
817 }
818
819 if (mPixels->has(1)) {
820 // Wire-ordered, colour-corrected, brightness-scaled pixel bytes.
821 //
822 // This used to branch on FASTLED_HD_COLOR_MIXING and call
823 // loadRGBScaleAndBrightness() here, treating its three outputs as the
824 // pixel. They are not the pixel: that call returns
825 // ColorAdjustment::color -- the correction *scale* at full
826 // brightness, a per-strip constant -- for a caller that will apply it
827 // to a pixel itself. So every LED came out the same colour and the
828 // frame never reached the wire (#4323). WS2816, the only user of this
829 // range, showed one flat tint on every non-AVR build.
830 //
831 // loadAndScaleRGB() is the call that loads the pixel. It applies
832 // `premixed`, which already carries brightness, so there is no
833 // separate brightness step left to do. The old branch existed to
834 // defer brightness to 16-bit precision, but it was producing a
835 // constant rather than extra precision, so nothing is given up by
836 // dropping it. Doing that properly needs a primitive that loads the
837 // pixel scaled at full brightness, which PixelIterator does not
838 // expose today.

841 // ordinary controller it is `map8_to_16` of the same 8-bit pixel, so
842 // an unbound channel is byte-identical to what this did before. A
843 // colour-managed source instead quantizes its s16.16 drive straight
844 // to 16 bits, which is what lets a 16-bit encoder reach more than 256
845 // levels.
846 u16 r16, g16, b16;
847 mPixels->loadAndScaleRGB16(&r16, &g16, &b16);
848#else
849 // TINY carries no colour pipeline, so there is nothing wider than the
850 // 8-bit pixel to load and the extra function pointer is not spent.
851 u8 b0, b1, b2;
852 mPixels->loadAndScaleRGB(&b0, &b1, &b2);
853
854 const u16 r16 = fl::map8_to_16(b0);
855 const u16 g16 = fl::map8_to_16(b1);
856 const u16 b16 = fl::map8_to_16(b2);
857#endif
858
859 mCurrent = array<u16, 3>{{r16, g16, b16}}; // Wire order 16-bit channels
860 mPixels->stepDithering();
861 mPixels->advanceData();
862 mHasValue = true;
863 } else {
864 mHasValue = false;
865 }
