34// All three sendAsync* methods touch mAsyncRequests, which Low-memory drops
35// (see #3224 Tier 1B RAM-side companion). On Low-memory these methods
36// become no-ops with a one-line FL_WARN; they remain in the API surface so
37// callers don't break, but they're link-DCE-friendly if no caller exists.

39#if FL_PLATFORM_HAS_LARGE_MEMORY
40void Remote::sendAsyncResponse(const char* method, const fl::json& result) {
41 fl::string methodName(method);
42 auto it = mAsyncRequests.find(methodName);
43 if (it == mAsyncRequests.end()) {
44 FL_WARN("No pending async request for method: %s", method);
45 return;
46 }
47
48 int requestId = it->second.requestId;
49 mAsyncRequests.erase(it);
50
51 // Build JSON-RPC response
52 fl::json response = fl::json::object();
53 response.set("jsonrpc", "2.0");
54 response.set("id", requestId);
55 response.set("result", result);
56
57 // Send via response sink
58 if (mResponseSink) {
59 mResponseSink(response);
60 FL_DBG("Sent async response for %s (id=%s)", method, requestId);
61 }
62}

153fl::json Remote::processRpc(const fl::json& request) {
154#if FL_PLATFORM_HAS_LARGE_MEMORY
155 // Extract optional timestamp field (0 = immediate, >0 = scheduled).
156 // Low-memory builds drop the scheduling path entirely (see #3224 Tier 1B):
157 // the LPC8xx integer-only RPC contract is invoked immediate-only.
158 u32 timestamp = 0;
159 if (request.contains("timestamp") && request["timestamp"].is_int()) {
160 timestamp = static_cast<u32>(request["timestamp"].as_int().value());
161 }
162#endif
163
164 u32 receivedAt = fl::millis();
165
166#if FL_PLATFORM_HAS_LARGE_MEMORY
167 // Execute or schedule
168 if (timestamp != 0) {
169 // Scheduled execution - result will be pushed to ResponseSink after execution
170 scheduleFunction(timestamp, receivedAt, request);
171 FL_DBG("RPC: Scheduled function - result will be pushed after execution");
172
173 // Return acknowledgment with null result and "scheduled" marker
174 fl::json response = fl::json::object();
175 if (request.contains("id")) {
176 response.set("id", request["id"]);
177 }
178 response.set("result", fl::json(nullptr));
179 response.set("scheduled", true); // Marker to not queue this response
180 return response;
181 }
182#endif
183
184 // Immediate execution path (always reached on Low-memory).
185
186#if FL_PLATFORM_HAS_LARGE_MEMORY
187 // Store request ID BEFORE invoking function (needed for async functions).
188 // This allows sendAsyncResponse() to find the request ID while function is
189 // running. Low-memory drops the async-tracking machinery (no production
190 // LowMemory sketch uses bindAsync / sendAsyncResponse today) -- see
191 // #3224 Tier 1B.
192 if (request.contains("id") && request.contains("method")) {
193 fl::string methodName = request["method"].as_string().value_or("");
194 int requestId = request["id"].as_int().value_or(0);
195 mAsyncRequests[methodName] = {requestId, receivedAt};
196 FL_DBG("Stored request ID for %s (id=%s)", methodName.c_str(), requestId);
197 }
198#endif
199
200 // Immediate execution - pass directly to Rpc
201 fl::json response = mRpc.handle(request);
202
203#if FL_PLATFORM_HAS_LARGE_MEMORY
204 // For async functions, response already sent via sendAsyncResponse().
205 // Envelope marker names renamed in #3228 -- accept legacy `__async` for
206 // one release of back-compat in case external Server forks set it.
207 bool ackAlready = (response.contains("ackOnly") && response["ackOnly"].as_bool().value_or(false))
208 || (response.contains("__async") && response["__async"].as_bool().value_or(false));
209 if (ackAlready) {
210 // Don't return response (ACK already sent by Rpc).
211 // Return a skip-envelope to prevent Server from queueing it.
212 fl::json nullResponse = fl::json::object();
213 nullResponse.set("noEnqueue", true);
214 return nullResponse;
215 }
216
217 // For sync functions, remove request ID (not needed)
218 if (request.contains("method")) {
219 fl::string methodName = request["method"].as_string().value_or("");
220 mAsyncRequests.erase(methodName);
221 }
222#endif
223
224#if FL_PLATFORM_HAS_LARGE_MEMORY
225 // Record result for the post-call results vector. LowMemory targets drop
226 // mResults entirely (see #3224 Tier 1B) -- RPC callers there receive
227 // results directly via the response sink, not through Remote::mResults.
228 if (response.contains("result") && request.contains("method")) {
229 fl::string funcName = request["method"].as_string().value_or("");
230 recordResult(funcName, response["result"], 0, receivedAt, receivedAt, false);
231 }
232#else
233 (void)receivedAt;
234#endif
235
236 return response;
237}

240void Remote::scheduleFunction(u32 timestamp, u32 receivedAt, const fl::json& jsonRpcRequest) {
241 // Make explicit copy for capture (avoid reference issues)
242 fl::json requestCopy = jsonRpcRequest;
243 fl::string funcName = requestCopy["method"].as_string().value_or("unknown");
244
245 // Wrap RPC execution in lambda and schedule it
246 mScheduler.schedule(timestamp, [this, requestCopy, timestamp, receivedAt, funcName]() {
247 u32 executedAt = fl::millis();
248
249 // Execute JSON-RPC request
250 fl::json response = mRpc.handle(requestCopy);
251
252 // Record result with timing metadata
253 if (response.contains("result") && requestCopy.contains("method")) {
254 recordResult(funcName, response["result"], timestamp, receivedAt, executedAt, true);
255 }
256 });
257
258 FL_DBG("Scheduled RPC: %s at %s", funcName, timestamp);
259}

347size_t Remote::update(u32 currentTimeMs) {
348 size_t processed = Server::pull(); // Pull requests from Server
349 size_t executed = tick(currentTimeMs); // Process scheduled tasks
350
351#if FL_PLATFORM_HAS_LARGE_MEMORY
352 // Push scheduled results as JSON-RPC responses. Gated on Low-memory
353 // because mResults is dropped there (see #3224 Tier 1B).
354 for (const auto& r : mResults) {
355 fl::json response = fl::json::object();
356 response.set("result", r.result);
357 // Note: We don't have the original request ID for scheduled calls
358 // This could be improved by storing the ID with RpcResult
359 mOutgoingQueue.push_back(response);
360 }
361#endif
362
363 size_t sent = Server::push(); // Push responses from Server
364 return processed + executed + sent;
365}

369fl::vector<Remote::MethodInfo> Remote::methods() const {
370 fl::vector<MethodInfo> result;
371
372 // Get flat JSON schema from underlying RPC
373 // Format: [["methodName", "returnType", [["param1", "type1"], ...]], ...]
374 fl::json jsonMethods = mRpc.methods();
375
376 if (!jsonMethods.is_array()) {
377 return result;
378 }
379
380 // Convert each flat method array to MethodInfo struct
381 for (fl::size i = 0; i < jsonMethods.size(); i++) {
382 fl::json method = jsonMethods[i];
383
384 if (!method.is_array() || method.size() < 3) {
385 continue; // Invalid method format
386 }
387
388 MethodInfo info;
389
390 // method[0] = method name
391 info.name = method[0].as_string().value_or("");
392
393 // method[1] = return type
394 info.returnType = method[1].as_string().value_or("void");
395
396 // method[2] = params array: [["param1", "type1"], ["param2", "type2"], ...]
397 if (method[2].is_array()) {
398 fl::json params = method[2];
399 for (fl::size j = 0; j < params.size(); j++) {
400 fl::json param = params[j];
401 if (param.is_array() && param.size() >= 2) {
402 ParamInfo paramInfo;
403 paramInfo.name = param[0].as_string().value_or("");
404 paramInfo.type = param[1].as_string().value_or("unknown");
405 info.params.push_back(fl::move(paramInfo));
406 }
407 }
408 }
409
410 // Flat schema doesn't include description/tags
411 info.description = "";
412 info.tags.clear();
413
414 result.push_back(fl::move(info));
415 }
416
417 return result;
418}
