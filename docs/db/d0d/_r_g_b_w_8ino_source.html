42void loop() {
43 uint32_t ms = millis();
44
45 for(int i = 0; i < NUM_LEDS; i++) {
46 // Use different noise functions for each LED and each color component
47 uint8_t hue = inoise16(ms * TIME_FACTOR_HUE, i * 1000, 0) >> 8;
48 uint8_t sat = inoise16(ms * TIME_FACTOR_SAT, i * 2000, 1000) >> 8;
49 uint8_t val = inoise16(ms * TIME_FACTOR_VAL, i * 3000, 2000) >> 8;
50
51 // Map the noise to full range for saturation and value
52 sat = map(sat, 0, 255, 30, 255);
53 val = map(val, 0, 255, 100, 255);
54
55 leds[i] = CHSV(hue, sat, val);
56 }
