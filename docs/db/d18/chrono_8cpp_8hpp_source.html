172fl::u64 millis64() {
173 Millis64State& state = get_millis64_state();
174 fl::u32 current_millis = fl::millis();
175 fl::unique_lock<fl::mutex> lock(state.mutex);
176
177 if (!state.initialized) {
178 // First call, set initial value
179 state.accumulated = current_millis;
180 state.last_millis = current_millis;
181 state.initialized = true;
182 return state.accumulated;
183 }
184
185 // Detect wraparound: current < last means 32-bit counter wrapped
186 // Delta calculation handles wraparound correctly via unsigned arithmetic
187 fl::u32 delta = current_millis - state.last_millis;
188
189 // Accumulate the delta into 64-bit counter
190 state.accumulated += delta;
191
192 // Update last value for next call
193 state.last_millis = current_millis;
194
195 return state.accumulated;
196}
