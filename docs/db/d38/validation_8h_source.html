65inline size_t captureEdgeCapacity(size_t shared_buffer_bytes,
66 size_t expected_data_bytes,
67 RxBackend backend) FL_NO_EXCEPT {
68 constexpr size_t kEdgesPerByte = 16;
69 constexpr size_t kFramingEdges = 2;
70 const size_t max_size = static_cast<size_t>(-1);
71
72 if (backend != RxBackend::ISR) {
73 if (shared_buffer_bytes > max_size / 8) {
74 return 0;
75 }
76 return shared_buffer_bytes * 8;
77 }
78
79 if (expected_data_bytes > (max_size - kFramingEdges) / kEdgesPerByte) {
80 return 0;
81 }
82 const size_t required =
83 expected_data_bytes * kEdgesPerByte + kFramingEdges;
84 size_t capacity = 1;
85 while (capacity < required) {
86 if (capacity > max_size / 2) {
87 return 0;
88 }
89 capacity *= 2;
90 }
91 return capacity;
92}

107inline bool isWholeCaptureFrame(size_t run_count, bool lead_run_is_low,
108 u32 lead_run_ns, u32 idle_threshold_ns,
109 size_t min_frame_runs) FL_NO_EXCEPT {
110 constexpr size_t kLeadPlusOneByteRuns = 1 + 8 + 7;
111 if (run_count >= min_frame_runs) {
112 return true;
113 }
114 return run_count >= kLeadPlusOneByteRuns && lead_run_is_low &&
115 lead_run_ns > idle_threshold_ns;

156inline size_t rpPioMaxWireBytes(size_t edge_capacity,
157 size_t dma_tail_words,
158 u32 samples_per_dma_word,
159 u32 sample_period_ns,
160 u32 idle_tail_ns,
161 u32 arming_lead_in_ns,
162 u32 bit_period_ns) FL_NO_EXCEPT {
163 constexpr size_t kPhasesPerByte = 16; // 8 bits x (high + low)
164 constexpr u64 kBitsPerByte = 8;
165
166 if (bit_period_ns == 0 || edge_capacity == 0 || samples_per_dma_word == 0 ||
167 sample_period_ns == 0) {
168 return 0;
169 }
170
171 const size_t by_pool = edge_capacity / kPhasesPerByte;
172
173 const u64 dma_words =
174 static_cast<u64>((edge_capacity + 1u) / 2u) + dma_tail_words;
175 const u64 budget_ns =
176 dma_words * samples_per_dma_word * sample_period_ns;
177 const u64 reserved_ns =
178 static_cast<u64>(idle_tail_ns) + arming_lead_in_ns;
179 if (budget_ns <= reserved_ns) {
180 return 0;
181 }
182 const u64 by_clock =
183 (budget_ns - reserved_ns) / (kBitsPerByte * bit_period_ns);
184
185 if (static_cast<u64>(by_pool) < by_clock) {
186 return by_pool;
187 }
188 return static_cast<size_t>(by_clock);
189}
