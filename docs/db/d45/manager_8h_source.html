350// Inline template definition for ChannelManager::setExclusiveDriver<B>().
351// Defined here so the ODR-use of `BusTraits<B>::instancePtr()` happens in the
352// caller's TU, which links the driver translation unit. The caller must have
353// the per-driver `bus_traits.h` (which provides `BusTraits<B>::instancePtr`)
354// visible at the call site.
355template<fl::Bus B, fl::u8 Which>
356inline void ChannelManager::setExclusiveDriver() FL_NO_EXCEPT {
357 // Priority above any platform default — see `bus_priorities.h` for the
358 // default-priority constants (highest there is ~10 for native LCD/I2S SPI).
359 constexpr int kExclusivePriority = 10000;
360 addDriver(kExclusivePriority, fl::BusTraits<B, Which>::instancePtr());
361}
