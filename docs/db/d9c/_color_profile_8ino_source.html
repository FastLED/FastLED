1// FL_AGENT_ALLOW_NEW_EXAMPLE -- the first example of the colour-profile API
2// (ChannelOptions::setColorProfile), which had none. It is also the probe
3// P9 (#4043) measures the colour pipeline's flash/RAM budget with: build it
4// with and without -DFASTLED_EXAMPLE_NO_COLOR_PROFILE=1 and the difference is
5// what binding a profile costs on that board.
6//
7// @filter: (memory is large)
8//
9// The Channels API alone is beyond an Uno's 32 KB. On small boards the colour
10// pipeline is compiled out entirely (FL_COLOR_PROFILE_RUNTIME, and the
11// shared-ownership parts under FL_COLOR_PIPELINE_SHARED), so they pay nothing
12// for it.
13//
14// What binding a profile does: every pixel goes through a streaming colour
15// pipeline -- decode the source (linear sRGB here), map it into the strip's
16// own gamut, solve the drives each emitter needs -- so a colour written as
17// CRGB lands at the same colour on the strip, rather than at whatever the
18// strip's primaries happen to make of those numbers. Brightness is applied
19// inside the pipeline as a linear flux scale, so dimming does not shift hue.

33void setup() {
34 fl::ChannelOptions opts;
35#if !defined(FASTLED_EXAMPLE_NO_COLOR_PROFILE)
36 // A profile describes the strip: its primaries' chromaticities and
37 // relative luminances. `WS2812B` is a placeholder until measured profiles
38 // land (P10); pass your own `fl::EmitterProfile` for a characterised part.
39 opts.setColorProfile(fl::profiles::WS2812B);
40#endif
41 // Dither. Binding a profile switches it off; asking for it again selects
42 // the pipeline's own temporal dither, for smoother fades at low
43 // brightness. Without a profile this is the default legacy dither, so
44 // both budget builds (see docs/color-pipeline-budgets.md) set it alike.
45 opts.mDitherMode = BINARY_DITHER;
46 FastLED.add(fl::ChannelConfig(
47 fl::makeClockless<fl::TIMING_WS2812_800KHZ>(PIN_DATA),
48 fl::span<CRGB>(leds, NUM_LEDS), GRB, opts));
49 FastLED.setBrightness(64);
50}

52void loop() {
53 // A slow hue sweep: with a profile bound, equal steps here are equal
54 // steps of colour on the strip.
55 const uint8_t hue = static_cast<uint8_t>(millis() / 32);
56 fill_rainbow(leds, NUM_LEDS, hue, 255 / NUM_LEDS);
57 FastLED.show();
58 delay(16);
59}
