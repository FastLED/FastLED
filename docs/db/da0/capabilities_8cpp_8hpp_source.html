37bool PinSet::acceptsAll(const PinBitset& request) const FL_NO_EXCEPT {
38 if (kind == PinSetKind::None) {
39 return request.none();
40 }
41 if (kind == PinSetKind::AnyOutputPin) {
42 return true;
43 }
44 // Allowlist: every requested bit must be in `pins`.
45 for (fl::u32 i = 0; i < kPinSetCapacity; ++i) {
46 if (request.test(i) && !pins.test(i)) {
47 return false;
48 }
49 }
50 return true;
51}
