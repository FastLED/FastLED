118inline fl::function<fl::optional<fl::json>()>
119createSerialRequestSource(const char* prefix = "") {
120 return [prefix]() -> fl::optional<fl::json> {
121 // Non-blocking check - any data available?
122 int avail = fl::available();
123 if (avail <= 0) {
124 return fl::nullopt;
125 }
126
127 // Data available - read a complete line.
128 // On Arduino platforms, fl::readLine() delegates to Serial.readStringUntil()
129 // which uses yield() (immediate context switch) for fast USB CDC multi-packet
130 // assembly. On other platforms, falls back to character-by-character reading.
131 auto line = fl::readLine('\n', '\r', fl::optional<u32>(1000));
132 if (!line.has_value() || line->empty()) {
133 return fl::nullopt;
134 }
135
136 // Use string_view for zero-copy prefix stripping and trimming
137 fl::string_view view = *line;
138
139 // Strip prefix if present
140 if (prefix && prefix[0] != '\0') {
141 if (view.starts_with(prefix)) {
142 view.remove_prefix(fl::strlen(prefix));
143 }
144 }
145
146 // Trim leading whitespace
147 while (!view.empty() && fl::isspace(view.front())) {
148 view.remove_prefix(1);
149 }
150
151 // Trim trailing whitespace
152 while (!view.empty() && fl::isspace(view.back())) {
153 view.remove_suffix(1);
154 }
155
156 // Only parse if input starts with '{'
157 if (view.empty() || view[0] != '{') {
158 // A short line here is ordinary serial noise -- boot banners,
159 // stray newlines -- and warning about it would drown the log.
160 // A long one is almost certainly a request that arrived damaged,
161 // and dropping that silently is expensive to debug: the caller
162 // sees no reply and no error, indistinguishable from an idle
163 // link, while the device keeps answering everything else. That
164 // is exactly how a receive-buffer truncation presented in
165 // FastLED#3956 -- 735 bytes transmitted, nothing returned, and
166 // no layer saying anything.
167 // Measure what arrived, not what survived normalisation: the
168 // view above has had its prefix stripped and whitespace trimmed,
169 // so a 64-byte line can fall under the threshold and stay silent,
170 // and the reported count would understate what was received.
171 if (line->size() >= kDroppedRequestWarnBytes) {
172 FL_WARN("[RPC] discarded a %u byte line that does not begin "
173 "with '{'; a request that arrived truncated looks "
174 "exactly like this",
175 static_cast<unsigned>(line->size()));
176 }
177 return fl::nullopt;
178 }
179
180 // Single copy when parsing JSON (unavoidable - JSON needs owned string)
181 fl::string input(view);
182 return fl::json::parse(input);
183 };
184}

203inline fl::ResponseStreamSink
204createSerialResponseStreamSink(const char* prefix = "REMOTE: ") FL_NO_EXCEPT {
205 return [prefix](fl::JsonStreamCallback writeJson) {
206 SerialWriter serial;
207 if (prefix && prefix[0] != '\0') {
208 serial.write(prefix, fl::strlen(prefix));
209 }
210
211 fl::JsonStreamWriter writer([&serial](const char* data, fl::size len) {
212 serial.write(data, len);
213 });
214 if (writeJson) {
215 writeJson(writer);
216 }
217 writer.flush();
218 serial.write("\n", 1);
219 fl::flush();
220 };
221}

248template<typename SerialIn>
249fl::optional<fl::string> readSerialStringUntil(SerialIn& serial, char delimiter, char skipChar, fl::optional<u32> timeoutMs) {
250 // Follows Arduino Serial.readStringUntil() API - blocks until delimiter found
251 fl::sstream buffer;
252
253 u32 startTime = fl::millis();
254
255 // Read characters until we find delimiter or timeout
256 while (true) {
257 // Check timeout (only if timeout is set)
258 if (timeoutMs.has_value()) {
259 if (fl::millis() - startTime >= timeoutMs.value()) {
260 // Timeout occurred - return nullopt
261 return fl::nullopt;
262 }
263 }
264
265 // Try to read next character
266 int c = serial.read();
267
268 // Handle -1 (no data available) like Arduino's timedRead():
269 // Keep trying until timeout (or forever if no timeout set)
270 if (c == -1) {
271 // Brief 1us yield instead of 1ms delay for fast USB CDC polling.
272 fl::delayMicroseconds(1);
273 continue;
274 }
275
276 // Found delimiter - complete
277 if (c == delimiter) {
278 break;
279 }
280
281 // Skip specified character (e.g., '\r' for cross-platform line endings)
282 if (c == skipChar) {
283 continue;
284 }
285
286 // Valid character - add to buffer
287 buffer << static_cast<char>(c);
288 }
289
290 // Convert to string and trim whitespace
291 fl::string result = buffer.str();
292 result.trim();
293 return result;
294}
