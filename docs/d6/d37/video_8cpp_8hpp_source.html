41 if (!mImpl) {
42 FL_WARN("Video::begin: mImpl is null, manually constructed videos "

46 // Drop a previous admission failure before evaluating this attempt.
47 // Rejecting one source says nothing about the next; only a persistent
48 // setError() failure keeps blocking.
49 if (mAdmissionError) {
50 mError.clear();
51 mAdmissionError = false;
52 }
53 if (!handle) {
54 setAdmissionError("filebuf is null");
55 FL_DBG("%s", mError.c_str());

60 return false;

63 setAdmissionError("unsupported or malformed FLED container");
64 return false;
