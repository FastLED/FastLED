13// StringHolder implementations
14//
15// OOM hardening (FastLED#3588): on classic ESP32 with WiFi + an HTTP
16// server active, free heap drops below 20 KB and fl::malloc genuinely
17// returns NULL. The previous "OOM is terminal" assumption turned that
18// into UB (writes through null; realloc leak plus null mData with a
19// stale nonzero capacity) and crashed unattended bench devices. On
20// failure a holder now degrades to a coherent empty state
21// (mData=null, mLength=0, mCapacity=0) and grow() keeps the old
22// buffer.
23

66void StringHolder::grow(size newLength) {
67 if (newLength + 1 <= mCapacity) {
68 // We have enough capacity for newLength + null terminator
69 mLength = newLength;
70 mData[mLength] = '\0';
71 return;
72 }
73
74 // Use fl::realloc for efficient growth without memory move.
75 // Keep the old buffer if the allocation fails — the string simply
76 // does not grow (callers observe the unchanged length).
77 char *grown = (char*)fl::realloc(mData, newLength + 1);
78 if (grown == nullptr) {
79 return;
80 }
81 mData = grown;
82 mLength = newLength;
83 mCapacity = newLength + 1;
84 mData[mLength] = '\0'; // Ensure null-termination
85}
