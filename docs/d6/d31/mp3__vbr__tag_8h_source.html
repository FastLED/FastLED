75inline bool Mp3ParseVbrTag(fl::span<const fl::u8> bytes,
76 Mp3VbrTag* out) FL_NO_EXCEPT {
77 out->present = false;
78 out->encoderDelay = 0;
79 out->encoderPadding = 0;
80 if (bytes.size() < 40) {
81 return false;
82 }
83 const fl::u8* const frame = bytes.data();
84 const fl::size length = bytes.size();
85 if (frame[0] != 0xFF || (frame[1] & 0xE0) != 0xE0) {
86 return false;
87 }
88
89 const int version_bits = (frame[1] >> 3) & 0x03;
90 const int layer_bits = (frame[1] >> 1) & 0x03;
91 if (layer_bits != 0x01) {
92 return false; // Layer III only; Xing/VBRI are not defined elsewhere.
93 }
94 const bool mpeg1 = version_bits == 0x03;
95 const bool mono = ((frame[3] >> 6) & 0x03) == 0x03;
96
97 // Side information sits between the header and the tag, and its size is
98 // fixed by version and channel count.
99 const fl::size side_info = mpeg1 ? (mono ? 17u : 32u) : (mono ? 9u : 17u);
100
101 // VBRI (Fraunhofer) always sits at a fixed offset and carries no delay we
102 // can trust to the sample, so it is reported as metadata without one.
103 if (length >= 40 && detail::mp3VbrMagic(frame + 36, "VBRI")) {
104 out->present = true;
105 return true;
106 }
107
108 const fl::size xing = 4 + side_info;
109 if (length < xing + 8) {
110 return false;
111 }
112 if (!detail::mp3VbrMagic(frame + xing, "Xing") &&
113 !detail::mp3VbrMagic(frame + xing, "Info")) {
114 return false;
115 }
116 out->present = true;
117
118 const fl::u32 flags = detail::mp3VbrRead32(frame + xing + 4);
119 fl::size cursor = xing + 8;
120 if (flags & 0x0001) { cursor += 4; } // frame count
121 if (flags & 0x0002) { cursor += 4; } // byte count
122 if (flags & 0x0004) { cursor += 100; } // seek table
123 if (flags & 0x0008) { cursor += 4; } // quality
124
125 // The encoder tag that follows is optional. Its delay and padding live 21
126 // bytes in, as 12 bits each packed across three bytes.
127 if (cursor + 24 > length) {
128 return true; // a tag frame, but with no usable delay information
129 }
130 const bool lame = detail::mp3VbrMagic(frame + cursor, "LAME") ||
131 detail::mp3VbrMagic(frame + cursor, "Lavc") ||
132 detail::mp3VbrMagic(frame + cursor, "Lavf");
133 if (!lame) {
134 return true;
135 }
136 const fl::u8* gap = frame + cursor + 21;
137 out->encoderDelay =
138 (static_cast<fl::u32>(gap[0]) << 4) | (static_cast<fl::u32>(gap[1]) >> 4);
139 out->encoderPadding =
140 ((static_cast<fl::u32>(gap[1]) & 0x0F) << 8) | static_cast<fl::u32>(gap[2]);
141 return true;
142}
