18// =============================================================================
19// variant lifecycle thunks (FastLED #3249 / #3244 Tier 3G)
20// =============================================================================
21//
22// Each (T x op) pair in `fl::variant<Ts...>` previously emitted its own
23// per-instantiation thunk -- on the json_value 10-alternative variant the
24// audit catalogued ~245 thunks totalling ~6.7 KB on LPC845.
25//
26// The collapse:
27// 1. Thunks now live at namespace scope (`fl::detail`) so two distinct
28// `variant<Ts...>` instantiations that share a type `T` also share
29// `typed_destroy_thunk<T>`, `typed_copy_thunk<T>`, `typed_move_thunk<T>`.
30// Previously each variant's outer template params re-mangled the symbol.
31// 2. Trivially-copyable T's route through a shared `noop_destroy` and
32// per-size `trivial_copy_n<sizeof(T)>` / `trivial_move_n<sizeof(T)>`
33// thunks. All `bool`, `i64`, `float`, `nullptr_t`, etc. alternatives
34// collapse to one no-op + a handful of memcpy bodies keyed on size,
35// regardless of how many variant types ride on them.
36//
37// `is_trivially_copyable<T>` is used as the gate for both the destroy AND
38// copy/move collapses: per the C++ standard, trivially-copyable implies
39// trivially-destructible, so this is strictly correct (and a touch more
40// conservative than the issue body's `is_trivially_destructible<T>` --
41// we don't have that trait in `fl/stl/type_traits.h` and the
42// conservative variant captures every json_value primitive alternative).
43//
44// `visit_fn<T, Visitor>` stays per-instantiation -- it encodes T-specific
45// dispatch into the visitor's `operator()(T const&)` and the collapse
46// would change behavior. See `variant<Ts...>::visit_fn` below.
47
48using variant_destroy_fn = void (*)(void* /*storage*/);
49using variant_copy_fn = void (*)(void* /*dst*/, const void* /*src*/);
50using variant_move_fn = void (*)(void* /*dst*/, void* /*src*/);
51
52// -- per-T thunks (used when T is not trivially-copyable) ---------------------
53
54template <typename T>
55void typed_destroy_thunk(void* storage) FL_NO_EXCEPT {
56 fl::bit_cast_ptr<T>(storage)->~T();
57}
58
59template <typename T>
60void typed_copy_thunk(void* dst, const void* src) FL_NO_EXCEPT {
61 const T* source = fl::bit_cast_ptr<const T>(src);
62 new (dst) T(*source);
63}
64
65template <typename T>
66void typed_move_thunk(void* dst, void* src) FL_NO_EXCEPT {
67 T* source = fl::bit_cast_ptr<T>(src);
68 new (dst) T(fl::move(*source));
69}

258 // –– visitor using O(1) function‐pointer table
259 template <typename Visitor> void visit(Visitor &visitor) FL_NO_EXCEPT {
260 if (_tag == Empty)
261 return;
262
263 // Fn is "a pointer to function taking (void* storage, Visitor&)"
264 using Fn = void (*)(void *, Visitor &);
265
266 // Build a constexpr array of one thunk per type in Types...
267 // Each thunk casts the storage back to the right T* and calls
268 // visitor.accept
269 static constexpr Fn table[] = {
270 &variant::template visit_fn<Types, Visitor>...};
271
272 // _tag is 1-based, so dispatch in O(1) via one indirect call:
273 // Check bounds to prevent out-of-bounds access
274 size_t index = _tag - 1;
275 if (index < sizeof...(Types)) {
276 table[index](&_storage, visitor);
277 }
278 }
