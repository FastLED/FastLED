81i32 VideoImpl::durationMicros() const {
82 if (!mStream) {
83 return -1;
84 }
85 i32 frames = mStream->framesRemaining();
86 if (frames < 0) {
87 return -1; // Stream case, duration unknown
88 }
89 fl::u32 micros_per_frame =
90 mFrameInterpolator->getFrameTracker().microsecondsPerFrame();
91 return (frames * micros_per_frame); // Convert to milliseconds

94bool VideoImpl::draw(fl::u32 now, fl::span<CRGB> leds) {
95 if (!mTime) {
96 mTime = fl::make_shared<TimeWarp>(now);
97 mTime->setSpeed(mTimeScale);
98 mTime->reset(now);
99 }
100 now = mTime->update(now);
101 if (!mStream) {
102 FL_WARN("no stream");
103 return false;
104 }
105 if (!mStream->isRgb8Playback()) {
106 for (fl::size_t i = 0; i < leds.size(); ++i) {
107 leds[i] = CRGB::Black;
108 }
109 return false;
110 }
111 bool ok = updateBufferIfNecessary(mPrevNow, now);
112 mPrevNow = now;
113 if (!ok) {
114 FL_WARN("updateBufferIfNecessary failed");
115 return false;
116 }
117 mFrameInterpolator->draw(now, leds);
118
119 fl::u32 time = mTime->time();
120 fl::u32 brightness = 255;
121 // Compute fade in/out brightness.
122 if (mFadeInTime || mFadeOutTime) {
123 brightness = 255;
124 if (time <= mFadeInTime) {
125 if (mFadeInTime == 0) {
126 brightness = 255;
127 } else {
128 brightness = time * 255 / mFadeInTime;
129 }
130 } else if (mFadeOutTime) {
131 i32 frames_remaining = mStream->framesRemaining();
132 if (frames_remaining < 0) {
133 // -1 means this is a stream.
134 brightness = 255;
135 } else {
136 FrameTracker &frame_tracker =
137 mFrameInterpolator->getFrameTracker();
138 fl::u32 micros_per_frame =
139 frame_tracker.microsecondsPerFrame();
140 fl::u32 millis_left =
141 (frames_remaining * micros_per_frame) / 1000;
142 if (millis_left < mFadeOutTime) {
143 brightness = millis_left * 255 / mFadeOutTime;
144 }
145 }
146 }
147 }
148 if (brightness < 255) {
149 if (brightness == 0) {
150 for (size_t i = 0; i < mPixelsPerFrame; ++i) {
151 leds[i] = CRGB::Black;
152 }
153 } else {
154 for (size_t i = 0; i < mPixelsPerFrame; ++i) {
155 leds[i].nscale8(brightness);
156 }
157 }
158 }
159 return true;
160}

174bool VideoImpl::updateBufferFromStream(fl::u32 now) {
175 FASTLED_ASSERT(mTime, "mTime is null");
176 if (!mStream) {
177 FL_WARN("no stream");
178 return false;
179 }
180 if (mStream->atEnd()) {
181 return false;
182 }
183
184 fl::u32 currFrameNumber = 0;
185 fl::u32 nextFrameNumber = 0;
186 bool needs_frame =
187 mFrameInterpolator->needsFrame(now, &currFrameNumber, &nextFrameNumber);
188 if (!needs_frame) {
189 return true;
190 }
191
192 if (mFrameInterpolator->capacity() == 0) {
193 FL_WARN("capacity == 0");
194 return false;
195 }
196
197 const bool has_current_frame = mFrameInterpolator->has(currFrameNumber);
198 const bool has_next_frame = mFrameInterpolator->has(nextFrameNumber);
199
200 fl::FixedVector<fl::u32, 2> frame_numbers;
201 if (!has_current_frame) {
202 frame_numbers.push_back(currFrameNumber);
203 }
204 size_t capacity = mFrameInterpolator->capacity();
205 if (capacity > 1 && !has_next_frame) {
206 frame_numbers.push_back(nextFrameNumber);
207 }
208
209 for (size_t i = 0; i < frame_numbers.size(); ++i) {
210 FramePtr recycled_frame;
211 if (mFrameInterpolator->full()) {
212 fl::u32 frame_to_erase = 0;
213 bool ok =
214 mFrameInterpolator->get_oldest_frame_number(&frame_to_erase);
215 if (!ok) {
216 FL_WARN("get_oldest_frame_number failed");
217 return false;
218 }
219 recycled_frame = mFrameInterpolator->erase(frame_to_erase);
220 if (!recycled_frame) {
221 FL_WARN("erase failed for frame: %s", frame_to_erase);
222 return false;
223 }
224 }
225 fl::u32 frame_to_fetch = frame_numbers[i];
226 if (!recycled_frame) {
227 // Happens when we are not full and we need to allocate a new frame.
228 recycled_frame = fl::make_shared<Frame>(mPixelsPerFrame);
229 }
230
231 if (!mStream->readFrame(recycled_frame.get())) {
232 if (mStream->atEnd()) {
233 if (!mStream->rewind()) {
234 FL_WARN("rewind failed");
235 return false;
236 }
237 mTime->reset(now);
238 frame_to_fetch = 0;
239 if (!mStream->readFrameAt(frame_to_fetch,
240 recycled_frame.get())) {
241 FL_WARN("readFrameAt failed");
242 return false;
243 }
244 } else {
245 // For streaming mode, buffer might not have data yet - this is OK
246 // Producer (effect renderer) will fill buffer on next call
247 // Just return success and we'll try again next frame
248 if (mStream->getType() == PixelStream::kStreaming) {
249 return true; // Gracefully handle empty streaming buffer
250 }
251 FL_WARN("We failed for some other reason");
252 return false;
253 }
254 }
255 bool ok = mFrameInterpolator->insert(frame_to_fetch, recycled_frame);
256 if (!ok) {
257 FL_WARN("insert failed");
258 return false;
259 }
260 }
261 return true;
262}

264bool VideoImpl::updateBufferFromFile(fl::u32 now, bool forward) {
265 fl::u32 currFrameNumber = 0;
266 fl::u32 nextFrameNumber = 0;
267 bool needs_frame =
268 mFrameInterpolator->needsFrame(now, &currFrameNumber, &nextFrameNumber);
269 if (!needs_frame) {
270 return true;
271 }
272 bool has_curr_frame = mFrameInterpolator->has(currFrameNumber);
273 bool has_next_frame = mFrameInterpolator->has(nextFrameNumber);
274 if (has_curr_frame && has_next_frame) {
275 return true;
276 }
277 if (mFrameInterpolator->capacity() == 0) {
278 FL_WARN("capacity == 0");
279 return false;
280 }
281
282 fl::FixedVector<fl::u32, 2> frame_numbers;
283 if (!mFrameInterpolator->has(currFrameNumber)) {
284 frame_numbers.push_back(currFrameNumber);
285 }
286 if (mFrameInterpolator->capacity() > 1 &&
287 !mFrameInterpolator->has(nextFrameNumber)) {
288 frame_numbers.push_back(nextFrameNumber);
289 }
290
291 for (size_t i = 0; i < frame_numbers.size(); ++i) {
292 FramePtr recycled_frame;
293 if (mFrameInterpolator->full()) {
294 fl::u32 frame_to_erase = 0;
295 bool ok = false;
296 if (forward) {
297 ok = mFrameInterpolator->get_oldest_frame_number(
298 &frame_to_erase);
299 if (!ok) {
300 FL_WARN("get_oldest_frame_number failed");
301 return false;
302 }
303 } else {
304 ok = mFrameInterpolator->get_newest_frame_number(
305 &frame_to_erase);
306 if (!ok) {
307 FL_WARN("get_newest_frame_number failed");
308 return false;
309 }
310 }
311 recycled_frame = mFrameInterpolator->erase(frame_to_erase);
312 if (!recycled_frame) {
313 FL_WARN("erase failed for frame: %s", frame_to_erase);
314 return false;
315 }
316 }
317 fl::u32 frame_to_fetch = frame_numbers[i];
318 if (!recycled_frame) {
319 // Happens when we are not full and we need to allocate a new frame.
320 recycled_frame = fl::make_shared<Frame>(mPixelsPerFrame);
321 }
322
323 do { // only to use break
324 if (!mStream->readFrameAt(frame_to_fetch, recycled_frame.get())) {
325 if (!forward) {
326 // nothing more we can do, we can't go negative.
327 return false;
328 }
329 if (mStream->atEnd()) {
330 if (!mStream->rewind()) { // Is this still
331 FL_WARN("rewind failed");
332 return false;
333 }
334 mTime->reset(now);
335 frame_to_fetch = 0;
336 if (!mStream->readFrameAt(frame_to_fetch,
337 recycled_frame.get())) {
338 FL_WARN("readFrameAt failed");
339 return false;
340 }
341 break; // we have the frame, so we can break out of the loop
342 }
343 FL_WARN("We failed for some other reason");
344 return false;
345 }
346 break;
347 } while (false);
348
349 bool ok = mFrameInterpolator->insert(frame_to_fetch, recycled_frame);
350 if (!ok) {
351 FL_WARN("insert failed");
352 return false;
353 }
354 }
355 return true;
356}

358bool VideoImpl::updateBufferIfNecessary(fl::u32 prev, fl::u32 now) {
359 const bool forward = now >= prev;
360
361 PixelStream::Type type = mStream->getType();
362 switch (type) {
363 case PixelStream::kFile:
364 return updateBufferFromFile(now, forward);
365 case PixelStream::kStreaming:
366 return updateBufferFromStream(now);
367 default:
368 FL_WARN("Unknown type: %s", fl::u32(type));
369 return false;
370 }
371}
