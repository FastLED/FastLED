222struct RxConfig {
223 // Hardware parameters
224 size_t buffer_size = 512;
225 fl::optional<u32> hz = fl::nullopt;
226
227 // Signal detection parameters
228 u32 signal_range_min_ns = 100;
229 u32 signal_range_max_ns = 100000;
230 u32 skip_signals = 0;
231 bool start_low = true;
232
233 // Internal loopback configuration (ESP32 RMT only)
234 // When true, RX receives from TX output internally (no external wire needed).
235 // IMPORTANT: For RMT TX → RMT RX loopback on ESP32-S3, both TX and RX MUST be
236 // configured on the SAME GPIO with io_loop_back=true. This is required because
237 // ESP32-S3 has a hardware limitation where RMT TX GPIO output stops when RMT RX
238 // is actively receiving on a different GPIO.
239 bool io_loop_back = false;
240
241 // DMA streaming mode (ESP32 RMT only). Non-DMA RMT RX caps a single receive
242 // at one mem-block fill (~4096 symbols, ≈170 WS2812B LEDs). DMA mode + ESP-IDF
243 // 5.3+ en_partial_rx fires the ISR multiple times per transmission — each fill
244 // copies into a DRAM accumulation buffer, extending capture to the full
245 // buffer_size. DMA bypasses on-chip RMT memory (uses DRAM), so it does NOT
246 // consume the shared RX pool and is safe to enable alongside other RMT
247 // channels. ESP32-S3 has one shared DMA slot; acquisition is coordinated via
248 // RmtMemoryManager::allocateDMA(). See issue #2254.
249 bool use_dma = false;
250
252 constexpr RxConfig() FL_NO_EXCEPT = default;
253};
