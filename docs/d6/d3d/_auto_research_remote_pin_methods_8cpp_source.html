122 // Register "setDebug" function - enable/disable runtime debug logging
123 remote.bind("setDebug", [this](const fl::json& args) -> fl::json {
124 fl::json response = fl::json::object();
125
126 // Validate args: expects [enabled: bool]
127 if (!args.is_array() || args.size() != 1) {
128 response.set("success", false);
129 response.set("error", "InvalidArgs");
130 response.set("message", "Expected [enabled: bool]");
131 return response;
132 }
133
134 if (!args[0].is_bool()) {
135 response.set("success", false);
136 response.set("error", "InvalidType");
137 response.set("message", "Argument must be boolean");
138 return response;
139 }
140
141 bool enabled = args[0].as_bool().value();
142 mState->debug_enabled = enabled;
143 fl::setLogLevel(static_cast<fl::u8>(
144 enabled ? fl::LogLevel::FL_LOG_LEVEL_DEBUG
145 : fl::LogLevel::FL_LOG_LEVEL_NONE));
146
147 response.set("success", true);
148 response.set("debug_enabled", enabled);
149 response.set("message", enabled ? "Debug logging enabled" : "Debug logging disabled");
150
151 return response;

154 // Register "testGpioConnection" function - test if TX and RX pins are electrically connected
155 // This is a pre-test to diagnose hardware connection issues before running validation
156 remote.bind("testGpioConnection", [](const fl::json& args) -> fl::json {
157 fl::json response = fl::json::object();
158
159 // Validate args: expects [txPin, rxPin]
160 if (!args.is_array() || args.size() != 2) {
161 response.set("error", "InvalidArgs");
162 response.set("message", "Expected [txPin, rxPin]");
163 return response;
164 }
165
166 if (!args[0].is_int() || !args[1].is_int()) {
167 response.set("error", "InvalidPinType");
168 response.set("message", "Pin numbers must be integers");
169 return response;
170 }
171
172 int tx_pin = static_cast<int>(args[0].as_int().value());
173 int rx_pin = static_cast<int>(args[1].as_int().value());
174
175 // Test 1: TX drives LOW, RX has pullup → RX should read LOW if connected
176 pinMode(tx_pin, OUTPUT);
177 pinMode(rx_pin, INPUT_PULLUP);
178 digitalWrite(tx_pin, LOW);
179 delay(5); // Allow signal to settle
180 int rx_when_tx_low = digitalRead(rx_pin);
181
182 // Test 2: TX drives HIGH → RX should read HIGH if connected
183 digitalWrite(tx_pin, HIGH);
184 delay(5); // Allow signal to settle
185 int rx_when_tx_high = digitalRead(rx_pin);
186
187 // Restore pins to safe state
188 pinMode(tx_pin, INPUT);
189 pinMode(rx_pin, INPUT);
190
191 // Analyze results
192 bool connected = (rx_when_tx_low == LOW) && (rx_when_tx_high == HIGH);
193
194 response.set("txPin", static_cast<int64_t>(tx_pin));
195 response.set("rxPin", static_cast<int64_t>(rx_pin));
196 response.set("rxWhenTxLow", rx_when_tx_low == LOW ? "LOW" : "HIGH");
197 response.set("rxWhenTxHigh", rx_when_tx_high == HIGH ? "HIGH" : "LOW");
198 response.set("connected", connected);
199
200 if (connected) {
201 response.set("success", true);
202 response.set("message", "GPIO pins are connected");
203 } else {
204 response.set("success", false);
205 if (rx_when_tx_low == HIGH && rx_when_tx_high == HIGH) {
206 response.set("message", "RX pin stuck HIGH - no connection detected (check jumper wire)");
207 } else if (rx_when_tx_low == LOW && rx_when_tx_high == LOW) {
208 response.set("message", "RX pin stuck LOW - possible short to ground");
209 } else {
210 response.set("message", "Unexpected GPIO behavior - check wiring");
211 }
212 }
213
214 return response;
215 });

221 // Register "getPins" function - query current and default pin configuration
222 remote.bind("getPins", [this](const fl::json& args) -> fl::json {
223 fl::json response = fl::json::object();
224 response.set("success", true);
225 response.set("txPin", static_cast<int64_t>(mState->pin_tx));
226 response.set("rxPin", static_cast<int64_t>(mState->pin_rx));
227
228 fl::json defaults = fl::json::object();
229 defaults.set("txPin", static_cast<int64_t>(mState->default_pin_tx));
230 defaults.set("rxPin", static_cast<int64_t>(mState->default_pin_rx));
231 response.set("defaults", defaults);
232
233 // Use the shared chipName() so this agrees with `status`. The
234 // hand-rolled chain here only knew four ESP32 variants and reported
235 // "unknown" for every other target, so an RP2350W answered
236 // getPins -> platform "unknown"
237 // status -> platform "Raspberry Pi Pico 2 W (RP2350)"
238 // on the same firmware (FastLED#3899, whose closeout requires exact
239 // recorded identities). chipName() covers stub, the ESP32 family,
240 // Teensy 4.x and both RP2350 variants.
241 response.set("platform", autoresearch::chipName());
242
243 return response;
244 });

246 // Register "setTxPin" function - set TX pin (regenerates test cases)
247 remote.bind("setTxPin", [this](const fl::json& args) -> fl::json {
248 fl::json response = fl::json::object();
249
250 if (!args.is_array() || args.size() != 1 || !args[0].is_int()) {
251 response.set("error", "InvalidArgs");
252 response.set("message", "Expected [pin: int]");
253 return response;
254 }
255
256 int new_pin = static_cast<int>(args[0].as_int().value());
257
258 // Validate pin range (ESP32 GPIO range)
259 if (new_pin < 0 || new_pin > 48) {
260 response.set("error", "InvalidPin");
261 response.set("message", "Pin must be 0-48");
262 return response;
263 }
264
265 int old_pin = mState->pin_tx;
266 if (new_pin != old_pin) {
267 mState->invalidateLedRxValidation();
268 }
269 mState->pin_tx = new_pin;
270
271 response.set("success", true);
272 response.set("txPin", static_cast<int64_t>(new_pin));
273 response.set("previousTxPin", static_cast<int64_t>(old_pin));
274 return response;
275 });
276
277 // Register "setRxPin" function - set RX pin (recreates RX channel)
278 remote.bind("setRxPin", [this](const fl::json& args) -> fl::json {
279 fl::json response = fl::json::object();
280
281 if (!args.is_array() || args.size() != 1 || !args[0].is_int()) {
282 response.set("error", "InvalidArgs");
283 response.set("message", "Expected [pin: int]");
284 return response;
285 }
286
287 int new_pin = static_cast<int>(args[0].as_int().value());
288
289 // Validate pin range (ESP32 GPIO range)
290 if (new_pin < 0 || new_pin > 48) {
291 response.set("error", "InvalidPin");
292 response.set("message", "Pin must be 0-48");
293 return response;
294 }
295
296 int old_pin = mState->pin_rx;
297 bool pin_changed = (new_pin != old_pin);
298 bool rx_recreated = false;
299
300 if (pin_changed) {
301 fl::shared_ptr<fl::RxChannel> replacement_rx;
302 if (mState->rx_factory) {
303 replacement_rx = mState->rx_factory(new_pin);
304 }
305 if (!replacement_rx) {
306 response.set("error", "RxChannelCreationFailed");
307 response.set("message", "Failed to create RX channel on new pin");
308 response.set("pinRx", static_cast<int64_t>(new_pin));
309 return response;
310 }
311
312 mState->invalidateLedRxValidation();
313 mState->pin_rx = new_pin;
314 mState->rx_channel = replacement_rx;
315 rx_recreated = true;
316 }
317
318 response.set("success", true);
319 response.set("rxPin", static_cast<int64_t>(new_pin));
320 response.set("previousRxPin", static_cast<int64_t>(old_pin));
321 response.set("rxChannelRecreated", rx_recreated);
322 return response;
323 });
324
325 // Register "setPins" function - set both TX and RX pins atomically
326 remote.bind("setPins", [this](const fl::json& args) -> fl::json {
327 fl::json response = fl::json::object();
328
329 // Accept either {txPin, rxPin} object or [txPin, rxPin] array
330 int new_tx_pin = -1;
331 int new_rx_pin = -1;
332
333 if (args.is_array() && args.size() == 1 && args[0].is_object()) {
334 // Object form: [{txPin: int, rxPin: int}]
335 fl::json config = args[0];
336 if (config.contains("txPin") && config["txPin"].is_int()) {
337 new_tx_pin = static_cast<int>(config["txPin"].as_int().value());
338 }
339 if (config.contains("rxPin") && config["rxPin"].is_int()) {
340 new_rx_pin = static_cast<int>(config["rxPin"].as_int().value());
341 }
342 } else if (args.is_array() && args.size() == 2) {
343 // Array form: [txPin, rxPin]
344 if (args[0].is_int()) {
345 new_tx_pin = static_cast<int>(args[0].as_int().value());
346 }
347 if (args[1].is_int()) {
348 new_rx_pin = static_cast<int>(args[1].as_int().value());
349 }
350 } else {
351 response.set("error", "InvalidArgs");
352 response.set("message", "Expected [{txPin, rxPin}] or [txPin, rxPin]");
353 return response;
354 }
355
356 // Validate pin ranges
357 if (new_tx_pin < 0 || new_tx_pin > 48) {
358 response.set("error", "InvalidTxPin");
359 response.set("message", "TX pin must be 0-48");
360 return response;
361 }
362 if (new_rx_pin < 0 || new_rx_pin > 48) {
363 response.set("error", "InvalidRxPin");
364 response.set("message", "RX pin must be 0-48");
365 return response;
366 }
367
368 int old_tx_pin = mState->pin_tx;
369 int old_rx_pin = mState->pin_rx;
370 bool tx_pin_changed = (new_tx_pin != old_tx_pin);
371 bool rx_pin_changed = (new_rx_pin != old_rx_pin);
372 bool rx_recreated = false;
373
374 fl::shared_ptr<fl::RxChannel> replacement_rx;
375 if (rx_pin_changed) {
376 if (mState->rx_factory) {
377 replacement_rx = mState->rx_factory(new_rx_pin);
378 }
379 if (!replacement_rx) {
380 response.set("error", "RxChannelCreationFailed");
381 response.set("message", "Failed to create RX channel - previous pins preserved");
382 response.set("pinRx", static_cast<int64_t>(new_rx_pin));
383 return response;
384 }
385 }
386
387 // Commit pins, validation, and channel as one state transition only
388 // after any replacement channel has been created successfully.
389 if (tx_pin_changed || rx_pin_changed) {
390 mState->invalidateLedRxValidation();
391 }
392 mState->pin_tx = new_tx_pin;
393 mState->pin_rx = new_rx_pin;
394 if (rx_pin_changed) {
395 mState->rx_channel = replacement_rx;
396 rx_recreated = true;
397 }
398
399 response.set("success", true);
400 response.set("txPin", static_cast<int64_t>(new_tx_pin));
401 response.set("rxPin", static_cast<int64_t>(new_rx_pin));
402 response.set("previousTxPin", static_cast<int64_t>(old_tx_pin));
403 response.set("previousRxPin", static_cast<int64_t>(old_rx_pin));
404 response.set("rxChannelRecreated", rx_recreated);
405 return response;
406 });
