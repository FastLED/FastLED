426 virtual void* beginShowLeds(int size) FL_NO_EXCEPT {
427 FASTLED_UNUSED(size);
428 // By default, emit an integer. This integer will, by default, be passed back.
429 // If you override beginShowLeds() then
430 // you should also override endShowLeds() to match the return state.
431 //
432 // For async led controllers this should be used as a sync point to block
433 // the caller until the leds from the last draw frame have completed drawing.
434 // for each controller:
435 // beginShowLeds();
436 // for each controller:
437 // showLeds();
438 // for each controller:
439 // endShowLeds();
440 uintptr_t d = getDither();
441 void* out = fl::int_to_ptr<void>(d);
442 return out;
443 }
444
445 virtual void endShowLeds(void* data) FL_NO_EXCEPT {
446 // By default recieves the integer that beginShowLeds() emitted.
447 //For async controllers this should be used to signal the controller
448 // to begin transmitting the current frame to the leds.
449 uintptr_t d = fl::ptr_to_int(data);
450 setDither(static_cast<fl::u8>(d));
451 }
