38// =====================================================================
39// FreqRange / NanosRange — inclusive [min, max] ranges with explicit
40// "unspecified" sentinels.
41//
42// All-zero is intentionally NOT the unspecified state — that would
43// silently match every legitimate value. Use any() / exactly() /
44// atLeast() / atMost() to construct.
45// =====================================================================
46
47struct FreqRange {
48 fl::u32 min_hz;
49 fl::u32 max_hz;
50
51 static constexpr FreqRange any() FL_NO_EXCEPT {
52 return {0u, 0xFFFFFFFFu};
53 }
54 static constexpr FreqRange exactly(fl::u32 hz) FL_NO_EXCEPT {
55 return {hz, hz};
56 }
57 static constexpr FreqRange atLeast(fl::u32 hz) FL_NO_EXCEPT {
58 return {hz, 0xFFFFFFFFu};
59 }
60 static constexpr FreqRange atMost(fl::u32 hz) FL_NO_EXCEPT {
61 return {0u, hz};
62 }
63
64 constexpr bool isUnspecified() const FL_NO_EXCEPT {
65 return min_hz == 0u && max_hz == 0xFFFFFFFFu;
66 }
67 constexpr bool contains(fl::u32 hz) const FL_NO_EXCEPT {
68 return hz >= min_hz && hz <= max_hz;
69 }
70};

98// =====================================================================
99// PinSet — discriminated pin allowlist (bitset-backed).
100//
101// Four conceptual shapes, three encodings:
102// None — slot not applicable (e.g. clock on a Clockless group)
103// AnyOutputPin — any GPIO output (RMT5-style flexibility)
104// Allowlist size = 1 — fixed pin (LPUART per-instance TX, SPI IO_MUX)
105// Allowlist size N>1 — flex within an allowlist (FlexIO shifters)
106// =====================================================================
107
111static constexpr fl::u32 kPinSetCapacity = 128u;
112
113using PinBitset = fl::bitset_fixed<kPinSetCapacity>;
114
115enum class PinSetKind : fl::u8 {
116 None = 0,
117 AnyOutputPin = 1,
118 Allowlist = 2,
119};
120
121struct PinSet {
122 PinSetKind kind;
123 PinBitset pins;
124
125 constexpr PinSet() FL_NO_EXCEPT : kind(PinSetKind::None), pins() {}
126 constexpr PinSet(PinSetKind k) FL_NO_EXCEPT : kind(k), pins() {}
127 constexpr PinSet(PinSetKind k, const PinBitset& b) FL_NO_EXCEPT : kind(k), pins(b) {}
128
130 static constexpr PinSet none() FL_NO_EXCEPT { return PinSet(PinSetKind::None); }
131
133 static constexpr PinSet anyOutput() FL_NO_EXCEPT { return PinSet(PinSetKind::AnyOutputPin); }
134
137 template <fl::size N>
138 static PinSet allowlist(const fl::i16 (&arr)[N]) FL_NO_EXCEPT {
139 PinSet ps(PinSetKind::Allowlist);
140 for (fl::size i = 0; i < N; ++i) {
141 if (arr[i] >= 0 && static_cast<fl::u32>(arr[i]) < kPinSetCapacity) {
142 ps.pins.set(static_cast<fl::u32>(arr[i]));
143 }
144 }
145 return ps;
146 }

151
154 bool accepts(fl::i16 pin) const FL_NO_EXCEPT;
155

161};

164// =====================================================================
165// ClocklessTimingCapability — driver's clock-tree model.
166//
167// Captures the physics: tick width = base_clock_hz / divider, phases
168// are held for an integer number of ticks. Compatibility with a
169// chipset's tolerance windows is determined by whether an integer N
170// exists such that N * tick_ns falls inside each window.
171//
172// DMA-fed drivers (PARLIO, I2S) also fit: their "tick" is the divided
173// peripheral clock, their "phase" is consecutive same-state bits in a
174// shaped data pattern.
175// =====================================================================
176
177struct ClocklessTimingCapability {
178 fl::u32 base_clock_hz;
179 fl::u16 min_divider;
180 fl::u16 max_divider;
181 fl::u16 max_phase_ticks;
182
183 static constexpr ClocklessTimingCapability unspecified() FL_NO_EXCEPT {
184 return {0u, 0u, 0u, 0u};
185 }
186 constexpr bool isUnspecified() const FL_NO_EXCEPT {
187 return base_clock_hz == 0u;
188 }
189};

314// =====================================================================
315// ChannelRequest — passive input to canMatch.
316//
317// One type for both single-pin and ganged-multi-pin requests: data_pins
318// is a bitset, so a single-pin request just has one bit set. Drivers
319// that fan out to N independent groups vs gang into one group is the
320// resolver's concern (not the request's).
321//
322// Requests are NOT a first-class API surface — sketches don't construct
323// them. The Channel Manager builds them from addLeds<>() calls and
324// hands them to fl::canMatch against each driver's published capability
325// data.
326// =====================================================================
327
