195 static FASTLED_FORCE_INLINE u16x16 pow(u16x16 base, u16x16 exp) FL_NO_EXCEPT {
196 constexpr u16x16 one(1.0f);
197 if (exp.mValue == 0) return one;
198 if (base == one) return one;
199 if (base.mValue == 0) return u16x16();
200 // Snap base values within ~2 ULPs of 1.0 to exactly 1.0 to dodge the
201 // log2(1+t) minimax polynomial's upper-endpoint residual (#2969).
202 constexpr u32 kOneRaw = static_cast<u32>(SCALE);
203 if (base.mValue >= (kOneRaw - 2u) && base.mValue <= kOneRaw) {
204 return one;
205 }
206 return exp2_fp(exp * log2_fp(base));
207 }

212 // Returns 0-based position of highest set bit, or -1 if v==0.
213 static constexpr FASTLED_FORCE_INLINE int highest_bit(u32 v) FL_NO_EXCEPT {
214 return v == 0 ? -1 : _highest_bit_step(v, 0);
215 }
216
217 static constexpr int _highest_bit_step(u32 v, int r) FL_NO_EXCEPT {
218 return (v & 0xFFFF0000u) ? _highest_bit_step(v >> 16, r + 16)
219 : (v & 0x0000FF00u) ? _highest_bit_step(v >> 8, r + 8)
220 : (v & 0x000000F0u) ? _highest_bit_step(v >> 4, r + 4)
221 : (v & 0x0000000Cu) ? _highest_bit_step(v >> 2, r + 2)
222 : (v & 0x00000002u) ? r + 1
223 : r;
224 }

226 // Fixed-point log base 2 for positive values.
227 // Uses 2-term polynomial for log2(1+t), t in [0,1) (simplified for unsigned).
228 // Horner evaluation uses u64 intermediates (24 frac bits) to minimize
229 // rounding error, then converts back to 16 frac bits.
230 static FASTLED_FORCE_INLINE u16x16 log2_fp(u16x16 x) FL_NO_EXCEPT {
231 u32 val = x.mValue;
232 int msb = highest_bit(val);
233 if (msb < 0) return u16x16(); // log2(0) = undefined, return 0
234
235 u32 int_part = msb - FRAC_BITS;
236 u32 t;
237 if (msb >= FRAC_BITS) {
238 t = (val >> (msb - FRAC_BITS)) - (SCALE);
239 } else {
240 t = (val << (FRAC_BITS - msb)) - (SCALE);
241 }
242
243 // 2-term polynomial coefficients for log2(1+t), t in [0,1).
244 // Stored as u64 with 24 fractional bits.
245 constexpr int IFRAC = 24;
246 constexpr u64 c0 = 24189248ULL; // 1.44179 * 2^24
247 constexpr u64 c1 = 5049984ULL; // 0.30093 * 2^24
248
249 // Extend t from 16 to 24 frac bits.
250 u64 t24 = static_cast<u64>(t) << (IFRAC - FRAC_BITS);
251
252 // Simplified Horner: c0 + c1*t, then multiply by t
253 u64 acc = c0 + ((c1 * t24) >> IFRAC);
254 u64 frac_part = (acc * t24) >> IFRAC;
255
256 // Convert from 24 frac bits back to 16.
257 u32 frac16 = static_cast<u32>(frac_part >> (IFRAC - FRAC_BITS));
258 return from_raw((int_part << FRAC_BITS) + frac16);
259 }
260
261 // Fixed-point 2^x. Uses 4-term minimax polynomial for 2^t, t in [0,1).
262 // Horner evaluation uses u64 intermediates (24 frac bits) to minimize
263 // rounding error, then converts back to 16 frac bits.
264 static FASTLED_FORCE_INLINE u16x16 exp2_fp(u16x16 x) FL_NO_EXCEPT {
265 u16x16 fl_val = floor(x);
266 u16x16 fr = x - fl_val;
267 u32 n = fl_val.mValue >> FRAC_BITS;
268
269 if (n >= INT_BITS) return from_raw(0xFFFFFFFF); // overflow
270
271 u32 int_pow = (SCALE) << n;
272
273 // 4-term minimax coefficients for 2^t - 1, t in [0,1).
274 // Stored as u64 with 24 fractional bits.
275 constexpr int IFRAC = 24;
276 constexpr u64 d0 = 11629376ULL; // 0.69316 * 2^24
277 constexpr u64 d1 = 4038400ULL; // 0.24071 * 2^24
278 constexpr u64 d2 = 895232ULL; // 0.05336 * 2^24
279 constexpr u64 d3 = 214016ULL; // 0.01276 * 2^24
280
281 // Extend fr from 16 to 24 frac bits.
282 u64 fr24 = static_cast<u64>(fr.mValue) << (IFRAC - FRAC_BITS);
283
284 // Horner: 1 + fr * (d0 + fr * (d1 + fr * (d2 + fr * d3)))
285 u64 acc = d3;
286 acc = d2 + ((acc * fr24) >> IFRAC);
287 acc = d1 + ((acc * fr24) >> IFRAC);
288 acc = d0 + ((acc * fr24) >> IFRAC);
289 constexpr u64 one24 = 1ULL << IFRAC;
290 u64 frac_pow24 = one24 + ((acc * fr24) >> IFRAC);
291
292 // Convert from 24 frac bits to 16 frac bits, then scale by int_pow.
293 u32 frac_pow16 = static_cast<u32>(frac_pow24 >> (IFRAC - FRAC_BITS));
294 u64 result = (static_cast<u64>(int_pow) * frac_pow16) >> FRAC_BITS;
295 return from_raw(static_cast<u32>(result));
296 }
