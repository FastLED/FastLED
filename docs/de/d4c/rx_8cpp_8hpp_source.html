71// Implementation of make4PhaseTiming function
72ChipsetTiming4Phase make4PhaseTiming(const ChipsetTiming& timing_3phase,
73 u32 tolerance_ns) FL_NO_EXCEPT {
74 // Calculate derived values from 3-phase timing
75 // The encoder uses:
76 // Bit 0: T1 high + (T2+T3) low
77 // Bit 1: (T1+T2) high + T3 low
78 u32 t0h = timing_3phase.T1; // Bit 0 high time
79 u32 t0l = timing_3phase.T2 + timing_3phase.T3; // Bit 0 low time
80 u32 t1h = timing_3phase.T1 + timing_3phase.T2; // Bit 1 high time
81 u32 t1l = timing_3phase.T3; // Bit 1 low time
82
83 ChipsetTiming4Phase result;
84
85 // Bit 0 timing thresholds
86 result.t0h_min_ns = (t0h > tolerance_ns) ? (t0h - tolerance_ns) : 0;
87 result.t0h_max_ns = t0h + tolerance_ns;
88 result.t0l_min_ns = (t0l > tolerance_ns) ? (t0l - tolerance_ns) : 0;
89 result.t0l_max_ns = t0l + tolerance_ns;
90
91 // Bit 1 timing thresholds
92 result.t1h_min_ns = (t1h > tolerance_ns) ? (t1h - tolerance_ns) : 0;
93 result.t1h_max_ns = t1h + tolerance_ns;
94 result.t1l_min_ns = (t1l > tolerance_ns) ? (t1l - tolerance_ns) : 0;
95 result.t1l_max_ns = t1l + tolerance_ns;
96
97 // Reset pulse threshold
98 result.reset_min_us = timing_3phase.RESET;
99
100 // gap_tolerance_ns retains its default value of 0
101
102 return result;
103}

114// RMT device specialization for ESP32.
115//
116// The `RmtRxChannel::create` impl only exists on chips that actually have RMT
117// silicon: the IDF 4.x impl in `rmt_rx_4/` gates on
118// `FASTLED_ESP32_HAS_RMT && !FASTLED_ESP32_RMT5_ONLY_PLATFORM && !FASTLED_RMT5`
119// and the IDF 5.x impl in `rmt_rx_5/` gates on `FASTLED_RMT5`. On ESP32-C2
120// the SoC has no RMT peripheral (`SOC_RMT_SUPPORTED == 0`, so
121// `FASTLED_ESP32_HAS_RMT == 0`), and the workflow forces
122// `FASTLED_RMT5 = 0` for that chip — so both impls skip and the specialisation
123// link-fails with `undefined reference to fl::RmtRxChannel::create(int)`.
124// Mirror the exact gates the impls use here so the specialisation collapses
125// to a DummyRxDevice on RMT-less chips.
126template <>
127fl::shared_ptr<RxDevice> RxDevice::create<RxDeviceType::RMT>(int pin) FL_NO_EXCEPT {
128#if (FASTLED_ESP32_HAS_RMT && !FASTLED_ESP32_RMT5_ONLY_PLATFORM && !FASTLED_RMT5) || FASTLED_RMT5
129 auto device = RmtRxChannel::create(pin);
130 if (!device) {
131 return fl::make_shared<DummyRxDevice>("RMT RX channel creation failed");
132 }
133 return device;
134#else
135 (void)pin;
136 return fl::make_shared<DummyRxDevice>("RMT RX not supported on this ESP32 SoC");
137#endif
138}
