65// Integer conversion (excluding bool) - dispatches through the shared
66// non-template `json_convert_to_i64_core` (FastLED #3247). Per-T narrowing
67// + overflow detection happen here; the heavy variant-walking lives once
68// in the core.
69template <typename T>
70struct JsonToType<T, typename fl::enable_if<fl::is_integral<T>::value && !fl::is_same<T, bool>::value>::type> {
71 static fl::tuple<T, TypeConversionResult> convert(const json& j) {
72 const json_value* val = j.internal_value();
73 if (!val) {
74 TypeConversionResult result;
75 result.setError("failed to access JSON value");
76 return fl::make_tuple(T(), result);
77 }
78 fl::i64 raw = 0;
79 TypeConversionResult result = json_convert_to_i64_core(*val, &raw);
80 T narrowed = static_cast<T>(raw);
81 if (static_cast<fl::i64>(narrowed) != raw) {
82 result.addWarning(fl::string("integer overflow/truncation: ") +
83 fl::to_string(raw) + " converted to " +
84 fl::to_string(static_cast<fl::i64>(narrowed)));
85 }
86 return fl::make_tuple(narrowed, result);
87 }
88};

164// ConstSpanWrapper conversion - converts JSON array to vector wrapper
165template <typename T>
166struct JsonToType<fl::ConstSpanWrapper<T>, void> {
167 static fl::tuple<fl::ConstSpanWrapper<T>, TypeConversionResult> convert(const json& j) {
168 TypeConversionResult result;
169
170 if (!j.is_array()) {
171 result.setError("expected array for span parameter");
172 return fl::make_tuple(fl::ConstSpanWrapper<T>(), result);
173 }
174
175 fl::vector<T> vec;
176 for (fl::size i = 0; i < j.size(); i++) {
177 auto elemResult = JsonToType<T>::convert(j[i]);
178 T elem = fl::get<0>(elemResult);
179 TypeConversionResult elemConvResult = fl::get<1>(elemResult);
180
181 if (elemConvResult.hasError()) {
182 result.setError("element " + fl::to_string(static_cast<fl::i64>(i)) + ": " + elemConvResult.errorMessage());
183 return fl::make_tuple(fl::ConstSpanWrapper<T>(), result);
184 }
185
186 vec.push_back(elem);
187 }
188
189 return fl::make_tuple(fl::ConstSpanWrapper<T>(fl::move(vec)), result);
190 }
191};

223// fl::vector<fl::u8> specialization - accepts base64 string OR integer array.
224// Base64 strings provide compact binary transport for JSON-RPC.
225template <>
226struct JsonToType<fl::vector<fl::u8>, void> {
227 static fl::tuple<fl::vector<fl::u8>, TypeConversionResult> convert(const json& j) {
228 TypeConversionResult result;
229
230 if (j.is_string()) {
231 // Decode base64 string to binary
232 fl::string encoded;
233 auto strResult = JsonToType<fl::string>::convert(j);
234 encoded = fl::get<0>(strResult);
235 TypeConversionResult strConvResult = fl::get<1>(strResult);
236 if (strConvResult.hasError()) {
237 return fl::make_tuple(fl::vector<fl::u8>(), strConvResult);
238 }
239 fl::vector<fl::u8> decoded = fl::base64_decode(encoded);
240 if (decoded.empty() && !encoded.empty()) {
241 result.setError("invalid base64 string");
242 return fl::make_tuple(fl::vector<fl::u8>(), result);
243 }
244 return fl::make_tuple(fl::move(decoded), result);
245 }
246
247 if (j.is_array()) {
248 // Fall back to integer array
249 fl::vector<fl::u8> vec;
250 for (fl::size i = 0; i < j.size(); i++) {
251 auto elemResult = JsonToType<fl::u8>::convert(j[i]);
252 fl::u8 elem = fl::get<0>(elemResult);
253 TypeConversionResult elemConvResult = fl::get<1>(elemResult);
254 if (elemConvResult.hasError()) {
255 result.setError("element " + fl::to_string(static_cast<fl::i64>(i)) + ": " + elemConvResult.errorMessage());
256 return fl::make_tuple(fl::vector<fl::u8>(), result);
257 }
258 vec.push_back(elem);
259 }
260 return fl::make_tuple(fl::move(vec), result);
261 }
262
263 result.setError("expected base64 string or integer array for byte vector");
264 return fl::make_tuple(fl::vector<fl::u8>(), result);
265 }
