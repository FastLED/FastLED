263 // Register "fftCqOctaveBench" - CQ_OCTAVE per-frame cost on device
264 // (#4540). Times the first frame (cold scratch) and the average warm
265 // frame, and reports free heap before construction, after the first
266 // frame, and after all warm frames, so per-frame heap growth shows up as
267 // a difference between the last two. The input is a fixed two-tone
268 // signal so results are comparable across builds.
269 remote.bind("fftCqOctaveBench", [](const fl::json& args) -> fl::json {
270 fl::i32 samples = 512;
271 fl::i32 bands = 16;
272 fl::i32 frames = 50;
273 const fl::json config = args.is_array() && args.size() > 0 ? args[0] : args;
274 if (!config.is_null() && config.is_object()) {
275 if (config.contains("samples") && config["samples"].is_int())
276 samples = static_cast<fl::i32>(config["samples"].as_int().value());
277 if (config.contains("bands") && config["bands"].is_int())
278 bands = static_cast<fl::i32>(config["bands"].as_int().value());
279 if (config.contains("frames") && config["frames"].is_int())
280 frames = static_cast<fl::i32>(config["frames"].as_int().value());
281 }
282 fl::json response = fl::json::object();
283 response.set("samples", static_cast<int64_t>(samples));
284 response.set("bands", static_cast<int64_t>(bands));
285 response.set("frames", static_cast<int64_t>(frames));
286 if (samples < 64 || samples > 4096 || bands < 2 || bands > 512 ||
287 frames < 1 || frames > 10000) {
288 response.set("success", false);
289 response.set("error", "out_of_range");
290 return response;
291 }
292 const fl::i32 kSampleRate = 44100;
293 fl::vector<fl::i16> signal(static_cast<fl::size>(samples));
294 for (fl::i32 i = 0; i < samples; ++i) {
295 const float t = static_cast<float>(i) / static_cast<float>(kSampleRate);
296 const float v = 0.5f * fl::sinf(2.0f * static_cast<float>(FL_M_PI) * 220.0f * t) +
297 0.3f * fl::sinf(2.0f * static_cast<float>(FL_M_PI) * 1500.0f * t);
298 signal[static_cast<fl::size>(i)] = static_cast<fl::i16>(v * 20000.0f);
299 }
300 const fl::u32 heapBefore = static_cast<fl::u32>(fl::getFreeHeap().free_sram);
301 fl::audio::fft::Args fftArgs(samples, bands, 174.6f, 4698.3f, kSampleRate,
302 fl::audio::fft::Mode::CQ_OCTAVE);
303 fl::audio::fft::Impl fft(fftArgs);
304 fl::audio::fft::Bins out(static_cast<fl::size>(bands));
305 const fl::u32 t0 = fl::micros();
306 fft.run(signal, &out);
307 const fl::u32 firstUs = fl::micros() - t0;
308 const fl::u32 heapAfterFirst = static_cast<fl::u32>(fl::getFreeHeap().free_sram);
309 const fl::u32 t1 = fl::micros();
310 for (fl::i32 f = 0; f < frames; ++f) {
311 fft.run(signal, &out);
312 }
313 const fl::u32 warmTotalUs = fl::micros() - t1;
314 const fl::u32 heapAfterWarm = static_cast<fl::u32>(fl::getFreeHeap().free_sram);
315 float checksum = 0.0f;
316 for (float b : out.raw()) checksum += b;
317 response.set("success", true);
318 response.set("first_frame_us", static_cast<int64_t>(firstUs));
319 response.set("warm_frame_us", static_cast<double>(warmTotalUs) / frames);
320 response.set("heap_before", static_cast<int64_t>(heapBefore));
321 response.set("heap_after_first", static_cast<int64_t>(heapAfterFirst));
322 response.set("heap_after_warm", static_cast<int64_t>(heapAfterWarm));
323 response.set("checksum", static_cast<double>(checksum));
324 return response;
325 });
