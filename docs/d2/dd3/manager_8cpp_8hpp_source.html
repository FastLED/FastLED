76u32 ChannelManager::pollNeededWaitSliceMs(u32 startTime, u32 timeoutMs) const FL_NO_EXCEPT {
77 constexpr u32 kPollNeededFallbackSliceMs = 1;
78 if (timeoutMs == 0) {
79 return kPollNeededFallbackSliceMs;
80 }
81 const u32 elapsed = millis() - startTime;
82 if (elapsed >= timeoutMs) {
83 return 0;
84 }
85 const u32 remaining = timeoutMs - elapsed;
86 return remaining < kPollNeededFallbackSliceMs ? remaining : kPollNeededFallbackSliceMs;
87}

89FL_NO_INLINE FL_COLD bool ChannelManager::addDriverSlow(
90 int priority,
91 const fl::shared_ptr<IChannelDriver>& driver,
92 const fl::string* engineName,
93 AddDriverSlowReason reason) FL_NO_EXCEPT {
94 if (reason == AddDriverSlowReason::NULL_DRIVER) {
95 FL_WARN("ChannelManager::addDriver() - Null driver provided");
96 return false;
97 }
98 if (reason == AddDriverSlowReason::EMPTY_NAME) {
99 FL_WARN("ChannelManager::addDriver() - Engine has empty name (driver->getName() returned empty string)");
100 return false;
101 }
102
103 for (const auto& entry : mDrivers) {
104 if (entry.name == *engineName) {
105 // True-duplicate fast path: same shared_ptr at same priority is
106 // a no-op (legacy clockless controllers may pre-bind the same
107 // driver singleton from many template instantiations). Skip the
108 // replace flow entirely so we don't waitForReady() or emit a
109 // spurious "Replacing" warning.
110 if (entry.driver == driver && entry.priority == priority) {
111 FL_DBG("ChannelManager::addDriver() - '" << engineName->c_str() << "' already registered at priority " << priority << " (idempotent no-op)");
112 return false;
113 }
114 FL_WARN("ChannelManager::addDriver() - Replacing existing driver '" << engineName->c_str() << "'");
115
116 FL_DBG("ChannelManager: Waiting for all drivers to become READY before replacement");
117 waitForReady();
118
119 // Re-scan after waiting: task pumping in waitForReady() can run
120 // callbacks, so do not retain an index or iterator across it.
121 for (size_t i = 0; i < mDrivers.size(); ++i) {
122 if (mDrivers[i].name != *engineName) {
123 continue;
124 }
125 FL_DBG("ChannelManager: Removing old driver '" << engineName->c_str() << "' (shared_ptr may delete)");
126 if (mDrivers[i].driver) {
127 mDrivers[i].driver->setPollNeededCallback(IChannelDriver::PollNeededCallback());
128 }
129 mDrivers.erase(mDrivers.begin() + i);
130 break;
131 }
132 return true;
133 }
134 }
135 return true;
136}

138void ChannelManager::addDriver(int priority, fl::shared_ptr<IChannelDriver> driver) {
139 if (!driver) {
140 (void)addDriverSlow(priority, driver, nullptr, AddDriverSlowReason::NULL_DRIVER);
141 return;
142 }
143
144 // Get driver name from the driver itself
145 fl::string engineName = driver->getName();
146
147 // Reject drivers with empty names
148 if (engineName.empty()) {
149 (void)addDriverSlow(priority, driver, &engineName, AddDriverSlowReason::EMPTY_NAME);
150 return;
151 }
152
153 // A duplicate name takes the uncommon replacement/idempotency path.
154 for (const auto& entry : mDrivers) {
155 if (entry.name == engineName) {
156 if (!addDriverSlow(priority, driver, &engineName, AddDriverSlowReason::DUPLICATE_NAME)) {
157 return;
158 }
159 break;
160 }
161 }
162
163 // Respect exclusive driver mode: auto-disable if name doesn't match exclusive driver
164 bool enabled = true; // Default: enabled
165 if (!mExclusiveDriver.empty()) {
166 enabled = (engineName == mExclusiveDriver); // Only enable if matches exclusive driver
167 }
168
169 mDrivers.push_back({priority, driver, engineName, enabled});
170 driver->setPollNeededCallback(mPollNeededCallback);
171
172 // Build capability string for debug output. Gate the entire block behind
173 // FASTLED_HAS_DBG because the `capStr` exists ONLY to feed the FL_DBG
174 // line below. On release builds (FASTLED_HAS_DBG=0 — i.e. the default
175 // SKETCH_HAS_LARGE_MEMORY=0 path AND any -DFASTLED_LOG_VERBOSITY=0
176 // opt-in build via the gating in fl/log/log.h) the FL_DBG itself is a
177 // no-op, but without this guard the `fl::string capStr` allocation +
178 // two `if` branches still emitted code. See #2773 item 2.3 follow-up.
179#if FASTLED_HAS_DBG
180 IChannelDriver::Capabilities caps = driver->getCapabilities();
181 fl::string capStr;
182 if (caps.supportsClockless) {
183 capStr += "CLOCKLESS";
184 }
185 if (caps.supportsSpi) {
186 if (!capStr.empty()) capStr += "|";
187 capStr += "SPI";
188 }
189 if (capStr.empty()) {
190 capStr = "NONE";
191 }
192
193 FL_DBG("ChannelManager: Added driver '" << engineName.c_str() << "' (priority " << priority << ", caps: " << capStr.c_str() << ")");
194#endif
195
196 // Sort drivers by priority descending (higher values first) after each insertion
197 // Higher priority values = higher precedence (e.g., priority 50 selected over priority 10)
198 // Only 1-4 drivers expected — sort_small skips the quicksort_impl
199 // instantiation entirely (see #2907 for the bloat motivation).
200 fl::sort_small(mDrivers.begin(), mDrivers.end());
201}

276bool ChannelManager::setExclusiveDriverByName(const char* name) {
277 // Handle null or empty name: disable everything.
278 if (!name || !name[0]) {
279 FL_ERROR("ChannelManager::setExclusiveDriverByName() - Null or empty driver name provided");
280 mExclusiveDriver.clear();
281 for (auto& entry : mDrivers) {
282 entry.enabled = false;

284 return false;
285 }
286
287 // Store exclusive driver name for forward compatibility.
288 // When non-empty, addDriver() will auto-disable non-matching drivers.
289 mExclusiveDriver = name;
290
291 // Single-pass: enable only drivers matching the given name.
292 bool found = false;
293 for (auto& entry : mDrivers) {
294 entry.enabled = (entry.name == name);
295 found = found || entry.enabled;
296 }
297
298 if (!found) {
299 FL_ERROR("ChannelManager::setExclusiveDriverByName() - Driver '" << name << "' not found in registry");
300 }

408fl::shared_ptr<IChannelDriver> ChannelManager::selectDriverForChannel(const ChannelDataPtr& data, const fl::string& affinity) {
409 if (!data) {
410 FL_ERROR("ChannelManager::selectDriverForChannel() - Null channel data");
411 return fl::shared_ptr<IChannelDriver>();
412 }
413
414 // If affinity is specified, look up by name. Misses fall through to
415 // priority dispatch below — per-frame logging is intentionally silent
416 // here because `Channel::showPixels` now emits a one-shot, actionable
417 // FL_ERROR with the enableDrivers<...>() / enableAllDrivers() hint
418 // (#2455). Use `findDriverByName` (silent) rather than `getDriverByName`
419 // (logs on miss) so the silent fall-through actually IS silent.
420 do {
421 if (affinity.empty()) {
422 break;
423 }
424 auto driver = findDriverByName(affinity);
425 if (!driver) {
426 break; // diagnostic emitted at the channel layer
427 }
428 if (!driver->canHandle(data)) {
429 FL_WARN_ONCE("ChannelManager: Affinity driver '" << affinity << "' cannot handle channel data (chipset/bus mismatch). Falling back to AUTO/priority dispatch.");
430 break;
431 }
432 return driver;
433 } while (false);
434
435
436 // No affinity: iterate drivers by priority (already sorted descending)
437 for (const auto& entry : mDrivers) {
438 if (!entry.enabled) continue;
439 if (entry.driver->canHandle(data)) {
440 return entry.driver; // Return shared_ptr
441 }
442 }
443
444 FL_ERROR("ChannelManager: No compatible driver found for channel data");
445 return fl::shared_ptr<IChannelDriver>();
446}

449template<typename Condition>
450bool ChannelManager::waitForCondition(Condition condition, u32 timeoutMs) {
451 const u32 startTime = timeoutMs > 0 ? millis() : 0;
452
453 // Tier 1: instant non-blocking check (avoid micros() / millis() cost on
454 // the common already-ready path).
455 if (condition()) {
456 return true;
457 }
458
459 // Tier 2: bounded microsecond spin (#2818). Catches short DMA tails
460 // (APA102 small strips, WS2812B <=8 LEDs) without paying the >=1-tick
461 // floor of the cooperator yield below. Budget is runtime-tunable via
462 // FastLED.setWaitSpinBudgetUs(N); set to 0 to disable.
463 {
464 const u32 spinBudget = fl::detail::getWaitSpinBudgetUs();
465 if (spinBudget > 0) {
466 const u32 spinStart = fl::micros();
467 while ((fl::micros() - spinStart) < spinBudget) {
468 if (condition()) {
469 return true;
470 }
471 if (timeoutMs > 0 && (millis() - startTime) >= timeoutMs) {
472 FL_ERROR("ChannelManager: Timeout occurred while waiting for condition");
473 return false;
474 }
475 }
476 }
477 }
478
479 while (!condition()) {
480 // Check timeout if specified
481 if (timeoutMs > 0 && (millis() - startTime >= timeoutMs)) {
482 FL_ERROR("ChannelManager: Timeout occurred while waiting for condition");
483 return false; // Timeout occurred
484 }
485
486 const u32 sliceMs = pollNeededWaitSliceMs(startTime, timeoutMs);
487 if (sliceMs == 0) {
488 return false;
489 }
490 if (waitForPollNeededSignal(sliceMs)) {
491 continue;
492 }
493
494 // Adaptive yield (refs #2815, generalizes the #2493 ESP32-P4 carve-out):
495 //
496 // The 1-tick (>=1 ms at CONFIG_FREERTOS_HZ=1000) floor only exists to
497 // keep WiFi / lwIP / BT controller tasks alive while we are inside the
498 // channel wait loop (#2254). When no radio is actually up, that floor
499 // is pure timing drift -- visible as the regression reported in #2420
500 // and as the per-frame cost the #2493 ESP32-P4 carve-out was avoiding.
501 //
502 // NetworkDetector::isAnyNetworkActive() is the runtime version of the
503 // "is a radio up?" question. On non-ESP32 platforms and on ESP32-P4
504 // (no radio silicon) it folds to a constant `false`, so this is
505 // strictly a perf win for the common single-strip / no-WiFi case
506 // without losing the WiFi-friendly behavior when a radio is active.
507 if (fl::NetworkDetector::isAnyNetworkActive()) {
508 // Radio active: keep WiFi/lwIP/BT alive with the deep yield.
509 task::run(250, task::ExecFlags::SYSTEM);
510 } else {
511 // No radio: fast yield, no FreeRTOS tick floor.
512 task::run(0, task::ExecFlags::SYSTEM);
513 }
514 }
515
516 return true; // Condition met
517}

519IChannelDriver::DriverState ChannelManager::poll() {
520 // Poll all registered drivers and return aggregate state
521 // Priority order: ERROR > BUSY > DRAINING > READY
522 bool anyBusy = false;
523 bool anyDraining = false;
524 fl::string firstError;
525
526 for (auto& entry : mDrivers) {
527 IChannelDriver::DriverState result = entry.driver->poll();
528 if (result.state == IChannelDriver::DriverState::BUSY) {
529 anyBusy = true;
530 } else if (result.state == IChannelDriver::DriverState::DRAINING) {
531 anyDraining = true;
532 }
533 // Capture first error encountered
534 if (result.state == IChannelDriver::DriverState::ERROR && firstError.empty()) {
535 firstError = result.error;
536 }
537 }
538
539 // Return error if any driver reported error
540 if (!firstError.empty()) {
541 return IChannelDriver::DriverState(IChannelDriver::DriverState::ERROR, firstError);
542 }
543
544 if (anyBusy) {
545 return IChannelDriver::DriverState(IChannelDriver::DriverState::BUSY);
546 }
547 if (anyDraining) {
548 return IChannelDriver::DriverState(IChannelDriver::DriverState::DRAINING);
549 }
550 return IChannelDriver::DriverState(IChannelDriver::DriverState::READY);
551}

564 bool ok = waitForCondition([this]() {
