48class WaveCrgbMapDefault : public WaveCrgbMap {

54 void mapWaveToLEDs(const XYMap &xymap, WaveSimulation2D &waveSim,
55 fl::span<CRGB> leds) override {
56 const fl::u32 width = waveSim.getWidth();
57 const fl::u32 height = waveSim.getHeight();
58 for (fl::u32 y = 0; y < height; y++) {
59 for (fl::u32 x = 0; x < width; x++) {
60 fl::u32 idx = xymap(x, y);
61 u8 value8 = waveSim.getu8(x, y);

370 void draw(DrawContext context) override {
371 // Update the wave simulation.
372 if (mAutoUpdates) {
373 mWaveSim.update();
374 }
375 // Map the wave values to the LEDs.
376 mCrgbMap->mapWaveToLEDs(mXyMap, mWaveSim, context.leds);
377 // #3122 A4 — post-process output blur as a cheaper alternative
378 // to SuperSample. Skipped entirely when amount == 0.
379 if (mOutputBlurAmount > 0) {
380 const fl::u32 w = mWaveSim.getWidth();
381 const fl::u32 h = mWaveSim.getHeight();
382 for (fl::u8 pass = 0; pass < mOutputBlurPasses; ++pass) {
383 fl::blur2d(context.leds, static_cast<fl::u8>(w),
384 static_cast<fl::u8>(h), mOutputBlurAmount,
385 mXyMap);
386 }
387 }
388 }
