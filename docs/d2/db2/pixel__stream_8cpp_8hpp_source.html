13// FLED v1 header layout (12 bytes). See
14// https://github.com/zackees/ledmapper/blob/main/docs/fled-format.md
15namespace {
16constexpr fl::u8 kFledMagic[4] = {'F', 'L', 'E', 'D'};
17constexpr fl::u8 kFledVersionV1 = 1;
18constexpr fl::size_t kFledHeaderBytes = 12;
19// Defensive cap on the embedded JSON to bound the worst-case heap
20// allocation if a malformed file claims a giant json_length. Real
21// screenmaps run ~500 B – ~50 KB; 1 MiB is well above that.
22constexpr fl::size_t kFledMaxJsonBytes = 1u * 1024u * 1024u;

24bool isFledMagicPrefix(const fl::u8* bytes, fl::size_t size) FL_NO_EXCEPT {
25 for (fl::size_t i = 0; i < size; ++i) {
26 if (bytes[i] != kFledMagic[i]) {
27 return false;
28 }
29 }
30 return true;

40bool PixelStream::begin(filebuf_ptr h) FL_NO_EXCEPT {
41 close();
42 if (!h || mBaseBytesPerFrame <= 0) {
43 return false;
44 }
45 mHandle = h;
46 mbytesPerFrame = mBaseBytesPerFrame;
47 mPayloadOffset = 0;
48 mEmbeddedScreenMapJson.clear();
49 mFledPixelFormat = 0;
50 mFledBytesPerLed = 0;
51 mHasVideoColor = false;
52 mHasFledContainer = false;
53 mStreamingPrefixSize = 0;
54 mStreamingPrefixPos = 0;
55 mStreamingProbePending = false;
56 mStreamingRejected = false;
57 // Probe seekability: if seek-to-start succeeds, this is a seekable file.
58 mType = mHandle->seek(0, seek_dir::beg) ? kFile : kStreaming;
59 if (mType == kFile) {
60 // A file without FLED magic remains a legacy headerless RGB stream.
61 // Once magic is recognized, though, every invalidity is a container
62 // error: falling back would render its header bytes as LEDs.
63 if (mHandle->size() >= 4) {
64 char hdr[kFledHeaderBytes];
65 fl::size_t got = mHandle->read(hdr, kFledHeaderBytes);
66 const bool hasFledMagic = got >= 4
67 && static_cast<fl::u8>(hdr[0]) == kFledMagic[0]
68 && static_cast<fl::u8>(hdr[1]) == kFledMagic[1]
69 && static_cast<fl::u8>(hdr[2]) == kFledMagic[2]
70 && static_cast<fl::u8>(hdr[3]) == kFledMagic[3];
71 if (hasFledMagic) {
72 const bool supportedFled = got == kFledHeaderBytes
73 && static_cast<fl::u8>(hdr[4]) == kFledVersionV1
74 && static_cast<fl::u8>(hdr[6]) == 0
75 && static_cast<fl::u8>(hdr[7]) == 0
76 && fled::bytesPerLed(static_cast<fl::u8>(hdr[5])) != 0;
77 if (!supportedFled) {
78 close();
79 return false;
80 }
81 const fl::u8 pixelFormat = static_cast<fl::u8>(hdr[5]);
82 const fl::u32 jsonLen =
83 static_cast<fl::u32>(static_cast<fl::u8>(hdr[8]))
84 | (static_cast<fl::u32>(static_cast<fl::u8>(hdr[9])) << 8)
85 | (static_cast<fl::u32>(static_cast<fl::u8>(hdr[10])) << 16)
86 | (static_cast<fl::u32>(static_cast<fl::u8>(hdr[11])) << 24);
87 // Cap the JSON length against a defensive maximum BEFORE
88 // doing any size arithmetic — guards against a malformed
89 // file declaring a multi-gigabyte json_length that would
90 // either overflow the offset calc or trigger a huge resize.
91 const fl::size_t jsonLenSz = static_cast<fl::size_t>(jsonLen);
92 const fl::size_t fileSize = mHandle->size();
93 const bool jsonInRange = jsonLenSz <= kFledMaxJsonBytes
94 && jsonLenSz <= fileSize - kFledHeaderBytes;
95 if (jsonInRange) {
96 mEmbeddedScreenMapJson.resize(static_cast<fl::size>(jsonLenSz));
97 fl::size_t jr = jsonLenSz > 0
98 ? mHandle->read(&mEmbeddedScreenMapJson[0], jsonLenSz)
99 : 0;
100 if (jr == jsonLenSz) {
101 mPayloadOffset = kFledHeaderBytes + jsonLenSz;
102 mFledPixelFormat = pixelFormat;
103 mFledBytesPerLed = fled::bytesPerLed(pixelFormat);
104 mHasFledContainer = true;
105 if (mbytesPerFrame % 3 != 0) {
106 close();
107 return false;
108 }
109 const fl::i32 ledCount = mbytesPerFrame / 3;
110 const fl::i32 bytesPerLed =
111 static_cast<fl::i32>(mFledBytesPerLed);
112 if (ledCount > (fl::numeric_limits<fl::i32>::max)() /
113 bytesPerLed) {
114 close();
115 return false;
116 }
117 mbytesPerFrame = ledCount * bytesPerLed;
118 if ((fileSize - mPayloadOffset) %
119 static_cast<fl::size_t>(mbytesPerFrame) != 0) {
120 close();
121 return false;
122 }
123 fl::json envelope;
124 if (!mEmbeddedScreenMapJson.empty()) {
125 envelope = fl::json::parse(mEmbeddedScreenMapJson);
126 }
127 const bool validEnvelope = mEmbeddedScreenMapJson.empty() ||
128 (envelope.has_value() && envelope.is_object());
129 if (!validEnvelope) {
130 close();
131 return false;
132 }
133 const bool colorOk =
134 fled::resolveVideoColor(envelope, mFledPixelFormat,
135 &mVideoColor) ==
136 fled::ColorStatus::Ok;
137 mHasVideoColor = colorOk;
138 if (!colorOk) {
139 fled::PixelStorage storage;
140 const bool advisoryRgb8 = pixelStorage(&storage) &&
141 storage.mFormat == fl::PixelFormat::Rgb8;
142 if (!mBestEffortFled || !advisoryRgb8) {
143 close();
144 return false;
145 }
146 FL_WARN("FLED video.color rejected; using explicit best-effort RGB8 playback");
147 }
148 // Stream is now positioned at the first frame byte.
149 return mHandle->available();
150 }
151 mEmbeddedScreenMapJson.clear();
152 }
153 // FLED magic was recognized, so malformed/truncated
154 // containers must not become headerless raw RGB.
155 close();
156 return false;
157 }
158 // No magic: rewind so subsequent reads see raw RGB triplets.
159 mHandle->seek(0, seek_dir::beg);
160 }
161 return mHandle->available();
162 }
163 // A partial FLED prefix stays buffered until enough bytes arrive to
164 // classify it. A raw mismatch is replayed through readBytes().
165 mStreamingProbePending = true;
166 if (!probeStreamingMagic()) {
167 close();
168 return false;
169 }
170 return true;
171}

188bool PixelStream::probeStreamingMagic() const FL_NO_EXCEPT {
189 if (!mStreamingProbePending) {
190 return !mStreamingRejected;
191 }
192 while (mStreamingPrefixSize < 4 && mHandle && mHandle->available()) {
193 const fl::size_t want = 4 - mStreamingPrefixSize;
194 const fl::size_t got = mHandle->read(mStreamingPrefix + mStreamingPrefixSize,
195 want);
196 if (got == 0) {
197 break;
198 }
199 mStreamingPrefixSize += got;
200 if (!isFledMagicPrefix(mStreamingPrefix, mStreamingPrefixSize)) {
201 mStreamingProbePending = false;
202 return true;
203 }
204 }
205 if (mStreamingPrefixSize == 4) {
206 mStreamingProbePending = false;
207 mStreamingRejected = true;
208 return false;
209 }
210 return true;
211}

225bool PixelStream::readSample(PixelSample *out) FL_NO_EXCEPT {
226 if (!mHandle || !out) {
227 return false;
228 }
229 fled::PixelStorage storage;
230 fled::VideoColor color;
231 if (!pixelStorage(&storage) || !videoColor(&color) ||
232 storage.mFormat != fl::PixelFormat::Rgb16 ||
233 storage.mComponentByteOrder != fled::ComponentByteOrder::LittleEndian) {
234 return false;
235 }
236 fl::u8 bytes[6];
237 if (readBytes(bytes, sizeof(bytes)) != sizeof(bytes)) {
238 return false;
239 }
240 out->mStorage = storage;
241 out->mColor = color;
242 for (fl::u8 i = 0; i < 3; ++i) {
243 const fl::size_t offset = static_cast<fl::size_t>(i) * 2;
244 out->mComponents[i] = static_cast<fl::u16>(bytes[offset]) |
245 (static_cast<fl::u16>(bytes[offset + 1]) << 8);
246 }
247 return true;
248}

274bool PixelStream::readFrame(Frame *frame) {
275 if (!mHandle || !frame) {
276 return false;
277 }
278 if (mType == kFile && !framesRemaining()) {
279 return false;
280 }
281 if (!isRgb8Playback()) {
282 return false;
283 }
284 size_t n = 0;
285 const fl::size_t pixels = static_cast<fl::size_t>(mbytesPerFrame) / 3;
286 fl::span<CRGB> dst = frame->rgb();
287 for (; n < pixels && n < dst.size() && readPixel(&dst[n]); ++n) {
288 }
289 if (mType == kFile) {
290 DBG("pos: " << mHandle->pos());
291 }
292 return n * 3 == size_t(mbytesPerFrame);
293}

295bool PixelStream::hasFrame(fl::u32 frameNumber) {
296 if (!mHandle) {
297 return false;
298 }
299 if (mType == kStreaming) {
300 // Streaming handle doesn't support seeking
301 DBG("Not implemented and therefore always returns true");
302 return true;
303 }
304 // Use size_t throughout so frameNumber * bytesPerFrame doesn't overflow
305 // u32 for high-LED-count grids past ~1M frames.
306 fl::size_t total_bytes = mHandle->size();
307 fl::size_t frameBytes = static_cast<fl::size_t>(frameNumber)
308 * static_cast<fl::size_t>(mbytesPerFrame);
309 fl::size_t target = mPayloadOffset + frameBytes;
310 return target < total_bytes;
311}

313bool PixelStream::readFrameAt(fl::u32 frameNumber, Frame *frame) {
314 if (!mHandle || !frame || mType == kStreaming) {
315 // Streaming handle doesn't support seeking
316 FL_DBG("Streaming handle doesn't support seeking");
317 return false;
318 }
319 if (!isRgb8Playback()) {
320 return false;
321 }
322 fl::size_t frameBytes = static_cast<fl::size_t>(frameNumber)
323 * static_cast<fl::size_t>(mbytesPerFrame);
324 mHandle->seek(mPayloadOffset + frameBytes);
325 if (mHandle->bytesLeft() == 0) {
326 return false;
327 }
328 size_t read =
329 mHandle->readRGB8(frame->rgb()) * 3;
330
331 bool ok = int(read) == mbytesPerFrame;
332 if (!ok) {
333 DBG("readFrameAt failed - read: "
334 << read << ", mbytesPerFrame: " << mbytesPerFrame << ", frame:"
335 << frameNumber << ", left: " << mHandle->bytesLeft());
336 }
337 return ok;
338}

423size_t PixelStream::readBytes(u8 *dst, size_t len) {
424 if (!mHandle || !dst) {
425 return 0;
426 }
427 if (mType == kStreaming &&
428 (!probeStreamingMagic() || mStreamingProbePending)) {
429 return 0;
430 }
431 size_t bytesRead = 0;
432 while (bytesRead < len && mStreamingPrefixPos < mStreamingPrefixSize) {
433 dst[bytesRead++] = mStreamingPrefix[mStreamingPrefixPos++];
434 }
435 if (mType == kStreaming) {
436 while (bytesRead < len && mHandle->available(len - bytesRead)) {
437 if (mHandle->read(dst + bytesRead, 1)) {
438 bytesRead++;
439 } else {
440 break;
441 }
442 }
443 } else {
444 while (bytesRead < len && mHandle->available()) {
445 if (mHandle->read(dst + bytesRead, 1)) {
446 bytesRead++;
447 } else {
448 break;
449 }
450 }
451 }
452 return bytesRead;
453}
