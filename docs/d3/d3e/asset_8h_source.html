52constexpr bool is_parent_at(const char* p, fl::size i) FL_NO_EXCEPT {
53 // Segment start: either we're at the beginning, or preceded by a separator.
54 return (i == 0 || p[i - 1] == '/' || p[i - 1] == '\\')
55 && p[i] == '.' && p[i + 1] == '.'
56 && (p[i + 2] == '\0' || p[i + 2] == '/' || p[i + 2] == '\\');
57}
