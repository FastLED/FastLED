137void loop() {
138 // Check for incoming JSON RPC via serial and queue it
139 if (Serial.available()) {
140 fl::string jsonRpc = readSerialJson();
141
142 // Parse JSON and queue the request
143 fl::json doc = fl::json::parse(jsonRpc);
144 requestQueue.push_back(doc);
145 }
146
147 // Process all queued requests: pull + tick + push
148 remote.update(millis());
149
150 // Drain response queue and output to serial
151 while (!responseQueue.empty()) {
152 const auto& response = responseQueue[0];
153 Serial.println(response.to_string().c_str());
154 responseQueue.erase(responseQueue.begin());
155 }
156
157 FastLED.show();
158 delay(10);
159}
