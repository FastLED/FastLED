119int minimp3Version(const fl::u8* header) FL_NO_EXCEPT {
120 const int version_bits = (header[1] >> 3) & 0x03;
121 if (version_bits == 0x03) {
122 return 0; // MPEG-1.
123 }
124 if (version_bits == 0x02) {
125 return 1; // MPEG-2.
126 }
127 return 2; // MPEG-2.5.
128}

177int Mp3Minimp3Decoder::findSyncWord(const fl::u8* buf,
178 fl::size len) FL_NO_EXCEPT {
179 if (!buf || len < 4) {
180 return -1;
181 }
182
183 for (fl::size i = 0; i <= len - 4; ++i) {
184 const fl::u8 version = (buf[i + 1] >> 3) & 0x03;
185 const fl::u8 layer = (buf[i + 1] >> 1) & 0x03;
186 const fl::u8 bitrate = (buf[i + 2] >> 4) & 0x0f;
187 const fl::u8 sample_rate = (buf[i + 2] >> 2) & 0x03;
188 if (buf[i] == 0xff && (buf[i + 1] & 0xe0) == 0xe0 &&
189 version != 0x01 && layer != 0x00 && bitrate != 0x0f &&
190 sample_rate != 0x03) {
191 return static_cast<int>(i);
192 }
193 }
194 return -1;
195}

197int Mp3Minimp3Decoder::decodeFrame(const fl::u8** inbuf,
198 fl::size* bytes_left) FL_NO_EXCEPT {
199 if (!mDecoder || !mScratch || !mPcmBuffer || !inbuf || !bytes_left ||
200 !*inbuf || *bytes_left == 0) {
201 return -1;
202 }
203
204 const fl::u8* input = *inbuf;
205 const fl::size max_input = 0x7fffffff;
206 const int input_bytes = static_cast<int>(
207 *bytes_left > max_input ? max_input : *bytes_left);
208 mp3dec_frame_info_t info = {};
209 const int samples = mp3dec_decode_frame_r(
210 mDecoder.get(), mScratch.get(), input, input_bytes, mPcmBuffer.get(),
211 &info);
212 if (samples <= 0 || info.frame_bytes <= 0 ||
213 static_cast<fl::size>(info.frame_bytes) > *bytes_left) {
214 return -1;
215 }
216
217 const fl::u8* header = input + info.frame_offset;
218 *inbuf += info.frame_bytes;
219 *bytes_left -= static_cast<fl::size>(info.frame_bytes);
220
221 int frame_bytes = info.frame_bytes - info.frame_offset;
222 if (frame_bytes > 0 && (header[2] & 0x02) != 0) {
223 --frame_bytes;
224 }
225 mFrameInfo.bitrate = info.bitrate_kbps * 1000;
226 if (mFrameInfo.bitrate == 0 && samples > 0 && frame_bytes > 0) {
227 mFrameInfo.bitrate =
228 (frame_bytes * info.hz * 8) / samples;
229 }
230 mFrameInfo.nChans = info.channels;
231 mFrameInfo.samprate = info.hz;
232 mFrameInfo.bitsPerSample = 16;
233 mFrameInfo.outputSamps = samples * info.channels;
234 mFrameInfo.layer = info.layer;
235 mFrameInfo.version = minimp3Version(header);
236 return 0;
237}

240Mp3Minimp3Decoder::decodeToAudioSamples(const fl::u8* data,
241 fl::size len) FL_NO_EXCEPT {
242 fl::vector<audio::Sample> samples;
243
244 decode(data, len, [&](const Mp3Frame& frame) {
245 if (frame.channels == 2) {
246 fl::vector<fl::i16> mono_pcm;
247 mono_pcm.reserve(frame.samples);
248 for (int i = 0; i < frame.samples; ++i) {
249 const fl::i32 left = frame.pcm[i * 2];

251 mono_pcm.push_back(static_cast<fl::i16>((left + right) / 2));

253 samples.push_back(audio::Sample(mono_pcm));
254 } else {
255 samples.push_back(

257 frame.samples)));
258 }

399bool Mp3StreamDecoderImpl::fillBuffer() {
400 // Shift remaining data to beginning of buffer
401 if (mBufferPos > 0 && mBufferFilled > mBufferPos) {
402 fl::size remaining = mBufferFilled - mBufferPos;
403 for (fl::size i = 0; i < remaining; i++) {
404 mBuffer[i] = mBuffer[mBufferPos + i];
405 }
406 mBufferFilled = remaining;
407 mBufferPos = 0;
408 } else if (mBufferPos >= mBufferFilled) {
409 mBufferPos = 0;
410 mBufferFilled = 0;
411 }
412
413 // Fill the rest of the buffer from stream
414 fl::size spaceAvailable = BUFFER_SIZE - mBufferFilled;
415 if (spaceAvailable > 0 && mStream && mStream->available(1)) {
416 fl::size bytesRead =
417 mStream->read(mBuffer.get() + mBufferFilled, spaceAvailable);
418 mBufferFilled += bytesRead;
419 return bytesRead > 0;
420 }
421
422 return false;
423}

425bool Mp3StreamDecoderImpl::findAndDecodeFrame(audio::Sample* out_sample) {
426 if (!mDecoder) {
427 return false;
428 }
429
430 // Try to decode from current buffer
431 const fl::u8* inptr = mBuffer.get() + mBufferPos;
432 fl::size bytes_left = mBufferFilled - mBufferPos;
433
434 if (bytes_left == 0) {
435 return false;
436 }
437
438 // Find sync word
439 int offset = mDecoder->findSyncWord(inptr, bytes_left);
440 if (offset < 0) {
441 // No sync word found, consume buffer and try again
442 mBufferPos = mBufferFilled;
443 return false;
444 }
445
446 inptr += offset;
447 bytes_left -= offset;
448 mBufferPos += offset;
449
450 // Try to decode one frame
451 const fl::u8* decode_ptr = inptr;
452 fl::size decode_bytes = bytes_left;
453
454 /* The first frame may be a Xing/Info/VBRI header rather than audio. It is
455 a complete, valid MPEG frame, so decoding it emits noise before the
456 music; LAME also records how much encoder priming follows it. Inspect
457 before decoding so the frame can be dropped whole (FastLED#4129). */
458 if (!mInspectedFirstFrame) {
459 mInspectedFirstFrame = true;
460 Mp3VbrTag tag;
461 if (Mp3ParseVbrTag(fl::span<const fl::u8>(inptr, bytes_left), &tag) &&
462 tag.present) {
463 mSkipTagFrame = true;
464 mSkipSamples = tag.encoderDelay + MP3D_DECODER_DELAY;
465 }
466 }
467
468 int result = mDecoder->decodeFrame(&decode_ptr, &decode_bytes);
469
470 // Update buffer position based on how many bytes were consumed
471 fl::size consumed = (decode_ptr - inptr);
472 if (result != 0 && consumed == 0 && bytes_left == BUFFER_SIZE) {
473 // A full buffer cannot grow to complete this frame. Drop one byte so
474 // corrupt input cannot pin the streaming decoder on the same sync.
475 consumed = 1;
476 }
477 mBufferPos += consumed;
478 mBytesProcessed += consumed;
479
480 if (result == 0 && mSkipTagFrame) {
481 // The metadata frame itself: consumed above, never emitted.
482 mSkipTagFrame = false;
483 return false;
484 }
485
486 if (result == 0) {
487 // Successfully decoded a frame
488 Mp3Frame frame;
489 frame.pcm = mDecoder->mPcmBuffer.get();
490 frame.samples = mDecoder->mFrameInfo.outputSamps / mDecoder->mFrameInfo.nChans;
491 frame.channels = mDecoder->mFrameInfo.nChans;
492 frame.sample_rate = mDecoder->mFrameInfo.samprate;
493 frame.bitrate = mDecoder->mFrameInfo.bitrate;
494 frame.version = mDecoder->mFrameInfo.version;
495 frame.layer = mDecoder->mFrameInfo.layer;
496
497 // Update stream info on first successful decode
498 if (!mHasDecodedFirstFrame) {
499 mInfo.sampleRate = frame.sample_rate;
500 mInfo.channels = static_cast<fl::u8>(frame.channels);
501 mInfo.bitrate = frame.bitrate;
502 mInfo.version = static_cast<fl::u8>(frame.version);
503 mInfo.layer = static_cast<fl::u8>(frame.layer);
504 mInfo.isValid = true;
505 mHasDecodedFirstFrame = true;
506 }
507
508 /* Encoder priming plus the filterbank's own latency: not signal, and
509 audible as a burst at the start of playback if emitted. */
510 int first = 0;
511 if (mSkipSamples > 0) {
512 const fl::u32 available = static_cast<fl::u32>(frame.samples);
513 const fl::u32 drop =
514 mSkipSamples < available ? mSkipSamples : available;
515 first = static_cast<int>(drop);
516 mSkipSamples -= drop;
517 if (first >= frame.samples) {
518 return false; // whole frame was priming
519 }
520 }
521 const int emitted = frame.samples - first;
522
523 // Convert to audio::Sample (convert stereo to mono if needed)
524 if (frame.channels == 2) {
525 fl::vector<fl::i16> mono_pcm;
526 mono_pcm.reserve(emitted);
527
528 for (int i = first; i < frame.samples; i++) {
529 fl::i32 left = frame.pcm[i * 2];
530 fl::i32 right = frame.pcm[i * 2 + 1];
531 fl::i32 avg = (left + right) / 2;
532 mono_pcm.push_back(static_cast<fl::i16>(avg));
533 }
534
535 *out_sample = audio::Sample(mono_pcm);
536 } else {
537 // Mono audio - use directly
538 *out_sample = audio::Sample(
539 fl::span<const fl::i16>(frame.pcm + first, emitted));
540 }
541
542 return true;
543 }
544
545 return false;
546}

548bool Mp3StreamDecoderImpl::decodeNextFrame(audio::Sample* out_sample) {
549 if (!isReady()) {
550 mErrorMsg = "Decoder not ready";
551 mHasError = true;
552 return false;
553 }
554
555 if (mEndOfStream) {
556 return false;
557 }
558
559 while (true) {
560 const fl::size previous_position = mBufferPos;
561 if (findAndDecodeFrame(out_sample)) {
562 return true;
563 }
564
565 if (mBufferPos > previous_position) {
566 // The decoder rejected or skipped bytes. Retry the remainder before
567 // refilling so a valid frame after corruption is not discarded.
568 continue;
569 }
570
571 if (fillBuffer()) {
572 continue;
573 }
574
575 if (mBufferPos < mBufferFilled) {
576 // No more bytes are available to complete the current candidate.
577 // Advance through the tail so a later sync can still be recovered.
578 ++mBufferPos;
579 ++mBytesProcessed;
580 continue;
581 }
582
583 mEndOfStream = true;
584 return false;
585 }

638Mp3Info Mp3::parseMp3Info(fl::span<const fl::u8> data, fl::string* error_message) {
639 Mp3Info info;
640
641 // Validate input data
642 if (data.empty()) {
643 if (error_message) {
644 *error_message = "Empty MP3 data";
645 }
646 return info; // returns invalid info
647 }
648
649 // Minimum size check - need at least an MP3 frame header (4 bytes) + some data
650 if (data.size() < 128) {
651 if (error_message) {
652 *error_message = "MP3 data too small";
653 }
654 return info;
655 }
656
657 // Look for MP3 sync word (11 bits set: 0xFFE or 0xFFF)
658 bool foundSync = false;
659 fl::size syncOffset = 0;
660
661 for (fl::size i = 0; i <= data.size() - 4; i++) {
662 if (data[i] == 0xFF && (data[i + 1] & 0xE0) == 0xE0) {
663 foundSync = true;
664 syncOffset = i;
665 break;
666 }
667 }
668
669 if (!foundSync) {
670 if (error_message) {
671 *error_message = "Invalid MP3 stream - no sync word found";
672 }
673 return info;
674 }
675
676 // Use the decoder to parse the first frame
677 third_party::Mp3SelectedDecoder decoder;
678 if (!decoder.init()) {
679 if (error_message) {
680 *error_message = "Failed to initialize MP3 decoder";
681 }
682 return info;
683 }
684
685 const fl::u8* inptr = data.data() + syncOffset;
686 fl::size bytes_left = data.size() - syncOffset;
687
688 // Try to decode first frame to get metadata
689 int offset = decoder.findSyncWord(inptr, bytes_left);
690 if (offset >= 0) {
691 inptr += offset;
692 bytes_left -= offset;
693
694 int result = decoder.decodeFrame(&inptr, &bytes_left);
695 if (result == 0) {
696 // Successfully decoded - extract metadata
697 info.sampleRate = decoder.mFrameInfo.samprate;
698 info.channels = static_cast<fl::u8>(decoder.mFrameInfo.nChans);
699 info.bitrate = decoder.mFrameInfo.bitrate;
700 info.version = static_cast<fl::u8>(decoder.mFrameInfo.version);
701 info.layer = static_cast<fl::u8>(decoder.mFrameInfo.layer);
702 info.isValid = true;
703 } else if (error_message) {
704 *error_message = "Failed to decode MP3 frame header";
705 }
706 } else if (error_message) {
707 *error_message = "Failed to find MP3 sync word";
708 }
709
710 return info;
711}
