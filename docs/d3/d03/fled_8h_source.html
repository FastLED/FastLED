3// fl::Fled - value-type handle to a parsed .fled v1 container.
4// PIMPL; cheap to copy. Default-constructed instances are in a null state
5// and evaluate to false. See FLED_FORMAT.md for the on-disk format.
6//
7// What Fled exposes:
8// - Header info: version(), pixelFormat().
9// - Raw JSON envelope: json(), sectionCount().
10// - Raw byte ranges: blob(name, &outLen) for the post-JSON frame payload.
11// - Fully constructed objects: screenMap(), channels().
12//
13// What Fled deliberately does NOT expose (purged from v1):
14// - No video() accessor. Video access is via blob("frame_payload", &n) plus
15// json()["video"] for metadata. No Video object is reconstructed here.
16// - No script() accessor. Scripting is a future-v2 surface (docs only).

35// A typed, zero-copy video frame. The payload owner keeps Fled's owned bytes
36// alive; static-input callers still retain the lifetime obligation documented
37// by loadFromStatic(). Rgb16 components are decoded from little-endian bytes,
38// never reinterpreted as native-endian words.
39struct VideoFrameView {
40 PixelStorage mStorage = {fl::PixelFormat::Rgb8,
41 ComponentByteOrder::NotApplicable};
42 VideoColor mColor = {ColorPrimaries::Bt709, ColorTransfer::Srgb,
43 ColorMatrix::Rgb, ColorRange::Full, false, {}};
44 fl::shared_ptr<const fl::u8> mPayload;
45 fl::size mPayloadBytes = 0;
46 fl::size mStride = 0;
47
48 fl::u16 component16(fl::size led, fl::u8 component) const FL_NO_EXCEPT {
49 if (!mPayload || mStorage.mFormat != fl::PixelFormat::Rgb16 ||
50 mStorage.mComponentByteOrder != ComponentByteOrder::LittleEndian ||
51 component >= 3 || led >= mPayloadBytes / 6) {
52 return 0;
53 }
54 const fl::size offset = led * 6 + static_cast<fl::size>(component) * 2;
55 if (offset >= mPayloadBytes - 1) {
56 return 0;
57 }
58 return static_cast<fl::u16>(mPayload.get()[offset]) |
59 (static_cast<fl::u16>(mPayload.get()[offset + 1]) << 8);
60 }
61};
