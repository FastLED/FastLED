51 private:
54 static fl::vector<CRGB>& getReorderBufferTLS() {
55 return SingletonThreadLocal<fl::vector<CRGB>>::instance();
56 }
57 fl::optional<PixelController<RGB, 1, 0xFFFFFFFF>> mAddressedController;
58 PixelIteratorAny mPixelIterator;
59#if FL_COLOR_PROFILE_RUNTIME
60 // Present only when a colour profile is bound. The managed source keeps
61 // a reference to the controller it wraps, so it has to outlive the
62 // iterator built over it -- both live here, for the frame.
63 //
64 // Raw storage rather than the types themselves, because naming them here
65 // is what made the whole colour pipeline reachable from `show()` and cost
66 // every sketch ~3.5 KB of flash. `colorPipelineHooks()` builds into this,
67 // and only `setColorProfile` installs those hooks -- so a program that
68 // never binds a profile never references the pipeline and the linker
69 // drops it.
70 FL_ALIGNAS(8) unsigned char mManagedSourceStorage[kColorPipelineSourceStorage];
71 FL_ALIGNAS(8) unsigned char mManagedIteratorStorage[kColorPipelineIteratorStorage];
72 PixelIterator* mManagedIterator = nullptr;
73#endif

75 public:
82 ReorderingPixelIteratorAny(
83 PixelController<RGB, 1, 0xFFFFFFFF>& pixels,
84 const XYMap* addressing,
85 EOrder rgbOrder,
86 Rgbw rgbw,
87 Rgbww rgbww,
88 const fl::string& channelName,
89#if FL_COLOR_PIPELINE_SHARED
90 const StreamingPipelineQ16* pipeline,
91#else
92 StreamingPipelineQ16* pipeline,
93#endif
94 u8 ditherPhase, u8 brightness) FL_NO_EXCEPT
95 : mPixelIterator(pixels, rgbOrder, rgbw, rgbww) {
96 FL_UNUSED(pipeline);
97 FL_UNUSED(ditherPhase); // read only by the colour-managed source
98 FL_UNUSED(brightness); // read only by the colour-managed source
99 FL_UNUSED(channelName); // only consumed by FL_ERROR, a no-op on small platforms
100
101 // Apply addressing transformation if configured
102 if (addressing) {
103 u16 numLeds = pixels.size();
104 u16 width = addressing->getWidth();
105 u16 height = addressing->getHeight();
106 u16 expectedLeds = width * height;
107
108 // Validate that XYMap dimensions match channel LED count
109 if (expectedLeds != numLeds) {
110 FL_ERROR("Channel '%s': XYMap dimensions (%sx%s=%s) don't match LED count (%s). Addressing transformation may produce unexpected results.", channelName, width, height, expectedLeds, numLeds);
111 }
112
113 // Cast mData to CRGB array
114 const CRGB* pixelArray = (const CRGB*)pixels.mData;
115
116 // Get thread-local buffer (reuses allocation if size unchanged)
117 fl::vector<CRGB>& buffer = getReorderBufferTLS();
118 buffer.clear();
119 buffer.resize(numLeds);
120
121 // Fill buffer by mapping each physical index to its source
122 for (u16 physicalIdx = 0; physicalIdx < numLeds; physicalIdx++) {
123 u16 x = physicalIdx % width;
124 u16 y = physicalIdx / width;
125 u16 sourceIdx = addressing->mapToIndex(x, y);
126 buffer[physicalIdx] = (sourceIdx < numLeds) ? pixelArray[sourceIdx] : CRGB::Black;
127 }
128
129 // Construct PixelController with reordered buffer
130 mAddressedController.emplace(
131 buffer.data(), buffer.size(),
132 pixels.mColorAdjustment, DISABLE_DITHER);
133 mPixelIterator = PixelIteratorAny(mAddressedController.value(), rgbOrder, rgbw, rgbww);
134 }
135
136#if FL_COLOR_PROFILE_RUNTIME
137 const ColorPipelineHooks& hooks = colorPipelineHooks();
138 if (pipeline != nullptr && hooks.makeIterator != nullptr) {
139 // Colour-managed: build the iterator over a streaming source
140 // instead of over the order-variant controller. The source
141 // applies the colour order itself, so it wraps the RGB-ordered
142 // controller -- whichever of the two that is after addressing.
143 PixelController<RGB, 1, 0xFFFFFFFF>& base =
144 mAddressedController ? mAddressedController.value() : pixels;
145 mManagedIterator = hooks.makeIterator(
146 mManagedSourceStorage, mManagedIteratorStorage, base, rgbOrder,
147 *pipeline, rgbw, rgbww, ditherPhase, brightness);
148 }
149#endif
150 }

433bool Channel::reconcileColorProfile(const ChannelOptions& options) FL_NO_EXCEPT {
434 // A strict-mode rejection disables the channel. Remember whether the
435 // current disabled state is ours, so that withdrawing the rejection
436 // re-enables the channel without also overriding a caller's own
437 // setEnabled(false).
438 const bool wasRejectedByUs = mColorProfileFallback && !mProfileBindingAccepted;
439
440 // Recompute from scratch. Reconfiguration must not inherit the verdict of
441 // the configuration it replaces: a valid profile applied over a rejected
442 // one has to clear the fallback, not keep reporting it.
443 mColorProfileFallback = false;
444 mProfileBindingAccepted = true;
445
446 // C5's fallback, raised by the two conditions that mean the same thing to
447 // a caller: management was asked for and is not happening. The second one
448 // -- a profile that binds but builds no pipeline -- is raised further
449 // down, once the build has been attempted.
450 if (options.mColorProfile.mRequested && !options.hasColorProfile()) {
451 raiseColorProfileFallback("no profile bound");
452 }
453 if (options.mColorProfile.mUseGlobalSourceDefault) {
454 mSettings.mColorProfile.mSource = detail::defaultSourceProfile();
455 }
456 // Guards the build below: a binding already rejected by strict mode must
457 // not be built. It is deliberately *not* reused for the enablement
458 // branch -- the build can raise the fallback itself (#4345), and reading
459 // this stale value there left a strict-mode channel reporting Rejected
460 // while still enabled and rendering through the legacy path, which is the
461 // one thing strict mode exists to prevent.
462 const bool rejectedBeforeBuild = mColorProfileFallback && !mProfileBindingAccepted;
463
464 // Derive the pipeline once, here, rather than per frame: it inverts
465 // matrices and bisects a lightness bound. A binding that does not
466 // describe a usable pipeline leaves the channel on the legacy path
467 // rather than failing -- an unbound channel is the ordinary case, not an
468 // error.
469 mPipeline.reset();
470 const ColorPipelineHooks& hooks = colorPipelineHooks();
471 if (!rejectedBeforeBuild && hooks.build != nullptr) {
472 // Through the hook, not by name. Calling `buildPipelineForBinding`
473 // directly from here is what kept the entire pipeline alive in every
474 // build; the pointer is null until `setColorProfile` installs it.
475 // Built on the stack first, so a binding that does not describe a
476 // usable pipeline costs no allocation -- that is the failing half of
477 // an ordinary unbound channel, not an exceptional path.
478 StreamingPipelineQ16 pipeline;
479 if (hooks.build(mSettings.mColorProfile, mSettings, mChipset,
480 &pipeline)) {
481#if FL_COLOR_PIPELINE_SHARED
482 mPipeline = fl::make_shared<StreamingPipelineQ16>(pipeline);
483#else
484 mPipeline = fl::make_unique<StreamingPipelineQ16>(pipeline);
485#endif
486 } else if (options.hasColorProfile()) {
487 // R6: a profile outside the numerical bounds must "fail
488 // explicitly" and "not become native-drive input". A binding that
489 // cannot produce a pipeline used to leave every reporting surface
490 // saying the channel was fine -- Configured, no fallback flag, no
491 // warning -- while it rendered through the legacy path (#4345).
492 //
493 // This reports; it does not reject. Refusing the binding at
494 // setColorProfile() time is R6's fuller reading and needs the
495 // published coefficient bounds it also asks for, which do not
496 // exist yet.
497 raiseColorProfileFallback("bound profile builds no usable pipeline");
498 }
499 }
500
501 // Recomputed after the build, so a rejection raised by a failed build
502 // reaches this branch.
503 const bool rejectedNow = mColorProfileFallback && !mProfileBindingAccepted;
504 if (rejectedNow) {
505 setEnabled(false);
506 } else if (wasRejectedByUs) {
507 setEnabled(true);
508 }
509 return mColorProfileFallback;
510}

552void writeUCS7604(fl::vector_psram<u8>* data, PixelIterator& pixelIterator,
553 ClocklessEncoder encoder, const ChannelOptions& settings,
554 EOrder rgbOrder, bool managed) FL_NO_EXCEPT {
555 // Map encoder enum to UCS7604Mode
556 UCS7604Mode mode;
557 switch (encoder) {
558 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_8BIT:
559 mode = UCS7604Mode::UCS7604_MODE_8BIT_800KHZ;
560 break;
561 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_16BIT:
562 mode = UCS7604Mode::UCS7604_MODE_16BIT_800KHZ;
563 break;
564 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_16BIT_1600:
565 mode = UCS7604Mode::UCS7604_MODE_16BIT_1600KHZ;
566 break;
567 default:
568 // Should never happen â€” caller already checked
569 mode = UCS7604Mode::UCS7604_MODE_16BIT_800KHZ;
570 break;
571 }
572
573 // Get current control from global UCS7604 brightness
574 UCS7604CurrentControl current = ucs7604::brightness();
575
576 // Reorder current control values to match wire order (same logic as UCS7604ControllerT)
577 u8 rgb_currents[3] = {current.r, current.g, current.b};
578 u8 pos0 = (static_cast<int>(rgbOrder) >> 6) & 0x3;
579 u8 pos1 = (static_cast<int>(rgbOrder) >> 3) & 0x3;
580 u8 pos2 = (static_cast<int>(rgbOrder) >> 0) & 0x3;
581 UCS7604CurrentControl wire_current(
582 rgb_currents[pos0], rgb_currents[pos1], rgb_currents[pos2], current.w);
583
584 // Get gamma LUT.
585 //
586 // Only the unbound path uses it. With a profile bound the pixel source
587 // has already produced the device drive and quantized it once, to 16
588 // bits, so widening it again through a curve is the second shaping stage
589 // B1 and section 6 forbid after the device solve -- and the 2.8 here is
590 // `value_or`'s default rather than anything the caller asked for, since
591 // `setColorProfile` clears `mGamma`. #4326.
592 // The iterator's own state, not the options'. `hasColorProfile()` says a
593 // profile is configured; it does not say the managed iterator was built.
594 // A rejected binding leaves the legacy controller in place, and taking
595 // the no-gamma path over its 8-bit pixels would drop a stage that path
596 // still needs.
597 const bool wide_source = managed;
598 float gamma = settings.mGamma.value_or(2.8f);
599 fl::shared_ptr<const Gamma8> gamma8 = Gamma8::getOrCreate(gamma);
600
601 bool is_rgbw = pixelIterator.get_rgbw().active();
602 size_t num_leds = pixelIterator.size();
603
604 // (#2558) UCS7604 is a 4-channel chipset (always RGBW). If the user
605 // configured this channel for 5-channel RGBWW (warm-W + cool-W), the
606 // chipset can't carry the second W byte. Warn loudly â€” silently dropping
607 // the white channels would be very hard to diagnose. The pixel iterator
608 // continues to emit RGB-only bytes (is_rgbw=false) so the strip still
609 // lights up, just without the W diode.
610 if (pixelIterator.get_rgbww().active() && !is_rgbw) {
611 FL_WARN_ONCE("UCS7604 cannot carry 5-channel RGBWW â€” this chipset "
612 "always emits 4-channel RGBW. The warm/cool white "
613 "channels in your Rgbww config will be dropped. "
614 "Set ChannelOptions.mWhiteCfg = Rgbw{...} instead to "
615 "silence this warning.");
616 }
617
618 // Encode into the data buffer
619 encodeUCS7604(pixelIterator, num_leds, fl::back_inserter(*data),
620 mode, wire_current, is_rgbw, gamma8.get(), wide_source);
621}

876// point stays small and the bodies exist exactly once.
877void Channel::showPixels(PixelController<RGB, 1, 0xFFFFFFFF> &pixels) {

909// Encode `pixels` into mChannelData with the construction-bound encoder and
910// fire onChannelDataEncoded (#4566: out of line from showPixels()).
911void Channel::encodeFrame(PixelController<RGB, 1, 0xFFFFFFFF> &pixels) FL_NO_EXCEPT {
912 // Build pixel iterator with optional addressing transformation
913 // (#2558) Pass both Rgbw and Rgbww from the channel options; the iterator
914 // carries both, and the encoder dispatch below picks the right path based
915 // on which variant alternative ChannelOptions::mWhiteCfg holds.
916#if FL_COLOR_PROFILE_RUNTIME
917 // Brightness reaches the pipeline as C4's flux scalar and nowhere else.
918 // `premixed` is the brightness on this path: binding a profile sets
919 // correction and temperature to identity, so nothing else is folded into
920 // it -- and reading it rather than `ColorAdjustment::brightness` keeps
921 // this working when FASTLED_HD_COLOR_MIXING is off, where that field
922 // does not exist.
923 //
924 // The shared pipeline is never written after it is built (#4440): this
925 // managed iterator takes a reference, copies it, and sets flux on the copy, so a
926 // concurrent reader -- the power estimate copies the same pipeline --
927 // never sees a half-written flux, and a reconfiguration replaces
928 // `mPipeline` without touching what this frame encodes through. Small
929 // tiers encode synchronously and keep single ownership, mutated in place.
930#if FL_COLOR_PIPELINE_SHARED
931 const ColorPipelineStorage pipeline_ref = mPipeline;
932 const StreamingPipelineQ16* pipeline = pipeline_ref.get();
933#else
934 StreamingPipelineQ16* pipeline = mPipeline.get();
935#endif
936#else
937 StreamingPipelineQ16* pipeline = nullptr;
938#endif
939 // Dither advances on presentation, not on attempt (#4347, R8): this
940 // channel's phase moves only when its driver accepts a frame (submitFrame()), so
941 // a dropped submission -- no driver, a disabled one, a busy buffer --
942 // does not consume a phase and the cycle's average stays unbiased. Both
943 // the legacy offsets and the colour-managed temporal dither read it.
944 if (pixels.ditherActive()) {
945 pixels.reseed_binary_dithering(mDitherPhase);
946 }
947 ReorderingPixelIteratorAny iterator(pixels, mScreenMap.getXYMap(), mRgbOrder,
948 mSettings.rgbw(), mSettings.rgbww(),
949 mName, pipeline, mDitherPhase,
950 pixels.mColorAdjustment.premixed.r);
951 PixelIterator& pixelIterator = iterator.get();
952
953 // Encode pixels with the writer selected once from the immutable chipset.
954 auto& data = mChannelData->getData();
955 data.clear();
956 if (mSettings.isRgbww()) {
957 mChannelData->setPixelFormat(ChannelPixelFormat::RGBWW);
958 } else if (mSettings.isRgbw()) {
959 mChannelData->setPixelFormat(ChannelPixelFormat::RGBW);
960 } else {
961 mChannelData->setPixelFormat(ChannelPixelFormat::RGB);
962 }
963
964 if (mPixelEncoder != nullptr) {
965 mPixelEncoder(*this, pixelIterator, iterator.isManaged(), data);
966 }
967
968 // Fire event after encoding completes
969 {
970 auto& events = ChannelEvents::instance();
971 events.onChannelDataEncoded(*this, *mChannelData);
972 }
973}

975// Submit the encoded frame to `driver` (#4566: out of line from showPixels()).
976void Channel::submitFrame(const fl::shared_ptr<IChannelDriver>& driver) FL_NO_EXCEPT {
977 // #2517: detect the silent-drop scenario before enqueuing â€” if the
978 // resolved driver is registered with ChannelManager but currently
979 // disabled (typically by `FastLED.setExclusiveDriver<OtherBus>()`),
980 // `ChannelManager::onEndFrame()` will skip its `show()` call and the
981 // frame is silently dropped on the floor. Emit an actionable
982 // FL_ERROR so users can diagnose the missing output.
983 //
984 // One-shot via `mDisabledDriverWarned` to avoid per-frame spam.
985 // Reset the latch whenever the driver returns to ENABLED so a later
986 // disable re-emits the diagnostic.
987 fl::string driverName = driver->getName();
988 auto status = ChannelManager::instance().driverStatus(driverName);
989 if (status == ChannelManager::DriverStatus::STATUS_DISABLED) {
990#if FASTLED_LOG_RUNTIME_ENABLED
991 if (!mDisabledDriverWarned) {
992 emitDisabledDriverError(
993 mName, driverName,
994 ChannelManager::instance().exclusiveDriverName());
995 mDisabledDriverWarned = true;
996 }
997#endif
998 // Skip the enqueue â€” the data wouldn't be sent anyway, and dropping
999 // it here matches the existing behaviour (rather than leaving stale
1000 // bytes in the driver's pending queue across an enable/disable flip).
1001 // The drop happens in both debug and release; only the diagnostic
1002 // and the one-shot latch are gated. In release, the empty
1003 // emitDisabledDriverError + the exclusiveDriverName() call + the
1004 // 3-arg fl::string chain dead-strip (see #2950).
1005 return;
1006 } else if (status == ChannelManager::DriverStatus::STATUS_ENABLED) {
1007#if FASTLED_LOG_RUNTIME_ENABLED
1008 // Reset the one-shot guard so a future disable re-emits the diagnostic.
1009 mDisabledDriverWarned = false;
1010#endif
1011 }
1012 // NOT_REGISTERED: driver is not managed by ChannelManager (e.g. a custom
1013 // test driver, or one that was removed). Fall through to enqueue and let
1014 // the driver decide what to do â€” this is the historic behaviour.
1015
1016 // Enqueue for transmission (will be sent when driver->show() is called)
1017 driver->enqueue(mChannelData);
1018 // Presented, per the driver contract: an accepted enqueue is the frame.
1019 ++mDitherPhase;
1020 auto& events = ChannelEvents::instance();
1021 // Reuse the name fetched for driverStatus() instead of a second getName().
1022 events.onChannelEnqueued(*this, driverName);
1023}
