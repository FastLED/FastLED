8// NOTE: SD card support (FileSystem::beginSd and make_sdcard_filesystem)
9// lives in a SEPARATE translation unit at
10// `src/fl/fs/sd/file_system_sd.cpp.hpp`, compiled into
11// `fl.system.sd+.cpp.o` via `src/fl/build/fl.system.sd+.cpp`.
12//
13// This split lets the linker tree-shake the entire SD chain (libSD.a,
14// libFS.a, Arduino's VFSImpl, the printf engine VFSFileImpl drags in
15// via snprintf) AUTOMATICALLY when the user never calls
16// `FileSystem::beginSd()` -- no FASTLED_USE_SDCARD opt-in required, and
17// the macro that the earlier macro-gate PR (#2778 v1) shipped is
18// removed by this PR. See FastLED #2773 item 1.2 and the SD TU header
19// for the mechanism.
20//
21// All other FileSystem methods (begin, openRead, readText, ...) and the
22// NullFileSystem stub stay here, so existing sketches that use only
23// `FileSystem::begin(platform_filesystem)` are unaffected.

191bool FileSystem::readText(const char *path, fl::string *out) {
192 fl::ifstream file = openRead(path);
193 if (!file.is_open()) {
194 FL_WARN("Failed to open file: %s", path);
195 return false;
196 }
197 fl::size size = file.size();
198 out->reserve(size + out->size());
199 bool wrote = false;
200 while (file.available()) {
201 u8 buf[64];
202 fl::size n = file.read(buf, sizeof(buf));
203 out->append((const char *)buf, n);
204 wrote = true;
205 }
206 file.close();
207 FL_DBG_IF(!wrote, "Failed to write any data to the output string.");
208 return wrote;
209}

212// Convenience wrapper: mount an SD card and decode a JPEG in one call.
213// Out of line because the API header wraps, it does not implement
214// (agents/docs/cpp-standards.md -> API Object Pattern, rule 2; FastLED #4003).
215FramePtr loadJpegFromSD(int cs_pin, const char *filepath,
216 const JpegConfig &config,
217 fl::string *error_message) FL_NO_EXCEPT {
218 FileSystem fs;
219 if (!fs.beginSd(cs_pin)) {
220 if (error_message) {
221 *error_message = "Failed to initialize SD card on CS pin ";
222 error_message->append(static_cast<fl::u32>(cs_pin));
223 }
224 return FramePtr();
225 }
226 return fs.loadJpeg(filepath, config, error_message);
227}
