97 public:
98 PollNeededCallbackSlot() FL_NO_EXCEPT
99 : mSnapshot(nullptr), mRetired(nullptr) {}
100
101 ~PollNeededCallbackSlot() FL_NO_EXCEPT {
102 Snapshot* active =
103 mSnapshot.exchange(nullptr, fl::memory_order_acq_rel);
104 destroySnapshots(active);
105 destroySnapshots(mRetired);
106 mRetired = nullptr;
107 }
108
109 void set(PollNeededCallback callback) FL_NO_EXCEPT {
110 if (callback.callback == nullptr) {
111 clear();
112 return;
113 }
114 Snapshot* snapshot = new Snapshot(callback); // ok bare allocation
115 retire(mSnapshot.exchange(snapshot, fl::memory_order_acq_rel));
116 }
117
118 void clear() FL_NO_EXCEPT {
119 retire(mSnapshot.exchange(nullptr, fl::memory_order_acq_rel));
120 }
121
122 void invoke() const FL_NO_EXCEPT {
123 Snapshot* snapshot = mSnapshot.load(fl::memory_order_acquire);
124 if (snapshot == nullptr) {
125 return;
126 }
127 snapshot->callback.invoke();
128 }
129
130 private:
131 void retire(Snapshot* snapshot) FL_NO_EXCEPT {
132 if (snapshot == nullptr) {
133 return;
134 }
135 // An ISR may already have loaded this pointer, so reclaim only when
136 // the slot is destroyed and driver teardown has quiesced callbacks.
137 snapshot->next = mRetired;
138 mRetired = snapshot;
139 }
140
141 static void destroySnapshots(Snapshot* snapshot) FL_NO_EXCEPT {
142 while (snapshot != nullptr) {
143 Snapshot* next = snapshot->next;
144 delete snapshot; // ok bare allocation
145 snapshot = next;
146 }
147 }
148
149 fl::atomic<Snapshot*> mSnapshot;
150 Snapshot* mRetired;
151 };
