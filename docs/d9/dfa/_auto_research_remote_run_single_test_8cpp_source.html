222// Diagnostic probe (Teensy 4.x): manually drive TX via the standard GPIO1-4
223// alias and read RX with digitalRead, so we can tell whether the
224// IOMUXC_GPR_GPR26..29 fast->standard remap that ObjectFLED relies on
225// actually reaches the pad on the connected board. Saves and restores all
226// touched IOMUXC/GPIO state; never modifies FlexPWM or DMA.
227fl::json probeStandardGpioPad(int tx_pin, int rx_pin) {
228 fl::json probe = fl::json::object();
229 probe.set("supported", false);
230 probe.set("txPin", static_cast<int64_t>(tx_pin));
231 probe.set("rxPin", static_cast<int64_t>(rx_pin));
232#if !defined(FL_IS_TEENSY_4X)
233 probe.set("message", "Teensy 4.x only");
234 return probe;
235#else
236 probe.set("supported", true);
237 if (tx_pin < 0 || tx_pin >= NUM_DIGITAL_PINS ||
238 rx_pin < 0 || rx_pin >= NUM_DIGITAL_PINS) {
239 probe.set("success", false);
240 probe.set("error", "InvalidPin");
241 return probe;
242 }
243
244 volatile uint32_t* fast_output = portOutputRegister(tx_pin);
245 volatile uint32_t* standard_output = reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
246 reinterpret_cast<uintptr_t>(fast_output) - 0x01E48000u);
247 volatile uint32_t* fast_mode = portModeRegister(tx_pin);
248 volatile uint32_t* standard_mode = reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
249 reinterpret_cast<uintptr_t>(fast_mode) - 0x01E48000u);
250 volatile uint32_t* tx_mux = portConfigRegister(tx_pin);
251 volatile uint32_t* tx_pad = portControlRegister(tx_pin);
252 volatile uint32_t* rx_mux = portConfigRegister(rx_pin);
253 volatile uint32_t* rx_pad = portControlRegister(rx_pin);
254
255 const uint32_t gpio6_base =
256 static_cast<uint32_t>(reinterpret_cast<uintptr_t>(&GPIO6_DR));
257 const uint32_t output_addr =
258 static_cast<uint32_t>(reinterpret_cast<uintptr_t>(fast_output));
259 const uint8_t offset =
260 static_cast<uint8_t>((output_addr - gpio6_base) >> 14);
261 const uint8_t bit = digitalPinToBit(tx_pin);
262 const uint32_t mask = 1u << bit;
263 probe.set("txBit", static_cast<int64_t>(bit));
264 probe.set("txMask", static_cast<int64_t>(mask));
265 probe.set("txOffset", static_cast<int64_t>(offset));
266 if (offset > 3) {
267 probe.set("success", false);
268 probe.set("error", "UnsupportedGpioBank");
269 return probe;
270 }
271
272 volatile uint32_t* gpr = &IOMUXC_GPR_GPR26 + offset;
273 // Save state we are about to clobber.
274 const uint32_t saved_gpr = *gpr;
275 const uint32_t saved_tx_mux = *tx_mux;
276 const uint32_t saved_tx_pad = *tx_pad;
277 const uint32_t saved_rx_mux = *rx_mux;
278 const uint32_t saved_rx_pad = *rx_pad;
279 const uint32_t saved_fast_mode = *fast_mode;
280 const uint32_t saved_standard_mode = *standard_mode;
281 const uint32_t saved_fast_out = *fast_output;
282 const uint32_t saved_standard_out = *standard_output;
283
284 // Make sure standard GPIO1-4 banks are clocked. Teensy startup usually
285 // leaves them on, but assert it explicitly so this probe is independent
286 // of boot order.
287 CCM_CCGR1 |= CCM_CCGR1_GPIO1(CCM_CCGR_ON);
288 CCM_CCGR0 |= CCM_CCGR0_GPIO2(CCM_CCGR_ON);
289 CCM_CCGR2 |= CCM_CCGR2_GPIO3(CCM_CCGR_ON);
290 CCM_CCGR3 |= CCM_CCGR3_GPIO4(CCM_CCGR_ON);
291
292 // RX pin: simple input, no pull, so we can sample what the pad sees.
293 *rx_mux = 5 | 0x10;
294 *rx_pad = (5u << 3) | (3u << 6); // DSE=5, SPEED=3, no pull/keeper
295 pinMode(rx_pin, INPUT);
296
297 // TX pin: ALT5 (GPIO) + SION, mapped to standard bank.
298 *tx_mux = 5 | 0x10;
299 *tx_pad = (3u << 3) | (3u << 6); // DSE=3 (matches ObjectFLED), SPEED=3
300 *gpr &= ~mask; // route pad to standard alias
301 *standard_mode |= mask; // output mode in standard bank
302 // DR_CLEAR = +0x88, DR_SET = +0x84
303 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
304 reinterpret_cast<uintptr_t>(standard_output) + 0x88u) = mask;
305 delayMicroseconds(50);
306 const int rx_initial = digitalReadFast(rx_pin);
307
308 int rx_high_count = 0;
309 int rx_low_count = 0;
310 for (int i = 0; i < 5; ++i) {
311 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
312 reinterpret_cast<uintptr_t>(standard_output) + 0x84u) = mask;
313 delayMicroseconds(40);
314 if (digitalReadFast(rx_pin) == HIGH) {
315 ++rx_high_count;
316 }
317 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
318 reinterpret_cast<uintptr_t>(standard_output) + 0x88u) = mask;
319 delayMicroseconds(40);
320 if (digitalReadFast(rx_pin) == LOW) {
321 ++rx_low_count;
322 }
323 }
324
325 probe.set("standardGpioRxInitial",
326 rx_initial == HIGH ? "HIGH" : "LOW");
327 probe.set("standardGpioRxHighCount", static_cast<int64_t>(rx_high_count));
328 probe.set("standardGpioRxLowCount", static_cast<int64_t>(rx_low_count));
329 probe.set("standardGpioConnected",
330 (rx_high_count == 5) && (rx_low_count == 5));
331
332 // Tight-loop fast toggle: write HIGH, read RX, write LOW, read RX, with
333 // no delay between write and read. Measures whether the standard GPIO
334 // alias propagates to the pad and back to digitalReadFast within a
335 // CPU-bus-cycle window comparable to ObjectFLED's DMA write rate.
336 int fast_high_count = 0;
337 int fast_low_count = 0;
338 int fast_high_zero_delay = 0;
339 int fast_low_zero_delay = 0;
340 constexpr int kFastIters = 50;
341 for (int i = 0; i < kFastIters; ++i) {
342 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
343 reinterpret_cast<uintptr_t>(standard_output) + 0x84u) = mask;
344 if (digitalReadFast(rx_pin) == HIGH) {
345 ++fast_high_zero_delay;
346 }
347 delayNanoseconds(200);
348 if (digitalReadFast(rx_pin) == HIGH) {
349 ++fast_high_count;
350 }
351 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
352 reinterpret_cast<uintptr_t>(standard_output) + 0x88u) = mask;
353 if (digitalReadFast(rx_pin) == LOW) {
354 ++fast_low_zero_delay;
355 }
356 delayNanoseconds(200);
357 if (digitalReadFast(rx_pin) == LOW) {
358 ++fast_low_count;
359 }
360 }
361 probe.set("fastToggleHighSettled", static_cast<int64_t>(fast_high_count));
362 probe.set("fastToggleLowSettled", static_cast<int64_t>(fast_low_count));
363 probe.set("fastToggleHighZeroDelay",
364 static_cast<int64_t>(fast_high_zero_delay));
365 probe.set("fastToggleLowZeroDelay",
366 static_cast<int64_t>(fast_low_zero_delay));
367 probe.set("fastToggleIterations", static_cast<int64_t>(kFastIters));
368
369 // Restore TX/RX pad state.
370 if ((saved_fast_out & mask) != 0) {
371 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
372 reinterpret_cast<uintptr_t>(fast_output) + 0x84u) = mask;
373 } else {
374 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
375 reinterpret_cast<uintptr_t>(fast_output) + 0x88u) = mask;
376 }
377 if ((saved_standard_out & mask) != 0) {
378 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
379 reinterpret_cast<uintptr_t>(standard_output) + 0x84u) = mask;
380 } else {
381 *reinterpret_cast<volatile uint32_t*>( // ok reinterpret cast - Teensy GPIO alias address map
382 reinterpret_cast<uintptr_t>(standard_output) + 0x88u) = mask;
383 }
384 *fast_mode = saved_fast_mode;
385 *standard_mode = saved_standard_mode;
386 *gpr = saved_gpr;
387 *tx_pad = saved_tx_pad;
388 *tx_mux = saved_tx_mux;
389 *rx_pad = saved_rx_pad;
390 *rx_mux = saved_rx_mux;
391
392 probe.set("success", true);
393 return probe;
394#endif
395}

397fl::json measureTightTiming(const fl::string& driver_name,
398 const fl::ChipsetTimingConfig& timing,
399 const fl::vector<fl::ChannelConfig>& tx_configs,
400 int iterations,
401 uint32_t max_allowed_overhead_us,
402 bool& out_passed) {
403 fl::json metric = fl::json::object();
404 out_passed = false;
405
406 metric.set("requested", true);
407 metric.set("supported", false);
408 metric.set("driver", driver_name.c_str());
409
410 if (iterations < 1) {
411 metric.set("message", "iterations must be >= 1");
412 return metric;
413 }
414 if (tx_configs.empty()) {
415 metric.set("message", "no channel configs");
416 return metric;
417 }
418 for (fl::size i = 0; i < tx_configs.size(); i++) {
419 if (tx_configs[i].isSpi()) {
420 metric.set("message", "clocked SPI timing metric is not supported");
421 return metric;
422 }
423 }
424
425 fl::vector<fl::ChannelConfig> sample_configs = tx_configs;
426 fl::vector<fl::shared_ptr<fl::Channel>> channels;
427 for (fl::size i = 0; i < sample_configs.size(); i++) {
428 fl::ChannelConfig channel_config(
429 sample_configs[i].getDataPin(),
430 timing,
431 sample_configs[i].mLeds,
432 sample_configs[i].rgb_order);
433 fl::shared_ptr<fl::Channel> channel = FastLED.add(channel_config);
434 if (!channel) {
435 FastLED.clear(ClearFlags::CHANNELS);
436 metric.set("message", "failed to create channel");
437 return metric;
438 }
439 channels.push_back(channel);
440 }
441
442 ScopedFastLedBrightness scoped_brightness(255);
443 for (fl::size lane = 0; lane < sample_configs.size(); lane++) {
444 fill_solid(sample_configs[lane].mLeds.data(),
445 sample_configs[lane].mLeds.size(),
446 CRGB::Black);
447 }
448 FastLED.show();
449 if (!FastLED.wait(1000)) {
450 FastLED.clear(ClearFlags::CHANNELS);
451 metric.set("message", "warmup wait timeout");
452 return metric;
453 }
454 delay(2);
455
456 const uint32_t expected_wire_us =
457 expectedClocklessWireUs(timing, maxLaneLeds(sample_configs));
458 uint32_t min_show_us = 0xFFFFFFFFu;
459 uint32_t max_show_us = 0;
460 uint32_t min_wait_us = 0xFFFFFFFFu;
461 uint32_t max_wait_us = 0;
462 uint32_t min_total_us = 0xFFFFFFFFu;
463 uint32_t max_total_us = 0;
464 uint32_t min_overhead_us = 0xFFFFFFFFu;
465 uint32_t max_overhead_us = 0;
466 uint64_t sum_show_us = 0;
467 uint64_t sum_wait_us = 0;
468 uint64_t sum_total_us = 0;
469 uint64_t sum_overhead_us = 0;
470 int samples = 0;
471 bool timed_out = false;
472
473 for (int sample = 0; sample < iterations; sample++) {
474 for (fl::size lane = 0; lane < sample_configs.size(); lane++) {
475 const uint8_t r = static_cast<uint8_t>(31 + sample * 17 + lane * 11);
476 const uint8_t g = static_cast<uint8_t>(67 + sample * 23 + lane * 7);
477 const uint8_t b = static_cast<uint8_t>(103 + sample * 29 + lane * 5);
478 fill_solid(sample_configs[lane].mLeds.data(),
479 sample_configs[lane].mLeds.size(),
480 CRGB(r, g, b));
481 }
482
483 const uint32_t t0 = micros();
484 FastLED.show();
485 const uint32_t t1 = micros();
486 const bool ok = FastLED.wait(1000);
487 const uint32_t t2 = micros();
488 if (!ok) {
489 timed_out = true;
490 break;
491 }
492
493 const uint32_t show_us = t1 - t0;
494 const uint32_t wait_us = t2 - t1;
495 const uint32_t total_us = t2 - t0;
496 const uint32_t overhead_us =
497 total_us > expected_wire_us ? total_us - expected_wire_us : 0;
498
499 if (show_us < min_show_us) min_show_us = show_us;
500 if (show_us > max_show_us) max_show_us = show_us;
501 if (wait_us < min_wait_us) min_wait_us = wait_us;
502 if (wait_us > max_wait_us) max_wait_us = wait_us;
503 if (total_us < min_total_us) min_total_us = total_us;
504 if (total_us > max_total_us) max_total_us = total_us;
505 if (overhead_us < min_overhead_us) min_overhead_us = overhead_us;
506 if (overhead_us > max_overhead_us) max_overhead_us = overhead_us;
507
508 sum_show_us += show_us;
509 sum_wait_us += wait_us;
510 sum_total_us += total_us;
511 sum_overhead_us += overhead_us;
512 samples++;
513 }
514
515 FastLED.clear(ClearFlags::CHANNELS);
516
517 metric.set("supported", true);
518 metric.set("samples", static_cast<int64_t>(samples));
519 metric.set("iterations", static_cast<int64_t>(iterations));
520 metric.set("expected_wire_us", static_cast<int64_t>(expected_wire_us));
521 metric.set("max_allowed_overhead_us",
522 static_cast<int64_t>(max_allowed_overhead_us));
523
524 if (samples == 0) {
525 metric.set("passed", false);
526 metric.set("message", timed_out ? "timing wait timeout" : "no samples");
527 return metric;
528 }
529
530 metric.set("min_show_us", static_cast<int64_t>(min_show_us));
531 metric.set("max_show_us", static_cast<int64_t>(max_show_us));
532 metric.set("avg_show_us",
533 static_cast<int64_t>(sum_show_us / static_cast<uint64_t>(samples)));
534 metric.set("min_wait_us", static_cast<int64_t>(min_wait_us));
535 metric.set("max_wait_us", static_cast<int64_t>(max_wait_us));
536 metric.set("avg_wait_us",
537 static_cast<int64_t>(sum_wait_us / static_cast<uint64_t>(samples)));
538 metric.set("min_total_us", static_cast<int64_t>(min_total_us));
539 metric.set("max_total_us", static_cast<int64_t>(max_total_us));
540 metric.set("avg_total_us",
541 static_cast<int64_t>(sum_total_us / static_cast<uint64_t>(samples)));
542 metric.set("min_overhead_us", static_cast<int64_t>(min_overhead_us));
543 metric.set("max_overhead_us", static_cast<int64_t>(max_overhead_us));
544 metric.set("avg_overhead_us",
545 static_cast<int64_t>(sum_overhead_us / static_cast<uint64_t>(samples)));
546
547 out_passed = !timed_out && max_overhead_us <= max_allowed_overhead_us;
548 metric.set("passed", out_passed);
549 if (timed_out) {
550 metric.set("message", "timing wait timeout");
551 } else {
552 metric.set("message", out_passed ? "tight timing within budget"
553 : "tight timing exceeded budget");
554 }
555 return metric;
556}

1048 // built next to it fails begin() with `dma_buffer_busy` and captures

1050 // actualRxPin as though the change took effect. `setPins` never hit this

1052 // releasing the claim. Confirmed on an RP2350W: the failing run reports
1053 // rxBeginError='dma_buffer_busy'. See FastLED#4374.

1056 // is what a caller probing alternative wiring means by them -- adopting
1057 // the pin as standing state would trade a silent failure for a silent
1058 // reconfiguration.

1085 // 103 LEDs, and 101, 102 and 103 were all accepted and then failed in
1086 // capture with no error at all -- the outcome a bounds check exists to
1087 // prevent (FastLED#4371).
1088 //
1089 // Two things decide where this check can live. The real ceiling needs the
1090 // resolved timing and encoder, so it cannot sit with the rest of the
1091 // argument validation; and it belongs to the backend that *captures*, not
1092 // the driver that transmits, so it has to follow the RX channel. On RP,
1093 // PLATFORM_DEFAULT resolves to PIO RX whatever the TX driver is -- keying
1094 // off `driver_name` would skip the bound for a UART0 request, which is
1095 // captured by the same pool and hits the same ceiling, while applying a
1096 // PIO-only limit to a PIO TX pointed at some other backend.

1098 rx_channel_to_use ? rx_channel_to_use->backend()
1099 : fl::RxBackend::PLATFORM_DEFAULT;

1102 // Reserve for the gap between arming the capture and the transmitter's
1103 // first edge, which the sampler also spends words on. Measured on an
1104 // RP2350W it sits between 4 us and 64 us: a 63-LED 400 kHz frame passes
1105 // with 64 us of budget to spare and a 64-LED one fails with 4 us. Take
1106 // the upper end -- advertising a maximum that does not work is the
1107 // defect being removed here, so erring long is not symmetric with
1108 // erring short.
1109 constexpr fl::u32 kArmingLeadInNs = 64000;

1112 static_cast<fl::u32>(resolved_timing.total_period_ns());

1115 fl::kRpPioRxSamplesPerDmaWord,
1116 1000000000u / fl::kRpPioRxClockHz,
1117 fl::RxChannelConfig(pin_rx).signal_range_max_ns,
