74 template <typename Fn>
75 int decode(const fl::u8* data, fl::size len,
76 Fn on_frame) FL_NO_EXCEPT {
77 if (!mDecoder) {
78 return 0;
79 }
80
81 const fl::u8* inptr = data;
82 fl::size bytes_left = len;
83 int frames_decoded = 0;
84
85 while (bytes_left > 0) {
86 int offset = findSyncWord(inptr, bytes_left);
87 if (offset < 0) {
88 break;
89 }
90
91 inptr += offset;
92 bytes_left -= offset;
93
94 int result = decodeFrame(&inptr, &bytes_left);
95 if (result == 0) {
96 Mp3Frame frame;
97 frame.pcm = mPcmBuffer.get();
98 frame.samples = mFrameInfo.outputSamps / mFrameInfo.nChans;
99 frame.channels = mFrameInfo.nChans;
100 frame.sample_rate = mFrameInfo.samprate;
101 frame.bitrate = mFrameInfo.bitrate;
102 frame.version = mFrameInfo.version;
103 frame.layer = mFrameInfo.layer;
104
105 on_frame(frame);
106 frames_decoded++;
107 } else if (bytes_left > 0) {
108 inptr++;
109 bytes_left--;
110 }
111 }
112
113 return frames_decoded;
114 }
