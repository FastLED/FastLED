270// ============================================================================
271// FL_WATCHDOG_AUTO(...) — the canonical zero-setup loop()-top guard.
272//
273// Variadic so the same macro covers both the no-argument default (15 000 ms)
274// and the explicit-timeout form:
275//
276// void loop() {
277// FL_WATCHDOG_AUTO(); // 15 000 ms default
278// doWork();
279// }
280//
281// void loop() {
282// FL_WATCHDOG_AUTO(5000); // 5-second timeout
283// doWork();
284// }
285//
286// "AUTO" because this macro does more than just feed: it lazily arms the WDT
287// on first call, prints the prior-boot reset/crash diagnostic, pauses 3 s on
288// a crash so the developer can read the message, and feeds the WDT on both
289// construction AND destruction. Single source-line use; the macro stamps the
290// local variable name from `__LINE__` so two `FL_WATCHDOG_AUTO()` on the same
291// line are a compile error. Two on different lines but simultaneously alive
292// (nested scopes) trigger a one-shot runtime warning — see ScopedWatchdog.
293//
294// Future variants stay grouped under the `FL_WATCHDOG_*` prefix:
295// FL_WATCHDOG_AUTO_QUIET / FL_WATCHDOG_AUTO_VERBOSE — log-level variants
296// FL_WATCHDOG_MANUAL — non-RAII setup helper
297// ============================================================================
298
299#define FL_WATCHDOG__CONCAT_INNER(a, b) a##b
300#define FL_WATCHDOG__CONCAT(a, b) FL_WATCHDOG__CONCAT_INNER(a, b)
301#define FL_WATCHDOG_AUTO(...) \
302 ::fl::ScopedWatchdog FL_WATCHDOG__CONCAT(_fl_wdt_, __LINE__) { __VA_ARGS__ }

304// ============================================================================
305// FL_WATCHDOG_* capability macros — defined by the per-platform impl.
306// The noop fallback defines NONE of the boolean flags and defaults numerics to 0.
307// User code may #ifdef on the boolean flags or static_assert on the numerics.
308// ============================================================================
309//
310// #define FL_WATCHDOG_HAS_HARDWARE // 1 wherever a real or emulated WDT runs
311// #define FL_WATCHDOG_PERSIST_BYTES N // small persistent buffer size, >= 8
312// #define FL_WATCHDOG_HAS_PRE_TIMEOUT_IRQ // pre-reset callback honored at HW level
313// #define FL_WATCHDOG_HAS_WINDOW_MODE // setWindow() honored
314// #define FL_WATCHDOG_HAS_CRASH_REPORT // hasCrashReport()/readCrashReport() honored
315// #define FL_WATCHDOG_HAS_BOOTLOADER_REBOOT // rebootIntoBootloader() honored
316// #define FL_WATCHDOG_MAX_TIMEOUT_MS N // upper bound, in ms
317//
318// The Tier 0 surface is always callable; Tier 1/2 methods return false on
319// platforms whose impl does not honor them, so user code that checks the
320// return value works portably.
