33struct AudioSamplePool {
34 void put(SampleImplPtr&& impl) {
35 if (impl.unique()) {
36 // There is no more shared_ptr to this object, so we can recycle it.
37 fl::unique_lock<fl::mutex> lock(mutex);
38 if (impl && pool.size() < MAX_POOL_SIZE) {
39 // Reset the impl for reuse (clear internal state)
40 impl->reset();
41 pool.push_back(impl);
42 return;
43 }
44 }
45 // Pool is full, discard the impl
46 impl.reset();
47 }
48 SampleImplPtr getOrCreate() {
49 {
50 fl::unique_lock<fl::mutex> lock(mutex);
51 if (!pool.empty()) {
52 SampleImplPtr impl = pool.back();
53 pool.pop_back();
54 return impl;
55 }
56 }
57 return fl::make_shared<SampleImpl>();
58 }
59
60 fl::vector<SampleImplPtr> pool;
61 static constexpr fl::size MAX_POOL_SIZE = 8;
62 fl::mutex mutex;
63};

157void SoundLevelMeter::processBlock(const fl::i16 *samples, fl::size count) {
158 if (count == 0) {
159 return;
160 }
161
162 // 1) compute block power → dBFS
163 double sum_sq = 0.0;
164 for (fl::size i = 0; i < count; ++i) {
165 double s = samples[i] / 32768.0; // normalize to ±1
166 sum_sq += s * s;
167 }
168 double p = sum_sq / count; // mean power
169 double dbfs = 10.0 * log10(p + 1e-12);
170 mCurrentDbfs = dbfs;
171
172 // 2) update global floor (with optional smoothing)
173 if (dbfs < mDbfsFloorGlobal) {
174 if (mDbfsFloorGlobal == FL_INFINITY_DOUBLE || mSmoothingAlpha <= 0.0) {
175 mDbfsFloorGlobal = dbfs;
176 } else {
177 mDbfsFloorGlobal = mSmoothingAlpha * dbfs +
178 (1.0 - mSmoothingAlpha) * mDbfsFloorGlobal;
179 }
180 mOffset = mSplFloor - mDbfsFloorGlobal;
181 }
182
183 // 3) estimate SPL
184 mCurrentSpl = dbfs + mOffset;
185}

200void Sample::applyGain(float gain) {
201 if (!isValid() || gain == 1.0f) return;
202 auto& samples = mImpl->pcm_mutable();
203 for (fl::size i = 0; i < samples.size(); ++i) {
204 fl::i32 val = static_cast<fl::i32>(static_cast<float>(samples[i]) * gain);
205 if (val > 32767) val = 32767;
206 if (val < -32768) val = -32768;
207 samples[i] = static_cast<fl::i16>(val);
208 }
209 mImpl->pcmDidChange();
210}
