88// The TX clock edge field was renamed within IDF 5.5: 5.5.1 has only
89// `sample_edge`; 5.5.5 has `shift_edge` and keeps `sample_edge` as a
90// deprecated alias. Which one exists is detected here rather than guessed
91// from a version number. The two enums are ordered differently
92// (PARLIO_SAMPLE_EDGE_POS == 1, PARLIO_SHIFT_EDGE_POS == 0), and the newer
93// enumerator is not declared on older IDFs, so the new branch names the
94// member's own type and the value's meaning is in the comment.
95template <typename Cfg>
96auto setParlioRawTxEdge(Cfg& cfg, int) -> decltype(cfg.shift_edge, void()) {
97 cfg.shift_edge = static_cast<decltype(cfg.shift_edge)>(0); // PARLIO_SHIFT_EDGE_POS
98}
99template <typename Cfg>
100void setParlioRawTxEdge(Cfg& cfg, long) {
101 cfg.sample_edge = PARLIO_SAMPLE_EDGE_POS; // IDF before the rename
102}

105fl::json autoResearchDeviceJson(const fl::string& name) {
106 if (name == "RMT") return fl::deviceJson<fl::Bus::RMT>();
107 if (name == "SPI" || name == "SPI_UNIFIED") return fl::deviceJson<fl::Bus::SPI>();
108 // RP registers its two fixed PL022 blocks under the concrete names
109 // "SPI0"/"SPI1". Without these they fell through to the generic tail
110 // below, which hardcodes which=0 and echoes the driver name into
111 // bus_name/vendor_name/device_name -- so `drivers` reported SPI1 with
112 // which=0 and no real bus metadata, unlike UART0/UART1 and PIO0/1/2.
113 // RP-only. SAMD21 (spi_hw_manager_samd21.cpp.hpp:38-39) and nRF52
114 // (init_channel_driver_nrf52.cpp.hpp:49) also register drivers called
115 // "SPI0"/"SPI1", but DeviceInfoResolver<Bus::SPI, 0|1> is specialized
116 // only in the RP branch of bus_info.h. Routing those platforms here
117 // would hand them the unspecialized no-op resolver and report
118 // is_noop=true / available=false -- the very regression this mapping
119 // was added to fix, just moved to a different chip.
120#if defined(FL_IS_RP)
121 if (name == "SPI0") return fl::deviceJson<fl::Bus::SPI, 0>();
122 if (name == "SPI1") return fl::deviceJson<fl::Bus::SPI, 1>();
123#endif
124 if (name == "UART" || name == "LPUART" || name == "UART0") {
125 return fl::deviceJson<fl::Bus::UART, 0>();
126 }
127 if (name == "UART1") return fl::deviceJson<fl::Bus::UART, 1>();
128 if (name == "BIT_BANG" || name == "STUB") return fl::deviceJson<fl::Bus::BIT_BANG>();
129
130 if (name == "FLEX_IO" || name == "LCD_RGB") {
131 return fl::deviceJson<fl::Bus::FLEX_IO, 1>();
132 }
133 if (name == "OBJECT_FLED" || name == "PARLIO" || name == "LCD_SPI" ||
134 name == "LCD_CLOCKLESS" || name == "I2S" || name == "I2S_SPI") {
135 return fl::deviceJson<fl::Bus::FLEX_IO, 0>();
136 }
137 if (name == "PIO0") return fl::deviceJson<fl::Bus::FLEX_IO, 0>();
138 if (name == "PIO1") return fl::deviceJson<fl::Bus::FLEX_IO, 1>();
139 if (name == "PIO2") return fl::deviceJson<fl::Bus::FLEX_IO, 2>();
140
141 fl::json info = fl::json::object();
142 info.set("bus", static_cast<int64_t>(0));
143 info.set("bus_name", name.c_str());
144 info.set("vendor_name", name.c_str());
145 info.set("device_name", name.c_str());
146 info.set("which", static_cast<int64_t>(0));
147 info.set("is_noop", false);
148 info.set("notes", "registered runtime driver");
149 fl::json runtime = fl::json::object();
150 runtime.set("available", true);
151 runtime.set("initialized", false);
152 runtime.set("ready", true);
153 runtime.set("busy", false);
154 runtime.set("error", "");
155 info.set("runtime", runtime);
156 return info;
157}

225 // Returns: {success, passed, totalTests, passedTests, duration_ms, driver,

228 // NOTE: runSingleTestImpl() may return early (error cases) without calling

237 }
238 return fl::json(nullptr);

329 // memory-mapped register space without resetting the chip (esptool

339 remote.bind("parlioRawTest", [](const fl::json& args) -> fl::json {
340 fl::json response = fl::json::object();
341 int pin = 1;
342 int hz = 8000000;
343 int width = 1;
344 int pattern = 0xAA;
345 int qnb = 0;
346 int mts = 1024;
347 int burst = 0;
348 int depth = 4;
349 int chunks = 1;
350 if (args.is_object()) {
351 if (args.contains("pin") && args["pin"].is_int()) pin = (int)args["pin"].as_int().value();
352 if (args.contains("hz") && args["hz"].is_int()) hz = (int)args["hz"].as_int().value();
353 if (args.contains("width") && args["width"].is_int()) width = (int)args["width"].as_int().value();
354 if (args.contains("pattern") && args["pattern"].is_int()) pattern = (int)args["pattern"].as_int().value();
355 if (args.contains("qnb") && args["qnb"].is_int()) qnb = (int)args["qnb"].as_int().value();
356 if (args.contains("mts") && args["mts"].is_int()) mts = (int)args["mts"].as_int().value();
357 if (args.contains("burst") && args["burst"].is_int()) burst = (int)args["burst"].as_int().value();
358 if (args.contains("depth") && args["depth"].is_int()) depth = (int)args["depth"].as_int().value();
359 if (args.contains("chunks") && args["chunks"].is_int()) chunks = (int)args["chunks"].as_int().value();
360 }
361 parlio_tx_unit_config_t cfg = {};
362 cfg.clk_src = PARLIO_CLK_SRC_DEFAULT;
363 cfg.data_width = (size_t)width;
364 cfg.clk_in_gpio_num = (gpio_num_t)-1;
365 cfg.clk_out_gpio_num = (gpio_num_t)-1;
366 cfg.valid_gpio_num = (gpio_num_t)-1;
367 cfg.trans_queue_depth = (size_t)depth;
368 cfg.max_transfer_size = (size_t)mts;
369 cfg.output_clk_freq_hz = (uint32_t)hz;
370 setParlioRawTxEdge(cfg, 0); // shift_edge where it exists, sample_edge before
371 cfg.bit_pack_order = PARLIO_BIT_PACK_ORDER_MSB;
372 if (burst > 0) cfg.dma_burst_size = (size_t)burst;
373 for (int i = 0; i < 16; ++i) cfg.data_gpio_nums[i] = (gpio_num_t)-1;
374 cfg.data_gpio_nums[0] = (gpio_num_t)pin;
375 parlio_tx_unit_handle_t unit = nullptr;
376 esp_err_t err = parlio_new_tx_unit(&cfg, &unit);
377 if (err != ESP_OK) {
378 response.set("success", false);
379 response.set("error", esp_err_to_name(err));
380 return response;
381 }
382 err = parlio_tx_unit_enable(unit);
383 uint8_t* buf = parlioRawTestState().buffer;
384 for (int i = 0; i < 512; ++i) buf[i] = (uint8_t)pattern;
385 parlio_transmit_config_t tcfg = {};
386 tcfg.idle_value = 0;
387 tcfg.flags.queue_nonblocking = qnb ? 1 : 0;
388 if (burst > 0) {
389 // dma_burst_size is a unit-config field; already applied below via cfg2 hack? No —
390 // it must be set before creation; handled via the 'burst' arg at cfg time.
391 }
392 esp_err_t terr = ESP_OK;
393 const size_t chunk_bytes = 512 / (chunks > 0 ? chunks : 1);
394 // Repeat for ~3 s so the edge-probe task (which re-arms 150 ms
395 // after the RPC) always overlaps live transmissions.
396 for (int rep = 0; rep < 300 && terr == ESP_OK; ++rep) {
397 for (int c = 0; c < chunks && terr == ESP_OK; ++c) {
398 terr = parlio_tx_unit_transmit(unit, buf + c * chunk_bytes, chunk_bytes * 8, &tcfg);
399 }
400 if (terr == ESP_OK) {
401 (void)parlio_tx_unit_wait_all_done(unit, 100);
402 }
403 vTaskDelay(pdMS_TO_TICKS(10));
404 }
405 esp_err_t werr = parlio_tx_unit_wait_all_done(unit, 1000);
406 parlio_tx_unit_disable(unit);
407 parlio_del_tx_unit(unit);
408 response.set("success", true);
409 response.set("enableErr", esp_err_to_name(err));
410 response.set("txErr", esp_err_to_name(terr));
411 response.set("waitErr", esp_err_to_name(werr));
412 return response;
413 });

439 remote.bind("edgeProbeRead", [](const fl::json& args) -> fl::json {
440 fl::json response = fl::json::object();
441 response.set("success", true);
442 fl::u32 count = 0;
443 const fl::u32 *stamps = edgeProbeStamps(count);
444 response.set("edges", static_cast<int64_t>(count));
445 response.set("probeState", static_cast<int64_t>(edgeProbeState()));
446 {
447 const fl::u32 *rl = edgeProbeRmtLive();
448 fl::json live = fl::json::array();
449 for (int k = 0; k < 8; ++k) live.push_back(static_cast<int64_t>(rl[k]));
450 response.set("rmtLive", live);
451 }
452 fl::json runs = fl::json::array();
453 // Convert consecutive timestamps to durations in ns. Entry i
454 // encodes (level << 31) | cycle_stamp.
455 for (fl::u32 i = 1; i < count && i < 200; ++i) {
456 const fl::u32 prev = stamps[i - 1];
457 const fl::u32 cur = stamps[i];
458 const fl::u32 d_cycles = (cur & 0x7FFFFFFFu) - (prev & 0x7FFFFFFFu);
459 const fl::u32 level = prev >> 31;
460 // cycles -> ns via configured CPU MHz
461 const fl::u64 ns = (static_cast<fl::u64>(d_cycles) * 1000u) / edgeProbeCpuMhz();
462 fl::json e = fl::json::object();
463 e.set("level", static_cast<int64_t>(level));
464 e.set("ns", static_cast<int64_t>(ns));
465 runs.push_back(e);
466 }
467 response.set("runs", runs);
468 return response;
469 });
