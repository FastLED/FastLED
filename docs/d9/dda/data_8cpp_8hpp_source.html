70void ChannelData::writeWithPadding(fl::span<u8> dst) FL_NO_EXCEPT {
71 size_t targetSize = dst.size();
72 size_t currentSize = mEncodedData.size();
73
74 // Destination must be at least as large as current data
75 if (targetSize < currentSize) {
76 return; // or throw? For now, silently fail
77 }
78
79 // Create source span from encoded data
80 fl::span<const u8> src(mEncodedData.data(), currentSize);
81
82 if (mPaddingGenerator) {
83 // Use custom padding generator (writes directly to dst)
84 mPaddingGenerator(src, dst);
85 } else {
86 // Default behavior: left-pad with zeros, then memcopy data
87 // Padding bytes go out first to non-existent pixels
88 size_t paddingSize = targetSize - currentSize;
89 if (paddingSize > 0) {
90 fl::fill(dst.begin(), dst.begin() + paddingSize, u8(0));
91 }
92 fl::memcopy(dst.data() + paddingSize, src.data(), currentSize);
93 }
94}
