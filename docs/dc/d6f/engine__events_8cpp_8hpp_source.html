91void EngineEvents::_addListener(Listener *listener, int priority) {
92 if (_hasListener(listener)) {
93 return;
94 }
95 for (auto it = mListeners.begin(); it != mListeners.end(); ++it) {
96 if (it->priority < priority) {
97 // this is now the highest priority in this spot.
98 EngineEvents::Pair pair = EngineEvents::Pair(listener, priority);
99 mListeners.insert(it, pair);
100 return;
101 }
102 }
103 EngineEvents::Pair pair = EngineEvents::Pair(listener, priority);
104 mListeners.push_back(pair);
105}
