98 resolved_encoder = fl::encoder_for<fl::TIMING_UCS7604_800KHZ>();

106 struct DriverEntry {

108 fl::vector<int> lane_sizes;
109 int pin_tx;
110 };

117 response.set("error", "InvalidDriverEntry");

128 response.set("success", false);

131 msg << "drivers[" << i << "] missing 'driver' (string) field";

146 response.set("success", false);
147 response.set("error", "UnknownDriver");

150 response.set("message", msg.str().c_str());
151 return response;

154 // Validate laneSizes
155 if (!entry.contains("laneSizes") || !entry["laneSizes"].is_array()) {
156 response.set("success", false);
157 response.set("error", "MissingLaneSizes");
158 fl::sstream msg;
159 msg << "drivers[" << i << "] missing 'laneSizes' (array) field";
160 response.set("message", msg.str().c_str());

165 fl::vector<int> lane_sizes;
166 for (fl::size li = 0; li < lane_sizes_json.size(); li++) {
167 if (!lane_sizes_json[li].is_int()) {
168 response.set("success", false);
169 response.set("error", "InvalidLaneSizeType");
170 return response;

173 if (size <= 0) {
174 response.set("success", false);
175 response.set("error", "InvalidLaneSize");

179 }

181 // Extract optional pinTx per driver (default: auto-assign consecutive pins)
182 int pin_tx = next_pin;

184 pin_tx = static_cast<int>(entry["pinTx"].as_int().value());
185 }
186
187 DriverEntry de;
188 de.name = driver_name;
189 de.lane_sizes = lane_sizes;
190 de.pin_tx = pin_tx;
191 driver_entries.push_back(de);

193 // Advance pin for next driver (skip past this driver's lanes)
194 next_pin = pin_tx + (int)lane_sizes.size();

197 // ========== EXECUTION ==========
198 uint32_t start_ms = millis();

200 // Step 1: Enable all requested drivers (not exclusive)
201 // First disable all, then enable only the ones we want
202 for (fl::size i = 0; i < mState->drivers_available.size(); i++) {
203 FastLED.setDriverEnabled(mState->drivers_available[i].name.c_str(), false);

209 // Step 2: Create channels for each driver using affinity binding
210 fl::vector<fl::unique_ptr<fl::vector<CRGB>>> all_led_arrays;
211 fl::vector<fl::ChannelPtr> all_channels;

213 for (fl::size di = 0; di < driver_entries.size(); di++) {
214 const auto& de = driver_entries[di];
215 for (fl::size li = 0; li < de.lane_sizes.size(); li++) {
216 auto leds = fl::make_unique<fl::vector<CRGB>>(de.lane_sizes[li]);

218 // Set up channel with typed driver selection (#2459).
219 // The pre-#2459 string `mAffinity` field is gone — translate the
220 // discovered driver name back to a `fl::Bus` enum value.
221 fl::ChannelOptions opts;
222 {
223 const fl::string& n = de.name;
224 if (n == "RMT") opts.mBus = fl::Bus::RMT;
225 else if (n == "PARLIO") opts.mBus = fl::Bus::FLEX_IO;
226 else if (n == "SPI") opts.mBus = fl::Bus::SPI;
227 else if (n == "I2S") opts.mBus = fl::Bus::FLEX_IO;
228 else if (n == "I2S_SPI") opts.mBus = fl::Bus::FLEX_IO;
229 else if (n == "LCD_RGB") { opts.mBus = fl::Bus::FLEX_IO; opts.mBusWhich = 1; }
230 else if (n == "LCD_SPI") opts.mBus = fl::Bus::FLEX_IO;
231 else if (n == "LCD_CLOCKLESS") opts.mBus = fl::Bus::FLEX_IO;
232 else if (n == "UART") opts.mBus = fl::Bus::UART;
233 else if (n == "UART0") { opts.mBus = fl::Bus::UART; opts.mBusWhich = 0; }
234 else if (n == "UART1") { opts.mBus = fl::Bus::UART; opts.mBusWhich = 1; }
235 else if (n == "FLEX_IO") { opts.mBus = fl::Bus::FLEX_IO; opts.mBusWhich = 1; }
236 else if (n == "PIO0") { opts.mBus = fl::Bus::FLEX_IO; opts.mBusWhich = 0; }
237 else if (n == "PIO1") { opts.mBus = fl::Bus::FLEX_IO; opts.mBusWhich = 1; }
238 else if (n == "PIO2") { opts.mBus = fl::Bus::FLEX_IO; opts.mBusWhich = 2; }
239 else if (n == "OBJECT_FLED") opts.mBus = fl::Bus::FLEX_IO;
240 else if (n == "LPUART") opts.mBus = fl::Bus::UART;
241 else if (n == "BIT_BANG") opts.mBus = fl::Bus::BIT_BANG;
242 else if (n == "STUB") opts.mBus = fl::Bus::BIT_BANG;
243 // else: leave Bus::AUTO; priority dispatch will pick.
244 }
245
246 fl::ChannelConfig channel_config(

248 timing_config.timing,

250 RGB,

252 );

254 auto channel = FastLED.add(channel_config);

268 all_led_arrays.push_back(fl::move(leds));
269 }
270 }

272 // Step 3: Set LED data patterns and call show()
273 // Use a simple pattern: fill each driver's LEDs with a known color
274 int array_idx = 0;
275 for (fl::size di = 0; di < driver_entries.size(); di++) {
276 const auto& de = driver_entries[di];
277 for (fl::size li = 0; li < de.lane_sizes.size(); li++) {

279 for (int led = 0; led < de.lane_sizes[li]; led++) {
280 // Pattern: alternate colors per driver for visual distinction
281 if (di == 0) {
282 leds[led] = CRGB(0xFF, 0x00, 0x00); // Red for first driver
283 } else {
284 leds[led] = CRGB(0x00, 0xFF, 0x00); // Green for second driver
285 }
286 }
287 array_idx++;
288 }
289 }

291 // Step 4: Call show() - both drivers transmit simultaneously
292 bool show_success = true;
293 uint32_t show_start = micros();

295 FastLED.show();
296 show_success = FastLED.wait(5000) && show_success; // 5 second timeout for DMA completion
297 }
298 uint32_t show_duration_us = micros() - show_start;
299
300 // Step 5: Validate first driver's output via RX loopback (if available)
301 // Only the first driver (PARLIO) is typically connected to the RX pin
302 bool rx_validation_passed = true;
303 bool rx_validation_attempted = false;
304 fl::vector<fl::RunResult> run_results;
305 const bool is_rp_pio_pair = driver_entries.size() == 2 &&
306 ((driver_entries[0].name == "PIO0" && driver_entries[1].name == "PIO1") ||
307 (driver_entries[0].name == "PIO1" && driver_entries[1].name == "PIO0"));
308
309 // PIO0+PIO1 is a resource-concurrency test. Its two transmitters have
310 // already been exact-byte loopback-validated independently; re-entering
311 // capture after their shared show() creates another channel while both
312 // engines remain configured and tests a different resource topology.
313 if (mState->rx_channel && driver_entries.size() > 0 && !is_rp_pio_pair) {
314 const auto& primary_driver = driver_entries[0];
315
316 // Only attempt RX validation if the primary driver's pin matches our TX pin
317 if (primary_driver.pin_tx == mState->pin_tx) {
318 rx_validation_attempted = true;
319
320 // Create validation config for the primary driver
321 fl::vector<fl::ChannelConfig> tx_configs;
322 int led_array_offset = 0;
323 for (fl::size li = 0; li < primary_driver.lane_sizes.size(); li++) {
324 tx_configs.push_back(fl::ChannelConfig(
325 primary_driver.pin_tx + (int)li,
326 timing_config.timing,
327 fl::span<CRGB>(all_led_arrays[led_array_offset + li]->data(),
328 all_led_arrays[led_array_offset + li]->size()),
329 RGB
330 ));
331 }
332
333 fl::AutoResearchConfig autoresearch_config(
334 timing_config.timing,
335 timing_config.name,
336 tx_configs,
337 primary_driver.name.c_str(),
338 mState->rx_channel,
339 mState->rx_buffer,
340 primary_driver.lane_sizes[0],
341 fl::RxDeviceType::RMT,
342 timing_config.encoder
343 );
344
345 int total_tests = 0;
346 int passed_tests = 0;
347 uint32_t val_show_duration_ms = 0;
348
349 autoResearchChipsetTiming(autoresearch_config, total_tests, passed_tests,
350 val_show_duration_ms, &run_results);
351
352 rx_validation_passed = (total_tests > 0) && (passed_tests == total_tests);
353 }
354 }
355
356 // Step 6: Clean up channels
357 FastLED.clear(ClearFlags::CHANNELS);
358
359 uint32_t duration_ms = millis() - start_ms;
360 int total_tests = 1;
361 int passed_tests = show_success ? 1 : 0;
362 const int primary_tx_pin = driver_entries.empty()
363 ? mState->pin_tx
364 : driver_entries[0].pin_tx;
365 int capture_evidence_bytes = 0;
366 int capture_evidence_raw_edges = 0;

368 const auto& rr = run_results[ri];
369 if (rr.capturedBytes > 0) {
370 capture_evidence_bytes += rr.capturedBytes;
371 }
372 if (rr.rawEdgesAfterWait > capture_evidence_raw_edges) {
373 capture_evidence_raw_edges = rr.rawEdgesAfterWait;
374 }
375 }
376 if (rx_validation_attempted) {
377 total_tests++;
378 if (rx_validation_passed) {
379 passed_tests++;
380 }
381 }
382
383 // ========== RESPONSE ==========
384 response.set("success", true);
385 response.set("passed", show_success && rx_validation_passed);
386 response.set("totalTests", static_cast<int64_t>(total_tests));
387 response.set("passedTests", static_cast<int64_t>(passed_tests));
388 response.set("failedTests", static_cast<int64_t>(total_tests - passed_tests));
389 response.set("requestedTxPin", static_cast<int64_t>(primary_tx_pin));
390 response.set("requestedRxPin", static_cast<int64_t>(mState->pin_rx));
391 response.set("actualTxPin", static_cast<int64_t>(primary_tx_pin));
392 response.set("actualRxPin", static_cast<int64_t>(mState->pin_rx));

397 response.set("captureEvidenceExpected", rx_validation_attempted);
398 response.set("captureEvidenceBytes",
399 static_cast<int64_t>(capture_evidence_bytes));
400 response.set("captureEvidenceRawEdges",
401 static_cast<int64_t>(capture_evidence_raw_edges));
402 response.set("duration_ms", static_cast<int64_t>(duration_ms));
403 response.set("show_duration_us", static_cast<int64_t>(show_duration_us));
404 response.set("show_success", show_success);
405 response.set("concurrent_resource_test", is_rp_pio_pair);
406 response.set("iterations", static_cast<int64_t>(iterations));
407 response.set("rx_validation_attempted", rx_validation_attempted);
408 response.set("rx_validation_passed", rx_validation_passed);
409
410 // List drivers tested
411 fl::json drivers_tested = fl::json::array();
412 for (fl::size i = 0; i < driver_entries.size(); i++) {
413 fl::json drv = fl::json::object();
414 drv.set("driver", driver_entries[i].name.c_str());

423 }
424 response.set("drivers", drivers_tested);
425 response.set("pattern", pattern.c_str());
426
427 return response;
