93inline fl::vector<fl::string> registeredDriverNames() FL_NO_EXCEPT {
94 fl::vector<fl::string> names;

97 for (const auto& info : infos) {

127inline fl::json buildDeviceInfo(fl::u8 instance) FL_NO_EXCEPT {
128 fl::json info = fl::json::object();
129 info.set("bus", "FLEX_IO");
130 info.set("instance", static_cast<int64_t>(instance));
131 // firstDriverName gives the most-preferred registered driver's
132 // name — on classic ESP32 this is typically "I2S_SPI" for the
133 // FLEX_IO slot; on P4/C6/etc. "PARLIO"; on S3 "LCD_CLOCKLESS".
134 info.set("driverName", firstDriverName());
135 // Full driver list so the host can pick the FLEX_IO-bound one by
136 // name if firstDriverName isn't the FLEX_IO owner.
137 fl::json names_arr = fl::json::array();
138 for (const auto& n : registeredDriverNames()) {
139 names_arr.push_back(n);
140 }
141 info.set("registeredDriverNames", names_arr);
142 info.set("silicon", siliconName());
143 info.set("idfVersion", idfVersion());
144 info.set("supportedByBuild",
145 firstDriverName() != fl::string::from_literal("<none>"));
146 return info;
147}
