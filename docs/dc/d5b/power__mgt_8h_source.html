141 constexpr PowerModelRGB toRGB() const {
142 // Distribute white-channel mW evenly across R/G/B. Clamp to 255 (u8
143 // max) on overflow — a wrapping static_cast<u8> here would make the
144 // brightness limiter *less* conservative on the highest-draw configs
145 // (where it most needs to be conservative). The worst-case
146 // under-budget per channel from the floor division is 2 mW.
147 return PowerModelRGB(
148 static_cast<fl::u8>(
149 (red_mW + (white_mW + warm_white_mW) / 3) > 255
150 ? 255 : (red_mW + (white_mW + warm_white_mW) / 3)),
151 static_cast<fl::u8>(
152 (green_mW + (white_mW + warm_white_mW) / 3) > 255
153 ? 255 : (green_mW + (white_mW + warm_white_mW) / 3)),
154 static_cast<fl::u8>(
155 (blue_mW + (white_mW + warm_white_mW) / 3) > 255
156 ? 255 : (blue_mW + (white_mW + warm_white_mW) / 3)),
157 dark_mW,
158 exponent);
159 }
