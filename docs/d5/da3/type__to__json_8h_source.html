29// Explicit unsigned-int specialization. Without this, `json(u32_value)` in
30// the primary template above is ambiguous between `json(int)` and
31// `json(i64)` for a `u32` argument — GCC 15 reports:
32// error: call of overloaded 'json(const long unsigned int&)' is ambiguous
33// Route u32 → i64 explicitly so the RPC framework can carry unsigned 32-bit
34// return values / arg types (e.g. `fl::u32 rgb` in the LPC PWM+DMA
35// AutoResearch RPCs from #3517).
36template <>
37struct TypeToJson<fl::u32> {
38 static json convert(const fl::u32& value) FL_NO_EXCEPT {
39 return json(static_cast<fl::i64>(value));
40 }
41};
