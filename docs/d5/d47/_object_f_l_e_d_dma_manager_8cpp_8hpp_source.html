14// #3416 OF-HIGH-1 / CodeRabbit-flagged: save/restore PRIMASK rather than
15// unconditionally re-enabling interrupts, so we don't clobber the
16// caller's IRQ-disabled state if we're invoked from inside another
17// critical section. Inline asm avoids needing the CMSIS header.
18static inline uint32_t fl_save_primask_disable() {
19 uint32_t pm;
20 __asm volatile ("mrs %0, primask\n\tcpsid i" : "=r"(pm) :: "memory");
21 return pm;
22}
23static inline void fl_restore_primask(uint32_t pm) {
24 __asm volatile ("msr primask, %0" :: "r"(pm) : "memory");
25}

38 if (mCurrentOwner != owner) {
39 fl_restore_primask(pm);

46void ObjectFLEDDmaManager::waitForCompletion() {
47 // Spin-wait for DMA completion with periodic delays.
48 while (!dma3.complete()) {
49 delayMicroseconds(10);
50 }
51 // #3416 OF-MED-6: dma3.complete() reflects DMA major-loop done but
52 // does NOT include the WS2812 reset latch period. If the caller is
53 // about to destroy the LED buffer or hand the line off to another
54 // peripheral, returning here can corrupt the final frame's tail
55 // bits. The destructor wait path adds its own latch margin (see
56 // ObjectFLED::~ObjectFLED) but defensive callers should also gate
57 // on busy()==0 which factors in LATCH_DELAY. Documented here so
58 // the next reader doesn't mistake "DMA done" for "frame done".
59}
