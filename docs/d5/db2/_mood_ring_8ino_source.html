118void setup() {
119 Serial.begin(115200);
120
121 FastLED.addLeds<WS2812, PIN_DATA>(leds, NUM_LEDS)
122 .setCorrection(TypicalLEDStrip)
123 .setScreenMap(screenmap);
124 FastLED.setBrightness(brightness.value());
125
126 // Route audio through FastLED.add() for auto-pump when available.
127 // gAutoPump may only be true when FastLED.add() returned a live
128 // processor; otherwise loop() would skip the manual pump and the
129 // listener would never see a sample.
130 auto input = audio.audioInput();
131 if (input) {
132 gProcessor = FastLED.add(input);
133 if (gProcessor) {
134 gAutoPump = true;
135 fl::printf("MoodRing: audio routed via FastLED.add() (auto-pump)\n");
136 }
137 }
138 if (!gProcessor) {
139 gProcessor = fl::make_shared<fl::audio::Processor>();
140 gAutoPump = false;
141 fl::printf("MoodRing: audio using manual pump (fallback)\n");
142 }
143
144 gListener = fl::make_shared<mood_ring::RoomListener>(gProcessor);
145 gListener->begin();
146
147 Serial.println("MoodRing setup complete");
148}

178void loop() {
179 const fl::u32 now = millis();
180
181 // Manual audio pump fallback (e.g. WASM / when FastLED.add() didn't take).
182 if (!gAutoPump) {
183 fl::audio::Sample sample = audio.next();
184 if (sample.isValid() && listen.value()) {
185 gProcessor->update(sample);
186 }
187 }
188
189 applyUiTuning();
190
191 // Listen, then paint. With listening off every signal eases to calm at
192 // its normal release rate, so the sketch fades into a slow ambient lamp
193 // rather than cutting or going dark.
194 if (listen.value()) {
195 gListener->update(now);
196 } else {
197 gListener->rest(now);
198 }
199 const mood_ring::RoomSense &sense = gListener->sense();
200 gPainter.draw(sense, now, leds);
201
202 // Log regime changes so the listener is observable in the field.
203 static const char *sLastRegime = "";
204 const char *regime = gListener->regimeName();
205 if (regime != sLastRegime) {
206 sLastRegime = regime;
207 fl::printf("MoodRing: regime -> %s\n", regime);
208 }
209 if (debugPrint.value()) {
210 static fl::u32 sFrame = 0;
211 if ((sFrame++ % 30) == 0) printSense(sense);
212 }
213
214 // Brightness.
215 uint8_t finalBrightness;
216 if (autoBrightness.value()) {
217 const float avg = getAverageBrightness(leds);
218 finalBrightness = applyBrightnessCompression(
219 avg, static_cast<uint8_t>(autoBrightnessMax.value()),
220 autoBrightnessLowThreshold.value(),
221 autoBrightnessHighThreshold.value());
222 } else {
223 finalBrightness = static_cast<uint8_t>(brightness.value());
224 }
225 FastLED.setBrightness(finalBrightness);
226 FastLED.show();
227}
