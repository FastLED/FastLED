98 void showPixels(PixelController<RGB_ORDER>& pixels) FL_NO_EXCEPT override {
99 fl::shared_ptr<IChannelDriver> driver = resolveDriver();
100 if (!driver) {
101 return;
102 }
103
104 // (A1) Respect runtime driver enable/disable: ChannelManager is the
105 // single source of truth. A disabled driver means "drop this frame",
106 // not "enqueue and hope" -- silently enqueuing to a disabled driver
107 // is the #2517 silent-drop failure mode.
108 if (!ChannelManager::registry().isDriverEnabled(driver->getName().c_str())) {
109 FL_WARN_ONCE("SlimSpiBridgeController: driver '%s' is disabled - dropping frame",
110 driver->getName().c_str());
111 return;
112 }
113
114 // (A8) If the previous frame's buffer is still in flight, block
115 // (bounded) until the driver reports it can accept more work rather
116 // than racing the driver's in-progress transfer.
117 if (mData->isInUse()) {
118 if (!driver->waitForReady()) {
119 FL_WARN_ONCE("SlimSpiBridgeController: driver '%s' did not become ready in "
120 "time - dropping frame", driver->getName().c_str());
121 return;
122 }
123 }
124
125 // (A5) Publish the pixel byte layout so the driver (and any
126 // downstream padding/diagnostic code) knows the encoded stride.
127 if (this->getRgbww().active()) {
128 mData->setPixelFormat(ChannelPixelFormat::RGBWW);
129 } else if (this->getRgbw().active()) {
130 mData->setPixelFormat(ChannelPixelFormat::RGBW);
131 } else {
132 mData->setPixelFormat(ChannelPixelFormat::RGB);
133 }
134
135 // Dither reseed mirrors Channel::encodeFrame(): only reseed when a
136 // non-zero dither mode is set.
137 if (pixels.ditherActive()) {
138 pixels.reseed_binary_dithering(mDitherPhase);
139 }
140
141 // (A3) Re-encode into the same buffer every frame -- clear() keeps
142 // capacity, so steady-state operation is allocation-free.
143 mData->getData().clear();
144 fl::PixelIterator it(&pixels, this->getRgbw(), this->getRgbww());
145 encode(it, mData->getData());
146
147 driver->enqueue(mData);
148 // Presented, per the driver contract: an accepted enqueue is the
149 // frame -- mirrors Channel::submitFrame()'s ++mDitherPhase placement
150 // (dither only advances when a frame is actually handed to the
151 // driver, not on every showPixels() attempt).
152 ++mDitherPhase;
153 }

162 fl::shared_ptr<IChannelDriver> resolveDriver() FL_NO_EXCEPT {
163#if defined(FASTLED_DISABLE_DYNAMIC_DRIVER) && FASTLED_DISABLE_DYNAMIC_DRIVER
164 return {};
165#else
166 // Build busKey only when we actually need it (kBus != AUTO).
167 fl::string busKey;
168 if (kBus != Bus::AUTO) {
169 busKey = fl::string::from_literal(busDriverName(kBus, B_WHICH, /*spi=*/true));
170 }
171
172 fl::shared_ptr<IChannelDriver> driver =
173 ChannelManager::registry().selectDriverForChannel(mData, busKey);
174
175#if FASTLED_LOG_RUNTIME_ENABLED
176 if (!driver && !mBusWarned) {
177 FL_ERROR("SlimSpiBridgeController: no compatible driver found for bus '%s' - "
178 "cannot transmit", busKey.c_str());
179 mBusWarned = true;
180 }
181#endif // FASTLED_LOG_RUNTIME_ENABLED
182 return driver;
183#endif // !FASTLED_DISABLE_DYNAMIC_DRIVER
184 }

190 static void encode(fl::PixelIterator& it, fl::vector_psram<u8>& out) FL_NO_EXCEPT {
191 switch (CHIPSET) {
192 case fl::SpiChipset::APA102:
193 case fl::SpiChipset::DOTSTAR:
194 case fl::SpiChipset::HD107:

199 case fl::SpiChipset::HD107HD:
200 it.writeAPA102(&out, true);
201 break;
202 case fl::SpiChipset::SK9822:
203 it.writeSK9822(&out, false);
204 break;
205 case fl::SpiChipset::SK9822HD:
206 it.writeSK9822(&out, true);
207 break;
208 case fl::SpiChipset::WS2801:
209 it.writeWS2801(&out);
210 break;
211 case fl::SpiChipset::WS2803:
212 it.writeWS2803(&out);
213 break;
214 case fl::SpiChipset::P9813:
215 it.writeP9813(&out);
216 break;
217 case fl::SpiChipset::LPD8806:
218 it.writeLPD8806(&out);
219 break;
220 case fl::SpiChipset::LPD6803:
221 it.writeLPD6803(&out);
222 break;
223 case fl::SpiChipset::SM16716:
224 it.writeSM16716(&out);
225 break;
226 case fl::SpiChipset::HD108:
227 it.writeHD108(&out, false);
228 break;

234 break;
