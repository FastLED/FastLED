21// Append a u32 little-endian to a byte vector.
22void appendU32LE(fl::vector<fl::u8>& out, fl::u32 v) FL_NO_EXCEPT {
23 out.push_back(static_cast<fl::u8>(v & 0xff));
24 out.push_back(static_cast<fl::u8>((v >> 8) & 0xff));
25 out.push_back(static_cast<fl::u8>((v >> 16) & 0xff));
26 out.push_back(static_cast<fl::u8>((v >> 24) & 0xff));
27}

29// Append the characters of a fl::string to a byte vector.
30void appendString(fl::vector<fl::u8>& out, const fl::string& s) FL_NO_EXCEPT {
31 const fl::size n = s.size();
32 for (fl::size i = 0; i < n; ++i) {
33 out.push_back(static_cast<fl::u8>(s[i]));

37// Build the JSON envelope by concatenating the configured sections under
38// their canonical top-level keys. If no sections are configured the
39// envelope is "{}".
40fl::string buildEnvelope(const fl::string& screenMapJson,
41 const fl::string& channelsJson,
42 const fl::string& videoColorJson) FL_NO_EXCEPT {
43 fl::string out;
44 out += '{';
45 bool first = true;
46 if (!screenMapJson.empty()) {
47 out += "\"map\":";
48 out += screenMapJson;
49 first = false;
50 }
51 if (!channelsJson.empty()) {
52 if (!first) out += ',';
53 out += "\"channels\":";
54 out += channelsJson;
55 first = false;
56 }
57 if (!videoColorJson.empty()) {
58 if (!first) out += ',';
59 out += "\"video\":{\"color\":";
60 out += videoColorJson;
61 out += '}';
62 first = false;
63 }
64 out += '}';
65 return out;
66}

68// Spellings are the ones FLED_FORMAT.md defines. They are written here rather
69// than derived from the enum so that renaming a C++ enumerator cannot silently
70// change the wire format.
71const char* primariesName(ColorPrimaries p) FL_NO_EXCEPT {
72 switch (p) {

75 case ColorPrimaries::Bt2020: return "bt2020";

90// A CIE xy pair, four decimals -- enough for the D65 white point (0.3127,
91// 0.3290) to survive the round trip exactly.
92void appendXy(fl::string& out, float x, float y) FL_NO_EXCEPT {
93 out += '[';
94 out += fl::to_string(x, 4);

100fl::string serializeVideoColor(const VideoColor& c) FL_NO_EXCEPT {
101 fl::string out;
102 out += "{\"primaries\":";
103 const char* named = primariesName(c.primaries);
104 if (named != nullptr) {
105 out += '"';
106 out += named;
107 out += '"';
108 } else {
109 out += "{\"red\":";
110 appendXy(out, c.customPrimaries[0], c.customPrimaries[1]);
111 out += ",\"green\":";
112 appendXy(out, c.customPrimaries[2], c.customPrimaries[3]);
113 out += ",\"blue\":";
114 appendXy(out, c.customPrimaries[4], c.customPrimaries[5]);
115 out += ",\"white\":";
116 appendXy(out, c.customPrimaries[6], c.customPrimaries[7]);
117 out += '}';
118 }
119 out += ",\"transfer\":\"";
120 out += transferName(c.transfer);
121 // v1 defines exactly one value for each of matrix and range; the parser
122 // rejects anything else, so emitting them literally keeps producer and
123 // parser from drifting apart.
124 out += "\",\"matrix\":\"rgb\",\"range\":\"full\"}";
125 return out;
126}

168fl::Fled FledBuilder::build() const FL_NO_EXCEPT {
169 const fl::string envelope =
170 buildEnvelope(mScreenMapJson, mChannelsJson, mVideoColorJson);
171 const fl::u32 jsonLen = static_cast<fl::u32>(envelope.size());
172
173 fl::vector<fl::u8> buf;
174 buf.reserve(12 + envelope.size() + mPayload.size());
175
176 // 12-byte header: 4-byte magic, version, pixel_format, 2 reserved,
177 // u32 little-endian JSON length.
178 buf.push_back('F');
179 buf.push_back('L');
180 buf.push_back('E');
181 buf.push_back('D');
182 buf.push_back(mVersion);
183 buf.push_back(mPixelFormat);
184 buf.push_back(0); // reserved
185 buf.push_back(0); // reserved
186 appendU32LE(buf, jsonLen);
187
188 // JSON envelope.
189 appendString(buf, envelope);
190
191 // Optional frame payload (already a copy held by the builder).
192 for (fl::size i = 0; i < mPayload.size(); ++i) {
193 buf.push_back(mPayload[i]);
194 }
195
196 return fl::Fled::loadFromVector(fl::move(buf));
197}
