70template <typename OutputIterator>
71void buildUCS7604Preamble(OutputIterator out, UCS7604Mode mode,
72 u8 r_current, u8 g_current,
73 u8 b_current, u8 w_current) {
74 // Sync pattern (6 bytes)
75 *out++ = 0xFF;
76 *out++ = 0xFF;
77 *out++ = 0xFF;
78 *out++ = 0xFF;
79 *out++ = 0xFF;
80 *out++ = 0xFF;
81
82 // Header (2 bytes)
83 *out++ = 0x00;
84 *out++ = 0x03;
85
86 // Mode byte
87 *out++ = static_cast<u8>(mode);
88
89 // Current control (4 bytes, 4-bit each, wire order RGBW)
90 *out++ = r_current & 0x0F;
91 *out++ = g_current & 0x0F;
92 *out++ = b_current & 0x0F;
93 *out++ = w_current & 0x0F;
94
95 // Reserved (2 bytes)
96 *out++ = 0x00;
97 *out++ = 0x00;
98}

145template <typename InputIterator, typename OutputIterator>
146void encodeUCS7604_16bit_RGB(InputIterator first, InputIterator last, OutputIterator out,
147 const Gamma8& gamma) FL_NO_EXCEPT {
148 while (first != last) {
149 const auto& pixel = *first;
150
151 // Apply gamma correction for 16-bit output
152 u8 rgb_in[3] = { pixel[0], pixel[1], pixel[2] };
153 u16 rgb_out[3];
154 gamma.convert(fl::span<const u8>(rgb_in, 3), fl::span<u16>(rgb_out, 3));
155
156 // Write big-endian 16-bit values
157 *out++ = rgb_out[0] >> 8;
158 *out++ = rgb_out[0] & 0xFF;
159 *out++ = rgb_out[1] >> 8;
160 *out++ = rgb_out[1] & 0xFF;
161 *out++ = rgb_out[2] >> 8;
162 *out++ = rgb_out[2] & 0xFF;
163 ++first;
164 }
165}

241 if (mode == UCS7604Mode::UCS7604_MODE_8BIT_800KHZ) {

253inline size_t ucs7604FrameBytes(size_t num_leds, UCS7604Mode mode,
254 bool is_rgbw) FL_NO_EXCEPT {
255 constexpr size_t kPreambleLen = 15;
256 const size_t total = kPreambleLen + num_leds * ucs7604BytesPerLed(mode, is_rgbw);
257 // The protocol requires the total to be divisible by 3.
258 return total + (3u - (total % 3u)) % 3u;
259}

272template <typename OutputIterator>
273void encodeUCS7604(PixelIterator& pixel_iter, size_t num_leds, OutputIterator out,
274 UCS7604Mode mode, const UCS7604CurrentControl& current, bool is_rgbw,
275 const Gamma8* gamma = nullptr,
276 bool wide_source = false) FL_NO_EXCEPT {
277 constexpr size_t PREAMBLE_LEN = 15;
278
279#if FL_PLATFORM_HAS_TINY_MEMORY
280 // No wide encoder exists on TINY, so the flag selects nothing there.
281 FL_UNUSED(wide_source);
282#endif
283
284 // Padding comes from ucs7604FrameBytes() so the length callers predict and
285 // the length written here are the same number by construction.
286 const size_t led_data_size = num_leds * ucs7604BytesPerLed(mode, is_rgbw);
287 const size_t padding =
288 ucs7604FrameBytes(num_leds, mode, is_rgbw) - PREAMBLE_LEN - led_data_size;
289
290 // Build preamble (15 bytes) with current control
291 buildUCS7604Preamble(out, mode, current.r, current.g, current.b, current.w);
292
293 // Add padding (0-2 zero bytes)
294 for (size_t i = 0; i < padding; ++i) {
295 *out++ = 0;
296 }
297
298 // Encode LED data based on mode and RGB/RGBW
299 if (mode == UCS7604Mode::UCS7604_MODE_8BIT_800KHZ) {
300 if (is_rgbw) {
301 auto range = makeScaledPixelRangeRGBW(&pixel_iter);
302 encodeUCS7604_8bit_RGBW(range.first, range.second, out);
303 } else {
304 auto range = makeScaledPixelRangeRGB(&pixel_iter);
305 encodeUCS7604_8bit_RGB(range.first, range.second, out);
306 }
307 } else {
308 // 16-bit modes -- fall back to gamma 2.8 if no gamma provided
309 static fl::shared_ptr<const Gamma8> default_gamma;
310 const Gamma8& g = gamma ? *gamma : *(default_gamma ? default_gamma : (default_gamma = Gamma8::getOrCreate(2.8f)));
311 if (is_rgbw) {
312 // RGBW stays on the gamma path: `ColorManagedPixelSource`
313 // delegates its RGBW entry point to the legacy controller, so
314 // there is no wide drive to consume here yet.
315 auto range = makeScaledPixelRangeRGBW(&pixel_iter);
316 encodeUCS7604_16bit_RGBW(range.first, range.second, out, g);
317#if !FL_PLATFORM_HAS_TINY_MEMORY
318 } else if (wide_source) {
319 encodeUCS7604_16bit_RGB_wide(pixel_iter, out);
320#endif
321 } else {
322 auto range = makeScaledPixelRangeRGB(&pixel_iter);
323 encodeUCS7604_16bit_RGB(range.first, range.second, out, g);
324 }
325 }
326}
