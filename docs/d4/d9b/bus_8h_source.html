153inline const char* busName(Bus b, fl::u8 which = 0) FL_NO_EXCEPT {
154 (void)which;
155 switch (b) {
156 case Bus::AUTO: return "AUTO";
157 case Bus::BIT_BANG: return "BIT_BANG";
158 case Bus::UART: return "UART";
159 case Bus::SPI: return "SPI";
160 case Bus::DUAL_SPI: return "DUAL_SPI";
161 case Bus::QUAD_SPI: return "QUAD_SPI";
162 case Bus::OCTAL_SPI: return "OCTAL_SPI";
163 case Bus::RMT: return "RMT";
164 case Bus::FLEX_IO: return "FLEX_IO";
165 }
166 return "";
167}

170inline const char* busDriverName(Bus b, fl::u8 which = 0, bool spi = false) FL_NO_EXCEPT {
171 switch (b) {
172 case Bus::FLEX_IO:
173#if defined(FL_IS_TEENSY_4X)
174 (void)spi;
175 return which == 0 ? "OBJECT_FLED" : "FLEX_IO";
176#elif defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
177 (void)spi;
178 if (which == 0) return "PIO0";
179 if (which == 1) return "PIO1";
180 #if defined(FL_IS_RP2350)
181 if (which == 2) return "PIO2";
182 #endif
183 return "FLEX_IO_UNAVAILABLE";
184#elif defined(FL_IS_ESP32)
185#if FASTLED_HAS_PARLIO
186 (void)which;
187 (void)spi;
188 return "PARLIO";
189#elif FASTLED_HAS_LCD_SPI || FASTLED_HAS_LCD_CAM_CLOCKLESS
190 (void)which;
191 return spi ? "LCD_SPI" : "LCD_CLOCKLESS";
192#elif FASTLED_HAS_I2S_SPI || FASTLED_HAS_I2S_LED
193 (void)which;
194 return spi ? "I2S_SPI" : "I2S";
195#else
196 (void)which;
197 (void)spi;
198 return "FLEX_IO";
199#endif
200#else
201 (void)which;
202 (void)spi;
203 return "FLEX_IO";
204#endif
205 case Bus::AUTO:
206 case Bus::BIT_BANG:
207 case Bus::DUAL_SPI:
208 case Bus::QUAD_SPI:
209 case Bus::OCTAL_SPI:
210 case Bus::RMT:
211 (void)spi;
212 return busName(b, which);
213 case Bus::SPI:
214#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
215 (void)spi;
216 return which == 0 ? "SPI0" : "SPI1";
217#else
218 (void)spi;
219 return busName(b, which);
220#endif
221 case Bus::UART:
222 (void)spi;
223#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
224 return which == 0 ? "UART0" : "UART1";
225#elif defined(FL_IS_TEENSY_4X)
226 (void)which;
227 return "LPUART";
228#else
229 (void)which;
230 return "UART";
231#endif
232 }
233 return "";
234}
