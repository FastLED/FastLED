33// Write `value` as a lowercase hex string into `out`, prefixed with "0x".
34// Returns bytes written (always <= 10, since u32 max is 0xFFFFFFFF = 8 hex
35// digits + 2 prefix chars). Does not write a NUL. Truncates if out is small.
36inline fl::size writeHex(fl::span<char> out, fl::u32 value) FL_NO_EXCEPT {
37 char tmp[10];
38 tmp[0] = '0';
39 tmp[1] = 'x';
40 static const char hex[] = "0123456789abcdef";
41 for (int i = 7; i >= 0; --i) {
42 tmp[2 + (7 - i)] = hex[(value >> (i * 4)) & 0xFu];
43 }
44 fl::size n = (out.size() < 10) ? out.size() : 10;
45 for (fl::size i = 0; i < n; ++i) out[i] = tmp[i];
46 return n;
47}
48
49// Copy a string_view into out[]. Returns bytes copied, truncated to out.size().
50inline fl::size writeView(fl::span<char> out, fl::string_view sv) FL_NO_EXCEPT {
51 fl::size n = (out.size() < sv.size()) ? out.size() : sv.size();
52 for (fl::size i = 0; i < n; ++i) out[i] = sv[i];
53 return n;
54}

69fl::size ResetInfo::describe(fl::span<char> out, bool verbose) const FL_NO_EXCEPT {
70 fl::size n = 0;
71 // <cause>
72 n += platforms::writeView(out.subspan(n), causeName());
73 // ( <sub> ) if subcauseId != 0
74 if (subcauseId != 0) {
75 fl::string_view sub = subcauseName();
76 if (sub.size() != 0) {
77 n += platforms::writeView(out.subspan(n), fl::string_view(" ("));
78 n += platforms::writeView(out.subspan(n), sub);
79 n += platforms::writeChar(out.subspan(n), ')');
80 }
81 }
82 if (verbose) {
83 n += platforms::writeView(out.subspan(n), fl::string_view(" raw="));
84 n += platforms::writeHex(out.subspan(n), rawRegister);
85 }
86 return platforms::writeNulIfRoom(out, n);
87}

145// Counts simultaneously-alive ScopedWatchdog instances. >1 is almost always
146// a programmer error: nested guards mean the lazy-init's timeout argument
147// of the second instance is silently ignored, and timeout semantics across
148// the two scopes become ambiguous. Singleton-style accessor (function-local
149// static) so the counter has well-defined lifetime even if a guard is
150// constructed before main().
151inline int& scopedWatchdogActiveCount() FL_NO_EXCEPT {
152 static int count = 0;
153 return count;
154}
155
156// First-init helper, used by ScopedWatchdog's constructor. Runs only once
157// per process. Single-threaded loop() is the canonical caller — no atomics
158// are needed for the guard byte on supported MCUs.
159inline void scopedWatchdogFirstInit(fl::u32 timeout_ms) FL_NO_EXCEPT {
160 Watchdog& wdt = Watchdog::instance();
161 wdt.begin(timeout_ms);
162
163 ResetInfo info = wdt.lastResetInfo();
164 if (info.cause == ResetCause::WATCHDOG ||
165 info.cause == ResetCause::PANIC ||
166 info.cause == ResetCause::LOCKUP) {
167 char buf[128];
168 fl::size n = info.describe(fl::span<char>(buf, sizeof(buf)), /*verbose=*/true);
169 scopedWatchdogPrintLine(fl::string_view("[FastLED.watchdog] recovered from:"));
170 scopedWatchdogPrintLine(fl::string_view(buf, n));
171
172 if (wdt.hasCrashReport()) {
173 // Hook for Tier 2 platforms. Future revision: also emit a
174 // multi-line CrashReport::describe() result here. For now the
175 // bare cause + raw register is the documented contract.
176 }
177 scopedWatchdogPause3s();
178 }
179 scopedWatchdogPrintLine(fl::string_view("[FastLED.watchdog] armed via FL_WATCHDOG_AUTO()"));
180}

184ScopedWatchdog::ScopedWatchdog(fl::u32 timeout_ms) FL_NO_EXCEPT {
185 static bool sInitialized = false;
186 if (!sInitialized) {
187 sInitialized = true;
188 platforms::scopedWatchdogFirstInit(timeout_ms);
189 }
190
191 // Single-instance enforcement. A second simultaneously-alive ScopedWatchdog
192 // is almost always a bug — the timeout argument here was silently ignored
193 // (only the first instance gets to arm the WDT). Warn once per process so
194 // the developer notices, but still feed so the program survives.
195 int& count = platforms::scopedWatchdogActiveCount();
196 if (count >= 1) {
197 static bool sWarnedOnce = false;
198 if (!sWarnedOnce) {
199 sWarnedOnce = true;
200 platforms::scopedWatchdogPrintLine(fl::string_view(
201 "[FastLED.watchdog] WARN: nested FL_WATCHDOG_AUTO() detected — "
202 "only one scoped guard should be alive at a time"));
203 }
204 }
205 ++count;
206
207 Watchdog::instance().feed();
208}
