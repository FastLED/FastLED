82void dumpRawEdgeTiming(fl::shared_ptr<fl::RxChannel> rx_channel,
83 const fl::ChipsetTimingConfig& timing,
84 fl::EdgeRange range) {
85 if (!rx_channel) {
86 AR_FL_WARN("[RAW EDGE TIMING] ERROR: RX channel is null");
87 return;
88 }
89
90 // Heap-allocated edge buffer (was a 2 KB FixedVector on the stack —
91 // the capture path already runs the loopTask close to its limit,
92 // FastLED#3569).
93 fl::vector<fl::EdgeTime> edges;
94 size_t buffer_size = range.count < 256 ? range.count : 256;
95 edges.resize(buffer_size); // Default initializes to EdgeTime()
96
97 // Get edges starting at offset
98 size_t edge_count = rx_channel->getRawEdgeTimes(edges, range.offset);
99
100 if (edge_count == 0) {
101 AR_FL_WARN("[RAW EDGE TIMING] WARNING: No edge data captured at offset " << range.offset);
102 return;
103 }
104
105 // Calculate actual range printed (start from offset)
106 size_t start_idx = range.offset;
107 size_t end_idx = range.offset + edge_count;
108
109 // Single-line collated edge dump. Was one FL_WARN per edge (~32-256
110 // lines per call) which dominated the Teensy 4 UART TX FIFO on every
111 // decode-failure path. Compact format keeps the diagnostic value
112 // (each edge as `H580` or `L640`) while shipping in a single WS
113 // frame. See fbuild#755 follow-up FL_WARN audit.
114 {
115 fl::sstream edge_dump;
116 edge_dump << "[RAW EDGES " << start_idx << ".." << (end_idx - 1) << "]";
117 for (size_t i = 0; i < edge_count; i++) {
118 edge_dump << " " << (edges[i].high ? "H" : "L") << edges[i].ns;
119 }
120 AR_FL_WARN(edge_dump.str());
121 }
122
123 // Pattern analysis (only if showing edges from start)
124 if (range.offset == 0 && edge_count >= 16) {
125 // Calculate expected timings from config (3-phase to 4-phase conversion)
126 uint32_t expected_bit0_high = timing.t1_ns;
127 uint32_t expected_bit0_low = timing.t2_ns + timing.t3_ns;
128 uint32_t expected_bit1_high = timing.t1_ns + timing.t2_ns;
129 uint32_t expected_bit1_low = timing.t3_ns;
130
131 const uint32_t tolerance = 150;
132
133 bool has_short_high = false, has_long_high = false;
134 bool has_short_low = false, has_long_low = false;
135
136 for (size_t i = 0; i < edge_count; i++) {
137 uint32_t ns = edges[i].ns;
138 if (edges[i].high) {
139 if (ns >= expected_bit0_high - tolerance && ns <= expected_bit0_high + tolerance)
140 has_short_high = true;
141 if (ns >= expected_bit1_high - tolerance && ns <= expected_bit1_high + tolerance)
142 has_long_high = true;
143 } else {
144 if (ns >= expected_bit1_low - tolerance && ns <= expected_bit1_low + tolerance)
145 has_short_low = true;
146 if (ns >= expected_bit0_low - tolerance && ns <= expected_bit0_low + tolerance)
147 has_long_low = true;
148 }
149 }
150
151 fl::sstream ss;
152 ss << "\n[RAW EDGE TIMING] Pattern Analysis:\n";
153 ss << " Short HIGH (~" << expected_bit0_high << "ns, Bit 0): " << (has_short_high ? "FOUND ✓" : "MISSING ✗") << "\n";
154 ss << " Long HIGH (~" << expected_bit1_high << "ns, Bit 1): " << (has_long_high ? "FOUND ✓" : "MISSING ✗") << "\n";
155 ss << " Short LOW (~" << expected_bit1_low << "ns, Bit 1): " << (has_short_low ? "FOUND ✓" : "MISSING ✗") << "\n";
156 ss << " Long LOW (~" << expected_bit0_low << "ns, Bit 0): " << (has_long_low ? "FOUND ✓" : "MISSING ✗");
157 AR_FL_WARN(ss.str());
158
159 if (has_short_high && has_long_high && has_short_low && has_long_low) {
160 AR_FL_WARN("\n[RAW EDGE TIMING] ✓ Encoder appears to be working correctly (varied timing patterns)");
161 } else if (!has_short_high && !has_long_high) {
162 ss.clear();
163 ss << "[RAW EDGE TIMING] ✗ ENCODER BROKEN: No valid HIGH pulses detected!\n";
164 ss << "[RAW EDGE TIMING] Possible causes:\n";
165 ss << "[RAW EDGE TIMING] 1. Encoder not reading pixel buffer data\n";
166 ss << "[RAW EDGE TIMING] 2. Bytes encoder state machine stuck\n";
167 ss << "[RAW EDGE TIMING] 3. Data pointer not passed correctly to encoder";
168 FL_ERROR(ss.str());
169 } else if (!has_short_low && !has_long_low) {
170 // Use FL_WARN to avoid triggering bash autoresearch early exit
171 AR_FL_WARN("[RAW EDGE TIMING] ✗ ENCODER BROKEN: No valid LOW pulses detected!");
172 } else {
173 AR_FL_WARN("[RAW EDGE TIMING] ⚠ Partial pattern match - encoder may have issues");
174 }
175 }
176 AR_FL_WARN("");
177}

252// Decode SPI bit stream from raw RMT edges into LED RGB bytes
253// The SPI data pin carries APA102 encoded data clocked by PCLK.
254// RMT RX captures edges on the data pin. Each edge duration / bit_period_ns
255// gives the number of consecutive same-value bits.
256// Returns number of LED RGB bytes written to rx_buffer, or 0 on error.
257static size_t decodeSpiEdges(fl::shared_ptr<fl::RxChannel> rx_channel,
258 fl::span<uint8_t> rx_buffer,
259 uint32_t clock_hz) {
260 if (!rx_channel || clock_hz == 0) {
261 AR_FL_WARN("[SPI DECODE] Invalid parameters");
262 return 0;
263 }
264
265 const uint32_t bit_period_ns = static_cast<uint32_t>(1000000000ULL / clock_hz);
266 const uint32_t half_bit_ns = bit_period_ns / 2;
267 AR_FL_WARN("[SPI DECODE] clock=" << clock_hz << " Hz, bit_period=" << bit_period_ns << " ns");
268
269 // Read raw edges (up to 4096 to handle large strips)
270 constexpr size_t MAX_EDGES = 4096;
271 fl::vector<fl::EdgeTime> edges(MAX_EDGES);
272 fl::span<fl::EdgeTime> edge_span(edges.data(), edges.size());
273 size_t edge_count = rx_channel->getRawEdgeTimes(edge_span, 0);
274
275 if (edge_count == 0) {
276 AR_FL_WARN("[SPI DECODE] No edges captured");
277 return 0;
278 }
279 AR_FL_WARN("[SPI DECODE] Captured " << edge_count << " edges");
280
281 // Reconstruct bit stream from edges
282 // Each edge has a level (high/low) and duration in ns.
283 // Number of bits = round(duration_ns / bit_period_ns)
284 fl::vector<uint8_t> bits;
285 bits.reserve(edge_count * 4); // Rough estimate
286
287 for (size_t i = 0; i < edge_count; i++) {
288 uint32_t duration = edges[i].ns;
289 uint32_t num_bits = (duration + half_bit_ns) / bit_period_ns;
290 if (num_bits == 0) num_bits = 1; // At least 1 bit per edge
291 uint8_t bit_val = edges[i].high ? 1 : 0;
292 for (uint32_t b = 0; b < num_bits; b++) {
293 bits.push_back(bit_val);
294 }
295 }
296
297 AR_FL_WARN("[SPI DECODE] Reconstructed " << bits.size() << " bits");
298
299 if (bits.size() < 32) {
300 AR_FL_WARN("[SPI DECODE] Too few bits for APA102 frame");
301 return 0;
302 }
303
304 // Convert bits to bytes (MSB first)
305 size_t total_bytes = bits.size() / 8;
306 fl::vector<uint8_t> raw_bytes(total_bytes);
307 for (size_t i = 0; i < total_bytes; i++) {
308 uint8_t byte_val = 0;
309 for (int bit = 7; bit >= 0; bit--) {
310 size_t bit_idx = i * 8 + (7 - bit);
311 if (bit_idx < bits.size() && bits[bit_idx]) {
312 byte_val |= (1 << bit);
313 }
314 }
315 raw_bytes[i] = byte_val;
316 }
317
318 AR_FL_WARN("[SPI DECODE] Decoded " << total_bytes << " raw bytes");
319
320 // Log first few bytes for debugging
321 {
322 fl::sstream dbg;
323 dbg << "[SPI DECODE] First bytes:";
324 size_t show = total_bytes < 20 ? total_bytes : 20;
325 for (size_t i = 0; i < show; i++) {
326 dbg << " 0x";
327 uint8_t hi = raw_bytes[i] >> 4;
328 uint8_t lo = raw_bytes[i] & 0xF;
329 dbg << (char)(hi < 10 ? '0' + hi : 'A' + hi - 10);
330 dbg << (char)(lo < 10 ? '0' + lo : 'A' + lo - 10);
331 }
332 AR_FL_WARN(dbg.str());
333 }
334
335 // Find APA102 start frame (4 bytes of 0x00)
336 // The RMT may not capture the start frame since it's all zeros and
337 // the idle state is also LOW. In that case, the first captured data
338 // is the first LED frame header (0xE0 | brightness).
339 size_t data_start = 0;
340
341 // Check if we captured the start frame
342 if (total_bytes >= 4 && raw_bytes[0] == 0x00 && raw_bytes[1] == 0x00 &&
343 raw_bytes[2] == 0x00 && raw_bytes[3] == 0x00) {
344 data_start = 4; // Skip start frame
345 AR_FL_WARN("[SPI DECODE] Found start frame at offset 0");
346 } else {
347 // No start frame captured (RMT started at first edge = first HIGH bit)
348 // First byte should be 0xE0|brightness or 0xFF
349 AR_FL_WARN("[SPI DECODE] No start frame (first edge = first LED data)");
350 }
351
352 // Extract LED RGB data from APA102 frames
353 // Each LED: [0xE0|brightness] [color0] [color1] [color2] (4 bytes)
354 // With EOrder=RGB in the validation config, the wire order after the
355 // brightness header is [R, G, B] — we copy these directly to rx_buffer.
356 // NOTE: The APA102 end frame (all 0xFF) also has (0xFF & 0xE0) == 0xE0,
357 // so we cannot distinguish end frame from a max-brightness LED purely
358 // from the header byte. We stop after we've exhausted captured data.
359 size_t led_bytes_written = 0;
360 size_t pos = data_start;
361
362 while (pos + 4 <= total_bytes) {
363 uint8_t header = raw_bytes[pos];
364
365 // Check for valid LED frame header (top 3 bits must be 111)
366 if ((header & 0xE0) != 0xE0) {
367 AR_FL_WARN("[SPI DECODE] End of LED data at byte " << pos
368 << " (header=0x" << ((header >> 4) < 10 ? '0' + (header >> 4) : 'A' + (header >> 4) - 10)
369 << ((header & 0xF) < 10 ? '0' + (header & 0xF) : 'A' + (header & 0xF) - 10) << ")");
370 break;
371 }
372
373 uint8_t c0 = raw_bytes[pos + 1];
374 uint8_t c1 = raw_bytes[pos + 2];
375 uint8_t c2 = raw_bytes[pos + 3];
376
377 if (led_bytes_written + 3 <= rx_buffer.size()) {
378 rx_buffer[led_bytes_written + 0] = c0;
379 rx_buffer[led_bytes_written + 1] = c1;
380 rx_buffer[led_bytes_written + 2] = c2;
381 led_bytes_written += 3;
382 } else {
383 AR_FL_WARN("[SPI DECODE] rx_buffer full at LED " << (led_bytes_written / 3));
384 break;
385 }
386 pos += 4;
387 }
388
389 size_t num_leds = led_bytes_written / 3;
390 AR_FL_WARN("[SPI DECODE] Extracted " << num_leds << " LEDs (" << led_bytes_written << " RGB bytes)");
391 return led_bytes_written;
392}

421// Capture transmitted LED data via RX loopback
422// - rx_channel: Shared pointer to RX device (persistent across calls)
423// - rx_buffer: Buffer to store received bytes
424// - expected_data_bytes: bytes in the current frame
425// - tx_pin: GPIO driven by the tested TX controller
426// - timing: Chipset timing configuration for RX decoder
427// - driver_name: Name of the TX driver being tested (e.g., "RMT", "PARLIO")
428// Returns number of bytes captured, or 0 on error
429size_t capture(fl::shared_ptr<fl::RxChannel> rx_channel,
430 fl::span<uint8_t> rx_buffer,
431 size_t expected_data_bytes,
432 int tx_pin,
433 const fl::ChipsetTimingConfig& timing,
434 const char* driver_name,
435 fl::RunResult* diagnostics) {
436 // The RPC runner disables log output for normal tests. Keep successful
437 // captures allocation-quiet too: constructing diagnostic strings and
438 // copying edge snapshots on every frame defeats that guard during a
439 // repeated-frame soak. Failure paths still collect full evidence below.
440 const bool logs_enabled = fl::getLogLevel() >=
441 static_cast<fl::u8>(fl::LogLevel::FL_LOG_LEVEL_WARN);
442 if (diagnostics) {
443 diagnostics->captureWaitResult = -1;
444 diagnostics->rawEdgesAfterWait = 0;
445 diagnostics->rpPioGpioTransitions = -1;
446 diagnostics->rxBeginError.clear();
447 }
448
449 if (!rx_channel) {
450 FL_ERROR("RX channel is null");
451 return 0;
452 }
453
454 // Clear RX buffer
455 fl::memset(rx_buffer.data(), 0, rx_buffer.size());
456
457 // Prepare RX config (but don't arm yet to avoid locking TX resources)
458 fl::RxChannelConfig rx_config(rx_channel->getPin());
459 rx_config.hz = 40000000; // 40MHz for high-precision LED timing capture
460
461 // Buffer size: 1 LED byte = 8 bits = 8 RMT symbols
462 // UART with TX inversion produces standard WS2812 waveform, same symbol count
463 bool is_uart_driver = (fl::strcmp(driver_name, "UART") == 0)
464 || (fl::strcmp(driver_name, "LPUART") == 0)
465 || (fl::strcmp(driver_name, "UART0") == 0)
466 || (fl::strcmp(driver_name, "UART1") == 0);
467 bool is_object_fled_driver = (fl::strcmp(driver_name, "OBJECT_FLED") == 0);
468 rx_config.edge_capacity = fl::validation::captureEdgeCapacity(
469 rx_buffer.size(), expected_data_bytes, rx_channel->backend());
470 if (rx_config.edge_capacity == 0) {
471 FL_ERROR("[CAPTURE] RX edge-capacity overflow");
472 return 0;
473 }
474#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
475 // RP's platform-default RX resolves to PIO, while RxChannel retains the
476 // requested PLATFORM_DEFAULT enum. Recognize both that public default and
477 // an explicit PIO backend so the DMA capture is sized per frame.
478 const bool is_rp_pio_capture =
479 rx_channel->backend() == fl::RxBackend::PLATFORM_DEFAULT ||
480 rx_channel->backend() == fl::RxBackend::PIO ||
481 fl::strcmp(driver_name, "PIO0") == 0 || fl::strcmp(driver_name, "PIO1") == 0 ||
482 fl::strcmp(driver_name, "PIO2") == 0;
483 if (is_rp_pio_capture) {
484 // The shared result buffer is intentionally sized for the largest
485 // AutoResearch run, not this frame. PIO DMA stores one word for each
486 // high and low phase, so using that whole buffer here would request
487 // several MiB on a Pico and leave DMA with an invalid destination.
488 constexpr size_t kPioPhasesPerDataByte = 16;
489 // The guard must cover the headroom multiplier too: checking only
490 // against kPioPhasesPerDataByte would let an oversized byte count
491 // wrap during the *2 below and produce a small `requested` that
492 // sails past the clamp into RpPioRxDevice::begin().
493 constexpr size_t kPioPhaseHeadroomNumerator = 2;
494 constexpr size_t kPioPhasesPerDataByteWithHeadroom =
495 kPioPhasesPerDataByte * kPioPhaseHeadroomNumerator;
496 if (expected_data_bytes >
497 (static_cast<size_t>(-1) - 1u) / kPioPhasesPerDataByteWithHeadroom) {
498 FL_ERROR("[CAPTURE] PIO edge-capacity overflow");
499 return 0;
500 }
501 // 16 phases per byte is a *clean* frame: 8 bits x (high + low). Slower
502 // chipsets exceed it -- a 400 kHz capture overflows and is discarded
503 // (RxWaitResult::BUFFER_OVERFLOW), because long runs do not always
504 // land in one entry. Ask for headroom, then clamp to the static pool
505 // so the request can never exceed the FixedVector backing it:
506 // over-requesting authorises appendDuration() to write past the end.
507 size_t requested =
508 expected_data_bytes * kPioPhasesPerDataByteWithHeadroom + 1u;
509 if (requested > fl::kRpPioRxEdgeCapacity) {
510 requested = fl::kRpPioRxEdgeCapacity;
511 }
512 rx_config.edge_capacity = requested;
513 }
514#endif
515
516 bool is_rmt_driver = (fl::strcmp(driver_name, "RMT") == 0);
517 const int rx_pin = rx_channel->getPin();
518 // ESP-IDF's RMT io_loop_back path requires TX and RX to share one GPIO.
519 // AutoResearch normally discovers a physical TX->RX jumper on two
520 // different pins; enabling io_loop_back in that case disconnects the RX
521 // channel from the external GPIO path and yields a successful wait with
522 // zero edges on ESP32-C6. Keep the ESP32-S3 same-pin workaround, but use
523 // the physical jumper whenever TX and RX are distinct.
524 const bool use_rmt_internal_loopback =
525 fl::validation::useRmtInternalLoopback(
526 is_rmt_driver, tx_pin, rx_pin);
527 rx_config.io_loop_back = use_rmt_internal_loopback;
528 // RX DMA streaming: extends capture past the non-DMA cap by sizing
529 // mem_block_symbols = 14336 so ESP-IDF allocates ~14 DMA descriptor
530 // nodes (each 4092 bytes), yielding a 57 KB user-buffer cap — enough to
531 // capture ~583 WS2812B LEDs in a single rmt_receive() call. Safe for
532 // non-RMT TX paths (SPI, PARLIO, I2S, UART, LCD_*) that don't contend
533 // for the shared ESP32-S3 DMA slot. RMT loopback stays non-DMA.
534 // See issue #2254.
535 rx_config.use_dma = !is_rmt_driver;
536 // Preserve the channel's explicitly-selected capture backend —
537 // RxChannel::begin() recreates the device when the backend field
538 // differs, and this config is built fresh (default =
539 // PLATFORM_DEFAULT), which silently reverted rxBackend overrides
540 // like I2S_RX back to RMT (FastLED#3576 Phase 3).
541 rx_config.backend = rx_channel->backend();
542 if (use_rmt_internal_loopback) {
543 AR_FL_WARN("[CAPTURE] RMT TX -> RMT RX: Same-pin internal loopback enabled");
544 } else {
545 // The RX peripheral is platform-dependent (RMT on ESP32, FlexPWM on
546 // Teensy 4, LPC_SCT on LPC845, …). Don't claim "RMT RX" on platforms
547 // that have no RMT — that label burned an hour of Teensy-4 debug
548 // (FastLED#3059). Just say "external RX" so the label stays correct
549 // regardless of backend.
550 if (logs_enabled) {
551 AR_FL_WARN("[CAPTURE] " << driver_name
552 << " TX GPIO " << tx_pin << " -> RX GPIO " << rx_pin
553 << ": External jumper "
554 << "(io_loop_back=false, use_dma="
555 << (rx_config.use_dma ? "true" : "false") << ")");
556 }
557 }

559 // Driver-aware capture strategy:
560 // - RMT: Two-TX approach (ESP32-S3 workaround - TX GPIO blocked when RX active)
561 // - PARLIO/SPI/other: Single-TX approach (arm RX first, then TX)
562
563 // TX wait timeout: 1 second max per frame - prevents infinite hang if driver stalls
564 // Even a 10000-LED strip at 800kbps takes only ~300ms, so 1s is very safe
565 const uint32_t TX_WAIT_TIMEOUT_MS = 1000;
566
567 if (is_rmt_driver) {
568 // RMT: Two-TX approach for ESP32-S3 compatibility
569 // First TX without RX armed (diagnostics), then arm RX, then second TX
570 if (logs_enabled) {
571 AR_FL_WARN("[CAPTURE] RMT: Two-TX approach (ESP32-S3 workaround)");
572 }
573 FastLED.show();
574 if (!FastLED.wait(TX_WAIT_TIMEOUT_MS)) {
575 FL_ERROR("[CAPTURE] TX wait timeout (pre-arm) - driver may be stalled");
576 return 0;
577 }
578
579 // Arm RX for capture
580 if (!rx_channel->begin(rx_config)) {
581 if (diagnostics) diagnostics->rxBeginError = rx_channel->lastBeginError();
582 FL_ERROR("Failed to arm RX receiver");
583 return 0;
584 }
585 // Second TX with RX armed
586 FastLED.show();
587 if (!FastLED.wait(TX_WAIT_TIMEOUT_MS)) {
588 FL_ERROR("[CAPTURE] TX wait timeout (capture) - driver may be stalled");
589 return 0;
590 }
591 } else {
592 // Non-RMT (PARLIO, SPI, etc.): Single-TX approach
593 if (!rx_channel->begin(rx_config)) {
594 if (diagnostics) diagnostics->rxBeginError = rx_channel->lastBeginError();
595 AR_FL_WARN("[CAPTURE] RX begin() failed for pin " << rx_channel->getPin());
596 return 0;
597 }
598 if (logs_enabled) {
599 AR_FL_WARN("[CAPTURE] RX armed, calling FastLED.show()...");
600 }
601
602#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
603 fl::atomic_int rising_edges(0);
604 fl::isr::handle edge_counter;
605 bool edge_counter_attached = false;
606 const bool measure_rp_pio =
607 (fl::strcmp(driver_name, "PIO0") == 0 || fl::strcmp(driver_name, "PIO1") == 0 ||
608 fl::strcmp(driver_name, "PIO2") == 0) &&
609 diagnostics != nullptr;
610 if (measure_rp_pio) {
611 // FastLED.show() drives the channel engine until it reaches its
612 // terminal state, so polling GPIO after show() is too late to
613 // observe a short clockless frame. Count rising edges through the
614 // shared FastLED ISR dispatcher while show() is running instead.
615 // This is deliberately a binary physical oracle: an IRQ count
616 // above zero proves the jumper carries TX activity, while the PIO
617 // RX backend remains responsible for exact-byte verification.
618 fl::isr::config edge_config;
619 edge_config.handler = countRpPioRxRisingEdges;
620 edge_config.user_data = &rising_edges;
621 edge_config.flags = fl::isr::ISR_FLAG_EDGE_RISING;
622 edge_counter_attached = fl::isr::attach_external_handler(
623 static_cast<fl::u8>(rx_channel->getPin()), edge_config, &edge_counter) == 0;
624 }
625#endif
626 FastLED.show();
627 if (logs_enabled) {
628 AR_FL_WARN("[CAPTURE] FastLED.show() returned, calling wait...");
629 }
630#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
631 if (edge_counter_attached && edge_counter.is_valid()) {
632 fl::isr::detach_handler(edge_counter);
633 }
634 if ((fl::strcmp(driver_name, "PIO0") == 0 || fl::strcmp(driver_name, "PIO1") == 0 ||
635 fl::strcmp(driver_name, "PIO2") == 0) &&
636 diagnostics != nullptr) {
637 diagnostics->rpPioGpioTransitions = edge_counter_attached ? rising_edges.load() : -1;
638 }
639#endif
640 if (!FastLED.wait(TX_WAIT_TIMEOUT_MS)) {
641 if (is_object_fled_driver) {
642 fl::objectFledDiagnosticsRecord("afterFastLedWaitTimeout");
643 }
644 AR_FL_WARN("[CAPTURE] FastLED.wait() timed out");
645 return 0;
646 }
647 if (is_object_fled_driver) {
648 fl::objectFledDiagnosticsRecord("afterFastLedWait");
649 }
650 if (logs_enabled) {
651 AR_FL_WARN("[CAPTURE] FastLED.wait() done");
652 }
653 }

658 // Wait for RX completion
659 // WS2812B: ~30μs per LED → 3000 LEDs = 90ms, use 150ms
660 // UART with inverted TX produces same waveform timing as WS2812
661 const uint32_t rx_wait_ms = is_uart_driver ? 500 : 150;
662 if (logs_enabled) {
663 AR_FL_WARN("[CAPTURE] Waiting for RX completion (" << rx_wait_ms
664 << "ms timeout)...");
665 }
666 auto wait_result = rx_channel->wait(rx_wait_ms);
667 if (diagnostics) {
668 diagnostics->captureWaitResult = static_cast<int>(wait_result);
669 diagnostics->decodeOutputCapacity = static_cast<int>(rx_buffer.size());
670 if (wait_result != fl::RxWaitResult::SUCCESS) {
671 collectRawEdgeDiagnostics(rx_channel, diagnostics);
672 }
673 }
674 if (logs_enabled) {
675 AR_FL_WARN("[CAPTURE] RX wait returned: "
676 << static_cast<int>(wait_result));
677 }
678
679 if (wait_result != fl::RxWaitResult::SUCCESS) {
680 AR_FL_WARN("RX wait failed (timeout or no data received)");
681 fl::sstream ss;
682 ss << "\n⚠️ TROUBLESHOOTING:\n";
683 ss << " 1. Connect physical jumper wire from TX GPIO to RX GPIO " << rx_channel->getPin() << "\n";
684 ss << " 2. Check that both TX and RX pins are correctly configured\n";
685 ss << " 3. Verify the GPIO connection is working (GPIO baseline test should pass)\n";
686 ss << " 4. RMT internal loopback requires TX and RX to use the same GPIO";
687 AR_FL_WARN(ss.str());
688 if (!is_uart_driver) {
689 return 0;
690 }
691 // UART: still attempt decode with captured edges (timeout may be expected)
692 AR_FL_WARN("[CAPTURE] UART: attempting decode with captured edges despite timeout");
693 }

695 // UART TX with inversion puts a WS2812-shaped waveform on the wire, but
696 // quantized onto the UART pulse grid -- exactly like the wave8 and wave3
697 // cases handled above, and for the same reason: the decoder has to be told
698 // what the transmitter actually emitted, not what the datasheet says.
699 //
700 // This used to hardcode T0H=250 / T1H=750 / T1L=500, the shape an ESP32
701 // UART produces at 4 Mbps for a WS2812. That is right for one chipset on
702 // one platform and wrong everywhere else. At 400 kHz the real wire is
703 // 625 / 1250 / 1250, every phase of which falls outside those hardcoded
704 // windows -- so a correct frame decoded to nothing. See FastLED#4379.
705 if (is_uart_driver) {
706 AR_FL_WARN("[CAPTURE] UART (inverted TX): using standard WS2812 decoder with UART timing...");
707 // Take the geometry the engine recorded when it transmitted, rather
708 // than re-deriving it. Which geometry gets chosen depends on the
709 // backend's *maximum* baud, not on anything visible here, and the
710 // achieved baud is the wrong input: it is allowed to sit up to 1%
711 // under the requested one, and fed back as a ceiling that shortfall
712 // makes the transmitted geometry look infeasible and selects the
713 // other. Re-deriving from the generic ceiling is the fallback for
714 // backends that do not report one.
715 fl::ChipsetTiming uart_timing{0, 0, 0, 0, nullptr};
716#if defined(FL_IS_RP2040) || defined(FL_IS_RP2350)
717 if (fl::strcmp(driver_name, "UART0") == 0) {
718 uart_timing =
719 fl::BusTraits<fl::Bus::UART, 0>::instance().lastWireTiming();
720 } else if (fl::strcmp(driver_name, "UART1") == 0) {
721 uart_timing =
722 fl::BusTraits<fl::Bus::UART, 1>::instance().lastWireTiming();
723 }
724#endif
725 if (uart_timing.T1 == 0) {
726 // uartWireTiming() carries timing.reset_us and names the result
727 // "uart_wire", so nothing further needs setting here.
728 uart_timing = fl::uartWireTiming(timing, fl::kMaxUartBaudRate);
729 }
730 if (uart_timing.T1 == 0 && uart_timing.T2 == 0) {
731 // No representable geometry. Decoding against zeros would reject
732 // every symbol and report it as a capture fault; say what it is.
733 AR_FL_WARN("[CAPTURE] UART: no representable wave geometry for "
734 << timing.name);
735 return 0;
736 }
737 // Use wider tolerance (250ns) for UART because the UART clock and RMT
738 // sample clock are asynchronous, and GPIO matrix adds ~10-20ns jitter
739 auto rx_timing = fl::make4PhaseTiming(uart_timing, 250);
740 rx_timing.gap_tolerance_ns = 100000; // 100µs for UART inter-frame gaps
741
742 AR_FL_WARN("[CAPTURE] UART RX timing: T0H=" << uart_timing.T1
743 << " T1H=" << (uart_timing.T1 + uart_timing.T2)
744 << " T0L=" << (uart_timing.T2 + uart_timing.T3)
745 << " T1L=" << uart_timing.T3);
746
747 auto decode_result = rx_channel->decode(rx_timing, rx_buffer);
748 if (!decode_result.ok()) {
749 AR_FL_WARN("[CAPTURE] UART decode failed (error: " << static_cast<int>(decode_result.error()) << ")");
750 // Was EdgeRange(0, 256) -- 257 FL_WARN lines per decode failure
751// drowned the Teensy 4 UART TX FIFO and stalled autoresearch
752// between patterns (fbuild#755). 32 edges is enough to see whether
753// the first transition is a clean bit-0 or a glitch.
754dumpRawEdgeTiming(rx_channel, timing, fl::EdgeRange(0, 32));
755 return 0;
756 }
757 AR_FL_WARN("[CAPTURE] UART decoded " << decode_result.value() << " LED bytes");
758 return decode_result.value();
759 }
760
761 // SPI chipset drivers: decode raw SPI bit stream.
762 // These drivers use platform hardware SPI/I80/I2S to output APA102 data.
763 // RMT RX captures edges on the data pin; clock pin is ignored.
764 bool is_lcd_spi_driver = (fl::strcmp(driver_name, "LCD_SPI") == 0);
765 bool is_i2s_spi_driver = (fl::strcmp(driver_name, "I2S_SPI") == 0);
766 bool is_spi_unified_driver = (fl::strcmp(driver_name, "SPI_UNIFIED") == 0);
767 if (is_lcd_spi_driver || is_i2s_spi_driver || is_spi_unified_driver) {
768 // SPI clock used for validation: 2.4MHz (matches ValidationRemote.cpp)
769 const uint32_t spi_clock_hz = 2400000;
770 AR_FL_WARN("[CAPTURE] SPI chipset decode: clock=" << spi_clock_hz << " Hz");
771 size_t decoded = decodeSpiEdges(rx_channel, rx_buffer, spi_clock_hz);
772 if (decoded == 0) {
773 AR_FL_WARN("[CAPTURE] SPI decode failed");
774 // Was EdgeRange(0, 256) -- 257 FL_WARN lines per decode failure
775// drowned the Teensy 4 UART TX FIFO and stalled autoresearch
776// between patterns (fbuild#755). 32 edges is enough to see whether
777// the first transition is a clean bit-0 or a glitch.
778dumpRawEdgeTiming(rx_channel, timing, fl::EdgeRange(0, 32));
779 }
780 return decoded;
781 }
782
783 // Decode received data directly into rx_buffer
784 // Create 4-phase RX timing from the TX timing
785 //
786 // Wave8 drivers (SPI, PARLIO, I2S) use 8-bit expansion encoding:
787 // Bit 0: round(T1/(T1+T2+T3)*8) HIGH pulses, rest LOW
788 // Bit 1: round((T1+T2)/(T1+T2+T3)*8) HIGH pulses, rest LOW
789 // The actual pulse widths are quantized to tick boundaries, so we must
790 // compute the exact wave8 timing for RX decode thresholds.
791 // Without this correction, the RX decoder may reject valid waveforms when
792 // the nominal timing period differs from the quantized 8-tick period.
793 // Example: WS2812_800KHZ has T0L=1000ns nominal but PARLIO wave8 produces 750ns.
794 //
795 // Clock sources differ by driver:
796 // SPI/I2S: clock = 8/(T1+T2+T3) Hz, tick = (T1+T2+T3)/8 ns (variable)
797 // PARLIO: clock = 8 MHz fixed, tick = 125 ns (fixed)
798 bool is_spi_driver = (fl::strcmp(driver_name, "SPI") == 0);
799 bool is_parlio_driver = (fl::strcmp(driver_name, "PARLIO") == 0);
800 bool is_i2s_driver = (fl::strcmp(driver_name, "I2S") == 0);
801 // LCD_CLOCKLESS auto-selects wave3 vs wave8 based on canUseWave3() of the
802 // chipset timing. Compute the same eligibility check on the host side so
803 // the RX decoder reconstructs the correct quantized waveform.
804 bool is_lcd_clockless_driver = (fl::strcmp(driver_name, "LCD_CLOCKLESS") == 0);
805 bool lcd_clockless_uses_wave3 = false;
806 if (is_lcd_clockless_driver) {
807 fl::ChipsetTiming probe{timing.t1_ns, timing.t2_ns, timing.t3_ns, timing.reset_us, timing.name};
808 lcd_clockless_uses_wave3 = fl::canUseWave3(probe);
809 }
810 bool uses_wave8 = is_spi_driver || is_parlio_driver || is_i2s_driver
811 || (is_lcd_clockless_driver && !lcd_clockless_uses_wave3);
812 bool uses_wave3 = is_lcd_clockless_driver && lcd_clockless_uses_wave3;
813 fl::ChipsetTiming tx_timing;
814 if (uses_wave8) {
815 // Compute actual wave8 timing from chipset timing
816 const uint32_t period = timing.t1_ns + timing.t2_ns + timing.t3_ns;
817 // PARLIO uses fixed 8MHz clock (125ns/tick), not derived from period
818 // SPI/I2S/LCD_CLOCKLESS derive clock from period: tick = period/8
819 const uint32_t tick_ns = is_parlio_driver ? 125 : (period / 8);
820 // Wave8 LUT computes: pulses = round(fraction * 8)
821 const uint32_t pulses_bit0 = static_cast<uint32_t>(
822 static_cast<float>(timing.t1_ns) / period * 8.0f + 0.5f);
823 const uint32_t pulses_bit1 = static_cast<uint32_t>(
824 static_cast<float>(timing.t1_ns + timing.t2_ns) / period * 8.0f + 0.5f);
825 // Convert back to 3-phase timing (T1=T0H, T2=T1H-T0H, T3=actual_period-T1H)
826 const uint32_t actual_t0h = pulses_bit0 * tick_ns;
827 const uint32_t actual_t1h = pulses_bit1 * tick_ns;
828 const uint32_t actual_period = 8 * tick_ns;
829 const char* wave8_name = is_spi_driver ? "SPI_wave8" :
830 is_parlio_driver ? "PARLIO_wave8" :
831 is_lcd_clockless_driver ? "LCD_CLOCKLESS_wave8" : "I2S_wave8";
832 tx_timing = fl::ChipsetTiming{
833 actual_t0h, // T1 = T0H
834 actual_t1h - actual_t0h, // T2 = T1H - T0H
835 actual_period - actual_t1h, // T3 = actual_period - T1H
836 timing.reset_us,
837 wave8_name
838 };
839 AR_FL_WARN("[RX TIMING] " << wave8_name << ": pulses_bit0=" << pulses_bit0
840 << " pulses_bit1=" << pulses_bit1
841 << " tick_ns=" << tick_ns
842 << " -> T1=" << tx_timing.T1 << " T2=" << tx_timing.T2
843 << " T3=" << tx_timing.T3);
844 } else if (uses_wave3) {
845 // Wave3 encoding: 3 ticks per LED bit, clock = 3/(T1+T2+T3) Hz
846 const uint32_t period = timing.t1_ns + timing.t2_ns + timing.t3_ns;
847 const uint32_t tick_ns = period / 3;
848 const uint32_t ticks_bit0 = static_cast<uint32_t>(
849 static_cast<float>(timing.t1_ns) / period * 3.0f + 0.5f);
850 const uint32_t ticks_bit1 = static_cast<uint32_t>(
851 static_cast<float>(timing.t1_ns + timing.t2_ns) / period * 3.0f + 0.5f);
852 const uint32_t actual_t0h = ticks_bit0 * tick_ns;
853 const uint32_t actual_t1h = ticks_bit1 * tick_ns;
854 const uint32_t actual_period = 3 * tick_ns;
855 tx_timing = fl::ChipsetTiming{
856 actual_t0h,
857 actual_t1h - actual_t0h,
858 actual_period - actual_t1h,
859 timing.reset_us,
860 "LCD_CLOCKLESS_wave3"
861 };
862 AR_FL_WARN("[RX TIMING] LCD_CLOCKLESS_wave3: ticks_bit0=" << ticks_bit0
863 << " ticks_bit1=" << ticks_bit1
864 << " tick_ns=" << tick_ns
865 << " -> T1=" << tx_timing.T1 << " T2=" << tx_timing.T2
866 << " T3=" << tx_timing.T3);
867 } else {
868 tx_timing = fl::ChipsetTiming{timing.t1_ns, timing.t2_ns, timing.t3_ns, timing.reset_us, timing.name};
869 }
870 // Wave8/wave3 encoding has timing jitter due to clock quantization and GPIO matrix latency
871 // Use wider tolerance (200ns) to accommodate clock rounding
872 const uint32_t tolerance = (uses_wave8 || uses_wave3) ? 200 : 170;
873 auto rx_timing = fl::make4PhaseTiming(tx_timing, tolerance);
874
875 // Enable gap tolerance for PARLIO/SPI DMA gaps
876 // PARLIO: ~20µs typical gaps during buffer transitions
877 // SPI: Can have longer inter-frame gaps due to software encoding timing
878 // Increased to 100µs to accommodate SPI driver timing variations
879 rx_timing.gap_tolerance_ns = 100000; // 100µs (was 30µs)
880
881 AR_FL_WARN("[CAPTURE] Decoding...");
882 auto decode_result = rx_channel->decode(rx_timing, rx_buffer);
883 if (diagnostics) {
884 diagnostics->decodeOk = decode_result.ok() ? 1 : 0;
885 diagnostics->decodeError =
886 decode_result.ok() ? -1 : static_cast<int>(decode_result.error());
887 diagnostics->decodeBytes =
888 decode_result.ok() ? static_cast<int>(decode_result.value()) : 0;
889 }
890
891 if (!decode_result.ok()) {
892 // Use FL_WARN instead of FL_ERROR to avoid triggering bash autoresearch exit
893 // This can happen during warmup/setup and is not fatal
894 AR_FL_WARN("Decode failed (error code: " << static_cast<int>(decode_result.error()) << ")");
895 // Print raw edge timing on decode failure to diagnose the issue
896 // Was EdgeRange(0, 256) -- 257 FL_WARN lines per decode failure
897// drowned the Teensy 4 UART TX FIFO and stalled autoresearch
898// between patterns (fbuild#755). 32 edges is enough to see whether
899// the first transition is a clean bit-0 or a glitch.
900dumpRawEdgeTiming(rx_channel, timing, fl::EdgeRange(0, 32));
901 return 0;
902 }
903
904 AR_FL_WARN("[CAPTURE] Decoded " << decode_result.value() << " bytes");
905 return decode_result.value();
906}

908// Generic driver-agnostic autoresearch test runner
909// Tests all channels using the specified configuration
910void runTest(const char* test_name,
911 fl::AutoResearchConfig& config,
912 int& total, int& passed) {
913 // Multi-lane limitation: Only test Lane 0 (first channel)
914 // Hardware constraint: Only one TX channel can be read from via RX loopback
915 size_t channels_to_test = config.tx_configs.size() > 1 ? 1 : config.tx_configs.size();
916
917 if (config.tx_configs.size() > 1) {
918 AR_FL_WARN("\n[MULTI-LANE] Testing " << config.tx_configs.size() << " lanes, testing Lane 0 only (hardware limitation)");
919 }
920
921 // Test enabled configs (Lane 0 only for multi-lane)
922 for (size_t config_idx = 0; config_idx < channels_to_test; config_idx++) {
923 total++;
924
925 // Build test context for detailed error reporting
926 const auto& leds = config.tx_configs[config_idx].mLeds;
927 size_t num_leds = leds.size();
928 const bool lane_uses_rgbw =
929 legacyLaneUsesRgbw(config, config_idx);
930 // UCS7604 puts a 15-byte preamble and 16-bit channels on the wire, so
931 // its frame runs several times the pixel byte count. This number sizes
932 // the RX edge capacity as well as the comparison below, so deriving it
933 // from the pixel count truncates the capture (FastLED#4373).
934 const bool lane_uses_ucs7604 = isUCS7604(config.encoder);
935 fl::vector<uint8_t> expected_encoded;
936 if (lane_uses_ucs7604) {
937 expected_encoded = buildExpectedUCS7604(
938 config.tx_configs[config_idx].mLeds, config.encoder);
939 }
940 const size_t expected_data_bytes =
941 lane_uses_ucs7604 ? expected_encoded.size()
942 : num_leds * (lane_uses_rgbw ? 4u : 3u);
943
944 fl::TestContext ctx{
945 config.driver_name,
946 config.timing_name,
947 fl::toString(config.rx_type),
948 test_name,
949 static_cast<int>(config.tx_configs.size()),
950 static_cast<int>(config_idx),
951 config.base_strip_size,
952 static_cast<int>(num_leds),
953 config.tx_configs[config_idx].getDataPin()
954 };
955
956 AR_FL_WARN("\n=== " << test_name << " [Lane " << config_idx << "/" << config.tx_configs.size()
957 << ", Pin " << config.tx_configs[config_idx].getDataPin()
958 << ", LEDs " << config.tx_configs[config_idx].mLeds.size() << "] ===");
959
960 // Use RX channel provided via config (created in .ino file, never created dynamically here)
961 if (!config.rx_channel) {
962 FL_ERROR("[" << ctx.driver_name << "/" << ctx.timing_name << "/" << ctx.pattern_name
963 << " | Lane " << ctx.lane_index << "/" << ctx.lane_count
964 << " (Pin " << ctx.pin_number << ", " << ctx.num_leds << " LEDs) | RX:" << ctx.rx_type_name << "] "
965 << "RX channel is null - must be created in .ino and passed via AutoResearchConfig");
966 FL_ERROR("Result: FAIL ✗ (RX channel not provided)");
967 continue;
968 }
969
970 size_t bytes_captured = capture(config.rx_channel, config.rx_buffer,
971 expected_data_bytes,
972 config.tx_configs[config_idx].getDataPin(),
973 config.timing, config.driver_name);
974
975 if (bytes_captured == 0) {
976 FL_ERROR("[" << ctx.driver_name << "/" << ctx.timing_name << "/" << ctx.pattern_name
977 << " | Lane " << ctx.lane_index << "/" << ctx.lane_count
978 << " (Pin " << ctx.pin_number << ", " << ctx.num_leds << " LEDs) | RX:" << ctx.rx_type_name << "] "
979 << "Result: FAIL ✗ (capture failed)");
980 continue;
981 }
982
983 int mismatches = 0;
984
985 if (lane_uses_ucs7604) {
986 // UCS7604: Compare full encoded frame (preamble + padding + pixel data) byte-for-byte
987 size_t expected_len = expected_encoded.size();
988
989 AR_FL_WARN("UCS7604 encoded comparison: expected " << expected_len << " bytes, captured " << bytes_captured);
990
991 size_t compare_len = (bytes_captured < expected_len) ? bytes_captured : expected_len;
992 for (size_t i = 0; i < compare_len; i++) {
993 if (expected_encoded[i] != config.rx_buffer[i]) {
994 mismatches++;
995 }
996 }
997 // Count any missing bytes as mismatches
998 if (bytes_captured < expected_len) {
999 mismatches += static_cast<int>(expected_len - bytes_captured);
1000 }
1001
1002 AR_FL_WARN("Bytes Captured: " << bytes_captured << " (expected: " << expected_len << ")");
1003 int total_bytes = static_cast<int>(expected_len);
1004 AR_FL_WARN("Accuracy: " << (100.0 * (total_bytes - mismatches) / total_bytes) << "% ("
1005 << (total_bytes - mismatches) << "/" << total_bytes << " bytes match)");
1006 } else if (lane_uses_rgbw) {
1007 const fl::vector<uint8_t> expected =
1008 buildExpectedClockless(
1009 config.tx_configs[config_idx].mLeds, true);
1010 const size_t expected_len = expected.size();
1011 const size_t compare_len =
1012 bytes_captured < expected_len ? bytes_captured : expected_len;
1013 for (size_t i = 0; i < compare_len; ++i) {
1014 if (expected[i] != config.rx_buffer[i]) {
1015 ++mismatches;
1016 }
1017 }
1018 if (bytes_captured < expected_len) {
1019 mismatches +=
1020 static_cast<int>(expected_len - bytes_captured);
1021 }
1022
1023 AR_FL_WARN("RGBW bytes captured: " << bytes_captured
1024 << " (expected: " << expected_len << ")");
1025 AR_FL_WARN("Accuracy: "
1026 << (100.0 * (expected_len - mismatches) / expected_len)
1027 << "% (" << (expected_len - mismatches) << "/"
1028 << expected_len << " bytes match)");
1029 } else {
1030 // WS2812: Compare raw RGB per-LED
1031 size_t bytes_expected = num_leds * 3;
1032
1033 // NEVER EVER CHANGE THIS!!!!!
1034 const size_t front_padding_bytes = 0; // No front padding: PARLIO FRONT_PAD_BYTES=0
1035 const size_t rx_buffer_offset = front_padding_bytes;
1036
1037 if (bytes_captured > bytes_expected + front_padding_bytes) {
1038 AR_FL_WARN("Info: Captured " << bytes_captured << " bytes ("
1039 << front_padding_bytes << " front pad + "
1040 << bytes_expected << " LED data + "
1041 << (bytes_captured - bytes_expected - front_padding_bytes) << " back pad/RESET)");
1042 }
1043
1044 // Compare: byte-level comparison (COLOR_ORDER is RGB, so no reordering)
1045 size_t bytes_to_check = (bytes_captured < bytes_expected + rx_buffer_offset) ?
1046 (bytes_captured > rx_buffer_offset ? bytes_captured - rx_buffer_offset : 0) :
1047 bytes_expected;
1048 (void)bytes_to_check;
1049
1050 for (size_t i = 0; i < num_leds; i++) {
1051 size_t byte_offset = rx_buffer_offset + i * 3; // Skip front padding
1052 if (byte_offset + 2 >= bytes_captured) { // Check against total captured bytes
1053 FL_ERROR("[" << ctx.driver_name << "/" << ctx.timing_name << "/" << ctx.pattern_name
1054 << " | Lane " << ctx.lane_index << "/" << ctx.lane_count
1055 << " (Pin " << ctx.pin_number << ", " << ctx.num_leds << " LEDs) | RX:" << ctx.rx_type_name << "] "
1056 << "Incomplete data for LED[" << static_cast<int>(i)
1057 << "] (only " << bytes_captured << " bytes captured)");
1058 break;
1059 }
1060
1061 uint8_t expected_r = leds[i].r;
1062 uint8_t expected_g = leds[i].g;
1063 uint8_t expected_b = leds[i].b;
1064
1065 uint8_t actual_r = config.rx_buffer[byte_offset + 0];
1066 uint8_t actual_g = config.rx_buffer[byte_offset + 1];
1067 uint8_t actual_b = config.rx_buffer[byte_offset + 2];
1068
1069 if (expected_r != actual_r || expected_g != actual_g || expected_b != actual_b) {
1070 mismatches++;
1071 }
1072 }
1073
1074 AR_FL_WARN("Bytes Captured: " << bytes_captured << " (expected: " << bytes_expected << ")");
1075 AR_FL_WARN("Accuracy: " << (100.0 * (num_leds - mismatches) / num_leds) << "% ("
1076 << (num_leds - mismatches) << "/" << num_leds << " LEDs match)");
1077 }
1078
1079 if (mismatches == 0) {
1080 AR_FL_WARN("Result: PASS ✓");
1081 passed++;
1082 } else {
1083 FL_ERROR("[" << ctx.driver_name << "/" << ctx.timing_name << "/" << ctx.pattern_name
1084 << " | Lane " << ctx.lane_index << "/" << ctx.lane_count
1085 << " (Pin " << ctx.pin_number << ", " << ctx.num_leds << " LEDs) | RX:" << ctx.rx_type_name << "] "
1086 << "Result: FAIL ✗");
1087 }
1088 }
1089}

1091// Multi-run autoresearch test runner
1092// Runs the same test multiple times and tracks errors across runs
1093void runMultiTest(const char* test_name,
1094 fl::AutoResearchConfig& config,
1095 const fl::MultiRunConfig& multi_config,
1096 int& total, int& passed,
1097 fl::vector<fl::RunResult>* out_results) {
1098
1099 // PR #3421 framework-hang fix:
1100 // FL_WARN expansion ALWAYS constructs `fl::sstream() << X` regardless
1101 // of log level -- `ScopedLogDisable` only suppresses the OUTPUT inside
1102 // `log_emit`. Under autoresearch's RPC ASYNC handler (which wraps the
1103 // test in `ScopedLogDisable`), the cumulative `fl::sstream`/`fl::string`
1104 // heap alloc/free cycles in runMultiTest's banner + per-run + summary
1105 // FL_WARNs were leaving the Teensy 4 heap in a state where the
1106 // function-exit destructor of the heap-backed `fl::string` hangs
1107 // indefinitely (reproduced ~80% of runs on LPUART, FlexIO, and
1108 // ObjectFLED autoresearch tests). Skipping the sstream construction
1109 // when log level is below WARN avoids the heap thrash entirely.
1110 // See: marker-narrowed investigation in PR #3421 / session research.
1111 const bool kLogsEnabled = fl::getLogLevel() >=
1112 static_cast<fl::u8>(fl::LogLevel::FL_LOG_LEVEL_WARN);
1113
1114 fl::sstream ss;
1115 if (kLogsEnabled) {
1116 ss << "\n╔════════════════════════════════════════════════════════════════╗\n";
1117 ss << "║ MULTI-RUN TEST: " << test_name << "\n";
1118 ss << "║ Runs: " << multi_config.num_runs << " | Print Mode: "
1119 << (multi_config.print_all_runs ? "All" : "Errors ONLY") << "\n";
1120 ss << "╚════════════════════════════════════════════════════════════════╝";
1121 AR_FL_WARN(ss.str());
1122 }
1123
1124 fl::vector<fl::RunResult> run_results;
1125 int total_passed = 0;
1126 int total_failed = 0;
1127 bool retained_success = false;
1128
1129 // Multi-lane limitation: Only test Lane 0
1130 size_t channels_to_test = config.tx_configs.size() > 1 ? 1 : config.tx_configs.size();
1131
1132 if (config.tx_configs.size() > 1) {
1133 AR_FL_WARN("[MULTI-LANE] Testing " << config.tx_configs.size() << " lanes, testing Lane 0 only");
1134 }
1135
1136 // The tested lane and its pixel buffer are stable across repeated frames,
1137 // so construct the RGBW wire-order expectation once instead of allocating
1138 // and rebuilding it for every capture.
1139 fl::vector<uint8_t> expected_rgbw;
1140 if (channels_to_test > 0 && legacyLaneUsesRgbw(config, 0)) {
1141 expected_rgbw =
1142 buildExpectedClockless(config.tx_configs[0].mLeds, true);
1143 }
1144
1145 // Same reasoning for the UCS7604 encoded frame, plus one more: its length
1146 // is what sizes the RX edge capacity in capture(). UCS7604 carries a
1147 // 15-byte preamble and 16-bit channels, so the pixel byte count is several
1148 // times short of the wire frame and truncates the capture (FastLED#4373).
1149 const bool lane_uses_ucs7604 = isUCS7604(config.encoder);
1150 fl::vector<uint8_t> expected_ucs7604;
1151 if (channels_to_test > 0 && lane_uses_ucs7604) {
1152 expected_ucs7604 =
1153 buildExpectedUCS7604(config.tx_configs[0].mLeds, config.encoder);
1154 }
1155
1156 // Execute multiple runs
1157 for (int run = 1; run <= multi_config.num_runs; run++) {
1158 // runSingleTest executes synchronously inside the RPC task, so the
1159 // sketch loop cannot feed its five-second watchdog while a legitimate
1160 // repeated-frame soak is in progress. Each capture is independently
1161 // bounded by the TX/RX waits below; feed between them rather than
1162 // relaxing the watchdog or adding serial progress chatter.
1163 FastLED.watchdog().feed();
1164 // Do not construct progress strings under ScopedLogDisable: a long
1165 // hardware soak otherwise performs throwaway heap allocations.
1166 if (kLogsEnabled && (run % 3 == 1 || multi_config.num_runs <= 5)) {
1167 AR_FL_WARN("[Run " << run << "/" << multi_config.num_runs << "] Testing...");
1168 }
1169
1170 fl::RunResult result;
1171 result.run_number = run;
1172
1173 // Test Lane 0 only
1174 for (size_t config_idx = 0; config_idx < channels_to_test; config_idx++) {
1175 const auto& leds = config.tx_configs[config_idx].mLeds;
1176 size_t num_leds = leds.size();
1177 const bool lane_uses_rgbw =
1178 legacyLaneUsesRgbw(config, config_idx);
1179 const size_t expected_data_bytes =
1180 lane_uses_ucs7604 ? expected_ucs7604.size()
1181 : num_leds * (lane_uses_rgbw ? 4u : 3u);
1182 result.total_leds = num_leds;
1183 result.totalBytes = static_cast<int>(expected_data_bytes);
1184
1185 // Capture RX data
1186 size_t bytes_captured =
1187 capture(config.rx_channel, config.rx_buffer,
1188 expected_data_bytes,
1189 config.tx_configs[config_idx].getDataPin(),
1190 config.timing,
1191 config.driver_name, &result);
1192 result.capturedBytes = static_cast<int>(bytes_captured);
1193
1194 if (bytes_captured == 0) {
1195 AR_FL_WARN("[Run " << run << "] Capture failed");
1196 result.mismatches = static_cast<int>(num_leds);
1197 result.mismatchedBytes = static_cast<int>(result.totalBytes);
1198 result.captureFailed = true;
1199 result.passed = false;
1200 // This also covers decoder failures after a successful RX
1201 // wait: capture() reports zero bytes, so preserve the raw
1202 // edge snapshot before this run exits.
1203 collectRawEdgeDiagnostics(config.rx_channel, &result);
1204 break;
1205 }

1207 // Check pixel data
1208 int mismatches = 0;
1209
1210 // Diagnostic: single-line driver + first-24-bytes dump. Was 26
1211 // FL_WARNs (one per byte) -- on a 4-pattern test that produced 104
1212 // serial lines of pure hex dump, which dominated the Teensy 4 UART
1213 // TX FIFO under bursty autoresearch loads (fbuild#755 followup,
1214 // FastLED#3219 stack of fixes). Collapsing into a single
1215 // space-separated hex string keeps the diagnostic value while
1216 // dropping per-pattern serial overhead from ~2.1 KB to ~80 bytes.
1217 if (kLogsEnabled) {
1218 fl::sstream hex_dump;
1219 size_t dump_count = (bytes_captured < 24) ? bytes_captured : 24;
1220 for (size_t i = 0; i < dump_count; i++) {
1221 if (i > 0) hex_dump << " ";
1222 hex_dump << fl::hex << static_cast<int>(config.rx_buffer[i]) << fl::dec;
1223 }
1224 AR_FL_WARN("[RUN " << run << "] Driver=" << config.driver_name
1225 << " bytes_captured=" << bytes_captured
1226 << " first" << static_cast<int>(dump_count) << "=" << hex_dump.str());
1227 }
1228
1229 if (lane_uses_ucs7604) {
1230 // UCS7604: Compare full encoded frame byte-for-byte
1231 const fl::vector<uint8_t>& expected_encoded = expected_ucs7604;
1232 size_t expected_len = expected_encoded.size();
1233 result.totalBytes = static_cast<int>(expected_len);
1234
1235 size_t compare_len = (bytes_captured < expected_len) ? bytes_captured : expected_len;
1236 for (size_t i = 0; i < compare_len; i++) {
1237 if (expected_encoded[i] != config.rx_buffer[i]) {
1238 result.mismatchedBytes++;
1239 if ((expected_encoded[i] ^ config.rx_buffer[i]) == 0x01) {
1240 result.lsbOnlyErrors++;
1241 }

1243

1245 if (mismatches == 1) {

1247 << "] expected=0x" << fl::hex << static_cast<int>(expected_encoded[i])
1248 << " actual=0x" << static_cast<int>(config.rx_buffer[i]) << fl::dec);
1249 }
1250
1251 // Store first N errors (using byte-level reporting)
1252 if (result.errors.size() < static_cast<size_t>(multi_config.max_errors_per_run)) {
1253 result.errors.push_back(fl::LEDError(
1254 static_cast<int>(i), expected_encoded[i], 0, 0,
1255 config.rx_buffer[i], 0, 0
1256 ));
1257 }
1258 }
1259 }
1260 // Count missing bytes as mismatches
1261 if (bytes_captured < expected_len) {
1262 mismatches += static_cast<int>(expected_len - bytes_captured);
1263 }
1264 } else if (lane_uses_rgbw) {
1265 const fl::vector<uint8_t>& expected = expected_rgbw;
1266 size_t verified_leds = 0;
1267 for (size_t i = 0; i < num_leds; ++i) {
1268 const size_t byte_offset = i * 4;
1269 if (byte_offset + 3 >= bytes_captured) {
1270 break;
1271 }
1272 verified_leds = i + 1;
1273
1274 bool led_mismatch = false;
1275 for (size_t channel = 0; channel < 4; ++channel) {
1276 const uint8_t expected_byte =
1277 expected[byte_offset + channel];
1278 const uint8_t actual_byte =
1279 config.rx_buffer[byte_offset + channel];
1280 if (expected_byte != actual_byte) {
1281 ++result.mismatchedBytes;
1282 if ((expected_byte ^ actual_byte) == 0x01) {
1283 ++result.lsbOnlyErrors;
1284 }
1285 led_mismatch = true;
1286 }
1287 }
1288
1289 if (led_mismatch) {
1290 if (mismatches == 0) {
1291 AR_FL_WARN("\n[RGBW CORRUPTION @ LED "
1292 << static_cast<int>(i) << ", Run "
1293 << run << "]");
1294 const size_t corruption_edge_index = i * 64;
1295 const size_t offset =
1296 corruption_edge_index > 4
1297 ? corruption_edge_index - 4
1298 : 0;
1299 dumpRawEdgeTiming(
1300 config.rx_channel, config.timing,
1301 fl::EdgeRange(offset, 9));
1302 }
1303 ++mismatches;
1304
1305 if (result.errors.size() <
1306 static_cast<size_t>(
1307 multi_config.max_errors_per_run)) {
1308 result.errors.push_back(fl::LEDError(
1309 static_cast<int>(i),
1310 expected[byte_offset + 0],
1311 expected[byte_offset + 1],
1312 expected[byte_offset + 2],
1313 expected[byte_offset + 3],
1314 config.rx_buffer[byte_offset + 0],
1315 config.rx_buffer[byte_offset + 1],
1316 config.rx_buffer[byte_offset + 2],
1317 config.rx_buffer[byte_offset + 3]));
1318 }
1319 }
1320 }
1321

1323 const size_t unchecked = num_leds - verified_leds;
1324 mismatches += static_cast<int>(unchecked);
1325 result.mismatchedBytes +=
1326 static_cast<int>(unchecked * 4);
1327 AR_FL_WARN("[TRUNCATED RGBW CAPTURE] Only verified "
1328 << verified_leds << "/" << num_leds
1329 << " LEDs (" << bytes_captured
1330 << " bytes captured, needed "
1331 << expected_data_bytes << "). Marking "
1332 << unchecked
1333 << " unchecked LEDs as mismatches.");
1334 }
1335 } else {
1336 // WS2812: Per-LED RGB comparison
1337 size_t bytes_expected = num_leds * 3;
1338
1339 // Determine front padding offset (PARLIO only)
1340 const size_t rx_buffer_offset = 0; // No front padding: PARLIO FRONT_PAD_BYTES=0
1341
1342 size_t bytes_to_check = (bytes_captured < bytes_expected + rx_buffer_offset) ?
1343 (bytes_captured > rx_buffer_offset ? bytes_captured - rx_buffer_offset : 0) :
1344 bytes_expected;
1345 (void)bytes_to_check;
1346
1347 size_t verified_leds = 0;
1348 for (size_t i = 0; i < num_leds; i++) {
1349 size_t byte_offset = rx_buffer_offset + i * 3; // Apply offset for PARLIO front padding
1350 if (byte_offset + 2 >= bytes_captured) { // Check against total captured bytes
1351 break;
1352 }
1353 verified_leds = i + 1;
1354
1355 uint8_t expected_r = leds[i].r;
1356 uint8_t expected_g = leds[i].g;
1357 uint8_t expected_b = leds[i].b;
1358
1359 uint8_t actual_r = config.rx_buffer[byte_offset + 0];
1360 uint8_t actual_g = config.rx_buffer[byte_offset + 1];
1361 uint8_t actual_b = config.rx_buffer[byte_offset + 2];
1362

1364 uint8_t exp_bytes[3] = {expected_r, expected_g, expected_b};
1365 uint8_t act_bytes[3] = {actual_r, actual_g, actual_b};
1366 bool led_mismatch = false;
1367 for (int ch = 0; ch < 3; ch++) {
1368 if (exp_bytes[ch] != act_bytes[ch]) {
1369 result.mismatchedBytes++;
1370 if ((exp_bytes[ch] ^ act_bytes[ch]) == 0x01) {
1371 result.lsbOnlyErrors++;
1372 }
1373 led_mismatch = true;
1374 }
1375 }
1376
1377 if (led_mismatch) {
1378 // Print corruption context for first mismatch only
1379 if (mismatches == 0) {
1380 AR_FL_WARN("\n[CORRUPTION @ LED " << static_cast<int>(i) << ", Run " << run << "]");
1381
1382 // Calculate edge index and print timing around corruption point
1383 size_t corruption_edge_index = i * 48;
1384 size_t offset = corruption_edge_index > 4 ? corruption_edge_index - 4 : 0;
1385
1386 // Dump raw edge timing around corruption point (9 edges: -4 to +4)
1387 dumpRawEdgeTiming(config.rx_channel, config.timing, fl::EdgeRange(offset, 9));
1388 }
1389
1390 mismatches++;
1391
1392 // Store first N errors
1393 if (result.errors.size() < static_cast<size_t>(multi_config.max_errors_per_run)) {
1394 result.errors.push_back(fl::LEDError(
1395 i, expected_r, expected_g, expected_b,
1396 actual_r, actual_g, actual_b
1397 ));
1398 }
1399 }
1400 }
1401
1402 // Truncated capture: RX returned fewer bytes than expected. Count
1403 // the unchecked LEDs as mismatches rather than silently passing.
1404 // A short capture means we did NOT verify those LEDs, so the run
1405 // MUST fail. The old silent break hid real bugs on strips longer
1406 // than the RMT DMA buffer (~170 LEDs for WS2812B). See issue #2254.
1407 if (verified_leds < num_leds) {
1408 size_t unchecked = num_leds - verified_leds;
1409 mismatches += static_cast<int>(unchecked);
1410 result.mismatchedBytes += static_cast<int>(unchecked * 3);
1411 AR_FL_WARN("[TRUNCATED CAPTURE] Only verified " << verified_leds
1412 << "/" << num_leds << " LEDs (" << bytes_captured
1413 << " bytes captured, needed " << (num_leds * 3)
1414 << "). Marking " << unchecked
1415 << " unchecked LEDs as mismatches.");
1416 }
1417 }
1418
1419 result.mismatches = mismatches;
1420 result.passed = (mismatches == 0);
1421 if (!result.passed) {
1422 collectRawEdgeDiagnostics(config.rx_channel, &result);
1423 }
1424 }
1425
1426 // Print run result if configured (sstream construction skipped
1427 // when logs disabled -- see kLogsEnabled note at function top).
1428 if (kLogsEnabled && (multi_config.print_all_runs || !result.passed)) {
1429 AR_FL_WARN("[Run " << run << "/" << multi_config.num_runs << "] "
1430 << (result.passed ? "PASS" : "FAIL")
1431 << " | Errors: " << result.mismatches << "/" << result.total_leds
1432 << " (" << (100.0 * (result.total_leds - result.mismatches) / result.total_leds) << "%)");
1433
1434 // Print error details if enabled
1435 if (!result.passed && multi_config.print_per_led_errors && !result.errors.empty()) {
1436 AR_FL_WARN(" First " << result.errors.size() << " error(s):");
1437 for (size_t i = 0; i < result.errors.size(); i++) {
1438 const auto& err = result.errors[i];
1439 AR_FL_WARN(" LED[" << err.led_index << "]: expected RGB("
1440 << static_cast<int>(err.expected_r) << ","
1441 << static_cast<int>(err.expected_g) << ","
1442 << static_cast<int>(err.expected_b) << ") got RGB("
1443 << static_cast<int>(err.actual_r) << ","
1444 << static_cast<int>(err.actual_g) << ","
1445 << static_cast<int>(err.actual_b) << ")");
1446 }
1447 }
1448 }
1449
1450 if (result.passed) {
1451 ++total_passed;
1452 } else {
1453 ++total_failed;
1454 }
1455
1456 // The RPC response includes only failed pattern details. Keeping every
1457 // successful result during a long frame soak needlessly grows the
1458 // transient heap (and used to make the 16-frame RP PIO run wedge).
1459 // Retain one representative successful capture for byte-evidence;
1460 // retain every failure so its full postmortem remains available.
1461 if (!result.passed || !retained_success || multi_config.num_runs == 1) {
1462 if (result.passed) {
1463 retained_success = true;
1464 }
1465 run_results.push_back(fl::move(result));
1466 }
1467 FastLED.watchdog().feed();
1468 }
1469
1470 // Summary statistics
1471 // Summary FL_WARN -- gated on log level (see kLogsEnabled note above)
1472 // to prevent heap thrash under autoresearch ScopedLogDisable.
1473 if (kLogsEnabled) {
1474 ss.clear();
1475 ss << "\n╔════════════════════════════════════════════════════════════════╗\n";
1476 ss << "║ MULTI-RUN SUMMARY\n";
1477 ss << "╚════════════════════════════════════════════════════════════════╝\n";
1478 ss << "Total Runs: " << multi_config.num_runs << "\n";
1479 ss << "Passed: " << total_passed << " (" << (100.0 * total_passed / multi_config.num_runs) << "%)\n";
1480 ss << "Failed: " << total_failed << " (" << (100.0 * total_failed / multi_config.num_runs) << "%)";
1481 AR_FL_WARN(ss.str());
1482
1483 if (total_failed > 0) {
1484 ss.clear();
1485 ss << "\nFailed Run Numbers:\n";
1486 for (const auto& r : run_results) {
1487 if (!r.passed) {
1488 ss << " Run #" << r.run_number << " - " << r.mismatches << " errors\n";
1489 if (!r.errors.empty()) {
1490 ss << " First error at LED[" << r.errors[0].led_index << "]: "
1491 << "expected RGB(" << static_cast<int>(r.errors[0].expected_r) << ","
1492 << static_cast<int>(r.errors[0].expected_g) << ","
1493 << static_cast<int>(r.errors[0].expected_b) << ") got RGB("
1494 << static_cast<int>(r.errors[0].actual_r) << ","
1495 << static_cast<int>(r.errors[0].actual_g) << ","
1496 << static_cast<int>(r.errors[0].actual_b) << ")\n";
1497 }
1498 }
1499 }
1500 AR_FL_WARN(ss.str());
1501 }
1502 }
1503
1504 // Copy results to caller if requested
1505 if (out_results) {
1506 for (const auto& r : run_results) {
1507 out_results->push_back(r);
1508 }
1509 }
1510
1511 // Update totals
1512 total++;
1513 if (total_failed == 0) {
1514 passed++;
1515 if (kLogsEnabled) {
1516 AR_FL_WARN("\n[OVERALL] PASS ✓ - All " << multi_config.num_runs << " runs succeeded");
1517 }
1518 } else {
1519 if (kLogsEnabled) {
1520 AR_FL_WARN("\n[OVERALL] FAIL ✗ - " << total_failed << "/" << multi_config.num_runs << " runs failed");
1521 }
1522 }
