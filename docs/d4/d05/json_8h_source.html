146inline u32 round_shift_right_u64(u64 value, int shift) FL_NO_EXCEPT {
147 if (shift <= 0) {
148 return static_cast<u32>(value);
149 }
150 if (shift >= 64) {
151 return 0;
152 }
153 const u64 half = u64(1) << (shift - 1);
154 const u64 mask = (u64(1) << shift) - 1;
155 const u64 truncated = value >> shift;
156 const u64 remainder = value & mask;
157 const bool round_up = remainder > half || (remainder == half && (truncated & 1));
158 return static_cast<u32>(truncated + (round_up ? 1 : 0));
159}

161inline u32 ieee754_bits_from_scaled_u64(
162 u32 sign, u64 magnitude, int binary_exp) FL_NO_EXCEPT {
163 if (magnitude == 0) {
164 return sign;
165 }
166
167 const int top_bit = json_floor_log2_u64(magnitude);
168 int exponent = top_bit + binary_exp;
169 if (exponent > 127) {
170 return sign | 0x7F800000u;
171 }
172
173 if (exponent >= -126) {
174 const int shift = top_bit - 23;
175 u32 significand = 0;
176 if (shift > 0) {
177 significand = round_shift_right_u64(magnitude, shift);
178 } else {
179 significand = static_cast<u32>(magnitude << -shift);
180 }
181 if (significand >= 0x01000000u) {
182 significand >>= 1;
183 ++exponent;
184 if (exponent > 127) {
185 return sign | 0x7F800000u;
186 }
187 }
188 return sign | (static_cast<u32>(exponent + 127) << 23) |
189 (significand & 0x007FFFFFu);
190 }
191
192 const int subnormal_shift = -(binary_exp + 149);
193 u32 mantissa = 0;
194 if (subnormal_shift > 0) {
195 mantissa = round_shift_right_u64(magnitude, subnormal_shift);
196 } else {
197 const int left_shift = -subnormal_shift;
198 if (left_shift >= 64 || magnitude > (u64(-1) >> left_shift)) {
199 return sign | 0x7F800000u;
200 }
201 const u64 shifted = magnitude << left_shift;
202 mantissa = shifted > 0x007FFFFFu ? 0x00800000u : static_cast<u32>(shifted);
203 }
204 if (mantissa >= 0x00800000u) {
205 return sign | 0x00800000u;
206 }
207 return sign | mantissa;
208}

210inline u32 ieee754_bits_from_fixed_raw(i64 raw, int frac_bits) FL_NO_EXCEPT {
211 const u32 sign = raw < 0 ? 0x80000000u : 0u;
212 const u64 magnitude = raw < 0
213 ? static_cast<u64>(-(raw + 1)) + 1u
214 : static_cast<u64>(raw);
215 return ieee754_bits_from_scaled_u64(sign, magnitude, -frac_bits);
216}

218inline bool fixed_raw_from_ieee754_bits(u32 bits, int frac_bits, i64* out) FL_NO_EXCEPT {
219 const bool negative = (bits & 0x80000000u) != 0;
220 const u32 exp_bits = (bits >> 23) & 0xFFu;
221 const u32 mantissa_bits = bits & 0x007FFFFFu;
222 if (exp_bits == 0xFFu) {
223 return false;
224 }
225
226 u64 significand = mantissa_bits;
227 int exponent = -126;
228 if (exp_bits != 0) {
229 significand |= 0x00800000u;
230 exponent = static_cast<int>(exp_bits) - 127;
231 }
232
233 const int shift = exponent + frac_bits - 23;
234 u64 magnitude = 0;
235 if (shift >= 0) {
236 if (shift >= 63 || significand > (u64(0x7FFFFFFFFFFFFFFF) >> shift)) {
237 return false;
238 }
239 magnitude = significand << shift;
240 } else {
241 magnitude = round_shift_right_u64(significand, -shift);
242 }
243 if ((!negative && magnitude > u64(0x7FFFFFFFFFFFFFFF)) ||
244 (negative && magnitude > (u64(0x7FFFFFFFFFFFFFFF) + 1u))) {
245 return false;
246 }
247 *out = negative
248 ? (magnitude == (u64(0x7FFFFFFFFFFFFFFF) + 1u)
249 ? static_cast<i64>(0x8000000000000000ull)
250 : -static_cast<i64>(magnitude))
251 : static_cast<i64>(magnitude);
252 return true;
253}

734 }

736 auto arrPtr = mValue->as_array();

739 return json((*arrPtr)[idx]);

742 }

754 json operator[](const fl::string &key) FL_NO_EXCEPT {
755 if (!mValue || !mValue->is_object()) {
756 mValue = fl::make_shared<json_value>(json_object{});
757 }
758 // FastLED#3588: make_shared may degrade to empty under OOM.
759 if (!mValue) { return json(nullptr); }
760 // Get reference to the json_value
761 auto objPtr = mValue->data.ptr<json_object>();
762 if (objPtr) {
763 // If key doesn't exist, create a new json_value and insert it
764 if (objPtr->find(key) == objPtr->end()) {
765 (*objPtr)[key] = fl::make_shared<json_value>(nullptr);
766 }
767 // Return a new json object that wraps the shared_ptr to the json_value
768 return json((*objPtr)[key]);
769 }
770 // Should not happen if mValue is properly initialized as an object
771 //return *reinterpret_cast<json*>(&get_null_value());
772 return json(nullptr);

787 }
