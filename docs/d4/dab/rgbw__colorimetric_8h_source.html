88// Native-gamut detection (#2748). Returns true when the source primaries
89// of `p` (input_xy_r/g/b) match the LED's measured primaries (xy_r/g/b)
90// within float tolerance — i.e. the user is feeding "native LED gamut"
91// drive coordinates rather than a foreign gamut like sRGB / Rec.2020.
92//
93// Native mode has stricter topology authority than named-gamut mode:
94//
95// * Single-channel inputs are exact drive identity:
96// (R,0,0)->(R,0,0,0), (0,G,0)->(0,G,0,0), (0,0,B)->(0,0,B,0).
97//
98// * Dual-channel outer edges are channel-set identity, not value identity.
99// For RG/RB/GB edges the inactive RGB channel and W must remain zero, but
100// the two active channels are still solved from the source target XYZ and
101// the measured diode XYZ/Y columns. This is why yellow_half is not simply
102// raw (0.5,0.5,0,0): measured R/G luminance and the D65 source matrix set
103// the correct active-channel ratio.
104//
105// * Three-channel inputs fall through to the strict RGBW sub-gamut routing
106// and may use W inside RGW/RBW/BGW.
107//
108// The W chromaticity (input_xy_w) is intentionally NOT compared here: native
109// authority is about RGB topology. Different target whites affect the source
110// matrix and full-3-channel routing, not whether a pure/edge native input is
111// allowed to introduce W.
112inline bool is_native_input_gamut(const DiodeProfile& p) FL_NO_EXCEPT {
113 constexpr float kPrimaryEps = 1e-6f;
114 auto close = [](const float a[2], const float b[2]) FL_NO_EXCEPT {
115 const float dx = a[0] - b[0];
116 const float dy = a[1] - b[1];
117 return (dx * dx + dy * dy) < kPrimaryEps;
118 };
119 return close(p.input_xy_r, p.xy_r)
120 && close(p.input_xy_g, p.xy_g)
121 && close(p.input_xy_b, p.xy_b);
122}

124// Topology activity classifier (#2748). Counts how many of `s_r, s_g, s_b`
125// are above the LSB-level epsilon. The solvers use this to select the native
126// authority fast path:
127// n == 1 : exact single-axis identity
128// n == 2 : fixed-topology two-emitter solve
129// n == 3 : normal strict / LP / overdrive interior solve
130inline int count_active_channels(float s_r, float s_g, float s_b) FL_NO_EXCEPT {
131 // 1 / 65535 — matches the 16-bit verifier precision used in the
132 // reference math model; anything below this is below noise floor.
133 constexpr float kTopoEps = 1.0f / 65535.0f;
134 int n = 0;
135 if (s_r > kTopoEps) ++n;
136 if (s_g > kTopoEps) ++n;
137 if (s_b > kTopoEps) ++n;
138 return n;
139}

143// Precomputed per-profile data: emitter XYZ columns + the four matrix inverses
144// the solvers need. Built once when the active (profile, cct) pair changes.
145// xy_w is the *effective* W chromaticity — equal to profile->xy_w when no CCT
146// override is active, otherwise the cct_to_xy() shift. The strict solvers
147// must route barycentric containment against this value (not the profile's
148// raw xy_w) or they will choose the wrong sub-gamut for CCT-shifted whites.
149struct ProfileCache {
150 const DiodeProfile* profile;
151 float xy_w[2]; // effective W chromaticity (incl. CCT shift)
152 float P_R[3], P_G[3], P_B[3], P_W[3]; // emitter XYZ at full drive
153 float P_RGB_inv[3][3]; // [R G B]^-1, used by wx_lp_legacy
154 float P_RGW_inv[3][3]; // [R G W]^-1
155 float P_RBW_inv[3][3]; // [R B W]^-1
156 float P_BGW_inv[3][3]; // [B G W]^-1
157 float d_W[3]; // P_RGB_inv * P_W (wx_lp_legacy cache)
158
159 // Source-space → measured-device XYZ matrix (#2705).
160 //
161 // build_source_matrix() first derives the standard normalized CIE source
162 // matrix from profile.input_xy_* (source white has Y=1). build_profile_cache()
163 // then scales that matrix by the reference white-fit factor used by the
164 // math model so M_src·[1,1,1] is in the same absolute XYZ/Y domain as the
165 // measured emitter columns P_R/P_G/P_B/P_W.
166 //
167 // This scaling is essential: all runtime solves compare M_src targets
168 // directly against measured diode columns. Leaving M_src at Y=1 would make
169 // native D65 dual-edge and LP solves underpowered and would not match the
170 // reference model/cube.
171 //
172 // When the source chromaticities are degenerate (input_xy_w[1] == 0, the
173 // default for value-initialized `DiodeProfile{}`), `has_source_space` is
174 // false and solvers fall back to direct device-emitter projection.
175 float M_src[3][3];
176 bool has_source_space;
177};

196// Convenience: derive eta from a target CCT relative to the warm/cool
197// reference CCTs. Linear interpolation in CCT space, clamped to [0, 1].
198inline float rgbcct_eta_for_cct(int target_cct, int warm_cct,
199 int cool_cct) FL_NO_EXCEPT {
200 if (cool_cct == warm_cct) return 0.5f;
201 const float t = (static_cast<float>(target_cct) - warm_cct)
202 / (static_cast<float>(cool_cct) - warm_cct);
203 return fl::clamp(t, 0.0f, 1.0f);
204}

223// Strict sub-gamut solver (gist sec 5).
224//
225// Runtime path:
226// 1. Black/near-black returns zero.
227// 2. Native single-axis inputs return exact identity.
228// 3. Native dual-edge inputs are locked to RG/RB/GB and solved as a measured
229// two-emitter least-squares problem — no W/third-channel bleed, but also
230// not raw passthrough.
231// 4. Interior inputs split source value from chroma, solve the full-chroma
232// endpoint in one of {RGW, RBW, BGW}, then scale by source value.
233//
234// Returns false only for numerically degenerate profiles / matrices where the
235// caller should fall back to a simpler path.
236bool solve_strict_subgamut(const ProfileCache& cache, float s_r,
237 float s_g, float s_b,
238 float out_rgbw[4]) FL_NO_EXCEPT;

249// Reference wx_lp_legacy solver.
250//
251// This is the analytical runtime counterpart of the reference math-model
252// "wx_lp_legacy" cube. It is not the boosted/overdrive mode. It solves a
253// chromaticity-preserving bounded endpoint:
254// 1. honor native single/dual topology authority;
255// 2. split source value from chroma;
256// 3. project the full-chroma target into the reachable hull if needed;
257// 4. solve the xy constraints while maximizing W;
258// 5. keep a small RGB floor when the four-channel manifold can represent the
259// target, normalize the endpoint, then apply the original source value.
260//
261// Native single-axis inputs remain exact identity. Native dual edges remain
262// locked to RG/RB/GB but are solved from measured/source XYZ instead of raw
263// passthrough. This is intentionally separate from boosted overdrive below.
264bool solve_wx_lp_legacy(const ProfileCache& cache, float s_r, float s_g,
265 float s_b, float out_rgbw[4]) FL_NO_EXCEPT;
