56 // Register "testAsync" function - verify that show() returns before TX completes (async DMA)
57 // This proves the SPI driver releases back to the main thread while draining.
58 remote.bind("testAsync", [this](const fl::json& args) -> fl::json {
59 fl::json response = fl::json::object();
60
61 // Parse optional parameters from args object
62 int num_leds = 300;
63 fl::string requested_driver;
64 fl::json config;
65 if (args.is_object()) {
66 config = args;
67 } else if (args.is_array() && args.size() >= 1 && args[0].is_object()) {
68 config = args[0];
69 }
70 if (!config.is_null()) {
71 if (config.contains("numLeds") && config["numLeds"].is_int()) {
72 num_leds = static_cast<int>(config["numLeds"].as_int().value());
73 }
74 if (config.contains("driver") && config["driver"].is_string()) {
75 requested_driver = config["driver"].as_string().value();
76 }
77 }
78
79 // Set exclusive driver if requested (by-name path: requested_driver from RPC)
80 if (!requested_driver.empty()) {
81 if (!autoResearchSetExclusiveDriverByName(requested_driver.c_str())) {
82 response.set("success", false);
83 response.set("error", "DriverSetupFailed");
84 fl::sstream msg;
85 msg << "Failed to set '" << requested_driver.c_str() << "' as exclusive driver";
86 response.set("message", msg.str().c_str());
87 return response;
88 }
89 }
90
91 if (num_leds < 10 || num_leds > 1000) {
92 response.set("success", false);
93 response.set("error", "InvalidNumLeds");
94 response.set("message", "numLeds must be 10-1000");
95 return response;
96 }
97
98 // Set up a channel with the specified number of LEDs
99 fl::vector<CRGB> leds(num_leds);
100 for (int i = 0; i < num_leds; i++) {
101 leds[i] = CRGB(0xFF, 0x00, 0x80); // Solid color pattern
102 }
103
104 fl::ChannelConfig channel_config(
105 mState->pin_tx,
106 fl::makeTimingConfig<fl::TIMING_WS2812B_V5>(),
107 leds,
108 RGB
109 );
110
111 auto channel = FastLED.add(channel_config);
112 if (!channel) {
113 response.set("success", false);
114 response.set("error", "ChannelCreationFailed");
115 response.set("message", "Failed to create channel");
116 return response;
117 }
118
119 // === DRAW 1: May include one-time SPI hardware init overhead ===
120 uint32_t t0 = micros();
121 FastLED.show();
122 uint32_t t1 = micros();
123 FastLED.wait(5000); // 5 second timeout
124 uint32_t t2 = micros();
125
126 uint32_t show1_us = t1 - t0;
127 uint32_t wait1_us = t2 - t1;
128 uint32_t total1_us = t2 - t0;
129
130 // === DRAW 2: Should be fast (no init overhead) ===
131 uint32_t t3 = micros();
132 FastLED.show();
133 uint32_t t4 = micros();
134 FastLED.wait(5000); // 5 second timeout
135 uint32_t t5 = micros();
136
137 uint32_t show2_us = t4 - t3;
138 uint32_t wait2_us = t5 - t4;
139 uint32_t total2_us = t5 - t3;
140
141 // Clean up channel
142 FastLED.clear(ClearFlags::CHANNELS);
143
144 // Determine if async behavior is working on draw 2 (no init overhead):
145 // If async: show_us << total_us (show returns quickly, wait blocks for remainder)
146 // If blocking: show_us ≈ total_us (show blocks for entire TX, wait returns instantly)
147 // Pass criterion: show_us < 50% of total_us (on draw 2)
148 bool passed = (total2_us > 0) && (show2_us < total2_us / 2);
149
150 // Determine driver name
151 fl::string driver_name = "unknown";
152 for (fl::size i = 0; i < mState->drivers_available.size(); i++) {
153 if (mState->drivers_available[i].enabled) {
154 driver_name = mState->drivers_available[i].name;
155 break;
156 }
157 }
158
159 response.set("success", true);
160 response.set("passed", passed);
161 // Draw 1 (with possible init overhead)
162 response.set("show1_us", static_cast<int64_t>(show1_us));
163 response.set("wait1_us", static_cast<int64_t>(wait1_us));
164 response.set("total1_us", static_cast<int64_t>(total1_us));
165 // Draw 2 (steady-state, no init)
166 response.set("show2_us", static_cast<int64_t>(show2_us));
167 response.set("wait2_us", static_cast<int64_t>(wait2_us));
168 response.set("total2_us", static_cast<int64_t>(total2_us));
169 response.set("num_leds", static_cast<int64_t>(num_leds));
170 response.set("driver", driver_name.c_str());
171
172 fl::sstream msg;
173 if (passed) {
174 msg << "Async OK: draw2 show()=" << show2_us
175 << "us, wait()=" << wait2_us
176 << "us, total=" << total2_us << "us"
177 << " (draw1 show=" << show1_us << "us)";
178 } else {
179 msg << "Async FAIL: draw2 show()=" << show2_us
180 << "us out of " << total2_us
181 << "us total (expected <50%)"
182 << " (draw1 show=" << show1_us << "us)";
183 }
184 response.set("message", msg.str().c_str());
185
186 return response;
187 });

424 // Register "decodeFile" - Decode a media file and return first 16 pixels
425 // Args: [base64_data_string, extension_string] (base64 auto-decoded to fl::vector<fl::u8>)
426 remote.bind("decodeFile", [](fl::vector<fl::u8> data, fl::string ext) -> fl::json {
427 fl::json response = fl::json::object();
428
429 if (ext != ".mp4") {
430 response.set("success", false);
431 response.set("error", "Only .mp4 supported for device decode");
432 return response;
433 }
434
435 // Parse MP4 container metadata
436 fl::string error;
437 fl::H264Info info = fl::H264::parseH264Info(data, &error);
438 if (!info.isValid) {
439 response.set("success", false);
440 response.set("error", error.c_str());
441 return response;
442 }
443 response.set("width", static_cast<int64_t>(info.width));
444 response.set("height", static_cast<int64_t>(info.height));
445
446 if (!fl::H264::isSupported()) {
447 response.set("success", false);
448 response.set("error", "H264 decoder not supported on this platform");
449 return response;
450 }
451
452 // Create decoder and decode first frame
453 fl::string dec_error;
454 auto decoder = fl::H264::createDecoder(fl::H264Config{}, &dec_error);
455 if (!decoder) {
456 response.set("success", false);
457 response.set("error", dec_error.c_str());
458 return response;
459 }
460
461 auto stream = fl::make_shared<fl::memorybuf>(data.size());
462 stream->write(data);
463
464 if (!decoder->begin(stream)) {
465 fl::string msg;
466 decoder->hasError(&msg);
467 response.set("success", false);
468 response.set("error", msg.empty() ? "Decoder begin() failed" : msg.c_str());
469 return response;
470 }
471
472 auto result = decoder->decode();
473 if (result != fl::DecodeResult::Success) {
474 response.set("success", false);
475 response.set("error", "Decode returned non-success");
476 response.set("decode_result", static_cast<int64_t>(static_cast<int>(result)));
477 decoder->end();
478 return response;
479 }
480
481 fl::Frame frame = decoder->getCurrentFrame();
482 decoder->end();
483
484 if (!frame.isValid()) {
485 response.set("success", false);
486 response.set("error", "Decoded frame is invalid");
487 return response;
488 }
489
490 response.set("success", true);
491 response.set("frame_width", static_cast<int64_t>(frame.getWidth()));
492 response.set("frame_height", static_cast<int64_t>(frame.getHeight()));
493
494 // Return first 16 pixels as [[r,g,b], ...]
495 auto pixels = frame.rgb();
496 fl::json pixel_array = fl::json::array();
497 int count = pixels.size() < 16 ? static_cast<int>(pixels.size()) : 16;
498 for (int i = 0; i < count; i++) {
499 fl::json px = fl::json::array();
500 px.push_back(static_cast<int64_t>(pixels[i].r));
501 px.push_back(static_cast<int64_t>(pixels[i].g));
502 px.push_back(static_cast<int64_t>(pixels[i].b));
503 pixel_array.push_back(px);
504 }
505 response.set("pixels", pixel_array);
506
507 return response;
508 });
