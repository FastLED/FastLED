19// Declared rather than included. Every path that binds a profile has to
20// install these -- there are four, and three of them do not go through
21// setColorProfile: Channel::create<Profile>,
22// ChannelOptions::withColorProfile<Profile> and
23// CLEDController::bindStaticEmitterProfile all set mStaticProfile directly.
24// Missing one leaves that channel silently on the legacy path, which is the
25// exact bug this whole change exists to fix.
26//
27// Declared rather than included. `pipeline_binding.h` pulls the whole gfx
28// pipeline in behind it, and reaching those headers from here puts them
29// ahead of whatever brings `EmitterProfile` into scope unqualified --
30// `device_solve.h`, `gamut_map.h` and `white_allocation.h` all stop
31// compiling. One declaration is all this file needs, and keeping it to one
32// also keeps `options.h` cheap for every translation unit that includes it.

117 bool setColorProfile(const EmitterProfile& profile, SourceProfile source,
118 bool global_source, GamutPolicy gamut = GamutPolicy::ChromaCompress) FL_NO_EXCEPT {
119#if !FL_COLOR_PROFILE_RUNTIME
120 FL_UNUSED(profile); FL_UNUSED(source); FL_UNUSED(global_source); FL_UNUSED(gamut);
121 return false;
122#else
123 // Captured before mStaticProfile is cleared below, so the warning
124 // condition still sees whether a profile was bound on entry.
125 const bool had_profile = hasColorProfile();
126 mColorProfile.mStaticProfile = nullptr;
127 if (profile.native_code_depth == 0 || profile.native_code_depth > 16 ||
128 !validChromaticity(profile.xy_r) || !validChromaticity(profile.xy_g) ||
129 !validChromaticity(profile.xy_b) || !validPrimaries(source.primaries) ||
130 !finitePositive(profile.lum_r) || !finitePositive(profile.lum_g) || !finitePositive(profile.lum_b) ||
131 (profile.topology != colorimetric_response::EmitterTopology::RGB &&
132 profile.topology != colorimetric_response::EmitterTopology::RGBW &&
133 profile.topology != colorimetric_response::EmitterTopology::RGBWW) ||
134 (profile.topology != colorimetric_response::EmitterTopology::RGB &&
135 (!validChromaticity(profile.xy_white1) ||
136 !finitePositive(profile.lum_white1) ||
137 (profile.response_lut_size != 0 &&
138 profile.response_lut_white1 == nullptr))) ||
139 (profile.topology == colorimetric_response::EmitterTopology::RGBWW &&
140 (!validChromaticity(profile.xy_white2) ||
141 !finitePositive(profile.lum_white2) ||
142 (profile.response_lut_size != 0 &&
143 profile.response_lut_white2 == nullptr))) ||
144 (profile.response_lut_size != 0 &&
145 (profile.response_lut_r == nullptr || profile.response_lut_g == nullptr || profile.response_lut_b == nullptr))) {
146 clearColorProfile();
147 return false;
148 }
149 if (!validResponseLut(profile.response_lut_r, profile.response_lut_size) ||
150 !validResponseLut(profile.response_lut_g, profile.response_lut_size) ||
151 !validResponseLut(profile.response_lut_b, profile.response_lut_size)) {
152 clearColorProfile();
153 return false;
154 }
155 if (profile.topology != colorimetric_response::EmitterTopology::RGB &&
156 !validResponseLut(profile.response_lut_white1,
157 profile.response_lut_size)) {
158 clearColorProfile();
159 return false;
160 }
161 if (profile.topology == colorimetric_response::EmitterTopology::RGBWW &&
162 !validResponseLut(profile.response_lut_white2,
163 profile.response_lut_size)) {
164 clearColorProfile();
165 return false;
166 }
167 // Warn only once validation has passed. Warning earlier consumed
168 // mWarnedLegacyCleared on a profile that was then rejected, and
169 // clearColorProfile() leaves mCorrection/mTemperature alone -- so the
170 // next *valid* profile cleared the caller's legacy settings silently.
171 // Every field cleared below whose loss is detectable has to be
172 // represented here, or the caller loses it in silence. This used to
173 // name mCorrection and mTemperature only, while the clear also resets
174 // mGamma -- so a caller who set just a gamma had it discarded with no
175 // warning, which is exactly what the note above is about (#4331).
176 //
177 // mDitherMode is deliberately absent, and cannot be added by
178 // comparison. Its default is BINARY_DITHER and the clear sets
179 // DISABLE_DITHER, so `mDitherMode != BINARY_DITHER` is true precisely
180 // when the caller already chose DISABLE_DITHER and nothing is being
181 // taken away, and false in the case that loses dithering. Detecting
182 // that needs a "the caller set this" flag, which mGamma gets for free
183 // by being an optional. Until then, binding a profile turns dithering
184 // off quietly (#4331).
185 const bool clearing_legacy =
186 mCorrection != UncorrectedColor ||
187 mTemperature != UncorrectedTemperature ||
188 mGamma.has_value();
189 if (!had_profile && clearing_legacy && !mWarnedLegacyCleared) {
190 FL_WARN("Color profile clears legacy correction/temperature/gamma");
191 ChannelEvents::instance().onColorProfileWarning(
192 ColorProfileEvent{-1, {}, ColorProfileWarning::LegacyClearedByProfile});
193 mWarnedLegacyCleared = true;
194 }
195 // The one call that makes the colour pipeline reachable. Everything
196 // downstream goes through function pointers these install, so a
197 // program that never gets here never references the pipeline and the
198 // linker drops it -- about 3.5 KB of flash for sketches that do not
199 // ask for colour management.
200 installColorPipelineHooks();
201 mColorProfile.mStorage = fl::make_shared<ColorProfileStorage>(profile);
202 mColorProfile.mSource = source;
203 mColorProfile.mGamut = gamut;
204 mColorProfile.mRequested = true;
205 mColorProfile.mUseGlobalSourceDefault = global_source;
206 mCorrection = UncorrectedColor;
207 mTemperature = UncorrectedTemperature;
208 mGamma.reset();
209 mDitherMode = DISABLE_DITHER;
210 return true;
211#endif
212 }

222 void clearColorProfile() FL_NO_EXCEPT {
223#if FL_COLOR_PROFILE_RUNTIME
224 mColorProfile.mStorage.reset();
225 mColorProfile.mStaticProfile = nullptr;
226 mColorProfile.mRequested = false;
227 // Also drop the global-source opt-in. Without this, a profile bound
228 // by setColorProfile() leaves the flag set, and a following
229 // requestColorManagement(explicitSource) has its source silently

231 // setColorProfile() re-establishes the flag itself when it succeeds.
232 mColorProfile.mUseGlobalSourceDefault = false;
233#endif
