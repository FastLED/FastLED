42// =============================================================================
43// FASTLED_LOG_VERBOSITY - release-mode knob to no-op FL_WARN/FL_INFO/
44// FL_ERROR/FL_PRINT and free ~20-40 KB of
45// `.flash.rodata` string-pool bloat on embedded
46// builds. See FastLED #2773 item 2.3.
47// =============================================================================
48//
49// Each `FL_WARN(...)` / `FL_INFO(...)` / `FL_ERROR(...)` / `FL_PRINT(...)`
50// call site bakes a `__FILE__ + __LINE__ + user-supplied message` literal
51// into `.flash.rodata`. On the ESP32-S3 NEOPIXEL Blink build that pool is
52// ~57 KB - the single biggest `.rodata` contributor (see the audit on
53// #2473). The strings are live only because their call sites are live.
54//
55// Levels (mirror standard log-verbosity conventions; higher = more output):
56//
57// * `FASTLED_LOG_VERBOSITY == 0` -> ALL of FL_WARN/FL_INFO/FL_ERROR/
58// FL_PRINT/FL_DBG expand to `do {} while(0)`. Strings, `fl::sstream`
59// uses, and any transitive `fl::println` references vanish; the
60// downstream printf chain may shrink further as a result.
61// * `FASTLED_LOG_VERBOSITY == 1` -> current behavior; FL_WARN /
62// FL_INFO / FL_ERROR / FL_PRINT fire on platforms where
63// `SKETCH_HAS_LARGE_MEMORY` is set. FL_DBG follows its existing
64// `FASTLED_HAS_DBG` gate.
65// * `FASTLED_LOG_VERBOSITY >= 2` -> reserved for future "even noisier
66// than debug" output; currently equivalent to level 1.
67//
68// Default selection (in resolution order):
69//
70// * `FASTLED_TESTING` is set -> 1 (host unit tests need full diagnostics)
71// * `NDEBUG` is set (release) -> 0 (drop ~55 KB of FL_WARN/FL_LOG strings
72// on ESP32-S3 NEOPIXEL Blink; see #2886)
73// * otherwise (debug builds) -> 1 (preserve current behavior)
74//
75// Users who want logs back on a release build define
76// `-DFASTLED_LOG_VERBOSITY=1` in their build flags or
77// `#define FASTLED_LOG_VERBOSITY 1` before `#include <FastLED.h>`. The
78// per-channel logging defines (`FASTLED_LOG_RMT_ENABLED`, `FASTLED_LOG_SPI_ENABLED`,
79// etc.) are still gated by their own `#define`s - only the verbosity floor
80// changes here.
81#ifdef FASTLED_TESTING
82 // Host unit tests always want the full diagnostic stream so assertion
83 // and warning messages can be checked in CI.
84 #if !defined(FASTLED_LOG_VERBOSITY) || FASTLED_LOG_VERBOSITY < 1
85 #undef FASTLED_LOG_VERBOSITY
86 #define FASTLED_LOG_VERBOSITY 1
87 #endif
88#else
89 #ifndef FASTLED_LOG_VERBOSITY
90 #ifdef NDEBUG
91 #define FASTLED_LOG_VERBOSITY 0
92 #else
93 #define FASTLED_LOG_VERBOSITY 1
94 #endif
95 #endif
96#endif

149// =============================================================================
150// Centralised log emit (#2963 Proposal B, Option 3)
151// =============================================================================
152//
153// `fl::detail::log_emit(kind, file, line, body)` consumes the
154// user-supplied sstream (passed by rvalue reference) and **rewrites
155// it in place** to hold the full prefixed message:
156//
157// "<file>(<line>): <KIND>: <user payload>"
158//
159// then emits via `fl::println(body.c_str())`.
160//
161// Lifetime-safe variant of Path C's "Option 1" attempt. The body
162// sstream temporary lives in the **caller's** frame - its
163// lifetime extends to the end of the FL_WARN/FL_ERROR/FL_INFO
164// full expression, exactly as the OLD inline macro's temporary
165// did. So `body.c_str()` passed to `println` has the same
166// async-handler-safe lifetime as before: whatever the test
167// harness or platform println did synchronously continues to
168// work, and async deferred-flush handlers still see a live
169// pointer until the semicolon.
170//
171// The legacy macro inlined the entire
172// `<< file << "(" << line << "): KIND: " << X`
173// chain at every FL_WARN site (1,372 sites × ~30-50 B of inlined
174// prefix-format code). With this helper, the prefix-format chain
175// is compiled ONCE into libFastLED - each call site only emits
176// the `fl::sstream() << X` ctor + the user's payload chain + a
177// single `log_emit` call.
178namespace fl { namespace detail {
179enum class log_kind : fl::u8 {
180 WARN = 0,
181 ERROR = 1,
182 INFO = 2,
183};
184
185const char* log_kind_name(log_kind kind) FL_NO_INLINE FL_NO_EXCEPT;
186void log_emit_prefix(log_kind kind, const char* file, int line) FL_NO_INLINE FL_NO_EXCEPT;
187void log_emit_newline() FL_NO_INLINE FL_NO_EXCEPT;

189template <typename... Args>
190void log_emit_f(log_kind kind, const char* file, int line, const char* format,
191 const Args&... args) FL_NO_EXCEPT {
192 log_emit_prefix(kind, file, line);
193 fl::printf(format, args...);
194 log_emit_newline();
195}

197template <typename... Args>
198fl::string log_format_string(const char* format, const Args&... args) FL_NO_EXCEPT {
199 char buffer[256];

202}
203// Takes `body` by non-const lvalue reference (not `&&`) because the
204// macro's `fl::sstream() << X` expression has type `sstream&` (the
205// returned lvalue ref from chained operator<<). The underlying
206// temporary's lifetime is the FL_WARN full-expression's, so mutating
207// it through this lvalue ref is well-defined and the c_str() handed
208// to println outlives the call exactly as the OLD inline macro's
209// `(fl::sstream() << ...).c_str()` did.
210//
211// `FL_NO_INLINE` enforces the centralisation. Without it, at -O2
212// with LTO the compiler often inlines `log_emit` at every FL_WARN
213// site - defeating the whole point of moving the prefix chain
214// out-of-line. The noinline attribute is what makes this proposal
215// actually win bytes; bare centralisation alone isn't enough.
216void log_emit(log_kind kind, const char* file, int line, fl::sstream& body) FL_NO_INLINE FL_NO_EXCEPT;
217void log_emit_literal(log_kind kind, const char* file, int line,
218 const char* body) FL_NO_INLINE FL_NO_EXCEPT;

220// A one-argument log macro cannot distinguish `"literal"` from
221// `"prefix" << value` by argument count. log_seed makes that distinction in
222// the expression's type instead: a lone string literal remains a lightweight
223// log_literal, while a subsequent << materializes an sstream. This preserves
224// legacy stream syntax without charging literal call sites for an sstream.
225template <fl::size N>
226class log_literal {
227public:
228 explicit log_literal(const char (&text)[N]) FL_NO_EXCEPT : mText(text) {}
229
230 const char* c_str() const FL_NO_EXCEPT { return mText; }
231
232 template <typename T>
233 fl::sstream operator<<(const T& value) const FL_NO_EXCEPT {
234 fl::sstream body;
235 body << mText << value;
236 return body;
237 }
238
239private:
240 const char* mText;
241};
242
243class log_seed {
244public:
245 template <fl::size N>
246 log_literal<N> operator<<(const char (&text)[N]) const FL_NO_EXCEPT {
247 return log_literal<N>(text);
248 }
249
250 template <typename T>
251 fl::sstream operator<<(const T& value) const FL_NO_EXCEPT {
252 fl::sstream body;
253 body << value;
254 return body;
255 }
256};
257

310// =============================================================================
311// Unified-dispatch helpers for FL_WARN / FL_ERROR / FL_INFO / FL_PRINT (#3272)
312// =============================================================================
313// One `FL_WARN(...)` macro accepts BOTH stream-style and printf-style:
314//
315// FL_WARN("plain literal") // 1 literal -> direct format path
316// FL_WARN("foo " << x << " bar") // 1 stream expression -> sstream
317// FL_WARN("got %d items", n) // 2 args -> printf form (log_emit_f)
318//
319// Macro-level dispatch (rather than C++ overload resolution) is required
320// because the single-arg stream form is a chained `<<` expression - it
321// only compiles when the macro injects `fl::sstream()` as the LHS.
322// Routing through a function call would force the `<<` chain to evaluate
323// standalone, which fails (`const char*` has no `operator<<(float)`).
324//
325// The dispatch supports up to 16 user arguments in the printf form. The
326// filler list `_FL_M*15, _FL_S` is positioned so the (17 - argc)-th element
327// of the merged list lands at the `NAME` slot:
328// 1 user arg -> NAME = _FL_S (lazy literal/stream dispatch)
329// 2..16 args -> NAME = _FL_M (multi -> printf)
330#define _FL_VA_PICK17( \
331 _1, _2, _3, _4, _5, _6, _7, _8, \
332 _9, _10, _11, _12, _13, _14, _15, _16, \
333 NAME, ...) NAME
334
335#define _FL_VA_DISPATCH(_S, _M, ...) \
336 _FL_VA_PICK17(__VA_ARGS__, \
337 _M, _M, _M, _M, _M, _M, _M, _M, \
338 _M, _M, _M, _M, _M, _M, _M, _S)
339
