43void encodeHD108(InputIterator first, InputIterator last, OutputIterator out,
44 u8 global_brightness = 255) FL_NO_EXCEPT {
45 // Start frame: 64 bits (8 bytes) of 0x00
46 for (int i = 0; i < 8; i++) {
47 *out++ = 0x00;
48 }
49
50 // Compute brightness header bytes (cached for all LEDs)
51 u8 f0, f1;
52 hd108BrightnessHeader(global_brightness, &f0, &f1);
53
54 // LED data: 2-byte header + 6-byte RGB16 (count as we go)
55 size_t num_leds = 0;
56 while (first != last) {
57 const fl::array<u8, BYTES_PER_PIXEL_RGB>& pixel = *first;
58
59 // Apply gamma correction (2.8) to 16-bit (RGB order: 0=R, 1=G, 2=B)
60 u16 r16 = hd108GammaCorrect(pixel[0]); // Red
61 u16 g16 = hd108GammaCorrect(pixel[1]); // Green
62 u16 b16 = hd108GammaCorrect(pixel[2]); // Blue
63
64 // Transmit: 2 header + 6 color bytes (16-bit RGB, big-endian)
65 *out++ = f0;
66 *out++ = f1;
67 *out++ = static_cast<u8>(r16 >> 8);
68 *out++ = static_cast<u8>(r16 & 0xFF);
69 *out++ = static_cast<u8>(g16 >> 8);
70 *out++ = static_cast<u8>(g16 & 0xFF);
71 *out++ = static_cast<u8>(b16 >> 8);
72 *out++ = static_cast<u8>(b16 & 0xFF);
73
74 ++first;
75 ++num_leds;
76 }
77
78 // End frame: (num_leds / 2) + 4 bytes of 0xFF
79 const size_t latch = num_leds / 2 + 4;

108 // argument -- `(void)brightness_8bit`, then three `constexpr` gains of 31

114 // per LED to fetch a number that reached nothing. See FastLED#4402.
