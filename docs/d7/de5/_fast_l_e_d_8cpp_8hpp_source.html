159fl::PowerCodecPolicy fl::powerChannelCodecPolicy(
160 const fl::CLEDController& controller) FL_NO_EXCEPT {
161 PowerCodecPolicy policy;
162 const fl::vector<fl::ChannelPtr>& registered = CFastLED::channels();
163 for (fl::size i = 0; i < registered.size(); ++i) {
164 const fl::Channel* channel = registered[i].get();
165 if (channel != &controller) continue;
166 policy.kind = PowerCodecKind::Byte;
167 const fl::SpiChipsetConfig* spi =
168 channel->getChipset().ptr<fl::SpiChipsetConfig>();
169 if (spi == nullptr) return policy;
170 switch (spi->timing.chipset) {
171 case fl::SpiChipset::LPD6803:
172 policy.kind = PowerCodecKind::Native5;
173 break;
174 case fl::SpiChipset::LPD8806:
175 policy.kind = PowerCodecKind::Native7;
176 break;
177 case fl::SpiChipset::APA102HD:
178 case fl::SpiChipset::DOTSTARHD:
179 case fl::SpiChipset::SK9822HD:
180 case fl::SpiChipset::HD107HD: {
181 policy.kind = PowerCodecKind::FiveBit;
182 const fl::EmitterProfile* profile = channel->emitterProfile();
183 const fl::FiveBitSemantics semantics = fl::fiveBitSemanticsFor(
184 spi->timing.chipset, profile != nullptr
185 ? profile->five_bit_semantics
186 : fl::FiveBitSemantics::NotApplicable);
187 policy.min_field = fl::hdMinimumField(
188 semantics, fl::detail::hdFieldFloor());
189 break;
190 }
191 default:
192 break;
193 }
194 return policy;
195 }
196 // A custom managed controller has no known wire encoding. Charging full
197 // code for every lit emitter is conservative, never a silent guess.
198 return policy;
199}

227void CFastLED::clear(ClearFlags flags) {
228 // Lambda to check if flag is set, clear it, and return true if it was set
229 auto clearFlag = [&flags](ClearFlags flag) -> bool {
230 if ((flags & flag) != ClearFlags::NONE) {
231 flags = static_cast<ClearFlags>(static_cast<fl::u32>(flags) & ~static_cast<fl::u32>(flag));
232 return true;
233 }
234 return false;
235 };
236
237
238 // Reset POWER_SETTINGS - reset power management to defaults
239 if (clearFlag(ClearFlags::POWER_SETTINGS)) {
240 FastLED.mPPowerFunc = nullptr; // No power limiting function
241 FastLED.mNPowerData = 0xFFFFFFFF; // No power limit (max value)
242 FastLED.mLastRequestedScale = 255;
243 FastLED.mLastShownScale = 255;
244 }
245
246 // Reset BRIGHTNESS - reset global brightness to 255 (full brightness)
247 if (clearFlag(ClearFlags::BRIGHTNESS)) {
248 FastLED.mScale = 255;
249 }
250
251 // Reset REFRESH_RATE - reset refresh rate limiting to unlimited
252 if (clearFlag(ClearFlags::REFRESH_RATE)) {
253 FastLED.mNMinMicros = 0; // No minimum delay between frames
254 }
255
256 // Reset FPS_COUNTER - reset FPS tracking counter to 0
257 if (clearFlag(ClearFlags::FPS_COUNTER)) {
258 FastLED.mNFPS = 0;
259 }
260
261 // Reset CHANNELS - remove all channels from controller list
262 if (clearFlag(ClearFlags::CHANNELS)) {
263 // Wait for engines to complete, with 2s timeout to prevent infinite hang
264 // if an engine is permanently stalled (e.g. PARLIO on ESP32-C6 hw bug)
265 FastLED.wait(2000);
266
267 // Remove all channels by iterating through a copy of the vector
268 // (we make a copy because remove() modifies channels())
269 fl::vector<fl::ChannelPtr> channelsCopy = channels();
270 for (auto& channel : channelsCopy) {
271 remove(channel);
272 }
273
274 // Reset bus manager state (clear enqueued and transmitting channels)
275 fl::ChannelManager& manager = fl::channelManager();
276 manager.reset();
277
278 // Clear the internal storage (should already be empty, but ensure it)
279 channels().clear();
280 }
281
282 // Reset CHANNEL_ENGINES - clear all registered channel drivers
283 if (clearFlag(ClearFlags::CHANNEL_ENGINES)) {
284 // Always wait for all channel bus transmissions to complete first
285 FastLED.wait();
286 fl::ChannelManager& manager = fl::channelManager();
287 manager.clearAllDrivers();
288 }
289
290 // Ensure all flags were handled - catches typos or missing implementations
291 FL_ASSERT(flags == ClearFlags::NONE, "Unhandled flags in FastLED.clear()");
292}

294// This is bad code. But it produces the smallest binaries for reasons
295// beyond mortal comprehensions.
296// Instead of iterating through the link list for onBeginFrame(), beginShowLeds()
297// and endShowLeds(): store the pointers in an array and iterate through that.
298//
299// static uninitialized gControllersData produces the smallest binary on attiny85.
300// FL_LINT_ALLOW_GLOBAL(intentional ATtiny85 code-size optimization — moving to Singleton<T> reverses the size win documented above)
301static void* gControllersData[MAX_CLED_CONTROLLERS];
302
303// Refresh-rate throttle shared by show() and showColor().
304//
305// This mirrors the tiered wait already shipped in
306// IChannelDriver::waitForCondition (refs #2818 for the spin tier, #2815/#2817
307// for the adaptive yield). show() was the last call site still taking an
308// UNCONDITIONAL deep yield, which is the root cause of FastLED#2994.
309//
310// Why it matters: fl::task::run(<non-zero>, SYSTEM) routes to
311// pumpCoroutines(), which on ESP32 floors at one FreeRTOS tick
312// (coroutine_esp32.impl.hpp, the `if (ticks == 0) ticks = 1` branch added for
313// the #2254 websocket-starvation fix). That is >=1 ms at
314// CONFIG_FREERTOS_HZ=1000 and >=10 ms at the 100 Hz default. With
315// setMaxRefreshRate(800) the throttle only needs 1250 us, so every frame
316// overshot to a tick boundary -- ~500 shows per sequence x sub-ms jitter is
317// the 63-300 ms drift reported in #2994, against a rock-steady 2495 ms on
318// 3.10.3 (which used a plain busy-wait).
319//
320// Two tiers:
321// - No radio up: taskYIELD() only. Equal-priority tasks still run, and there
322// is no tick floor, so the wait stays microsecond-accurate.
323// - Radio up: keep the deep yield -- WiFi/lwIP/BT genuinely need the CPU --
324// but stop taking it once less than the spin budget remains, so the final
325// stretch is spun out instead of overshooting past our deadline.
326static void throttleToMaxRefreshRate(fl::u32 minMicros) FL_NO_EXCEPT {
327 if (!minMicros) {
328 return;
329 }
330#if SKETCH_HAS_LARGE_MEMORY
331 const fl::u32 spinBudget = fl::detail::getWaitSpinBudgetUs();
332#endif
333 while (true) {
334 const fl::u32 elapsed = fl::micros() - lastshow;
335 if (elapsed >= minMicros) {
336 return;
337 }
338#if SKETCH_HAS_LARGE_MEMORY
339 const fl::u32 remaining = minMicros - elapsed;
340 // Deep yield only while a radio is up AND there is more than the spin
341 // budget left to burn. Once inside the budget we spin, because a deep
342 // yield costs a whole tick and would overshoot the deadline.
343 if (remaining > spinBudget && fl::NetworkDetector::isAnyNetworkActive()) {
344 fl::task::run(250, fl::task::ExecFlags::SYSTEM);
345 } else {
346 // taskYIELD()-equivalent: equal-priority tasks still get CPU, but
347 // there is no FreeRTOS tick floor, so this stays sub-microsecond.
348 fl::task::run(0, fl::task::ExecFlags::SYSTEM);
349 }
350#endif
351 }
352}

362FL_KEEP_ALIVE void CFastLED::show(fl::u8 scale) {
363 FL_SCOPED_TRACE;
364 onBeginFrame();
365 throttleToMaxRefreshRate(mNMinMicros);
366 lastshow = fl::micros();
367
368 mLastRequestedScale = scale;
369 // If we have a function for computing power, use it!
370 if(mPPowerFunc) {
371 scale = (*mPPowerFunc)(scale, mNPowerData);
372 }
373 mLastShownScale = scale;
374
375
376 int length = 0;
377 CLEDController *pCur = CLEDController::head();
378
379 while(pCur && length < MAX_CLED_CONTROLLERS) {
380 if (pCur->getEnabled()) {
381 gControllersData[length] = pCur->beginShowLeds(pCur->size());
382 } else {
383 gControllersData[length] = nullptr;
384 }
385 length++;
386 if (mNFPS < FL_DITHER_ENABLE_MIN_REFRESH_HZ) { pCur->setDither(0); }
387 pCur = pCur->next();
388 }
389
390 pCur = CLEDController::head();
391 for (length = 0; length < MAX_CLED_CONTROLLERS && pCur; length++) {
392 if (pCur->getEnabled()) {
393 pCur->showLedsInternal(scale);
394 }
395 pCur = pCur->next();
396
397 }
398
399 length = 0; // Reset length to 0 and iterate again.
400 pCur = CLEDController::head();
401 while(pCur && length < MAX_CLED_CONTROLLERS) {
402 if (pCur->getEnabled()) {
403 pCur->endShowLeds(gControllersData[length]);
404 }
405 length++;
406 pCur = pCur->next();
407 }
408 countFPS();
409 onEndFrame();
410 onEndShowLeds();
411}

556fl::u32 CFastLED::getEstimatedPowerInMilliWatts(bool apply_limiter) const {
557#if FL_COLOR_PIPELINE_SHARED
558 if (apply_limiter && mPPowerFunc && fl::activeFramePowerDispatch()) {
559 bool managed = false;
560 for (CLEDController* p = CLEDController::head(); p; p = p->next()) {
561 if (p->getEnabled() && p->colorPipeline()) {
562 managed = true;
563 break;
564 }
565 }
566 if (managed) {
567 const fl::FramePowerPlan plan =
568 fl::activeFramePowerDispatch()->calculate(mScale, mNPowerData);
569 if (plan.modeled_mW == 0xFFFFFFFFu) return plan.modeled_mW;
570 const fl::u32 mcu_mW = plan.mcu_mW;
571 return plan.modeled_mW > mcu_mW ? plan.modeled_mW - mcu_mW : 0;
572 }
573 }
574#endif
575 fl::u32 fixed_power_mW = 0;
576 fl::u32 controllable_power_mW = 0;
577 fl::u32 dither_reserve_power_mW = 0;
578
579 // Sum unscaled power from all LED controllers using visitor pattern.
580 // The dark-current baseline is kept apart from the part brightness scales,
581 // because it is drawn at every brightness including zero (#4156 R4).
582 const fl::u32 dark_mW_per_led = get_power_model().dark_mW;
583 CLEDController::visitControllers([&](const CLEDController* controller, fl::span<const CRGB> leds) {
584 if (!leds.empty()) {
585 // The controller's own estimate: its RGBW setting, so an RGBW strip
586 // is charged for the diode it actually lights, and for a bound
587 // colour profile the solved drives rather than the source (#4344).
588 const fl::u32 unscaled_mW = controller_unscaled_power_mW(*controller);
589 const fl::u32 dark_mW = dark_mW_per_led * static_cast<fl::u32>(leds.size());
590 fixed_power_mW += dark_mW;
591 controllable_power_mW += unscaled_mW > dark_mW ? unscaled_mW - dark_mW : 0;
592 // What dithering can add on top, at any non-zero brightness (#4342).
593 // Not on TINY, where the bound costs more flash than the tier has
594 // to spare (#4478); controller_dither_reserve_mW is 0 there.
595#if !FL_PLATFORM_HAS_TINY_MEMORY
596 dither_reserve_power_mW += controller_dither_reserve_mW(*controller);
597#endif
598 }
599 });
600
601 // Determine effective brightness
602 fl::u8 effective_brightness = mScale;
603
604 if (apply_limiter && mPPowerFunc) {
605 // Power limiting is enabled and user wants limited power - calculate brightness after limiting
606 // This calls calculate_max_brightness_for_power_mW(mScale, mNPowerData)
607 effective_brightness = (*mPPowerFunc)(mScale, mNPowerData);
608 }
609
610 // Scale only the emitter share by the configured power-brightness response
611 // and add the baseline back, so the estimate matches what the limiter now
612 // budgets against.
613 // Note: MCU power consumption is NOT included - caller should add platform-specific MCU power if needed
614 // The dither reserve only at a non-zero brightness: nothing dithers at 0,
615 // the same as the limiter, whose answer of 0 needs no reserve either.
616 if (effective_brightness != 0) {
617 fixed_power_mW += dither_reserve_power_mW;
618 }
619 return fixed_power_mW +
620 scale_power_for_brightness(controllable_power_mW, effective_brightness);
621}

625// uint32_t x, y, t;
626//
627// // Load the array and pack it into x and y.
628// y = *(unsigned int*)(A);
629// x = *(unsigned int*)(A+4);
630//
631// // x = (A[0]<<24) | (A[m]<<16) | (A[2*m]<<8) | A[3*m];
632// // y = (A[4*m]<<24) | (A[5*m]<<16) | (A[6*m]<<8) | A[7*m];
633//
634 // // pre-transform x
635 // t = (x ^ (x >> 7)) & 0x00AA00AA; x = x ^ t ^ (t << 7);
636 // t = (x ^ (x >>14)) & 0x0000CCCC; x = x ^ t ^ (t <<14);
637 //
638 // // pre-transform y
639 // t = (y ^ (y >> 7)) & 0x00AA00AA; y = y ^ t ^ (t << 7);
640 // t = (y ^ (y >>14)) & 0x0000CCCC; y = y ^ t ^ (t <<14);
641 //
642 // // final transform
643 // t = (x & 0xF0F0F0F0) | ((y >> 4) & 0x0F0F0F0F);
644 // y = ((x << 4) & 0xF0F0F0F0) | (y & 0x0F0F0F0F);
645 // x = t;
646//
647// B[7*n] = y; y >>= 8;

674void CFastLED::countFPS(int nFrames) {
675 static int br = 0;
676 static fl::u32 lastframe = 0; // fl::millis();
677
678 if(br++ >= nFrames) {
679 fl::u32 now = fl::millis();
680 now -= lastframe;
681 if(now == 0) {
682 now = 1; // prevent division by zero below
683 }
684 mNFPS = (br * 1000) / now;
685 br = 0;
686 lastframe = fl::millis();
687 }

800fl::ChannelPtr CFastLED::add(const fl::ChannelConfig& config) {
801 // Issue #2459: the non-template `FastLED.add(cfg)` path is the runtime-
802 // selection mode. To make sure `cfg.options.mBus` (or priority dispatch
803 // when `mBus == Bus::AUTO`) can actually find the requested driver at
804 // runtime, we eagerly enroll every driver available on this platform.
805 // That trades binary size for ergonomics â€” the user wanted runtime
806 // selection, so they get runtime selection.
807 //
808 // For minimum binary size, callers should use the compile-time path:
809 // FastLED.addLeds<CHIPSET, PIN, ORDER, fl::Bus::X>(leds, n);
810 // which ODR-uses only `BusTraits<X>::instancePtr()` and lets
811 // `--gc-sections` drop every other driver TU.
812 //
813 // The warning fires at most once per process (FL_WARN_ONCE) and can be
814 // disabled with `-DFASTLED_SUPPRESS_RUNTIME_DRIVER_WARNING`.
815 #ifndef FASTLED_SUPPRESS_RUNTIME_DRIVER_WARNING
816 FL_WARN_ONCE("FastLED.add(cfg): runtime-selection mode â€” enrolling every "
817 "available driver via fl::enableAllDrivers(). For minimum "
818 "binary size, prefer FastLED.addLeds<CHIPSET, PIN, ORDER, "
819 "fl::Bus::X>(leds, n) which links only the named driver. "
820 "Suppress this warning with -DFASTLED_SUPPRESS_RUNTIME_DRIVER_WARNING.");
821 #endif
822 fl::enableAllDrivers();
823
824 fl::ChannelManager& manager = fl::channelManager();
825 FL_ASSERT(manager.getDriverCount() > 0,
826 "No channel drivers available - channel API requires at least one registered driver");
827 auto channel = fl::Channel::create(config);
828 add(channel);
829 return channel;
830}
