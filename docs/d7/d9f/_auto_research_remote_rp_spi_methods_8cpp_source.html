29bool parseInt(const fl::json& config, const char* key, int* value) {
30 if (!config.contains(key) || !config[key].is_int()) return false;
31 const fl::i64 parsed = config[key].as_int().value();
32 if (parsed < static_cast<fl::i64>((fl::numeric_limits<int>::min)()) ||
33 parsed > static_cast<fl::i64>((fl::numeric_limits<int>::max)())) {
34 return false;
35 }
36 *value = static_cast<int>(parsed);
37 return true;
38}

40template <fl::u8 Which>
41fl::json runLoopback(const fl::shared_ptr<AutoResearchState>& runtime_state,
42 int mosi_pin, int miso_pin, int sck_pin, int clock_hz) {
43 const fl::u8 loopback_pattern[] = {
44 0x00, 0xFF, 0x55, 0xAA, 0x12, 0x34, 0x56, 0x78,
45 0x87, 0x65, 0x43, 0x21, 0x0F, 0xF0, 0xCC, 0x33,
46 0x11, 0xEE, 0x22, 0xDD, 0x44, 0xBB, 0x88, 0x77,
47 0x66, 0x99, 0x3C, 0xC3, 0x5A, 0xA5, 0x7E, 0x81,
48 };
49 fl::json response = fl::json::object();
50 const int expected_mosi = Which == 0 ? 3 : 11;
51 const int expected_miso = Which == 0 ? 0 : 8;
52 const int expected_sck = Which == 0 ? 2 : 10;
53 if (mosi_pin != expected_mosi || miso_pin != expected_miso ||
54 sck_pin != expected_sck) {
55 response.set("success", false);
56 response.set("error", "UnsupportedPins");
57 response.set("message", "Pins do not select the requested RP fixed SPI instance.");
58 return response;
59 }
60 if (clock_hz < 100000 || clock_hz > 62500000) {
61 response.set("success", false);
62 response.set("error", "InvalidClock");
63 response.set("message", "clock_hz must be in [100000, 62500000].");
64 return response;
65 }
66
67 // The request is now fully validated. Release AutoResearch's diagnostic
68 // PIO reservation only when the fixed-SPI operation will actually run.
69 if (runtime_state) {
70 runtime_state->rx_channel.reset();
71 runtime_state->invalidateLedRxValidation();
72 }
73
74 fl::vector_psram<fl::u8> tx;
75 tx.assign(loopback_pattern, loopback_pattern + sizeof(loopback_pattern));
76 fl::u8 captured[sizeof(loopback_pattern)] = {};
77 auto channel = fl::ChannelData::create(
78 fl::SpiChipsetConfig(mosi_pin, sck_pin,
79 fl::SpiEncoder::apa102(static_cast<fl::u32>(clock_hz))),
80 fl::move(tx));
81 auto& driver = fl::BusTraits<fl::Bus::SPI, Which>::instance();
82 if (!driver.captureNextRxBytes(captured, sizeof(captured))) {
83 response.set("success", false);
84 response.set("error", "DriverBusy");
85 response.set("message", "RP SPI driver was busy before the loopback transfer.");
86 return response;
87 }
88 driver.enqueue(channel);
89 driver.show();
90
91 fl::IChannelDriver::DriverState state = fl::IChannelDriver::DriverState::BUSY;
92 const unsigned long start_ms = millis();
93 while (millis() - start_ms < 2000) {
94 state = driver.poll();
95 if (state == fl::IChannelDriver::DriverState::READY ||
96 state == fl::IChannelDriver::DriverState::ERROR) {
97 break;
98 }
99 delay(1);
100 }
101
102 bool exact_match = state == fl::IChannelDriver::DriverState::READY;
103 fl::json expected = fl::json::array();
104 fl::json received = fl::json::array();
105 for (size_t index = 0; index < sizeof(loopback_pattern); ++index) {
106 expected.push_back(static_cast<int64_t>(loopback_pattern[index]));
107 received.push_back(static_cast<int64_t>(captured[index]));
108 exact_match = exact_match && captured[index] == loopback_pattern[index];
109 }
110
111 fl::json data = fl::json::object();
112 data.set("driver", Which == 0 ? "SPI0" : "SPI1");
113 data.set("mosiPin", static_cast<int64_t>(mosi_pin));
114 data.set("misoPin", static_cast<int64_t>(miso_pin));
115 data.set("sckPin", static_cast<int64_t>(sck_pin));
116 data.set("requestedClockHz", static_cast<int64_t>(clock_hz));
117 data.set("actualClockHz", static_cast<int64_t>(driver.lastActualClockHz()));
118 data.set("wireIdle", state == fl::IChannelDriver::DriverState::READY);
119 data.set("expected", expected);
120 data.set("received", received);
121 response.set("success", exact_match);
122 response.set("message", exact_match ? "RP SPI loopback passed." : "RP SPI loopback failed.");
123 response.set("data", data);
124 return response;
125}

140template <bool Sk9822>
141fl::json runPublicApiLoopback(
142 const fl::shared_ptr<AutoResearchState>& runtime_state,
143 int pattern) {
144 constexpr fl::u8 kMosiPin = 11;
145 constexpr fl::u8 kSckPin = 10;
146 constexpr size_t kLedCount = 257;
147 constexpr size_t kFrameBytes = 4 + (kLedCount * 4) + 36;
148 PublicApiLoopbackState<Sk9822>& state = publicApiLoopbackState<Sk9822>();
149 CRGB* leds = state.leds;
150 fl::json response = fl::json::object();
151
152 if (pattern < 0 || pattern > 5) {
153 response.set("success", false);
154 response.set("error", "InvalidPattern");
155 return response;
156 }
157
158 if (runtime_state) {
159 runtime_state->rx_channel.reset();
160 runtime_state->invalidateLedRxValidation();
161 }
162
163 if (!state.initialized) {
164 if constexpr (Sk9822) {
165 FastLED.addLeds<SK9822, kMosiPin, kSckPin, RGB, 8000000,
166 fl::Bus::SPI, 1>(leds, kLedCount);
167 } else {
168 FastLED.addLeds<APA102, kMosiPin, kSckPin, RGB, 8000000,
169 fl::Bus::SPI, 1>(leds, kLedCount);
170 }
171 state.initialized = true;
172 }
173
174 for (size_t index = 0; index < kLedCount; ++index) leds[index] = CRGB::Black;
175 switch (pattern) {
176 case 1: for (size_t index = 0; index < kLedCount; ++index) leds[index] = CRGB::Red; break;
177 case 2: for (size_t index = 0; index < kLedCount; ++index) leds[index] = CRGB::Green; break;
178 case 3: for (size_t index = 0; index < kLedCount; ++index) leds[index] = CRGB::Blue; break;
179 case 4: for (size_t index = 0; index < kLedCount; ++index) leds[index] = CRGB::White; break;
180 case 5: leds[16] = CRGB::Red; break;
181 default: break;
182 }
183 fl::u8 captured[kFrameBytes] = {};
184 auto& driver = fl::BusTraits<fl::Bus::SPI, 1>::instance();
185 if (!driver.captureNextRxBytes(captured, sizeof(captured))) {
186 response.set("success", false);
187 response.set("error", "DriverBusy");
188 return response;
189 }
190 FastLED.show();
191 FastLED.wait(2000);
192
193 bool exact = driver.lastActualClockHz() > 0;
194 for (size_t index = 0; index < 4; ++index) exact = exact && captured[index] == 0x00;
195 for (size_t led = 0; led < kLedCount; ++led) {
196 const size_t offset = 4 + (led * 4);
197 exact = exact && captured[offset] == 0xFF;
198 exact = exact && captured[offset + 1] == leds[led].r;
199 exact = exact && captured[offset + 2] == leds[led].g;
200 exact = exact && captured[offset + 3] == leds[led].b;
201 }
202 for (size_t index = 4 + (kLedCount * 4); index < kFrameBytes; ++index) {
203 exact = exact && captured[index] == 0xFF;
204 }
205
206 fl::json data = fl::json::object();
207 data.set("chipset", Sk9822 ? "SK9822" : "APA102");
208 data.set("pattern", static_cast<int64_t>(pattern));
209 data.set("actualClockHz", static_cast<int64_t>(driver.lastActualClockHz()));
210 data.set("wireIdle", true);
211 data.set("frameBytes", static_cast<int64_t>(kFrameBytes));
212 response.set("success", exact);
213 response.set("message", exact ? "RP SPI public API loopback passed."
214 : "RP SPI public API loopback failed.");
215 response.set("data", data);
216 return response;
217}
