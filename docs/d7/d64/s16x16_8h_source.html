70 // Extracts bits [47:16] of the 64-bit product for Q16.16 fixed-point.

72 i32 lo, hi;

87 constexpr FASTLED_FORCE_INLINE s16x16 operator/(s16x16 b) const FL_NO_EXCEPT {
88#if FL_FIXED_POINT_NARROW_DIVIDE
89 // Same value as the expression below, without the 64-bit divide that
90 // costs a libgcc call on a core whose divider is 32 bits wide.
91 // FastLED#4307 measured that call at 47x the cost of `s8x8`'s single
92 // `SDIV`, and 36x scalar float. Magnitudes go through the unsigned
93 // routine and the sign is reapplied, which also gives the
94 // truncate-toward-zero that C++ integer division specifies.
95 const u32 numerator = mValue < 0
96 ? static_cast<u32>(-static_cast<i64>(mValue))
97 : static_cast<u32>(mValue);
98 const u32 denominator =
99 b.mValue < 0 ? static_cast<u32>(-static_cast<i64>(b.mValue))
100 : static_cast<u32>(b.mValue);
101 const u32 magnitude = divide64By32(numerator >> (32 - FRAC_BITS),
102 numerator << FRAC_BITS, denominator);
103 const bool negative = (mValue < 0) != (b.mValue < 0);
104 // Negated in unsigned space. The quotient of INT32_MIN by one is
105 // INT32_MIN, which is a representable s16.16 value of -32768.0, and
106 // negating it as a signed int is undefined -- it happened to give the
107 // right answer, which is exactly why a value comparison could not
108 // find it and UBSan could.
109 return from_raw(static_cast<i32>(negative ? 0u - magnitude : magnitude));
110#else
111 return from_raw(static_cast<i32>(
112 (static_cast<i64>(mValue) * (static_cast<i64>(SCALE))) / b.mValue));
113#endif
114 }

207 static FASTLED_FORCE_INLINE s16x16 atan(s16x16 x) FL_NO_EXCEPT {
208 constexpr s16x16 one(1.0f);
209 constexpr s16x16 pi_over_2(1.5707963f);
210 bool neg = x.mValue < 0;
211 s16x16 ax = abs(x);
212 s16x16 result;
213 if (ax <= one) {
214 result = atan_unit(ax);
215 } else {
216 result = pi_over_2 - atan_unit(one / ax);
217 }
218 return neg ? -result : result;
219 }
220
221 static FASTLED_FORCE_INLINE s16x16 atan2(s16x16 y, s16x16 x) FL_NO_EXCEPT {
222 constexpr s16x16 pi(3.1415926f);
223 constexpr s16x16 pi_over_2(1.5707963f);
224 if (x.mValue == 0 && y.mValue == 0) return s16x16();
225 if (x.mValue == 0) return y.mValue > 0 ? pi_over_2 : -pi_over_2;
226 if (y.mValue == 0) return x.mValue > 0 ? s16x16() : pi;
227 s16x16 ax = abs(x);
228 s16x16 ay = abs(y);
229 s16x16 a;
230 if (ax >= ay) {
231 a = atan_unit(ay / ax);
232 } else {
233 a = pi_over_2 - atan_unit(ax / ay);
234 }
235 if (x.mValue < 0) a = pi - a;
236 if (y.mValue < 0) a = -a;
237 return a;
238 }

261 // Signed cube root, exact to within one ULP (it truncates toward zero).
262 //
263 // The identity is the same one `sqrt` uses one power down: for a Q16 raw
264 // value `r` representing r/2^16, the root `y` satisfies
265 // (y/2^16)^3 = r/2^16, i.e. y^3 = r * 2^32 -- so the root is the integer
266 // cube root of the raw value shifted left by 2*FRAC_BITS.
267 //
268 // The shift cannot overflow: the widest input, raw = INT32_MIN, gives
269 // 2^31 << 32 = 2^63, which is representable in u64, and the result
270 // 2^21 = 32.0 is well inside i32.
271 static constexpr FASTLED_FORCE_INLINE s16x16 cbrt(s16x16 x) FL_NO_EXCEPT {
272 return x.mValue < 0
273 ? from_raw(-static_cast<i32>(fl::icbrt64(
274 static_cast<u64>(-static_cast<i64>(x.mValue)) << (2 * FRAC_BITS))))
275 : from_raw(static_cast<i32>(fl::icbrt64(
276 static_cast<u64>(x.mValue) << (2 * FRAC_BITS))));
277 }

279 static FASTLED_FORCE_INLINE s16x16 pow(s16x16 base, s16x16 exp) FL_NO_EXCEPT {
280 if (base.mValue <= 0) return s16x16();
281 constexpr s16x16 one(1.0f);
282 if (exp.mValue == 0) return one;
283 if (base == one) return one;
284 // Snap base values within ~2 ULPs of 1.0 to exactly 1.0 to dodge the
285 // log2(1+t) minimax polynomial's upper-endpoint residual (#2969).
286 constexpr i32 kOneRaw = static_cast<i32>(SCALE);
287 if (base.mValue >= (kOneRaw - 2) && base.mValue <= kOneRaw) {
288 return one;
289 }
290 return exp2_fp(exp * log2_fp(base));
291 }

366 // Returns 0-based position of highest set bit, or -1 if v==0.
367 static constexpr FASTLED_FORCE_INLINE int highest_bit(u32 v) FL_NO_EXCEPT {
368 return v == 0 ? -1 : _highest_bit_step(v, 0);

371 static constexpr int _highest_bit_step(u32 v, int r) FL_NO_EXCEPT {
372 return (v & 0xFFFF0000u) ? _highest_bit_step(v >> 16, r + 16)
373 : (v & 0x0000FF00u) ? _highest_bit_step(v >> 8, r + 8)
374 : (v & 0x000000F0u) ? _highest_bit_step(v >> 4, r + 4)
375 : (v & 0x0000000Cu) ? _highest_bit_step(v >> 2, r + 2)
376 : (v & 0x00000002u) ? r + 1
377 : r;
378 }

380 // Fixed-point log base 2 for positive values.
381 // Uses 4-term minimax polynomial for log2(1+t), t in [0,1).
382 // Horner evaluation uses i64 intermediates (24 frac bits) to minimize
383 // rounding error, then converts back to 16 frac bits.
384 static FASTLED_FORCE_INLINE s16x16 log2_fp(s16x16 x) FL_NO_EXCEPT {
385 u32 val = static_cast<u32>(x.mValue);
386 int msb = highest_bit(val);
387 i32 int_part = msb - FRAC_BITS;
388 i32 t;
389 if (msb >= FRAC_BITS) {
390 t = static_cast<i32>(
391 (val >> (msb - FRAC_BITS)) - (SCALE));
392 } else {
393 t = static_cast<i32>(
394 (val << (FRAC_BITS - msb)) - (SCALE));
395 }
396 // 4-term minimax coefficients for log2(1+t), t in [0,1).
397 // Stored as i64 with 24 fractional bits. Max product ~2^49, fits i64.
398 constexpr int IFRAC = 24;
399 constexpr i64 c0 = 24189248LL; // 1.44179 * 2^24
400 constexpr i64 c1 = -11728384LL; // -0.69907 * 2^24
401 constexpr i64 c2 = 6098176LL; // 0.36348 * 2^24
402 constexpr i64 c3 = -1788416LL; // -0.10660 * 2^24
403 // Extend t from 16 to 24 frac bits.
404 i64 t24 = static_cast<i64>(t) << (IFRAC - FRAC_BITS);
405 // Horner: t * (c0 + t * (c1 + t * (c2 + t * c3)))
406 i64 acc = c3;
407 acc = c2 + ((acc * t24) >> IFRAC);
408 acc = c1 + ((acc * t24) >> IFRAC);
409 acc = c0 + ((acc * t24) >> IFRAC);
410 i64 frac_part = (acc * t24) >> IFRAC;
411 // Convert from 24 frac bits back to 16.
412 i32 frac16 = static_cast<i32>(frac_part >> (IFRAC - FRAC_BITS));
413 return from_raw(static_cast<i32>(static_cast<u32>(int_part) << FRAC_BITS) + frac16);
414 }
415
416 // Fixed-point 2^x. Uses 4-term minimax polynomial for 2^t, t in [0,1).
417 // Horner evaluation uses i64 intermediates (24 frac bits) to minimize
418 // rounding error, then converts back to 16 frac bits.
419 static FASTLED_FORCE_INLINE s16x16 exp2_fp(s16x16 x) FL_NO_EXCEPT {
420 s16x16 fl_val = floor(x);
421 s16x16 fr = x - fl_val;
422 i32 n = fl_val.mValue >> FRAC_BITS;
423 if (n >= INT_BITS - 1) return from_raw(0x7FFFFFFF);
424 if (n < -FRAC_BITS) return s16x16();
425 i32 int_pow;
426 if (n >= 0) {
427 int_pow = static_cast<i32>(SCALE) << n;
428 } else {
429 int_pow = static_cast<i32>(SCALE) >> (-n);
430 }
431 // 4-term minimax coefficients for 2^t - 1, t in [0,1).
432 // Stored as i64 with 24 fractional bits. Max product ~2^48, fits i64.
433 constexpr int IFRAC = 24;
434 constexpr i64 d0 = 11629376LL; // 0.69316 * 2^24
435 constexpr i64 d1 = 4038400LL; // 0.24071 * 2^24
436 constexpr i64 d2 = 895232LL; // 0.05336 * 2^24
437 constexpr i64 d3 = 214016LL; // 0.01276 * 2^24
438 // Extend fr from 16 to 24 frac bits.
439 i64 fr24 = static_cast<i64>(fr.mValue) << (IFRAC - FRAC_BITS);
440 // Horner: 1 + fr * (d0 + fr * (d1 + fr * (d2 + fr * d3)))
441 i64 acc = d3;
442 acc = d2 + ((acc * fr24) >> IFRAC);
443 acc = d1 + ((acc * fr24) >> IFRAC);
444 acc = d0 + ((acc * fr24) >> IFRAC);
445 constexpr i64 one24 = 1LL << IFRAC;
446 i64 frac_pow24 = one24 + ((acc * fr24) >> IFRAC);
447 // Convert from 24 frac bits to 16 frac bits, then scale by int_pow.
448 i32 frac_pow16 = static_cast<i32>(frac_pow24 >> (IFRAC - FRAC_BITS));
449 i64 result =
450 (static_cast<i64>(int_pow) * frac_pow16) >> FRAC_BITS;
451 return from_raw(static_cast<i32>(result));
452 }

458 return static_cast<u32>(

462 // Polynomial atan for t in [0, 1]. Returns [0, π/4].
463 // 7th-order minimax: atan(t) ≈ t * (c0 + t² * (c1 + t² * (c2 + t² * c3)))
464 // Coefficients optimized via coordinate descent on s16x16 quantization grid.
465 static FASTLED_FORCE_INLINE s16x16 atan_unit(s16x16 t) FL_NO_EXCEPT {
466 constexpr s16x16 c0(0.9998779297f);
467 constexpr s16x16 c1(-0.3269348145f);
468 constexpr s16x16 c2(0.1594085693f);
469 constexpr s16x16 c3(-0.0472106934f);
470 s16x16 t2 = t * t;
471 return t * (c0 + t2 * (c1 + t2 * (c2 + t2 * c3)));
472 }
