79u16 decodeTransferU16(TransferFunction transfer, u8 code) FL_NO_EXCEPT {
80 switch (transfer) {
81 case TransferFunction::Srgb:
82 return FL_PGM_READ_WORD_ALIGNED(&SRGB_DECODE_LUT[code]);
83 case TransferFunction::Bt709:
84 return FL_PGM_READ_WORD_ALIGNED(&BT709_DECODE_LUT[code]);
85 case TransferFunction::Linear:
86 break;
87 }
88 // Linear needs no table: 257 maps 0..255 onto 0..65535 exactly, so 0 and
89 // 255 land on the endpoints without a rounding step.
90 return static_cast<u16>(static_cast<u32>(code) * 257u);
91}
