1213 template<typename T>
1214 class array_iterator {
1215 private:
1216 using variant_t = typename json_value::variant_t;
1217 variant_t* mVariant;
1218 size_t mIndex;
1219
1220 // Helper to get the size of the array regardless of its type
1221 size_t get_size() const FL_NO_EXCEPT {
1222 if (!mVariant) return 0;
1223
1224 if (mVariant->is<json_array>()) {
1225 auto ptr = mVariant->ptr<json_array>();
1226 return ptr ? ptr->size() : 0;
1227 }
1228
1229 if (mVariant->is<fl::vector<i16>>()) {
1230 auto ptr = mVariant->ptr<fl::vector<i16>>();
1231 return ptr ? ptr->size() : 0;
1232 }
1233
1234 if (mVariant->is<fl::vector<u8>>()) {
1235 auto ptr = mVariant->ptr<fl::vector<u8>>();
1236 return ptr ? ptr->size() : 0;
1237 }
1238
1239 if (mVariant->is<fl::vector<float>>()) {
1240 auto ptr = mVariant->ptr<fl::vector<float>>();
1241 return ptr ? ptr->size() : 0;
1242 }
1243
1244 return 0;
1245 }
1246
1247 // Helper to convert current element to target type T
1248 parse_result<T> get_value() const FL_NO_EXCEPT {
1249 if (!mVariant || mIndex >= get_size()) {
1250 return parse_result<T>(task::Error("Index out of bounds"));
1251 }
1252
1253 if (mVariant->is<json_array>()) {
1254 auto ptr = mVariant->ptr<json_array>();
1255 if (ptr && mIndex < ptr->size() && (*ptr)[mIndex]) {
1256 auto& val = *((*ptr)[mIndex]);
1257
1258 // Try to convert to T using the json_value conversion methods
1259 // Using FastLED type traits instead of std:: ones
1260 if (fl::is_same<T, bool>::value) {
1261 auto opt = val.as_bool();
1262 if (opt) {
1263 return parse_result<T>(*opt);
1264 } else {
1265 return parse_result<T>(task::Error("Cannot convert to bool"));
1266 }
1267 } else if (fl::is_integral<T>::value && fl::is_signed<T>::value) {
1268 auto opt = val.template as_int<T>();
1269 if (opt) {
1270 return parse_result<T>(*opt);
1271 } else {
1272 return parse_result<T>(task::Error("Cannot convert to signed integer"));
1273 }
1274 } else if (fl::is_integral<T>::value && !fl::is_signed<T>::value) {
1275 // For unsigned types, we check that it's integral but not signed
1276 auto opt = val.template as_int<T>();
1277 if (opt) {
1278 return parse_result<T>(*opt);
1279 } else {
1280 return parse_result<T>(task::Error("Cannot convert to unsigned integer"));
1281 }
1282 } else if (fl::is_floating_point<T>::value) {
1283 auto opt = val.template as_float<T>();
1284 if (opt) {
1285 return parse_result<T>(*opt);
1286 } else {
1287 return parse_result<T>(task::Error("Cannot convert to floating point"));
1288 }
1289 }
1290 } else {
1291 return parse_result<T>(task::Error("Invalid array access"));
1292 }
1293 }
1294
1295 if (mVariant->is<fl::vector<i16>>()) {
1296 auto ptr = mVariant->ptr<fl::vector<i16>>();
1297 if (ptr && mIndex < ptr->size()) {
1298 return parse_result<T>(static_cast<T>((*ptr)[mIndex]));
1299 } else {
1300 return parse_result<T>(task::Error("Index out of bounds in i16 array"));
1301 }
1302 }
1303
1304 if (mVariant->is<fl::vector<u8>>()) {
1305 auto ptr = mVariant->ptr<fl::vector<u8>>();
1306 if (ptr && mIndex < ptr->size()) {
1307 return parse_result<T>(static_cast<T>((*ptr)[mIndex]));
1308 } else {
1309 return parse_result<T>(task::Error("Index out of bounds in u8 array"));
1310 }
1311 }
1312
1313 if (mVariant->is<fl::vector<float>>()) {
1314 auto ptr = mVariant->ptr<fl::vector<float>>();
1315 if (ptr && mIndex < ptr->size()) {
1316 return parse_result<T>(static_cast<T>((*ptr)[mIndex]));
1317 } else {
1318 return parse_result<T>(task::Error("Index out of bounds in float array"));
1319 }
1320 }
1321
1322 return parse_result<T>(task::Error("Unknown array type"));
1323 }
