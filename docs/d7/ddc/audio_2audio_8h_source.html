141 // "Zero crossing factor". High values > .4 indicate hissing
142 // sounds. For example a microphone rubbing against a clothing.
143 // These types of signals indicate the audio should be ignored.
144 // Low zero crossing factors (with loud sound) indicate that there
145 // is organized sound like that coming from music. This is so cheap
146 // to calculate it's done automatically. It should be one of the first
147 // signals to reject or accept a sound signal.
148 //
149 // Returns: a value -> [0.0f, 1.0f)
150 // O(1) - pre-computed in constructor
151 float zcf() const FL_NO_EXCEPT {
152 const fl::size n = pcm().size();
153 if (n < 2) {
154 return 0.f;
155 }
156 return float(mZeroCrossings) / static_cast<float>(n - 1);
157 }
158
159 // Root mean square amplitude of the audio signal
160 // Returns: RMS value (computed lazily on first access)
161 float rms() const FL_NO_EXCEPT {
162 if (!mRmsComputed) {
163 const_cast<SampleImpl*>(this)->initRms();
164 const_cast<SampleImpl*>(this)->mRmsComputed = true;
165 }
166 return mRms;
167 }
168
169 private:
170 void initZeroCrossings() FL_NO_EXCEPT {
171 mZeroCrossings = 0;
172 if (mSignedPcm.size() > 1) {
173 for (fl::size i = 1; i < mSignedPcm.size(); ++i) {
174 const bool crossed =
175 (mSignedPcm[i - 1] < 0 && mSignedPcm[i] >= 0) ||
176 (mSignedPcm[i - 1] >= 0 && mSignedPcm[i] < 0);
177 if (crossed) {
178 ++mZeroCrossings;
179 }
180 }
181 }
182 }
183
184 void initRms() FL_NO_EXCEPT {
185 if (mSignedPcm.empty()) {
186 mRms = 0.0f;
187 return;
188 }
189 fl::u64 sum_sq = 0;
190 const int N = mSignedPcm.size();
191 for (int i = 0; i < N; ++i) {
192 fl::i32 x32 = fl::i32(mSignedPcm[i]);
193 sum_sq += x32 * x32;
194 }
195 mRms = sqrtf(float(sum_sq) / N);
196 }
197
198 VectorPCM mSignedPcm;
199 fl::u16 mZeroCrossings = 0;
200 float mRms = 0.0f; // Lazily computed RMS value
201 fl::u32 mTimestamp = 0;
202 mutable bool mRmsComputed = false; // Track if RMS has been computed
203};
