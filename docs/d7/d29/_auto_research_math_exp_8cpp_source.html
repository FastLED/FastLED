47Result run(fl::u32 iterations) {
48 Result r = {};
49 r.iterations = iterations;
50 r.large_memory = FL_PLATFORM_HAS_LARGE_MEMORY ? 1 : 0;
51
52 // Accuracy: the finite float range in 0.05 steps and the double range in
53 // 0.5 steps, against libm.
54 fl::u32 points = 0;
55 for (float x = -103.9f; x < 88.7f; x += 0.05f) {
56 const float want = ::expf(x);
57 const float got = fl::expf(x);
58 if (want <= 0.0f || want != want || want > 3.0e38f) {
59 continue;
60 }
61 const float d = (got > want ? got - want : want - got) / ulp_of(want);
62 if (d > r.worst_ulp_float) {
63 r.worst_ulp_float = d;
64 r.worst_x_float = x;
65 }
66 ++points;
67 }
68 for (double x = -745.0; x < 709.7; x += 0.5) {
69 const double want = ::exp(x);
70 const double got = fl::exp(x);
71 if (want <= 0.0 || want != want) {
72 continue;
73 }
74 const double d = (got > want ? got - want : want - got) / ulp_of(want);
75 if (d > r.worst_ulp_double) {
76 r.worst_ulp_double = d;
77 r.worst_x_double = x;
78 }
79 ++points;
80 }
81 r.accuracy_points = points;
82
83 // Speed: inputs over |x| <= 10, the constant-Q / Gaussian-window range.
84 static float inputs[256];
85 for (int i = 0; i < 256; ++i) {
86 inputs[i] = -10.0f + 20.0f * static_cast<float>(i) / 255.0f;
87 }
88 float sink = 0.0f;
89 r.fl_expf_us = time_us(&call_fl_expf, inputs, iterations, &sink);
90 r.libm_expf_us = time_us(&call_libm_expf, inputs, iterations, &sink);
91 r.fl_exp_us = time_us(&call_fl_exp, inputs, iterations, &sink);
92 r.libm_exp_us = time_us(&call_libm_exp, inputs, iterations, &sink);
93 r.sink = sink;
94
95 r.success = r.worst_ulp_float <= 2.0f && r.worst_ulp_double <= 2.0 && points > 0;
96 return r;
97}
