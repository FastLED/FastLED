66 template<typename ToRep, typename ToPeriod, typename FromRep, typename FromPeriod>
67 static constexpr ToRep duration_cast_impl(FromRep count) FL_NO_EXCEPT {
68 // Convert from source period to destination period
69 // target_count = source_count * (FromPeriod / ToPeriod)
70 // Using: target = source * FromPeriod::num * ToPeriod::den / (FromPeriod::den * ToPeriod::num)
71 using ratio = fl::ratio_divide<FromPeriod, ToPeriod>;
72 return static_cast<ToRep>(
73 static_cast<FromRep>(count) * static_cast<FromRep>(ratio::num) / static_cast<FromRep>(ratio::den)
74 );
75 }
