30int Scheduler::add_task(Handle t) {
31 if (t.is_valid()) {
32 for (const Handle& existing : mTasks) {
33 if (existing.mImpl == t.mImpl) {
34 return existing._id();
35 }
36 }
37 int task_id = mNextTaskId.fetch_add(1);
38 t._set_id(task_id);
39 const bool is_frame_task = t._type() == TaskType::kBeforeFrame ||
40 t._type() == TaskType::kAfterFrame;
41 mTasks.push_back(fl::move(t));
42 if (is_frame_task && !mFrameListenerRegistered) {
43 // The dedicated final listener works even where the general
44 // EngineEvents listener list is disabled for memory reasons.
45 EngineEvents::setFrameTaskListener(this);
46 mFrameListenerRegistered = true;
47 }
48 return task_id;
49 }
50 return 0; // Invalid task
51}

63void Scheduler::update() {
64 u32 current_time = fl::millis();
65
66 // Use index-based iteration to avoid iterator invalidation issues
67 for (fl::size i = 0; i < mTasks.size();) {
68 Handle& t = mTasks[i];
69
70 if (!t.is_valid() || t._is_canceled()) {
71 // erase() returns bool in fl::vector, not iterator
72 mTasks.erase(mTasks.begin() + i);
73 // Don't increment i since we just removed an element
74 } else {
75 // Check if task is ready to run (frame tasks will return false here)
76 bool should_run = t._ready_to_run(current_time);
77
78 if (should_run) {
79 // Update last run time for recurring tasks
80 t._set_last_run_time(current_time);
81
82 // Execute the task
83 if (t._has_then()) {
84 t._execute_then();
85 } else {
86 warn_no_then(t._id(), t._trace_label());
87 }
88
89 // Remove one-shot tasks, keep recurring ones
90 bool is_recurring = (t._type() == TaskType::kEveryMs || t._type() == TaskType::kAtFramerate);
91 if (is_recurring) {
92 ++i; // Keep recurring tasks
93 } else {
94 // erase() returns bool in fl::vector, not iterator
95 mTasks.erase(mTasks.begin() + i);
96 // Don't increment i since we just removed an element
97 }
98 } else {
99 ++i;
100 }
101 }
102 }
103 update_frame_listener_registration();
104}

124void Scheduler::update_tasks_of_type(TaskType task_type) {
125 u32 current_time = fl::millis();
126
127 remove_inactive_tasks();
128
129 // Stable handles isolate dispatch from callbacks that mutate mTasks.
130 // Separate reusable buffers preserve before/after nesting without a heap
131 // allocation on every frame after each buffer reaches its high-water mark.
132 fl::vector<Handle>& snapshot = task_type == TaskType::kBeforeFrame
133 ? mBeforeFrameSnapshot
134 : mAfterFrameSnapshot;
135 snapshot.clear();
136 for (const Handle& task : mTasks) {
137 if (task._type() == task_type &&
138 task._ready_to_run_frame_task(current_time)) {
139 snapshot.push_back(task);
140 }
141 }
142
143 const u32 clear_generation = mClearGeneration;
144 for (Handle& task : snapshot) {
145 if (mClearGeneration != clear_generation) {
146 break;
147 }
148 if (!task.is_valid() || task._is_canceled()) {
149 continue;
150 }
151
152 task._set_last_run_time(current_time);
153 if (task._has_then()) {
154 task._execute_then();
155 } else {
156 warn_no_then(task._id(), task._trace_label());
157 }
158 }
159 snapshot.clear();
160 remove_inactive_tasks();
161 update_frame_listener_registration();
162}
