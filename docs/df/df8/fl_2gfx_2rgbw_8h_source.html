33// Per-strip diode chromaticity + peak luminance, used by the colorimetric
34// modes. The default constant (`kRgbwDefaultProfile`) ships with the library
35// and uses datasheet-derived values for SK6812 RGBW3535 at 6000K. Users with a
36// colorimeter can build their own profile and pass it on the Rgbw setting used
37// by their controller.
38//
39// Source-space fields (input_xy_*, issue #2705) define the color space the
40// input RGB triple is interpreted in. The default profile (#2710) ships
41// with **native LED gamut + D65 white** — input primaries equal the LED's
42// own xy_r/g/b — so a saturated input like RGB=(255,0,0) reaches the LED
43// red diode at full drive and RGB=(255,255,255) lands on D65. Solvers
44// compute `X_t = M_src · source_rgb` where M_src is the standard CIE
45// primary-matrix construction from these four chromaticities.
46//
47// Users wanting standard color-space input semantics (Rec709 / sRGB,
48// Rec2020, DCI-P3 D65/D60) should opt in explicitly when creating the
49// owned profile, for example
50// `fl::color::make_diode_profile(profile, fl::color::InputGamut::Rec709)`.
51//
52// If input_xy_w[1] is left at 0.0f (the default for value-initialized
53// profiles `DiodeProfile{}`), solvers fall back to the legacy
54// interpretation in which the input RGB triple IS the device-emitter
55// drive coordinates — preserved for backward compatibility.
56struct DiodeProfile {
57 float xy_r[2]; // R diode chromaticity (CIE 1931 xy)
58 float xy_g[2]; // G diode chromaticity
59 float xy_b[2]; // B diode chromaticity
60 float xy_w[2]; // W diode chromaticity (at nominal_cct)
61 float lum_r; // R diode peak luminance (relative to W = 1.0)
62 float lum_g;
63 float lum_b;
64 float lum_w; // typically 1.0 by convention
65 int nominal_cct; // CCT at which xy_w was measured (e.g. 6000)
66
67 // Source / input color space (#2705). See above for default behavior.
68 float input_xy_r[2]; // source R primary chromaticity (default: native LED R)
69 float input_xy_g[2]; // source G primary chromaticity (default: native LED G)
70 float input_xy_b[2]; // source B primary chromaticity (default: native LED B)
71 float input_xy_w[2]; // source white chromaticity (default D65)
72};

74// Named source color spaces for opting into standard input-gamut semantics
75// (#2710). The default `kRgbwDefaultProfile` ships as `Native`; pick a
76// named gamut here when feeding content authored for that space (e.g.
77// sRGB / Rec.709 for ambilight, Rec.2020 for wide-gamut video, DCI-P3
78// for cinema-mastered material).
79enum class InputGamut : u8 {
80 Native = 0, // Input primaries == this profile's LED primaries + D65 white
81 Rec709, // sRGB / Rec.709 primaries + D65 white
82 Rec2020, // Rec.2020 (UHDTV) primaries + D65 white
83 DciP3D65, // DCI-P3 D65 — consumer display variant (e.g. Apple)
84 DciP3D60, // DCI-P3 D60 — ACES / cinema mastering variant
85};

151// Interpolation scheme used by the colorimetric LUT (#2720). Bilinear stores
152// 4 channel values per grid cell (8 B/cell at i16 quantization) — cheapest in
153// flash/RAM. Hermite additionally stores ∂/∂t_x and ∂/∂t_y per channel
154// (24 B/cell, 3x storage) and reaches lower error at the same grid size,
155// enabling small grids (N=8..16) on memory-constrained targets.
156enum class RgbwLutInterp : u8 {
157 Bilinear = 0,
158 Hermite = 1,
159};
160
161// Enable the 2D + 1D factored LUT path (issue #2545 Phase 2). When enabled,
162// the colorimetric modes use an interpolated LUT instead of solving per pixel.
163// Available whenever FASTLED_RGBW_COLORIMETRIC=1 — no separate
164// FASTLED_RGBW_COLORIMETRIC_LUT flag exists; gc-sections drops the LUT path
165// for sketches that never call this. `grid_n` is clamped to [4, 256] and
166// controls the LUT edge length. LUT is rebuilt whenever the active profile,
167// CCT, grid size, or interp scheme changes. Returns false on allocation
168// failure (or always false when FASTLED_RGBW_COLORIMETRIC is undefined).
169//
170// Memory cost = grid_n * grid_n * (interp == Hermite ? 24 : 8) bytes:
171// N=8 -> Bilinear 512 B / Hermite 1 536 B
172// N=16 -> Bilinear 2 048 B / Hermite 6 144 B
173// N=32 -> Bilinear 8 192 B / Hermite 24 576 B
174// N=64 -> Bilinear 32 KB / Hermite ~96 KB
175//
176// The single-arg overload preserves source compatibility and forwards to
177// RgbwLutInterp::Hermite (the historical default since #2707).
178bool enable_rgbw_colorimetric_lut(int grid_n,
179 RgbwLutInterp interp) FL_NO_EXCEPT;
180bool enable_rgbw_colorimetric_lut(int grid_n) FL_NO_EXCEPT;
181
182// Free the LUT and revert colorimetric calls to the closed-form solver.
183void disable_rgbw_colorimetric_lut() FL_NO_EXCEPT;
184
185// Returns true if the LUT path is currently active.
186bool rgbw_colorimetric_lut_enabled() FL_NO_EXCEPT;
187
188// Compile-time memory-cost accessor. Mirrors the storage math used by the
189// LUT builder so users can size their RAM/flash budget before calling
190// enable_rgbw_colorimetric_lut(). The runtime API clamps grid_n to [4, 256],
191// so this helper applies the same clamp so callers do not under-estimate
192// for grid_n < 4 or over-estimate for grid_n > 256. Returns 0 for grid_n < 1.
193constexpr unsigned long rgbw_colorimetric_lut_memory_bytes(
194 int grid_n, RgbwLutInterp interp) FL_NO_EXCEPT {
195 return (grid_n < 1)
196 ? 0UL
197 : (static_cast<unsigned long>(
198 grid_n < 4 ? 4 : (grid_n > 256 ? 256 : grid_n))
199 * static_cast<unsigned long>(
200 grid_n < 4 ? 4 : (grid_n > 256 ? 256 : grid_n))
201 * (interp == RgbwLutInterp::Hermite ? 24UL : 8UL));
202}

236 static u32 size_as_rgb(u32 num_of_rgbw_pixels) FL_NO_EXCEPT {
237 // The ObjectFLED controller expects the raw pixel byte data in
238 // multiples of 3. In the case of src data not a multiple of 3, then we
239 // need to add pad bytes so that the delegate controller doesn't walk
240 // off the end of the array and invoke a buffer overflow panic.
241 num_of_rgbw_pixels = (num_of_rgbw_pixels * 4 + 2) / 3;
242 u32 extra = num_of_rgbw_pixels % 3 ? 1 : 0;
243 num_of_rgbw_pixels += extra;
244 return num_of_rgbw_pixels;
245 }
