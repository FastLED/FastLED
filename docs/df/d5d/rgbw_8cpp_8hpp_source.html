23// Default diode profile: SK6812 RGBW3535 at ~6000K. Datasheet wavelength
24// midpoints (R 625nm, G 523nm, B 468nm) converted to CIE 1931 xy; W at
25// nominal 6000K (xy = 0.32208, 0.33805). Luminance ratios normalized so
26// W = 1.0; R/G/B set from datasheet typical mcd ratios for a 5050-class
27// part.
28//
29// Source space defaults to **native LED gamut + D65 white** (#2710): the
30// input primaries equal the LED's measured primaries so a saturated input
31// like RGB=(255,0,0) reaches the full saturation of the LED red diode,
32// and RGB=(255,255,255) lands on D65. This is what most FastLED users
33// expect — full use of their hardware's chromatic range. Users who want
34// named-gamut input semantics (Rec709 / sRGB, Rec2020, DCI-P3 D65/D60)
35// should construct an owned profile with
36// `fl::color::make_diode_profile(profile, fl::color::InputGamut::Rec709)`.
37// See #2705 for the source-space transform itself.
38const DiodeProfile kRgbwDefaultProfile = {
39 /* xy_r */ { 0.700606f, 0.299300f },
40 /* xy_g */ { 0.097940f, 0.831593f },
41 /* xy_b */ { 0.129086f, 0.049450f },
42 /* xy_w */ { 0.322080f, 0.338050f },
43 /* lum_r */ 0.10f,
44 /* lum_g */ 0.37f,
45 /* lum_b */ 0.08f,
46 /* lum_w */ 1.00f,
47 /* nominal_cct */ 6000,
48 /* input_xy_r */ { 0.700606f, 0.299300f }, // native LED R
49 /* input_xy_g */ { 0.097940f, 0.831593f }, // native LED G
50 /* input_xy_b */ { 0.129086f, 0.049450f }, // native LED B
51 /* input_xy_w */ { 0.31272f, 0.32903f }, // D65 white
52};

56// Not named min3: ArduinoCore-zephyr's <zephyr/sys/util.h> defines min3() as
57// a function-like macro, which would expand this declaration (#4416).
58inline u8 rgbw_min3(u8 a, u8 b, u8 c) FL_NO_EXCEPT {
59 if (a < b) {
60 if (a < c) {
61 return a;
62 } else {
63 return c;
64 }
65 } else {
66 if (b < c) {
67 return b;
68 } else {
69 return c;
70 }
71 }

74inline u8 divide_by_3(u8 x) {
75 u16 y = (u16(x) * 85) >> 8;
76 return static_cast<u8>(y);

83void rgb_2_rgbw_exact(u16 w_color_temperature, u8 r, u8 g,
84 u8 b, u8 r_scale, u8 g_scale,
85 u8 b_scale, u8 *out_r, u8 *out_g,
86 u8 *out_b, u8 *out_w) {
87 (void)w_color_temperature;
88 r = scale8(r, r_scale);
89 g = scale8(g, g_scale);
90 b = scale8(b, b_scale);
91 u8 min_component = rgbw_min3(r, g, b);
92 *out_r = r - min_component;
93 *out_g = g - min_component;
94 *out_b = b - min_component;
95 *out_w = min_component;
96}

121void rgb_2_rgbw_white_boosted(u16 w_color_temperature, u8 r,
122 u8 g, u8 b, u8 r_scale,
123 u8 g_scale, u8 b_scale, u8 *out_r,
124 u8 *out_g, u8 *out_b, u8 *out_w) {
125 (void)w_color_temperature;
126 r = scale8(r, r_scale);
127 g = scale8(g, g_scale);
128 b = scale8(b, b_scale);
129 u8 min_component = rgbw_min3(r, g, b);
130 u8 w;
131 bool is_min = true;
132 if (min_component <= 84) {
133 w = 3 * min_component;
134 } else {
135 w = 255;
136 is_min = false;
137 }
138 u8 r_prime;
139 u8 g_prime;
140 u8 b_prime;
141 if (is_min) {
142 r_prime = r - min_component;
143 g_prime = g - min_component;
144 b_prime = b - min_component;
145 } else {
146 u8 w3 = divide_by_3(w);
147 r_prime = r - w3;
148 g_prime = g - w3;
149 b_prime = b - w3;
150 }
151
152 *out_r = r_prime;
153 *out_g = g_prime;
154 *out_b = b_prime;
155 *out_w = w;
156}

215// Standard primary chromaticities for the named input gamuts (#2710). The
216// numbers come straight from each gamut's published spec:
217// Rec.709 / sRGB — ITU-R BT.709, IEC 61966-2-1
218// Rec.2020 — ITU-R BT.2020
219// DCI-P3 D65 — SMPTE EG 432-1 (consumer display variant)
220// DCI-P3 D60 (ACES) — ACES AP1 white = (0.32168, 0.33767)
221// White-point chromaticities likewise: D65 = (0.31272, 0.32903),
222// ACES D60 = (0.32168, 0.33767).
223namespace {
224struct NamedGamut {
225 float xy_r[2];
226 float xy_g[2];
227 float xy_b[2];
228 float xy_w[2];
229};
230constexpr NamedGamut kRec709 = {{0.6400f, 0.3300f}, {0.3000f, 0.6000f},
231 {0.1500f, 0.0600f}, {0.31272f, 0.32903f}};
232constexpr NamedGamut kRec2020 = {{0.7080f, 0.2920f}, {0.1700f, 0.7970f},
233 {0.1310f, 0.0460f}, {0.31272f, 0.32903f}};
234constexpr NamedGamut kDciP3D65 = {{0.6800f, 0.3200f}, {0.2650f, 0.6900f},
235 {0.1500f, 0.0600f}, {0.31272f, 0.32903f}};
236constexpr NamedGamut kDciP3D60 = {{0.6800f, 0.3200f}, {0.2650f, 0.6900f},
237 {0.1500f, 0.0600f}, {0.32168f, 0.33767f}};
238} // namespace

240// Forward declaration: real implementation lives inside the colorimetric
241// branch below (and is a no-op when colorimetric math is compiled out).
242// Called from set_input_gamut to evict per-process caches keyed on a profile
243// whose input_xy_* fields were just mutated in place — without this hook the
244// pointer+CCT cache key stays equal and stale M_src/LUT data is reused
245// after a gamut switch on the currently active profile.
246namespace { void invalidate_colorimetric_caches_for(const DiodeProfile* profile) FL_NO_EXCEPT; }

248void set_input_gamut(DiodeProfile& profile_ref, InputGamut g,
249 const float* white_xy) FL_NO_EXCEPT {
250 DiodeProfile* const profile = &profile_ref;
251 auto apply = [profile](const float r[2], const float gp[2],
252 const float b[2], const float w[2]) {
253 profile->input_xy_r[0] = r[0]; profile->input_xy_r[1] = r[1];
254 profile->input_xy_g[0] = gp[0]; profile->input_xy_g[1] = gp[1];
255 profile->input_xy_b[0] = b[0]; profile->input_xy_b[1] = b[1];
256 profile->input_xy_w[0] = w[0]; profile->input_xy_w[1] = w[1];
257 };
258 switch (g) {
259 case InputGamut::Native: {
260 // For Native, the input gamut tracks this profile's LED primaries —
261 // copy them over rather than picking a fixed sRGB-like fallback.
262 const float d65[2] = {0.31272f, 0.32903f};
263 const float* w = (white_xy != nullptr) ? white_xy : d65;
264 apply(profile->xy_r, profile->xy_g, profile->xy_b, w);
265 invalidate_colorimetric_caches_for(profile);
266 return;
267 }
268 case InputGamut::Rec709: apply(kRec709.xy_r, kRec709.xy_g, kRec709.xy_b,
269 white_xy != nullptr ? white_xy : kRec709.xy_w);
270 invalidate_colorimetric_caches_for(profile); return;
271 case InputGamut::Rec2020: apply(kRec2020.xy_r, kRec2020.xy_g, kRec2020.xy_b,
272 white_xy != nullptr ? white_xy : kRec2020.xy_w);
273 invalidate_colorimetric_caches_for(profile); return;
274 case InputGamut::DciP3D65: apply(kDciP3D65.xy_r, kDciP3D65.xy_g, kDciP3D65.xy_b,
275 white_xy != nullptr ? white_xy : kDciP3D65.xy_w);
276 invalidate_colorimetric_caches_for(profile); return;
277 case InputGamut::DciP3D60: apply(kDciP3D60.xy_r, kDciP3D60.xy_g, kDciP3D60.xy_b,
278 white_xy != nullptr ? white_xy : kDciP3D60.xy_w);
279 invalidate_colorimetric_caches_for(profile); return;
280 }
281 // Default-fallthrough for forward-compat with future enum additions:
282 // leave the profile's input_xy_* untouched — also nothing to invalidate.
283}

325// Per-process cached profile data (XYZ columns + matrix inverses). Rebuilt
326// whenever the active profile pointer OR the requested CCT changes. The CCT
327// part of the key is what makes Rgbw::white_color_temp actually shift the
328// W vertex chromaticity at solve time.
329struct ColorimetricCacheHolder {
330 colorimetric_detail::ProfileCache cache;
331 const DiodeProfile* cached_for = nullptr;
332 int cached_cct = 0;
333};
334
335// Resolve the CCT to use for the W vertex: if the dispatch passed a value in
336// the valid Krystek range AND it differs from the profile's nominal CCT,
337// shift W. Otherwise use the profile's xy_w as-is (cct_override = 0).
338inline int resolve_cct_override(const DiodeProfile* p, int requested_cct) FL_NO_EXCEPT {
339 if (requested_cct < 1500 || requested_cct > 15000) return 0;
340 if (requested_cct == p->nominal_cct) return 0;
341 return requested_cct;
342}
343
344inline const DiodeProfile* resolve_rgbw_profile(
345 const DiodeProfile* profile) FL_NO_EXCEPT {
346 return profile != nullptr ? profile : get_rgbw_colorimetric_profile();
347}
348
349inline const DiodeProfile* resolve_rgbw_profile(
350 const fl::shared_ptr<const DiodeProfile>& profile) FL_NO_EXCEPT {
351 return resolve_rgbw_profile(profile.get());
352}

354inline const colorimetric_detail::ProfileCache& get_cache(
355 const DiodeProfile* profile, int cct) FL_NO_EXCEPT {
356 ColorimetricCacheHolder& h =
357 fl::Singleton<ColorimetricCacheHolder>::instance();
358 const DiodeProfile* active = resolve_rgbw_profile(profile);
359 const int override_cct = resolve_cct_override(active, cct);
360 if (h.cached_for != active || h.cached_cct != override_cct) {
361 colorimetric_detail::build_profile_cache(active, override_cct, &h.cache);
362 h.cached_for = active;
363 h.cached_cct = override_cct;
364 }
365 return h.cache;
366}

368inline const colorimetric_detail::ProfileCache& get_cache(
369 const fl::shared_ptr<const DiodeProfile>& profile, int cct) FL_NO_EXCEPT {
370 return get_cache(resolve_rgbw_profile(profile), cct);
371}

377// ===== LUT state (issue #2545 Phase 2) ==================================
378// Always-compiled when the outer FASTLED_RGBW_COLORIMETRIC gate is on. The
379// inner FASTLED_RGBW_COLORIMETRIC_LUT gate that used to wrap this block was
380// redundant — once you've opted into colorimetric math, the LUT state +
381// rebuild_lut_if_stale + build_lut reference are gc-section-droppable for
382// sketches that never call enable_rgbw_colorimetric_lut() (the `enabled`
383// flag stays false, the fl::unique_ptr<LutTable> stays empty, and only the
384// LutStateHolder singleton instance itself is retained — ~32 bytes).
385struct LutStateHolder {
386 fl::unique_ptr<colorimetric_detail::LutTable> table;
387 const DiodeProfile* built_for = nullptr;
388 int built_cct = 0;
389 int requested_grid_n = 0;
390 RgbwLutInterp requested_interp = RgbwLutInterp::Hermite;
391 bool enabled = false;
392};
393
394inline colorimetric_detail::LutInterp to_internal_interp(
395 RgbwLutInterp i) FL_NO_EXCEPT {
396 return (i == RgbwLutInterp::Hermite)
397 ? colorimetric_detail::LutInterp::Hermite
398 : colorimetric_detail::LutInterp::Bilinear;
399}
400
401inline void rebuild_lut_if_stale(LutStateHolder& s, const DiodeProfile* profile,
402 int cct) FL_NO_EXCEPT {
403 if (!s.enabled || s.requested_grid_n <= 0) return;
404 const DiodeProfile* active = resolve_rgbw_profile(profile);
405 const int override_cct = resolve_cct_override(active, cct);
406 const colorimetric_detail::LutInterp interp =
407 to_internal_interp(s.requested_interp);
408 if (s.table && s.built_for == active && s.built_cct == override_cct
409 && s.table->N == s.requested_grid_n
410 && s.table->interp == interp) {
411 return; // up-to-date
412 }
413 s.table = fl::make_unique<colorimetric_detail::LutTable>(
414 colorimetric_detail::build_lut(get_cache(active, cct), s.requested_grid_n,
415 interp));
416 s.built_for = active;
417 s.built_cct = override_cct;
418}
419
420inline void rebuild_lut_if_stale(LutStateHolder& s,
421 const fl::shared_ptr<const DiodeProfile>& profile,
422 int cct) FL_NO_EXCEPT {
423 rebuild_lut_if_stale(s, resolve_rgbw_profile(profile), cct);
424}
425
426inline void rebuild_lut_if_stale(LutStateHolder& s, int cct) FL_NO_EXCEPT {
427 rebuild_lut_if_stale(s, static_cast<const DiodeProfile*>(nullptr), cct);
428}

430// Drop both the ProfileCache and the LUT cache when `profile` matches the
431// current cache key. set_input_gamut() mutates input_xy_* in place without
432// touching the profile pointer or CCT, so the (pointer, cct) cache key
433// stays equal and would otherwise serve stale M_src / LUT data.
434void invalidate_colorimetric_caches_for(const DiodeProfile* profile) FL_NO_EXCEPT {
435 ColorimetricCacheHolder& ch =
436 fl::Singleton<ColorimetricCacheHolder>::instance();
437 if (ch.cached_for == profile) {
438 ch.cached_for = nullptr;
439 ch.cached_cct = 0;
440 }
441 LutStateHolder& lh = fl::Singleton<LutStateHolder>::instance();
442 if (lh.built_for == profile) {
443 lh.built_for = nullptr;
444 lh.built_cct = 0;
445 }
446}

468void rgb_2_rgbw_colorimetric(const Rgbw& cfg, u8 r,
469 u8 g, u8 b, u8 r_scale,
470 u8 g_scale, u8 b_scale, u8 *out_r,
471 u8 *out_g, u8 *out_b, u8 *out_w) FL_NO_EXCEPT {
472 const int w_color_temperature = cfg.white_color_temp;
473 r = scale8(r, r_scale);
474 g = scale8(g, g_scale);
475 b = scale8(b, b_scale);
476 if ((r | g | b) == 0) {
477 *out_r = *out_g = *out_b = *out_w = 0;
478 return;
479 }
480 const float s_r = r * (1.0f / 255.0f);
481 const float s_g = g * (1.0f / 255.0f);
482 const float s_b = b * (1.0f / 255.0f);
483 const colorimetric_detail::ProfileCache& cache =
484 get_cache(cfg.profile, w_color_temperature);
485 float rgbw[4];
486
487 // LUT fast path — gc-sections drops the branch + singleton + lookup_lut
488 // for sketches that never call enable_rgbw_colorimetric_lut().
489 LutStateHolder& lut_state = fl::Singleton<LutStateHolder>::instance();
490 if (lut_state.enabled) {
491 // The xy/Y LUT has no source active-channel mask, so it cannot enforce
492 // native single-axis or dual-edge topology on its own. Keep those rows
493 // on the analytical guard path and reserve LUT lookup for interiors.
494 if (colorimetric_detail::is_native_input_gamut(*cache.profile)
495 && colorimetric_detail::count_active_channels(s_r, s_g, s_b) < 3) {
496 const bool ok = colorimetric_detail::solve_strict_subgamut(
497 cache, s_r, s_g, s_b, rgbw);
498 if (!ok) {
499 rgb_2_rgbw_exact(w_color_temperature, r, g, b, 255, 255, 255,
500 out_r, out_g, out_b, out_w);
501 return;
502 }
503 *out_r = colorimetric_detail::quantize_u8(rgbw[0]);
504 *out_g = colorimetric_detail::quantize_u8(rgbw[1]);
505 *out_b = colorimetric_detail::quantize_u8(rgbw[2]);
506 *out_w = colorimetric_detail::quantize_u8(rgbw[3]);
507 return;
508 }
509 rebuild_lut_if_stale(lut_state, cfg.profile, w_color_temperature);
510 float X_t[3];
511 if (cache.has_source_space) {
512 // #2705: use source-space matrix so the LUT lookup targets the
513 // same chromaticity as the closed-form solver.
514 const float s_vec[3] = { s_r, s_g, s_b };
515 colorimetric_detail::matvec3(cache.M_src, s_vec, X_t);
516 } else {
517 X_t[0] = cache.P_R[0] * s_r + cache.P_G[0] * s_g + cache.P_B[0] * s_b;
518 X_t[1] = cache.P_R[1] * s_r + cache.P_G[1] * s_g + cache.P_B[1] * s_b;
519 X_t[2] = cache.P_R[2] * s_r + cache.P_G[2] * s_g + cache.P_B[2] * s_b;
520 }
521 const float sum = X_t[0] + X_t[1] + X_t[2];
522 if (sum < 1e-9f) {
523 *out_r = *out_g = *out_b = *out_w = 0;
524 return;
525 }
526 const float xy_t[2] = { X_t[0] / sum, X_t[1] / sum };
527 const float source_value = fl::max(fl::max(s_r, s_g), s_b);
528 colorimetric_detail::lookup_lut(*lut_state.table, xy_t, source_value, rgbw);
529 *out_r = colorimetric_detail::quantize_u8(rgbw[0]);
530 *out_g = colorimetric_detail::quantize_u8(rgbw[1]);
531 *out_b = colorimetric_detail::quantize_u8(rgbw[2]);
532 *out_w = colorimetric_detail::quantize_u8(rgbw[3]);
533 return;
534 }
535
536 const bool ok =
537 colorimetric_detail::solve_strict_subgamut(cache, s_r, s_g, s_b, rgbw);
538 if (!ok) {
539 rgb_2_rgbw_exact(w_color_temperature, r, g, b, 255, 255, 255,
540 out_r, out_g, out_b, out_w);
541 return;
542 }
543 *out_r = colorimetric_detail::quantize_u8(rgbw[0]);
544 *out_g = colorimetric_detail::quantize_u8(rgbw[1]);
545 *out_b = colorimetric_detail::quantize_u8(rgbw[2]);
546 *out_w = colorimetric_detail::quantize_u8(rgbw[3]);
547}

678void rgbw_partial_reorder(EOrderW w_placement, u8 b0, u8 b1,
679 u8 b2, u8 w, u8 *out_b0,
680 u8 *out_b1, u8 *out_b2, u8 *out_b3) {
681
682 u8 out[4] = {b0, b1, b2, 0};
683 switch (w_placement) {
684 // unrolled loop for speed.
685 case W3:
686 out[3] = w;
687 break;
688 case W2:
689 out[3] = out[2]; // memmove and copy.
690 out[2] = w;
691 break;
692 case W1:
693 out[3] = out[2];
694 out[2] = out[1];
695 out[1] = w;
696 break;
697 case W0:
698 out[3] = out[2];
699 out[2] = out[1];
700 out[1] = out[0];
701 out[0] = w;
702 break;
703 }
704 *out_b0 = out[0];
705 *out_b1 = out[1];
706 *out_b2 = out[2];
707 *out_b3 = out[3];
708}
