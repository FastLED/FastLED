102template <int DATA_PIN, typename TIMING, EOrder RGB_ORDER, int WAIT_TIME, typename DriverTraits,
103 int XTRA0 = 0, bool DIRECT_SHOW = (FL_SLIM_BRIDGE_DIRECT_SHOW != 0)>
104class SlimBridgeController : public CPixelLEDController<RGB_ORDER> {
105 using Policy = SlimBridgeShowPolicy<DIRECT_SHOW>;
106
107public:
108 SlimBridgeController() FL_NO_EXCEPT {
109 ChipsetTimingConfig timing = makeTimingConfig<TIMING>();
110 // (A?) Fold WAIT_TIME (microseconds) into reset_us, mirroring the
111 // historical ClocklessIdf4 behavior: the driver must honor whichever
112 // is longer -- the chipset's own latch/reset requirement, or the
113 // caller-requested minimum inter-frame wait.
114 if (WAIT_TIME > 0 && static_cast<u32>(WAIT_TIME) > timing.reset_us) {
115 timing.reset_us = static_cast<u32>(WAIT_TIME);
116 }
117 mData = ChannelData::create(DATA_PIN, timing, fl::vector_psram<u8>(),
118 ChannelPixelFormat::RGB);
119 FL_STATIC_ASSERT(XTRA0 >= 0 && XTRA0 <= 32, "XTRA0 out of range");
120 mData->setExtraZeroBitsPerByte(static_cast<u8>(XTRA0));
121 // Idempotent by contract -- safe to call from every instantiation's
122 // constructor (mirrors ClocklessIdf5's registerWithManager() call).
123 // A no-op in direct-show mode.
124 Policy::template attach<DriverTraits>();
125 }
126
127 void init() FL_NO_EXCEPT override {}
128
130 const ChannelDataPtr& channelData() const FL_NO_EXCEPT { return mData; }
131
132protected:
133 void showPixels(PixelController<RGB_ORDER>& pixels) FL_NO_EXCEPT override {
134 IChannelDriver& driver = DriverTraits::instance();
135
136 // (A1) Respect runtime driver enable/disable: ChannelManager is the
137 // single source of truth. A disabled driver means "drop this frame",
138 // not "enqueue and hope" -- silently enqueuing to a disabled driver
139 // is the #2517 silent-drop failure mode.
140 if (!Policy::enabled(driver)) {
141 FL_WARN_ONCE("SlimBridgeController: driver '" << driver.getName().c_str() << "' is disabled - dropping frame");
142 return;
143 }
144
145 // (A8) If the previous frame's buffer is still in flight, block
146 // (bounded) until the driver reports it can accept more work rather
147 // than racing the driver's in-progress DMA/RMT transfer.
148 if (mData->isInUse()) {
149 if (!driver.waitForReady()) {
150 FL_WARN_ONCE("SlimBridgeController: driver '" << driver.getName().c_str() << "' did not become ready in time - dropping frame");
151 return;
152 }
153 }
154
155 // Give subclasses a chance to observe the accepted frame before it is
156 // encoded (e.g. feeding ActiveStripTracker for stub/WASM channels).
157 onBeforeEncode(pixels);
158
159 // (A5) Publish the pixel byte layout so the driver (and any
160 // downstream padding/diagnostic code) knows the encoded stride.
161 if (this->getRgbww().active()) {
162 mData->setPixelFormat(ChannelPixelFormat::RGBWW);
163 } else if (this->getRgbw().active()) {
164 mData->setPixelFormat(ChannelPixelFormat::RGBW);
165 } else {
166 mData->setPixelFormat(ChannelPixelFormat::RGB);
167 }
168
169 // (A3) Re-encode into the same buffer every frame -- clear() keeps
170 // capacity, so steady-state operation is allocation-free.
171 mData->getData().clear();
172 fl::PixelIterator iterator(&pixels, this->getRgbw(), this->getRgbww());
173 iterator.writeWS2812(&mData->getData());
174
175 driver.enqueue(mData);
176 Policy::afterEnqueue(driver);
177 }
178
181 virtual void onBeforeEncode(PixelController<RGB_ORDER>& pixels) FL_NO_EXCEPT { (void)pixels; }
182
183private:
184 ChannelDataPtr mData;
185};
