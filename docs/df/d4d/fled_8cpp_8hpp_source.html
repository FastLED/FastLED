31Fled Fled::load(FileSystem &fs, const char *path) FL_NO_EXCEPT {
32 fl::ifstream in = fs.openRead(path);
33 if (!in.is_open()) {
34 return Fled();
35 }
36 const fl::size total = in.size();
37 fl::vector<fl::u8> buf;
38 buf.resize(total);
39 if (total > 0) {
40 const fl::size got = in.read(buf.data(), total);
41 if (got != total) {
42 return Fled();
43 }
44 }
45 return Fled::loadFromVector(fl::move(buf));
46}

97fl::size Fled::frameCount(fl::size ledCount) const FL_NO_EXCEPT {
98 if (!mImpl || ledCount == 0) return 0;
99 const fl::u8 bpp = fl::fled::bytesPerLed(mImpl->pixelFormatByte());
100 if (bpp == 0) return 0;
101 const fl::size perFrame = ledCount * static_cast<fl::size>(bpp);
102 if (perFrame == 0) return 0;
103 return mImpl->payloadLen() / perFrame;
104}

147bool Fled::videoFrame(fl::size frameIndex, fl::size ledCount,
148 fled::VideoFrameView *out) const FL_NO_EXCEPT {
149 if (!mImpl || !out || ledCount == 0) {
150 return false;
151 }
152 fled::PixelStorage storage;
153 if (!fled::toPixelStorage(static_cast<fled::PixelFormat>(pixelFormat()),
154 &storage)) {
155 return false;
156 }
157 fled::VideoColor color;
158 if (videoColor(&color) != fled::ColorStatus::Ok) {
159 return false;
160 }
161 const fl::size bytesPerLed = fled::bytesPerLed(pixelFormat());
162 const fl::size payloadBytes = mImpl->payloadLen();
163 if (bytesPerLed == 0 || ledCount > payloadBytes / bytesPerLed) {
164 return false;
165 }
166 const fl::size stride = ledCount * bytesPerLed;
167 if (stride == 0 || payloadBytes % stride != 0 ||
168 frameIndex >= payloadBytes / stride) {
169 return false;
170 }
171 fl::size completePayloadBytes = 0;
172 fl::shared_ptr<const fl::u8> payload = blob("frame_payload",
173 &completePayloadBytes);
174 if (!payload) {
175 return false;
176 }
177 const fl::size offset = frameIndex * stride;
178 out->mStorage = storage;
179 out->mColor = color;
180 out->mPayload = fl::shared_ptr<const fl::u8>(payload, payload.get() + offset);
181 out->mPayloadBytes = stride;
182 out->mStride = stride;
183 return true;
184}

188fl::shared_ptr<const fl::u8> Fled::blob(const char *sectionName,
189 fl::size *outLen) const FL_NO_EXCEPT {
190 if (outLen) *outLen = 0;
191 if (!mImpl) {
192 return fl::shared_ptr<const fl::u8>();
193 }
194 if (!fl::fled::sectionNameIsPayload(sectionName)) {
195 return fl::shared_ptr<const fl::u8>();
196 }
197 const fl::size n = mImpl->payloadLen();
198 if (n == 0) {
199 return fl::shared_ptr<const fl::u8>();
200 }
201 const fl::u8 *raw = mImpl->bytes() + mImpl->payloadOffset();
202 if (outLen) *outLen = n;
203 // Aliasing ctor: pins mImpl alive while the returned ptr is held.
204 // NOTE: for loadFromStatic, the bytes themselves still live in the
205 // caller's static span - see Fled::loadFromStatic contract.
206 return fl::shared_ptr<const fl::u8>(mImpl, raw);
207}

211fl::shared_ptr<ScreenMap> Fled::screenMap() const FL_NO_EXCEPT {
212 if (!mImpl) {
213 return fl::shared_ptr<ScreenMap>();
214 }
215 const fl::json &env = mImpl->envelope();
216 // The v1 envelope carries the screen map under the "map" key. The
217 // canonical parser is ScreenMap::ParseJson, which takes a JSON
218 // document containing a top-level "map" object - exactly the shape
219 // of the envelope, so we hand it straight to ParseJson.
220 if (!env.contains(fl::string("map")) || !env["map"].is_object()) {
221 return fl::shared_ptr<ScreenMap>();
222 }
223 fl::string envText = env.to_string();
224 fl::flat_map<fl::string, ScreenMap> segments;
225 if (!ScreenMap::ParseJson(envText.c_str(), &segments)) {
226 return fl::shared_ptr<ScreenMap>();
227 }
228 if (segments.empty()) {
229 return fl::shared_ptr<ScreenMap>();
230 }
231 // Single-strip path: return the first segment. Multi-strip support
232 // would be a dedicated accessor (out of scope here).
233 ScreenMap &first = segments.begin()->second;
234 return fl::make_shared<ScreenMap>(first);
235}
