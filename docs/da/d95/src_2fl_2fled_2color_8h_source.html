23// Chromaticities + white point of the source. Custom carries explicit CIE xy

32// Transfer function. `Srgb` is the piecewise sRGB function, NOT the BT.709

51// Outcome of resolving `video.color` against the header's pixel_format.
52// Every non-Ok value is a rejection: FLED_FORMAT.md requires a clear
53// diagnostic rather than a silent fallback.
54enum class ColorStatus : fl::u8 {
55 Ok = 0,
56 // Caller passed a null out-pointer. API misuse, not a bad file - kept
57 // distinct so a diagnostic never sends someone hunting a valid envelope.
58 NullOutput,
59 // `video.color` absent on a pixel format that defines no default tuple
60 // (gray8, rgbw8). Not malformed - just undeclared, and unresolvable.
61 NoDefaultTuple,
62 // `video.color` present but not a JSON object.
63 NotAnObject,
64 UnknownPrimaries,
65 UnknownTransfer,
66 UnknownMatrix,
67 UnknownRange,
68 // Rules 2/3: display-encoded formats reject linear/pq/hlg;
69 // rgb16_linear rejects anything but linear.
70 TransferConflictsWithFormat,
71 // Rule 4: `range: "limited"` is reserved in v1.
72 LimitedRangeUnsupported,
73 // Rule 5: any matrix other than `rgb` is reserved in v1.
74 MatrixUnsupported,
75 // Rule 6: gray8/rgbw8 must declare all four keys or none.
76 IncompleteForFormat,
77 // Rule 7: custom primaries object missing a key or with a malformed pair.
78 MalformedCustomPrimaries,
79};

81// A resolved source-color declaration. `declared` distinguishes "the file
82// said so" from "the default tuple was applied", which matters because the
83// default is a compatibility interpretation, not an author's statement.
84struct VideoColor {
85 ColorPrimaries primaries;
86 ColorTransfer transfer;
87 ColorMatrix matrix;
88 ColorRange range;
89 bool declared;
90 // Valid only when primaries == Custom. Layout:
91 // {red.x, red.y, green.x, green.y, blue.x, blue.y, white.x, white.y}
92 float customPrimaries[8];
93};
