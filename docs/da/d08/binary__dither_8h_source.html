120 static FASTLED_FORCE_INLINE u8 phase(u8 frame) FL_NO_EXCEPT {
121 frame &= (0x01 << VIRTUAL_BITS) - 1;
122 u8 q = 0;
123 if (frame & 0x01) { q |= 0x80; }
124 if (frame & 0x02) { q |= 0x40; }
125 if (frame & 0x04) { q |= 0x20; }
126 if (frame & 0x08) { q |= 0x10; }
127 if (frame & 0x10) { q |= 0x08; }
128 if (frame & 0x20) { q |= 0x04; }
129 if (frame & 0x40) { q |= 0x02; }
130 if (frame & 0x80) { q |= 0x01; }
131 if (VIRTUAL_BITS < 8) {
132 q += 0x01 << (7 - VIRTUAL_BITS);
133 }
134 return q;
135 }
136
138 static FASTLED_FORCE_INLINE u8 offset(u8 q, u8 range) FL_NO_EXCEPT {
139 u8 d = fl::scale8(q, range);
140#if (FASTLED_SCALE8_FIXED == 1)
141 // Adjust for the fixed scale8's rounding up.
142 if (d) { --d; }
143#endif
144 return d;
145 }
146
151 static FASTLED_FORCE_INLINE void init(u8 (&d)[3], u8 (&e)[3], const u8 (&scale)[3],
152 u8 frame) FL_NO_EXCEPT {
153 const u8 q = phase(frame);
154 for (int i = 0; i < 3; ++i) {
155 // Computed in a local and stored once; re-reading `e[i]` after
156 // the `d[i]` store cost 36 B on AVR (#4672).
157 const u8 s = scale[i];
158 u8 range = s ? (256 / s) + 1 : 0;
159 d[i] = offset(q, range);
160 if (range) { --range; }
161 e[i] = range;
162 }
163 }
164
167 static FASTLED_FORCE_INLINE void reseed(u8 (&d)[3], const u8 (&e)[3], u8 frame) FL_NO_EXCEPT {
168 const u8 q = phase(frame);
169 for (int i = 0; i < 3; ++i) {
170 // `e[i]` is init's range less one; zero only for an unlit
171 // channel, which carries no offset.
172 d[i] = e[i] ? offset(q, static_cast<u8>(e[i] + 1)) : 0;
173 }
174 }
175
176 static FASTLED_FORCE_INLINE void clear(u8 (&d)[3], u8 (&e)[3]) FL_NO_EXCEPT {
177 d[0] = d[1] = d[2] = e[0] = e[1] = e[2] = 0;
178 }
179
180 static FASTLED_FORCE_INLINE bool active(const u8 (&e)[3]) FL_NO_EXCEPT {
181 return (e[0] | e[1] | e[2]) != 0;
182 }
183
186 static FASTLED_FORCE_INLINE void step(u8 (&d)[3], const u8 (&e)[3]) FL_NO_EXCEPT {
187 d[0] = e[0] - d[0];
188 d[1] = e[1] - d[1];
189 d[2] = e[2] - d[2];
190 }
191
193 static FASTLED_FORCE_INLINE void stepChannel(u8 (&d)[3], const u8 (&e)[3], int c) FL_NO_EXCEPT {
194 d[c] = e[c] - d[c];
195 }
196
198 static FASTLED_FORCE_INLINE u8 apply(u8 b, u8 d) FL_NO_EXCEPT {
199 return b ? fl::qadd8(b, d) : 0;
200 }
201};
