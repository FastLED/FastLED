80// Helper: Case-insensitive string comparison
81bool iequals(const string& a, const string& b) {
82 if (a.size() != b.size()) return false;
83 for (size_t i = 0; i < a.size(); ++i) {
84 if (fl::tolower(a[i]) != fl::tolower(b[i])) return false;
85 }
86 return true;
87}

89// Helper: Trim whitespace from string
90string trim(const string& s) {
91 size_t start = 0;
92 while (start < s.size() && fl::isspace(s[start])) ++start;
93 size_t end = s.size();
94 while (end > start && fl::isspace(s[end - 1])) --end;
95 return s.substr(start, end - start);
96}

98// Helper: Split string by delimiter
99vector<string> split(const string& s, char delimiter) {
100 vector<string> tokens;
101 size_t start = 0;
102 size_t end = s.find(delimiter);
103 while (end != string::npos) {
104 tokens.push_back(s.substr(start, end - start));
105 start = end + 1;
106 end = s.find(delimiter, start);
107 }
108 tokens.push_back(s.substr(start));
109 return tokens;
110}

112// Helper: Parse query string (e.g., "?id=123&name=test")
113map<string, string> parse_query_string(const string& query) {
114 map<string, string> params;
115 if (query.empty() || query[0] != '?') return params;
116
117 vector<string> pairs = split(query.substr(1), '&');
118 for (const auto& pair : pairs) {
119 size_t eq_pos = pair.find('=');
120 if (eq_pos != string::npos) {
121 string key = pair.substr(0, eq_pos);
122 string value = pair.substr(eq_pos + 1);
123 params[key] = value;
124 }
125 }
126 return params;
127}

140// Helper: Set socket to non-blocking mode
141bool set_nonblocking(int fd) {
142#ifdef FL_IS_WIN
143 u_long mode = 1;
144 return ioctlsocket(fd, FIONBIO, &mode) == 0;
145#else
146 int flags = fcntl(fd, F_GETFL, 0);

232 result += "HTTP/1.0 ";
233 result += fl::to_string(mStatusCode);
234 result += " ";

236 result += "\r\n";
237

241 }

245
246 // End of headers
247 result += "\r\n";

250 result += mBody;

275bool Server::start(int port) {
276 if (mRunning) {
277 mLastError = "Server already running";
278 return false;
279 }
280
281 if (!setup_listen_socket(port)) {
282 return false;

288
289 // Register with async system for automatic updates
290 if (!mAsyncRunner) {
291 mAsyncRunner = fl::make_unique<ServerAsyncRunner>(this);
292 task::Executor::instance().register_runner(mAsyncRunner.get());
293 }

295 return true;

298void Server::stop() {
299 if (!mRunning) return;
300
301 // Unregister from async system
302 if (mAsyncRunner) {
303 task::Executor::instance().unregister_runner(mAsyncRunner.get());
304 mAsyncRunner.reset();
305 }
306
307 // Close all client connections
308 for (auto& client : mClientSockets) {
309 if (client.fd != -1) {
310 close(client.fd);
311 }
312 }
313 mClientSockets.clear();
314
315 // Close listen socket
316 if (mListenSocket != -1) {
317 close(mListenSocket);
318 mListenSocket = -1;
319 }
320
321 // Free route handlers to prevent leaks when server lives in a shared library
322 // (LSAN runs before shared library static destructors)
323 mRoutes.clear();
324
325 mRunning = false;
326}

356bool Server::setup_listen_socket(int port) {
357 // Create socket (initialization handled by socket wrapper)
358 mListenSocket = socket(AF_INET, SOCK_STREAM, 0);
359 if (mListenSocket < 0) {
360 mLastError = "Failed to create socket";
361 return false;
362 }
363
364 // Set socket options
365 int opt = 1;
366 if (setsockopt(mListenSocket, SOL_SOCKET, SO_REUSEADDR,
367 (const char*)(&opt), sizeof(opt)) < 0) {
368 close(mListenSocket);
369 mListenSocket = -1;
370 mLastError = "Failed to set SO_REUSEADDR";
371 return false;
372 }
373
374 // Set non-blocking
375 if (!set_nonblocking(mListenSocket)) {
376 close(mListenSocket);
377 mListenSocket = -1;
378 mLastError = "Failed to set non-blocking mode";
379 return false;
380 }
381
382 // Bind socket
383 sockaddr_in addr{};
384 addr.sin_family = AF_INET;
385 addr.sin_addr.s_addr = INADDR_ANY;
386 addr.sin_port = htons(static_cast<u16>(port));
387
388 if (bind(mListenSocket, (sockaddr*)(&addr), sizeof(addr)) < 0) {
389 close(mListenSocket);
390 mListenSocket = -1;
391 mLastError = "Failed to bind to port";
392 return false;
393 }
394
395 // Listen
396 if (listen(mListenSocket, 5) < 0) {
397 close(mListenSocket);
398 mListenSocket = -1;
399 mLastError = "Failed to listen on socket";
400 return false;
401 }
402
403 return true;
404}

406void Server::accept_connections() {
407 sockaddr_in client_addr{};
408 socklen_t addr_len = sizeof(client_addr);
409
410 // Accept all pending connections (non-blocking)
411 while (true) {
412 int client_fd = accept(mListenSocket,
413 (sockaddr*)(&client_addr),
414 &addr_len);
415
416 if (client_fd < 0) {
417#ifdef FL_IS_WIN
418 if (WSAGetLastError() == SOCKET_ERROR_WOULD_BLOCK) break;
419#else
420 if (errno == EWOULDBLOCK || errno == EAGAIN) break;
421#endif
422 // Other error - log and continue
423 break;
424 }
425
426 // Set client socket to non-blocking
427 if (!set_nonblocking(client_fd)) {
428 close(client_fd);
429 continue;
430 }
431
432 // Add to client list
433 ClientConnection client;
434 client.fd = client_fd;
435 client.connect_time = fl::platforms::millis();
436 mClientSockets.push_back(client);
437 }
438}

440size_t Server::process_requests() {
441 size_t requests_processed = 0;
442
443 // Process all clients (iterate backwards to allow removal)
444 for (size_t i = mClientSockets.size(); i > 0; --i) {
445 size_t index = i - 1;
446 ClientConnection& client = mClientSockets[index];
447
448 // Try to read request
449 optional<Request> req = read_request(client);
450 if (!req) {
451 continue; // No complete request yet
452 }
453
454 // Find handler
455 optional<RouteHandler> handler = find_handler(req->method(), req->path());
456
457 Response resp;
458 if (handler) {
459 // Call handler
460 resp = (*handler)(*req);
461 } else {
462 // No handler found - 404
463 resp = Response::not_found();
464 }
465
466 // Send response
467 send_response(client.fd, resp);
468
469 // Close connection (HTTP/1.0 - no keep-alive)
470 close_client(index);
471
472 requests_processed++;
473 }
474
475 return requests_processed;
476}

478optional<Request> Server::read_request(ClientConnection& client) {
479 // Read data from socket
480 char buffer[4096];
481
482#ifdef FL_IS_WIN
483 int bytes = recv(client.fd, buffer, sizeof(buffer), 0);
484 if (bytes <= 0) {
485 if (WSAGetLastError() != SOCKET_ERROR_WOULD_BLOCK) {
486 return nullopt; // Connection closed or error
487 }
488 return nullopt; // No data yet
489 }
490#else
491 ssize_t bytes = recv(client.fd, buffer, sizeof(buffer), MSG_DONTWAIT);
492 if (bytes <= 0) {
493 if (errno != EWOULDBLOCK && errno != EAGAIN) {
494 return nullopt; // Connection closed or error
495 }
496 return nullopt; // No data yet
497 }
498#endif
499
500 client.buffer.append(buffer, static_cast<size_t>(bytes));
501
502 // Check if we have complete HTTP request (ends with \r\n\r\n)
503 size_t header_end = client.buffer.find("\r\n\r\n");
504 if (header_end == string::npos) {
505 return nullopt; // Headers not complete yet
506 }
507
508 // Parse request
509 Request req;
510
511 // Split into lines
512 string header_section = client.buffer.substr(0, header_end);
513 vector<string> lines = split(header_section, '\n');
514
515 if (lines.empty()) {
516 return nullopt;
517 }
518
519 // Parse request line: "GET /path HTTP/1.1\r"
520 string request_line = trim(lines[0]);
521 vector<string> parts = split(request_line, ' ');
522 if (parts.size() < 3) {
523 return nullopt;
524 }
525
526 req.mMethod = parts[0];
527
528 // Split path and query string
529 string full_path = parts[1];
530 size_t query_pos = full_path.find('?');
531 if (query_pos != string::npos) {
532 req.mPath = full_path.substr(0, query_pos);
533 req.mQuery = parse_query_string(full_path.substr(query_pos));
534 } else {
535 req.mPath = full_path;
536 }
537
538 req.mHttpVersion = parts[2];
539
540 // Parse headers
541 for (size_t i = 1; i < lines.size(); ++i) {
542 string line = trim(lines[i]);
543 if (line.empty()) continue;
544
545 size_t colon_pos = line.find(':');
546 if (colon_pos != string::npos) {
547 string name = trim(line.substr(0, colon_pos));
548 string value = trim(line.substr(colon_pos + 1));
549 req.mHeaders[name] = value;
550 }
551 }
552
553 // Read body if Content-Length present
554 optional<string> content_length = req.header("Content-Length");
555 if (content_length) {
556 int body_len = atoi(content_length->c_str());
557 if (body_len > 0) {
558 size_t body_start = header_end + 4;
559 size_t total_needed = body_start + static_cast<size_t>(body_len);
560
561 if (client.buffer.size() < total_needed) {
562 return nullopt; // Body not complete yet
563 }
564
565 req.mBody = client.buffer.substr(body_start, static_cast<size_t>(body_len));
566 }
567 }
568
569 return req;
570}

572bool Server::send_response(int client_fd, const Response& response) {
573 string data = response.to_string();
574 const char* ptr = data.c_str();
575 size_t remaining = data.size();
576
577 while (remaining > 0) {
578#ifdef FL_IS_WIN
579 int sent = send(client_fd, ptr, static_cast<int>(remaining), 0);
580#else
581 ssize_t sent = send(client_fd, ptr, remaining, 0);
582#endif
583
584 if (sent <= 0) {
585 return false;
586 }
587
588 ptr += sent;
589 remaining -= static_cast<size_t>(sent);
590 }
591
592 return true;
593}

638// ========== Main-task request marshaling (FastLED#3588) ==========
639//
640// Route handlers used to execute ON the esp_http_server task,
641// concurrently with the main loop's fl:: object churn - reproducibly
642// corrupting main-task heap objects on classic ESP32 under load.
643// Requests are now marshaled: the httpd task extracts plain-C request
644// data, queues it, and blocks on a semaphore; the main task (via the
645// task::Executor pump) runs the fl:: handler and hands back a
646// heap-allocated Response. No fl:: object is ever constructed or
647// mutated off the main task.
648
649namespace {
650
651struct EspPendingRequest {
652 Server* server;
653 fl::size route_index;
654 char method[8];
655 char uri[160];
656 char* body; // fl::malloc'd raw body (may be null)
657 fl::size body_len;
658 Response* response; // filled by the main task
659 SemaphoreHandle_t done;
660 // Ownership handshake (the original stack-allocated design left a
661 // DANGLING pointer in the queue when the 5 s handler wait timed
662 // out — the runner then wrote through freed httpd stack memory).
663 // 0 = queued, 1 = runner processing, 3 = abandoned by httpd.
664 fl::atomic<int> state;
665};
666
667void espRequestCleanup(EspPendingRequest* item) {
668 if (!item) return;
669 if (item->response) {
670 delete item->response; // ok bare allocation - see marshaling protocol
671 }
672 if (item->body) {
673 fl::free(item->body);
674 }
675 if (item->done) {
676 vSemaphoreDelete(item->done);
677 }
678 delete item; // ok bare allocation - see marshaling protocol
679}

681// FL_LINT_ALLOW_GLOBAL(constant-initialized ESP-IDF queue handle; Singleton<T> adds pointer storage and a branch without improving linker elision)
682QueueHandle_t s_esp_request_queue = nullptr;
683
684} // anonymous namespace

687class Server::ServerAsyncRunner : public task::Runner {
688public:
689 explicit ServerAsyncRunner(Server* server) : mServer(server) {}
690
691 void update() override {
692 if (!s_esp_request_queue) {
693 return;
694 }
695 EspPendingRequest* item = nullptr;
696 while (xQueueReceive(s_esp_request_queue, &item, 0) == pdTRUE) {
697 if (!item) {
698 continue;
699 }
700 int expected = 0;
701 if (!item->state.compare_exchange_strong(expected, 1)) {
702 // httpd side abandoned this request (its wait timed
703 // out) — we own the cleanup.
704 espRequestCleanup(item);
705 continue;
706 }
707 Request fl_req;
708 fl_req.mMethod = item->method;
709 fl_req.mPath = item->uri;
710 fl_req.mHttpVersion = "HTTP/1.1";
711 if (item->body && item->body_len > 0) {
712 fl_req.mBody = string(item->body, item->body_len);
713 }
714 Response* resp = new Response(); // ok bare allocation - freed via espRequestCleanup
715 if (item->server && item->route_index < item->server->mRoutes.size()) {
716 *resp = item->server->mRoutes[item->route_index].handler(fl_req);
717 } else {
718 *resp = Response::not_found();
719 }
720 item->response = resp;
721 xSemaphoreGive(item->done);
722 }
723 }
724
725 bool has_active_tasks() const override {
726 return mServer && mServer->is_running();
727 }
728
729 size_t active_task_count() const override {
730 return (mServer && mServer->is_running()) ? 1 : 0;
731 }
732
733private:
734 Server* mServer;
735};

761// Static method on Server - has access to private members of Request/Response.
762// Signature uses void* to avoid ESP-IDF types in header; cast to httpd_req_t* here.
763// This is called from the ESP-IDF HTTP server task.
764int Server::handle_esp_request(void* raw_req) {
765 // Runs on the esp_http_server task. NO fl:: objects here - only
766 // plain C extraction; the fl:: handler executes on the main task
767 // (see ServerAsyncRunner::update, FastLED#3588).
768 auto* req = static_cast<httpd_req_t*>(raw_req);
769 auto* ctx = static_cast<EspRouteContext*>(req->user_ctx);
770 if (!ctx || !ctx->server || !s_esp_request_queue) {
771 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "No context");
772 return ESP_FAIL;
773 }
774
775 EspPendingRequest* item = new EspPendingRequest(); // ok bare allocation - freed via espRequestCleanup by whichever side owns it last
776 item->server = ctx->server;
777 item->route_index = ctx->route_index;
778 item->body = nullptr;
779 item->body_len = 0;
780 item->response = nullptr;
781 item->method[0] = '\0';
782 item->uri[0] = '\0';
783 item->state.store(0);
784 switch (req->method) {
785 case HTTP_POST: fl::strncpy(item->method, "POST", sizeof(item->method) - 1); break;
786 case HTTP_PUT: fl::strncpy(item->method, "PUT", sizeof(item->method) - 1); break;
787 case HTTP_DELETE: fl::strncpy(item->method, "DELETE", sizeof(item->method) - 1); break;
788 default: fl::strncpy(item->method, "GET", sizeof(item->method) - 1); break;
789 }
790 fl::strncpy(item->uri, req->uri, sizeof(item->uri) - 1);
791 // Strip query string from the path
792 for (char* c = item->uri; *c; ++c) {
793 if (*c == '?') { *c = '\0'; break; }
794 }
795
796 if (req->content_len > 0 && req->content_len <= 8192) {
797 int total_len = static_cast<int>(req->content_len);
798 char* buf = static_cast<char*>(fl::malloc(static_cast<fl::size>(total_len) + 1));
799 if (buf) {
800 int received = 0;
801 while (received < total_len) {
802 int ret = httpd_req_recv(req, buf + received, total_len - received);
803 if (ret <= 0) {
804 break;
805 }
806 received += ret;
807 }
808 buf[received] = '\0';
809 item->body = buf;
810 item->body_len = static_cast<fl::size>(received);
811 }
812 }
813
814 item->done = xSemaphoreCreateBinary();
815 if (!item->done) {
816 espRequestCleanup(item);
817 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "No memory");
818 return ESP_FAIL;
819 }
820
821 if (xQueueSend(s_esp_request_queue, &item, pdMS_TO_TICKS(100)) != pdTRUE) {
822 espRequestCleanup(item); // never queued — we own it
823 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "Server busy");
824 return ESP_FAIL;
825 }
826
827 esp_err_t result = ESP_OK;
828 if (xSemaphoreTake(item->done, pdMS_TO_TICKS(5000)) != pdTRUE) {
829 int expected = 0;
830 if (item->state.compare_exchange_strong(expected, 3)) {
831 // Runner has not started: it will see 'abandoned' and clean
832 // up. We must NOT free — the pointer is still in the queue.
833 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "Main loop busy");
834 return ESP_FAIL;
835 }
836 // Runner won the race and is (or was) processing — wait for its
837 // give, then fall through and serve normally.
838 (void)xSemaphoreTake(item->done, portMAX_DELAY);
839 }
840
841 if (item->response) {
842 Response* resp = item->response;
843 char status_str[32];
844 fl::snprintf(status_str, sizeof(status_str), "%d", resp->mStatusCode);
845 httpd_resp_set_status(req, status_str);
846 for (auto it = resp->mHeaders.begin(); it != resp->mHeaders.end(); ++it) {
847 httpd_resp_set_hdr(req, it->first.c_str(), it->second.c_str());
848 }
849 httpd_resp_send(req, resp->mBody.c_str(), resp->mBody.size());
850 } else {
851 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "No response");
852 result = ESP_FAIL;
853 }
854 espRequestCleanup(item); // served — we own the cleanup
855 return result;
856}

966bool Server::start(int port) {
967 if (mRunning) {
968 mLastError = "Server already running";
969 return false;
970 }
971
972 if (!s_esp_request_queue) {
973 s_esp_request_queue = xQueueCreate(4, sizeof(EspPendingRequest*));
974 }
975 if (!mAsyncRunner) {
976 mAsyncRunner = fl::make_unique<ServerAsyncRunner>(this);
977 task::Executor::instance().register_runner(mAsyncRunner.get());
978 }
979
980 httpd_config_t config = HTTPD_DEFAULT_CONFIG();
981 config.server_port = static_cast<u16>(port);
982 if (mRoutes.size() > (fl::numeric_limits<u16>::max)()) {
983 mLastError = "Too many HTTP routes";
984 FL_WARN("%s", "[HTTP] Too many routes for ESP-IDF HTTP server");
985 task::Executor::instance().unregister_runner(mAsyncRunner.get());
986 mAsyncRunner.reset();
987 return false;
988 }
989 if (mRoutes.size() > config.max_uri_handlers) {
990 config.max_uri_handlers = static_cast<u16>(mRoutes.size());
991 }
992 // Route handlers execute ON the httpd task and run full fl::json /
993 // fl::string / Response machinery. The IDF default stack (4 KB) is
994 // not enough for that: overflows landed in adjacent heap blocks and
995 // corrupted unrelated main-task objects (FastLED#3588 — classic
996 // ESP32 crashed building RPC responses while a client hammered the
997 // server). FreeRTOS heap-allocated task stacks give no early
998 // warning, so size generously.
999 config.stack_size = 8192;
1000
1001 esp_err_t err = httpd_start(&s_esp_httpd, &config);
1002 if (err != ESP_OK) {
1003 mLastError = "httpd_start failed";
1004 FL_WARN("[HTTP] httpd_start failed: %s", esp_err_to_name(err));
1005 task::Executor::instance().unregister_runner(mAsyncRunner.get());
1006 mAsyncRunner.reset();
1007 return false;
1008 }
1009
1010 // Register all routes that were added before start()
1011 for (size_t i = 0; i < mRoutes.size(); ++i) {
1012 auto ctx = fl::make_unique<EspRouteContext>(EspRouteContext{this, i});
1013
1014 httpd_uri_t uri_handler = {};
1015 uri_handler.uri = mRoutes[i].path.c_str();
1016 uri_handler.method = to_httpd_method(mRoutes[i].method);
1017 uri_handler.handler = esp_route_handler;
1018 uri_handler.user_ctx = ctx.get();
1019
1020 esp_err_t reg_err = httpd_register_uri_handler(s_esp_httpd, &uri_handler);
1021 if (reg_err != ESP_OK) {
1022 mLastError = "Failed to register route " + mRoutes[i].path;
1023 FL_WARN("[HTTP] Failed to register route %s: %s",
1024 mRoutes[i].path.c_str(), esp_err_to_name(reg_err));
1025 httpd_stop(s_esp_httpd);
1026 s_esp_httpd = nullptr;
1027 s_esp_route_contexts.clear();
1028 task::Executor::instance().unregister_runner(mAsyncRunner.get());
1029 mAsyncRunner.reset();
1030 return false;
1031 }
1032 s_esp_route_contexts.push_back(fl::move(ctx));
1033 }
1034
1035 mPort = port;
1036 mRunning = true;
1037 mLastError.clear();
1038
1039 FL_WARN("[HTTP] Server started on port %s", port);
1040 return true;
1041}
1042
1043void Server::stop() {
1044 if (!mRunning) return;
1045
1046 // Stop the httpd first so no request can be queued mid-teardown,
1047 // then drain and unregister the main-task runner (FastLED#3588).
1048 if (s_esp_httpd) {
1049 httpd_stop(s_esp_httpd);
1050 s_esp_httpd = nullptr;
1051 }
1052 if (mAsyncRunner) {
1053 mAsyncRunner->update(); // serve any request already queued
1054 task::Executor::instance().unregister_runner(mAsyncRunner.get());
1055 mAsyncRunner.reset();
1056 }
1057
1058 // Clean up route contexts (unique_ptr auto-deletes on clear)
1059 s_esp_route_contexts.clear();
1060
1061 // Free route handlers
1062 mRoutes.clear();
1063
1064 mRunning = false;
1065 FL_WARN("[HTTP] Server stopped");
1066}
