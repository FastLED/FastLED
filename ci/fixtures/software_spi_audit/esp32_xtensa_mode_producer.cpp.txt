extern "C" __attribute__((used, noinline)) void esp32_global_generic_producer() {
    selection_gpio_layout_sink = selection_gpio_config_size + selection_gpio_mode_offset
        + selection_public_pin_mode_input + selection_public_pin_mode_input_pullup + selection_public_pin_mode_input_pulldown + selection_gpio_mask_offset + selection_gpio_pull_up_offset + selection_gpio_pull_down_offset + selection_gpio_intr_offset + selection_public_pin_mode_output + selection_gpio_mode_input_output + selection_gpio_mode_output + selection_gpio_mode_input;
    fl::GenericSoftwareSPIOutput<2, 3, DATA_RATE_MHZ(1)> spi;
    spi.init(); spi.select(); spi.writeByte(selection_input);
    spi.writeBit<0>(selection_input); spi.release();

    fl::FastPin<2>::toggle();
}
