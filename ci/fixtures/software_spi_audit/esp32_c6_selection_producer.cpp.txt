#include <Arduino.h>
#include <FastLED.h>
#include "platforms/shared/spi_bitbang/generic_software_spi.h"
#include "driver/gpio.h"
#include <stddef.h>
// Read-only values derived from the real SDK type/enums, not a hand-written
// structure layout. Volatile reads below retain these data exports under LTO.
extern "C" {
extern const volatile fl::u32 selection_gpio_config_size __attribute__((used)) = sizeof(gpio_config_t);
extern const volatile fl::u32 selection_gpio_mode_offset __attribute__((used)) = offsetof(gpio_config_t, mode);
extern const volatile fl::u32 selection_gpio_mode_input_output __attribute__((used)) = GPIO_MODE_INPUT_OUTPUT;
extern const volatile fl::u32 selection_gpio_mode_output __attribute__((used)) = GPIO_MODE_OUTPUT;
extern const volatile fl::u32 selection_gpio_mode_input __attribute__((used)) = GPIO_MODE_INPUT;
}
volatile fl::u32 selection_gpio_layout_sink;
static_assert(fl::Esp32OutputPinValidity<2>::value, "SDK/user-mask GPIO2 output policy");
static_assert(fl::Esp32OutputPinValidity<3>::value, "SDK/user-mask GPIO3 output policy");
static_assert(!fl::Esp32OutputPinValidity<-1>::value, "negative GPIO index must be invalid without shift");
static_assert(!fl::Esp32OutputPinValidity<64>::value, "GPIO64 must be invalid without shift");
#if defined(FL_IS_ESP_32C6)
static_assert(!fl::Esp32OutputPinValidity<12>::value, "C6 USB GPIO12 must remain invalid");
static_assert(!fl::Esp32OutputPinValidity<13>::value, "C6 USB GPIO13 must remain invalid");
#endif
#if EXPECT_FORCED
#ifdef HAS_HARDWARE_PIN_SUPPORT
#error "forced selection must not advertise hardware pin support"
#endif
static_assert(__is_base_of(fl::detail::ForcedSoftwarePin<2>, fl::FastPin<2>), "forced GPIO2 must use ForcedSoftwarePin");
static_assert(__is_base_of(fl::detail::ForcedSoftwarePin<3>, fl::FastPin<3>), "forced GPIO3 must use ForcedSoftwarePin");
static_assert(!fl::detail::ForcedSoftwarePinMap<2>::kHasPinMap, "ESP32 default mapping policy must use portable provider");
static_assert(!fl::detail::ForcedSoftwarePinMap<3>::kHasPinMap, "ESP32 default clock mapping policy must use portable provider");
#else
#ifndef HAS_HARDWARE_PIN_SUPPORT
#error "native selection must advertise hardware pin support"
#endif
static_assert(__is_base_of(fl::_ESPPIN<2, (fl::u32(1) << 2), true>, fl::FastPin<2>), "native GPIO2 control");
static_assert(__is_base_of(fl::_ESPPIN<3, (fl::u32(1) << 3), true>, fl::FastPin<3>), "native GPIO3 control");
#endif
#if EXPECT_NO_PINMAP
#ifndef FASTLED_NO_PINMAP
#error "global mapper switch missing"
#endif
#else
#ifdef FASTLED_NO_PINMAP
#error "unexpected global mapper switch"
#endif
#endif
volatile fl::u8 selection_input = 0xa5;
extern "C" __attribute__((used, noinline)) void esp32_global_generic_producer() {
    selection_gpio_layout_sink = selection_gpio_config_size + selection_gpio_mode_offset
        + selection_gpio_mode_input_output + selection_gpio_mode_output + selection_gpio_mode_input;
    fl::GenericSoftwareSPIOutput<2, 3, DATA_RATE_MHZ(1)> spi;
    spi.init(); spi.select(); spi.writeByte(selection_input);
    spi.writeBit<0>(selection_input); spi.release();
    // Reachable direct pin API probe, separate from the SPI protocol phase.
    fl::FastPin<2>::toggle();
}
void setup() { esp32_global_generic_producer(); }
void loop() {}
